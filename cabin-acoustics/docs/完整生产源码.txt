using Cabin.Core.Domain;
using Cabin.Core.IO;

namespace Cabin.Core.Analysis;

public sealed record SeatDifference(string Name, double BaselinePeak, double CurrentPeak,
    double PeakRatioDb, double BaselineFrequency, double CurrentFrequency, double RmsRatioDb);
public sealed record ModeDifference(string Key, double BaselineFrequency, double CurrentFrequency, double Percent);
public sealed class ComparisonResult
{
    public List<string> Warnings { get; init; } = [];
    public List<SeatDifference> Seats { get; init; } = [];
    public List<ModeDifference> Modes { get; init; } = [];
    public double Reference { get; init; }
    public AnalysisResult BaselineResult { get; init; } = new();
}

public static class Comparison
{
    public static ComparisonResult Compare(Project current, AnalysisResult currentResult, Snapshot baseline)
    {
        var oldProject = Demo.FromSnapshot(baseline);
        var oldResult = Engine.Calculate(oldProject);
        var warnings = new List<string>();
        if (current.Model.MinimumFrequency != baseline.Model.MinimumFrequency ||
            current.Model.MaximumFrequency != baseline.Model.MaximumFrequency ||
            current.Model.FrequencyStep != baseline.Model.FrequencyStep)
            throw new ArgumentException("分析频带或步长不同，不能比较。请恢复基准频带与步长后重算。");
        if (current.Model.Cutoff != baseline.Model.Cutoff)
            throw new ArgumentException("模态截断不同，不能比较。请使用相同截断频率。");
        if (current.DataSource != baseline.DataSource || current.IsDemo != baseline.IsDemo)
            throw new ArgumentException("来源或模拟标记不同，不能比较。");
        if (current.Model.Damping != baseline.Model.Damping)
            warnings.Add("阻尼已变化；差值同时包含阻尼影响。");
        if (current.Model.SoundSpeed != baseline.Model.SoundSpeed)
            warnings.Add("声速已变化；差值同时包含介质条件影响。");
        if (current.Model.Length != baseline.Model.Length || current.Model.Width != baseline.Model.Width || current.Model.Height != baseline.Model.Height)
            warnings.Add("尺寸变化；模态按阶次匹配，耳点使用绝对米制坐标。");
        var differences = new List<SeatDifference>();
        foreach (var curve in currentResult.Curves)
        {
            var old = oldResult.Curves.FirstOrDefault(x => x.Id == curve.Id);
            if (old is null)
            {
                warnings.Add(curve.Name + "没有相同标识的基准耳点，跳过此耳点。");
                continue;
            }
            var oldSeat = baseline.Seats.First(x => x.Id == curve.Id);
            var seat = current.Seats.First(x => x.Id == curve.Id);
            if (oldSeat.Point.X != seat.Point.X || oldSeat.Point.Y != seat.Point.Y || oldSeat.Point.Z != seat.Point.Z)
                warnings.Add(curve.Name + "位置变化；差值包含测点变化。");
            if (old.Peak.Magnitude < 1e-12 || old.Rms < 1e-12)
            {
                warnings.Add(curve.Name + "基准近零，不能计算稳定比值。");
                continue;
            }
            differences.Add(new(curve.Name, old.Peak.Magnitude, curve.Peak.Magnitude,
                Engine.Decibels(curve.Peak.Magnitude, old.Peak.Magnitude), old.Peak.Frequency,
                curve.Peak.Frequency, Engine.Decibels(curve.Rms, old.Rms)));
        }
        if (differences.Count == 0)
            throw new ArgumentException("没有可匹配的非零基准耳点");
        var modes = new List<ModeDifference>();
        foreach (var mode in currentResult.Modes)
        {
            var old = oldResult.Modes.FirstOrDefault(x => x.Key == mode.Key);
            if (old is not null)
                modes.Add(new(mode.Key, old.Frequency, mode.Frequency, (mode.Frequency / old.Frequency - 1) * 100));
        }
        return new()
        {
            Warnings = warnings,
            Seats = differences,
            Modes = modes,
            Reference = Math.Max(oldResult.Reference, currentResult.Reference),
            BaselineResult = oldResult
        };
    }
}

public sealed record ScanSettings(string Parameter, double Start, double End, int Count, Guid SeatId);
public sealed record ScanRow(double Parameter, double PeakMagnitude, double PeakFrequency, double Rms, double FirstMode, string Error);
public sealed class ScanResult
{
    public ScanSettings Settings { get; init; } = new("length", 3, 4, 5, Guid.Empty);
    public List<ScanRow> Rows { get; init; } = [];
    public string Unit => Settings.Parameter is "damping" ? "无量纲" : "m";
}

public static class ParameterScan
{
    public static ScanResult Run(Project project, ScanSettings settings, CancellationToken cancellationToken = default, IProgress<int>? progress = null)
    {
        if (settings.Count is < 2 or > 21)
            throw new ArgumentException("扫描点数应为2至21");
        if (!double.IsFinite(settings.Start) || !double.IsFinite(settings.End) || settings.Start >= settings.End)
            throw new ArgumentException("扫描终值应大于初值，且均为有限数值");
        if (!project.Seats.Any(x => x.Id == settings.SeatId))
            throw new ArgumentException("请选择扫描耳点");
        if (settings.Parameter is not ("length" or "width" or "height" or "sourceX" or "sourceY" or "sourceZ" or "damping"))
            throw new ArgumentException("不支持的扫描参数");
        var rows = new List<ScanRow>();
        for (var i = 0; i < settings.Count; i++)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var value = settings.Start + (settings.End - settings.Start) * i / (settings.Count - 1);
            var candidate = ProjectCodec.Clone(project);
            candidate.Baselines.Clear();
            Apply(candidate, settings.Parameter, value);
            try
            {
                var result = Engine.Calculate(candidate, cancellationToken);
                var curve = result.Curves.First(x => x.Id == settings.SeatId);
                rows.Add(new(value, curve.Peak.Magnitude, curve.Peak.Frequency, curve.Rms,
                    result.Modes.FirstOrDefault()?.Frequency ?? 0, ""));
            }
            catch (ArgumentException error)
            {
                rows.Add(new(value, 0, 0, 0, 0, error.Message));
            }
            progress?.Report((i + 1) * 100 / settings.Count);
        }
        return new() { Settings = settings, Rows = rows };
    }

    public static void Apply(Project project, string parameter, double value)
    {
        switch (parameter)
        {
            case "length": project.Model.Length = value; break;
            case "width": project.Model.Width = value; break;
            case "height": project.Model.Height = value; break;
            case "damping": project.Model.Damping = value; break;
            default:
                var source = project.Sources.FirstOrDefault() ?? throw new ArgumentException("没有首个声源可扫描");
                switch (parameter)
                {
                    case "sourceX": source.Point.X = value; break;
                    case "sourceY": source.Point.Y = value; break;
                    case "sourceZ": source.Point.Z = value; break;
                    default: throw new ArgumentException("未知参数");
                }
                break;
        }
    }
}

using Cabin.Core.Domain;
using Cabin.Core.IO;

namespace Cabin.Core.Analysis;

public sealed record DensityBand(double Lower, double Upper, int Count, int Axial, int Tangential, int Oblique);
public sealed record CouplingDiagnostic(string ModeKey, double Frequency, double SourceMagnitude,
    double SeatShape, string Observation);
public sealed record SensitivityRow(string Seat, double OriginalRms, double ExtendedRms, double RatioDb,
    double OriginalPeak, double ExtendedPeak);
public sealed class TruncationStudy
{
    public double OriginalCutoff { get; init; }
    public double ExtendedCutoff { get; init; }
    public int OriginalModes { get; init; }
    public int ExtendedModes { get; init; }
    public List<SensitivityRow> Rows { get; init; } = [];
}

public static class Diagnostics
{
    public static List<DensityBand> Density(AnalysisResult result, double width = 25)
    {
        if (!double.IsFinite(width) || width <= 0)
            throw new ArgumentException("频段宽度应大于零");
        var maximum = result.Modes.Select(x => x.Frequency).DefaultIfEmpty(0).Max();
        if (maximum / width > 10000)
            throw new ArgumentException("模态密度频段不得超过10000个");
        var rows = new List<DensityBand>();
        for (var lower = 0.0; lower <= maximum; lower += width)
        {
            var upper = lower + width;
            var modes = result.Modes.Where(x => x.Frequency >= lower && x.Frequency < upper).ToList();
            rows.Add(new(lower, upper, modes.Count, modes.Count(x => x.Dimension == 1),
                modes.Count(x => x.Dimension == 2), modes.Count(x => x.Dimension == 3)));
        }
        return rows;
    }

    public static List<CouplingDiagnostic> Coupling(Project project, AnalysisResult result, EarPoint seat)
    {
        var rows = new List<CouplingDiagnostic>();
        foreach (var mode in result.Modes.Where(x => x.Frequency <= project.Model.MaximumFrequency))
        {
            var source = Modes.SourceCoupling(mode, project.Model, project.Sources).Magnitude;
            var shape = Modes.Shape(mode, project.Model, seat.Point);
            var maximumAmplitude = project.Sources.Where(x => x.Enabled).Sum(x => x.Amplitude);
            var weakSource = source < Math.Max(1e-12, maximumAmplitude * 0.01);
            var weakSeat = Math.Abs(shape) < 0.01;
            var observation = weakSource && weakSeat ? "源与耳点均弱耦合" :
                weakSource ? "源弱耦合或多源抵消" : weakSeat ? "耳点接近此模态节点" : "存在耦合";
            rows.Add(new(mode.Key, mode.Frequency, source, shape, observation));
        }
        return rows;
    }

    public static TruncationStudy Extend(Project project, AnalysisResult original)
    {
        var extended = ProjectCodec.Clone(project);
        extended.Baselines.Clear();
        var cutoff = Math.Min(1000, project.Model.Cutoff + 100);
        if (cutoff <= project.Model.Cutoff)
            throw new ArgumentException("当前截断已达1000Hz上限，不能继续增加");
        extended.Model.Cutoff = cutoff;
        var result = Engine.Calculate(extended);
        var rows = new List<SensitivityRow>();
        foreach (var curve in original.Curves)
        {
            var other = result.Curves.First(x => x.Id == curve.Id);
            var ratio = curve.Rms > 1e-12 ? Engine.Decibels(other.Rms, curve.Rms) : double.NaN;
            rows.Add(new(curve.Name, curve.Rms, other.Rms, ratio, curve.Peak.Magnitude, other.Peak.Magnitude));
        }
        return new()
        {
            OriginalCutoff = project.Model.Cutoff,
            ExtendedCutoff = cutoff,
            OriginalModes = original.Modes.Count,
            ExtendedModes = result.Modes.Count,
            Rows = rows
        };
    }

    public static List<string> Advice(Project project, AnalysisResult result)
    {
        var advice = new List<string>();
        if (result.Modes.Count == 0)
            advice.Add("截断内没有非零模态，考虑提高截断频率。");
        if (result.Reference < 1e-12)
            advice.Add("总响应近零，检查源相位抵消、零幅值与模态节点。");
        var modes = result.Modes.Where(x => x.Frequency >= project.Model.MinimumFrequency &&
            x.Frequency <= project.Model.MaximumFrequency).ToList();
        if (modes.Any(x => project.Model.FrequencyStep > project.Model.Damping * x.Frequency))
            advice.Add("部分模态的频率步长大于ζfn，可能漏掉窄峰。请细化步长，使用滑杆观察连续频率响应。");
        if (project.Model.Cutoff < project.Model.MaximumFrequency * 1.5)
            advice.Add("截断频率距分析终点较近，建议增加截断并检查RMS变化。");
        if (modes.Any(x => x.Degeneracy > 1))
            advice.Add("频带内存在退化模态。单个退化振型的基底不唯一，应关注其叠加响应。");
        if (project.Sources.Count(x => x.Enabled) > 1)
            advice.Add("多源使用同一相位参考，幅值叠加必须保留复数相位。");
        advice.Add("以上均为数值检查提示；没有提示不代表实车模型已校准。");
        return advice;
    }
}

using System.Numerics;
using Cabin.Core.Domain;

namespace Cabin.Core.Analysis;

public sealed record ResponsePoint(double Frequency, double Real, double Imaginary)
{
    public double Magnitude => Math.Sqrt(Real * Real + Imaginary * Imaginary);
    public double Phase => Math.Atan2(Imaginary, Real) * 180 / Math.PI;
    public double RelativeDb(double reference) => Engine.Decibels(Magnitude, reference);
}

public sealed record SeatCurve(Guid Id, string Name, List<ResponsePoint> Points)
{
    public ResponsePoint Peak => Points.OrderByDescending(x => x.Magnitude).First();
    public double Rms => Math.Sqrt(Points.Average(x => x.Magnitude * x.Magnitude));
}

public sealed class AnalysisResult
{
    public List<Mode> Modes { get; init; } = [];
    public List<SeatCurve> Curves { get; init; } = [];
    public List<double> Frequencies { get; init; } = [];
    public double Reference { get; init; }
    public DateTimeOffset Computed { get; init; } = DateTimeOffset.UtcNow;
    public int InBandModes(double minimum, double maximum) => Modes.Count(x => x.Frequency >= minimum && x.Frequency <= maximum);
}

public sealed record Contribution(string Key, double Frequency, double Magnitude, double Phase, double SourceMagnitude, double SeatShape);
public sealed record FieldCell(double U, double V, double Value);
public sealed record FieldResult(List<FieldCell> Cells, double Maximum, string Plane, double Slice, string Quantity);

public static class Engine
{
    public static List<double> FrequencyGrid(Cavity model)
    {
        var frequencies = new List<double>();
        var count = (int)Math.Floor((model.MaximumFrequency - model.MinimumFrequency) / model.FrequencyStep + 1e-8);
        for (var i = 0; i <= count; i++)
            frequencies.Add(model.MinimumFrequency + i * model.FrequencyStep);
        if (frequencies.Count == 0 || model.MaximumFrequency - frequencies[^1] > 1e-8)
            frequencies.Add(model.MaximumFrequency);
        return frequencies;
    }

    public static AnalysisResult Calculate(Project project, CancellationToken cancellationToken = default)
    {
        var issues = Validation.Check(project);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
        var modes = Modes.Enumerate(project.Model);
        var frequencies = FrequencyGrid(project.Model);
        if ((long)modes.Count * frequencies.Count * project.Seats.Count > 30_000_000)
            throw new ArgumentException("计算量超过3000万模态项，请减少截断、采样或耳点数量");
        var couplings = modes.Select(x => Modes.SourceCoupling(x, project.Model, project.Sources)).ToList();
        var curves = new List<SeatCurve>();
        foreach (var seat in project.Seats)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var points = new List<ResponsePoint>();
            foreach (var frequency in frequencies)
            {
                cancellationToken.ThrowIfCancellationRequested();
                var response = At(project.Model, modes, couplings, seat.Point, frequency);
                points.Add(new(frequency, response.Real, response.Imaginary));
            }
            curves.Add(new(seat.Id, seat.Name, points));
        }
        var reference = curves.SelectMany(x => x.Points).Select(x => x.Magnitude).DefaultIfEmpty(0).Max();
        return new()
        {
            Modes = modes,
            Frequencies = frequencies,
            Curves = curves,
            Reference = reference
        };
    }

    public static Complex At(Cavity model, IReadOnlyList<Mode> modes, IReadOnlyList<Complex> couplings, Position point, double frequency)
    {
        var sum = Complex.Zero;
        for (var i = 0; i < modes.Count; i++)
            sum += Modes.Term(modes[i], model, couplings[i], point, frequency);
        return sum;
    }

    public static Complex At(Project project, AnalysisResult result, Position point, double frequency)
    {
        var couplings = result.Modes.Select(x => Modes.SourceCoupling(x, project.Model, project.Sources)).ToList();
        return At(project.Model, result.Modes, couplings, point, frequency);
    }

    public static List<Contribution> Contributions(Project project, AnalysisResult result, EarPoint seat, double frequency)
    {
        return result.Modes.Select(mode =>
        {
            var source = Modes.SourceCoupling(mode, project.Model, project.Sources);
            var value = Modes.Term(mode, project.Model, source, seat.Point, frequency);
            return new Contribution(mode.Key, mode.Frequency, value.Magnitude,
                value.Phase * 180 / Math.PI, source.Magnitude, Modes.Shape(mode, project.Model, seat.Point));
        }).OrderByDescending(x => x.Magnitude).ToList();
    }

    public static FieldResult Field(Project project, AnalysisResult result, string plane, double fraction,
        bool shape, Mode? mode, double frequency, int columns = 48, int rows = 24)
    {
        if (plane is not ("xy" or "xz" or "yz"))
            throw new ArgumentException("剖面只能为xy、xz、yz");
        if (fraction < 0 || fraction > 1 || !double.IsFinite(fraction))
            throw new ArgumentException("剖面位置必须在0至1之间");
        if (columns is < 2 or > 128 || rows is < 2 or > 96)
            throw new ArgumentException("剖面网格列数2—128、行数2—96");
        if (!double.IsFinite(frequency) || frequency < project.Model.MinimumFrequency || frequency > project.Model.MaximumFrequency)
            throw new ArgumentException("研究频率必须位于分析频带内");
        if (shape && mode is null)
            throw new ArgumentException("振型显示需要选定模态");
        var cells = new List<FieldCell>();
        var couplings = result.Modes.Select(x => Modes.SourceCoupling(x, project.Model, project.Sources)).ToList();
        for (var row = 0; row < rows; row++)
        {
            for (var column = 0; column < columns; column++)
            {
                var u = (column + 0.5) / columns;
                var v = (row + 0.5) / rows;
                var point = PlanePoint(project.Model, plane, u, v, fraction);
                var value = shape && mode is not null ? Modes.Shape(mode, project.Model, point) :
                    At(project.Model, result.Modes, couplings, point, frequency).Magnitude;
                cells.Add(new(u, v, value));
            }
        }
        var maximum = shape ? 1 : cells.Select(x => x.Value).DefaultIfEmpty(0).Max();
        return new(cells, maximum, plane, fraction, shape ? "有符号归一化振型" : "剖面局部归一化响应幅值");
    }

    public static Position PlanePoint(Cavity model, string plane, double u, double v, double fraction)
    {
        return plane switch
        {
            "xy" => new(u * model.Length, v * model.Width, fraction * model.Height),
            "xz" => new(u * model.Length, fraction * model.Width, v * model.Height),
            "yz" => new(fraction * model.Length, u * model.Width, v * model.Height),
            _ => throw new ArgumentException("未知剖面")
        };
    }

    public static (double U, double V) Project(Position point, Cavity model, string plane)
    {
        return plane switch
        {
            "xy" => (point.X / model.Length, point.Y / model.Width),
            "xz" => (point.X / model.Length, point.Z / model.Height),
            "yz" => (point.Y / model.Width, point.Z / model.Height),
            _ => throw new ArgumentException("未知剖面")
        };
    }

    public static double Decibels(double value, double reference)
    {
        if (reference <= 1e-15)
            return -120;
        return Math.Max(-120, 20 * Math.Log10(Math.Max(value, reference * 1e-6) / reference));
    }

    public static string FieldColor(double value, double maximum, bool shape)
    {
        var t = maximum > 1e-15 ? Math.Clamp(value / maximum, shape ? -1 : 0, 1) : 0;
        if (shape)
        {
            var strength = Math.Abs(t);
            var target = t >= 0 ? (R: 204, G: 104, B: 56) : (R: 35, G: 117, B: 133);
            var r = (int)(245 + (target.R - 245) * strength);
            var g = (int)(241 + (target.G - 241) * strength);
            var b = (int)(225 + (target.B - 225) * strength);
            return $"rgb({r},{g},{b})";
        }
        return $"hsl({195 - 170 * t:F0},48%,{90 - 45 * t:F0}%)";
    }
}


===== src/Cabin.Core/Analysis/Modes.cs =====
using System.Numerics;
using Cabin.Core.Domain;

namespace Cabin.Core.Analysis;

public sealed record Mode(int Nx, int Ny, int Nz, double Frequency, int Degeneracy = 1)
{
    public string Key => $"{Nx},{Ny},{Nz}";
    public int Dimension => (Nx > 0 ? 1 : 0) + (Ny > 0 ? 1 : 0) + (Nz > 0 ? 1 : 0);
    public string Kind => Dimension switch { 1 => "轴向", 2 => "切向", _ => "斜向" };
    public double NormFactor => Math.Pow(2, Dimension);
}

public static class Modes
{
    public static List<Mode> Enumerate(Cavity model)
    {
        var issues = new List<Issue>();
        Validation.Model(model, issues);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
        var result = new List<Mode>();
        var maxX = (int)Math.Floor(2 * model.Cutoff * model.Length / model.SoundSpeed);
        var maxY = (int)Math.Floor(2 * model.Cutoff * model.Width / model.SoundSpeed);
        var maxZ = (int)Math.Floor(2 * model.Cutoff * model.Height / model.SoundSpeed);
        for (var nx = 0; nx <= maxX; nx++)
        {
            for (var ny = 0; ny <= maxY; ny++)
            {
                for (var nz = 0; nz <= maxZ; nz++)
                {
                    if (nx + ny + nz == 0)
                        continue;
                    var f = Frequency(model, nx, ny, nz);
                    if (f <= model.Cutoff + 1e-9)
                        result.Add(new(nx, ny, nz, f));
                    if (result.Count > 10000)
                        throw new ArgumentException("模态数量超过10000，请降低截断频率");
                }
            }
        }
        result = result.OrderBy(x => x.Frequency).ThenBy(x => x.Nx).ThenBy(x => x.Ny).ThenBy(x => x.Nz).ToList();
        var start = 0;
        while (start < result.Count)
        {
            var end = start + 1;
            var tolerance = Math.Max(1e-8, result[start].Frequency * 1e-8);
            while (end < result.Count && result[end].Frequency - result[start].Frequency <= tolerance)
                end++;
            for (var i = start; i < end; i++)
                result[i] = result[i] with { Degeneracy = end - start };
            start = end;
        }
        return result;
    }

    public static double Frequency(Cavity model, int nx, int ny, int nz)
    {
        return model.SoundSpeed / 2 * Math.Sqrt(
            Math.Pow(nx / model.Length, 2) +
            Math.Pow(ny / model.Width, 2) +
            Math.Pow(nz / model.Height, 2));
    }

    public static double Shape(Mode mode, Cavity model, Position point)
    {
        return Math.Cos(mode.Nx * Math.PI * point.X / model.Length) *
            Math.Cos(mode.Ny * Math.PI * point.Y / model.Width) *
            Math.Cos(mode.Nz * Math.PI * point.Z / model.Height);
    }

    public static Complex SourceCoupling(Mode mode, Cavity model, IEnumerable<Emitter> sources)
    {
        var sum = Complex.Zero;
        foreach (var source in sources.Where(x => x.Enabled))
        {
            var phase = source.PhaseDegrees * Math.PI / 180;
            sum += Complex.FromPolarCoordinates(source.Amplitude, phase) * Shape(mode, model, source.Point);
        }
        return sum;
    }

    public static Complex Term(Mode mode, Cavity model, Complex coupling, Position point, double frequency)
    {
        var fn = mode.Frequency;
        var denominator = new Complex(fn * fn - frequency * frequency, 2 * model.Damping * fn * frequency);
        return mode.NormFactor * Software.ReferenceVolume / model.Volume * Shape(mode, model, point) * coupling *
            (Software.ReferenceFrequency * Software.ReferenceFrequency) / denominator;
    }
}

                continue;

            Model(baseline.Model, issues);
            Points(baseline.Model, baseline.Sources, baseline.Seats, issues);
        }

    public static void Model(Cavity model, List<Issue> issues)
    {
        Range(model.Length, 0.5, 10, "长度 Lx / m", issues);
        Range(model.Width, 0.5, 10, "宽度 Ly / m", issues);
        Range(model.Height, 0.5, 10, "高度 Lz / m", issues);
        Range(model.SoundSpeed, 300, 380, "声速 / m/s", issues);
        Range(model.Damping, 0.002, 0.2, "模态阻尼比", issues);
        Range(model.MinimumFrequency, 5, 499, "频带起点 / Hz", issues);
        Range(model.MaximumFrequency, 6, 500, "频带终点 / Hz", issues);
        Range(model.FrequencyStep, 0.25, 10, "频率步长 / Hz", issues);
        Range(model.Cutoff, 100, 1000, "模态截断 / Hz", issues);
        if (model.MinimumFrequency >= model.MaximumFrequency)
            issues.Add(new("频带", "终点必须大于起点"));
        if (model.Cutoff < model.MaximumFrequency)
            issues.Add(new("模态截断", "截断频率不能低于分析终点"));
        if ((model.MaximumFrequency - model.MinimumFrequency) / model.FrequencyStep > 2000)
            issues.Add(new("频率步长", "频率采样最多2001点"));
        if (double.IsFinite(model.Cutoff) && double.IsFinite(model.SoundSpeed) && model.SoundSpeed > 0)
        {
            var estimate = 4 * Math.PI / 3 * model.Volume * Math.Pow(model.Cutoff / model.SoundSpeed, 3);
            if (estimate > 6000)
                issues.Add(new("模态截断", "模型模态数量过大，请降低截断频率或缩小尺寸"));
        }
    }

            issues.Add(new(path, $"最多{maximum}字符，不允许控制字符"));
    }

    {
        if (!double.IsFinite(value) || value < minimum || value > maximum)
            issues.Add(new(path, $"必须为{minimum}至{maximum}之间的有限数值"));
    }

namespace Cabin.Core.IO;

public static class NativeFiles
{
    public static Project Open(string path)
    {
        var info = new FileInfo(path);
        if (!info.Exists)
            throw new IOException("项目文件不存在：" + path);
        if (info.Length is <= 0 or > Software.MaximumBytes)
            throw new ArgumentException("项目文件必须为非空且不超过2MiB");
        return ProjectCodec.Decode(File.ReadAllBytes(path));
    }

    public static void Save(string path, Project project)
    {
        var bytes = ProjectCodec.Encode(project);
        _ = ProjectCodec.Decode(bytes);
        AtomicWrite(path, bytes);
    }

    public static void WriteText(string path, string text)
    {
        AtomicWrite(path, new UTF8Encoding(true).GetBytes(text));
    }

    public static string ReadCsv(string path)
    {
        var info = new FileInfo(path);
        if (!info.Exists || info.Length > 400_000)
            throw new ArgumentException("点位CSV不存在或超过400000字节");
        return File.ReadAllText(path, new UTF8Encoding(false, true));
    }

    private static void AtomicWrite(string path, byte[] bytes)
    {
        var absolute = Path.GetFullPath(path);
        var directory = Path.GetDirectoryName(absolute)!;
        if (!Directory.Exists(directory))
            throw new DirectoryNotFoundException("请先建立保存目录：" + directory);
        var temporary = absolute + "." + Guid.NewGuid().ToString("N") + ".tmp";
        try
        {
            File.WriteAllBytes(temporary, bytes);
            File.Move(temporary, absolute, true);
        }
        finally
        {
            if (File.Exists(temporary))
                File.Delete(temporary);
        }
    }
}

using System.Globalization;
using System.Text;
using Cabin.Core.Domain;

namespace Cabin.Core.IO;

public sealed record PointRow(string Kind, string Name, Position Point, double Amplitude, double Phase, string Note);
public sealed class CsvPreview
{
    public List<PointRow> Rows { get; } = [];
    public List<Issue> Issues { get; } = [];
    public bool CanApply => Rows.Count > 0 && Issues.Count == 0;
}

public static class PointCsv
{
    public const string Header = "kind,name,x_m,y_m,z_m,amplitude,phase_deg,note";

    public static CsvPreview Preview(string text, Cavity model)
    {
        var preview = new CsvPreview();
        if (text.Length > 100000)
        {
            preview.Issues.Add(new("CSV", "点位CSV最多100000字符"));
            return preview;
        }
        var lines = text.TrimStart('\uFEFF').Replace("\r\n", "\n").Split('\n');
        if (lines.Length == 0 || lines[0].Trim() != Header)
        {
            preview.Issues.Add(new("第1行", "表头必须为：" + Header));
            return preview;
        }
        for (var i = 1; i < lines.Length; i++)
        {
            if (string.IsNullOrWhiteSpace(lines[i]))
                continue;
            var path = "第" + (i + 1) + "行";
            try
            {
                var cells = ParseLine(lines[i]);
                if (cells.Count != 8)
                    throw new ArgumentException("需要8列");
                if (cells[0] is not ("source" or "seat"))
                    throw new ArgumentException("kind必须为source或seat");
                var point = new Position(Number(cells[2]), Number(cells[3]), Number(cells[4]));
                var amplitude = cells[0] == "source" ? Number(cells[5]) : 0;
                var phase = cells[0] == "source" ? Number(cells[6]) : 0;
                var row = new PointRow(cells[0], cells[1], point, amplitude, phase, cells[7]);
                var issues = new List<Issue>();
                var sources = row.Kind == "source" ? new List<Emitter> { new() { Name = row.Name, Point = row.Point, Amplitude = amplitude, PhaseDegrees = phase } } : [];
                var seats = row.Kind == "seat" ? new List<EarPoint> { new() { Name = row.Name, Point = row.Point, Note = row.Note } } : [];
                Validation.Points(model, sources, seats, issues);
                if (issues.Count > 0)
                    throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
                preview.Rows.Add(row);
            }
            catch (ArgumentException error)
            {
                preview.Issues.Add(new(path, error.Message));
            }
        }
        if (preview.Rows.Count(x => x.Kind == "source") > 8 || preview.Rows.Count(x => x.Kind == "seat") > 12)
            preview.Issues.Add(new("CSV", "最多8个声源、12个耳点"));
        if (preview.Rows.Count == 0)
            preview.Issues.Add(new("CSV", "没有有效点位"));
        return preview;
    }

    public static void Apply(Project project, CsvPreview preview)
    {
        if (!preview.CanApply)
            throw new ArgumentException("CSV预检未通过；项目未改变");
        var sources = new List<Emitter>();
        var seats = new List<EarPoint>();
        foreach (var row in preview.Rows)
        {
            if (row.Kind == "source")
                sources.Add(new() { Name = row.Name, Point = ProjectCodec.Clone(row.Point), Amplitude = row.Amplitude, PhaseDegrees = row.Phase });
            else
                seats.Add(new() { Name = row.Name, Point = ProjectCodec.Clone(row.Point), Note = row.Note });
        }
        var issues = new List<Issue>();
        Validation.Points(project.Model, sources, seats, issues);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
        project.Sources = sources;
        project.Seats = seats;
    }

    public static string Export(Project project)
    {
        var builder = new StringBuilder(Header + Environment.NewLine);
        foreach (var source in project.Sources)
            builder.AppendLine(Join("source", source.Name, N(source.Point.X), N(source.Point.Y), N(source.Point.Z), N(source.Amplitude), N(source.PhaseDegrees), source.Enabled ? "启用" : "停用；导入时需手动停用"));
        foreach (var seat in project.Seats)
            builder.AppendLine(Join("seat", seat.Name, N(seat.Point.X), N(seat.Point.Y), N(seat.Point.Z), "", "", seat.Note));
        return builder.ToString();
    }

    public static string Join(params string[] fields) => string.Join(",", fields.Select(Quote));
    public static string N(double value) => value.ToString("G12", CultureInfo.InvariantCulture);
    public static string Quote(string value)
    {
        if (value.Contains(',') || value.Contains('"') || value.Contains('\n') || value.Contains('\r'))
            return "\"" + value.Replace("\"", "\"\"") + "\"";
        return value;
    }

    public static List<string> ParseLine(string line)
    {
        var cells = new List<string>();
        var value = new StringBuilder();
        var quoted = false;
        var closed = false;
        for (var i = 0; i < line.Length; i++)
        {
            var c = line[i];
            if (quoted)
            {
                if (c == '"')
                {
                    if (i + 1 < line.Length && line[i + 1] == '"')
                    {
                        value.Append('"');
                        i++;
                    }
                    else
                    {
                        quoted = false;
                        closed = true;
                    }
                }
                else
                    value.Append(c);
            }
            else if (c == ',')
            {
                cells.Add(value.ToString());
                value.Clear();
                closed = false;
            }
            else if (c == '"' && value.Length == 0 && !closed)
                quoted = true;
            else if (closed || c == '"')
                throw new ArgumentException("引号格式错误；不支持单元格内换行");
            else
                value.Append(c);
        }
        if (quoted)
            throw new ArgumentException("引号未闭合；不支持单元格内换行");
        cells.Add(value.ToString());
        return cells;
    }

    private static double Number(string value)
    {
        if (!double.TryParse(value, NumberStyles.Float, CultureInfo.InvariantCulture, out var number) || !double.IsFinite(number))
            throw new ArgumentException("数值使用小数点，不能为空或非有限值");
        return number;
    }
}

using System.Text.Json;
using System.Text.Json.Serialization;
using Cabin.Core.Domain;

namespace Cabin.Core.IO;

public static class ProjectCodec
{
    public static readonly JsonSerializerOptions Options = new()
    {
        WriteIndented = true,
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow,
        MaxDepth = 24
    };

    public static byte[] Encode(Project project)
    {
        var issues = Validation.Check(project, false);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
        var bytes = JsonSerializer.SerializeToUtf8Bytes(project, Options);
        if (bytes.Length > Software.MaximumBytes)
            throw new ArgumentException("编码后的项目超过2MiB，未保存");
        return bytes;
    }

    public static Project Decode(byte[] bytes)
    {
        if (bytes.Length == 0 || bytes.Length > Software.MaximumBytes)
            throw new ArgumentException("项目应为非空JSON，且不超过2MiB");
        try
        {
            using var document = JsonDocument.Parse(bytes, new() { MaxDepth = 24 });
            var root = document.RootElement;
            Require(root, "schema", "id", "name", "revision", "dataSource", "isDemo", "created", "modified", "model", "sources", "seats", "baselines", "reviews", "journal");
            if (root.GetProperty("schema").GetInt32() != Software.Schema)
                throw new ArgumentException("只支持项目格式版本1；未执行迁移");
            Require(root.GetProperty("model"), "length", "width", "height", "soundSpeed", "damping", "minimumFrequency", "maximumFrequency", "frequencyStep", "cutoff");
            foreach (var source in root.GetProperty("sources").EnumerateArray())
            {
                Require(source, "id", "name", "point", "amplitude", "phaseDegrees", "enabled");
                Require(source.GetProperty("point"), "x", "y", "z");
            }
            foreach (var seat in root.GetProperty("seats").EnumerateArray())
            {
                Require(seat, "id", "name", "point");
                Require(seat.GetProperty("point"), "x", "y", "z");
            }
            foreach (var baseline in root.GetProperty("baselines").EnumerateArray())
            {
                Require(baseline, "id", "name", "model", "sources", "seats", "isDemo", "dataSource");
                Require(baseline.GetProperty("model"), "length", "width", "height", "soundSpeed", "damping", "minimumFrequency", "maximumFrequency", "frequencyStep", "cutoff");
                foreach (var source in baseline.GetProperty("sources").EnumerateArray())
                {
                    Require(source, "id", "name", "point", "amplitude", "phaseDegrees", "enabled");
                    Require(source.GetProperty("point"), "x", "y", "z");
                }
                foreach (var seat in baseline.GetProperty("seats").EnumerateArray())
                {
                    Require(seat, "id", "name", "point");
                    Require(seat.GetProperty("point"), "x", "y", "z");
                }
            }
            foreach (var item in root.GetProperty("reviews").EnumerateArray())
                Require(item, "id", "target", "action", "owner", "result");
            foreach (var entry in root.GetProperty("journal").EnumerateArray())
                Require(entry, "time", "action", "detail");
            var project = JsonSerializer.Deserialize<Project>(bytes, Options) ?? throw new ArgumentException("项目内容为null");
            var issues = Validation.Check(project, false);
            if (issues.Count > 0)
                throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
            return project;
        }
        catch (Exception error) when (error is JsonException or InvalidOperationException or FormatException or OverflowException)
        {
            throw new ArgumentException("JSON项目结构或数值格式错误：" + error.Message);
        }
    }

    public static T Clone<T>(T value)
    {
        return JsonSerializer.Deserialize<T>(JsonSerializer.SerializeToUtf8Bytes(value, Options), Options)!;
    }

    private static void Require(JsonElement element, params string[] fields)
    {
        if (element.ValueKind != JsonValueKind.Object)
            throw new ArgumentException("预期JSON对象");
        var names = element.EnumerateObject().Select(x => x.Name).ToList();
        if (names.Distinct(StringComparer.Ordinal).Count() != names.Count)
            throw new ArgumentException("JSON包含重复字段");
        foreach (var field in fields)
        {
            if (!element.TryGetProperty(field, out var value) || value.ValueKind == JsonValueKind.Null)
                throw new ArgumentException("缺少字段：" + field);
        }
    }
}

                row.Error.Length == 0 ? N(row.PeakFrequency) : "", row.Error.Length == 0 ? N(row.Rms) : "",
                row.Error.Length == 0 ? N(row.FirstMode) : "", row.Error));

    private static StringBuilder Start(Project project, string header)
    {
        var output = new StringBuilder();
        output.AppendLine("# " + Software.Name + " V" + Software.Version);
        output.AppendLine("# " + Software.Scope);
        output.AppendLine("# project=" + OneLine(project.Name) + "; id=" + project.Id + "; revision=" + OneLine(project.Revision));
        output.AppendLine("# source=" + OneLine(project.DataSource) + "; demo=" + project.IsDemo);
        output.AppendLine("# c_m_s=" + N(project.Model.SoundSpeed) + "; damping=" + N(project.Model.Damping) + "; cutoff_hz=" + N(project.Model.Cutoff));
        output.AppendLine(header);
        return output;
    }

    private static void Pair(StringBuilder output, string key, string value)
    {
        output.AppendLine(key + "：" + value);
    }

    private static void Section(StringBuilder output, string title)
    {
        output.AppendLine();
        output.AppendLine("=== " + title + " ===");
    }

    private static string OneLine(string value)
    {
        return value.Replace("\r", " ").Replace("\n", " ");
    }

    private static string N(double value)
    {
        return double.IsFinite(value) ? value.ToString("G12", Invariant) : "不可计算（近零基准）";
    }

using System.Globalization;
using System.Windows;
using System.Windows.Media;

namespace Cabin.Desktop.Controls;

internal static class DrawingTools
{
    public static Brush Brush(string color)
    {
        var brush = new SolidColorBrush((Color)ColorConverter.ConvertFromString(color));
        brush.Freeze();
        return brush;
    }

    public static void Text(DrawingContext context, string text, double x, double y,
        double size = 12, string color = "#395762", bool bold = false, double maximumWidth = 10000)
    {
        var formatted = new FormattedText(text, CultureInfo.GetCultureInfo("zh-CN"), FlowDirection.LeftToRight,
            new Typeface(new FontFamily("Microsoft YaHei UI, Segoe UI"), FontStyles.Normal,
                bold ? FontWeights.SemiBold : FontWeights.Normal, FontStretches.Normal),
            size, Brush(color), 1.0)
        {
            MaxTextWidth = Math.Max(1, maximumWidth),
            Trimming = TextTrimming.CharacterEllipsis,
            MaxLineCount = 2
        };
        context.DrawText(formatted, new Point(x, y));
    }

    public static string Number(double value)
    {
        if (Math.Abs(value) >= 10000 || (Math.Abs(value) < 0.001 && value != 0))
            return value.ToString("0.##E+0", CultureInfo.InvariantCulture);
        return value.ToString("0.###", CultureInfo.InvariantCulture);
    }

    public static void Line(DrawingContext context, double x1, double y1, double x2, double y2,
        string color, double width = 1)
    {
        context.DrawLine(new Pen(Brush(color), width), new Point(x1, y1), new Point(x2, y2));
    }
}

    private Rect region;
    private int columns;
    private int rows;

    public FieldCanvas()
    {
        MinHeight = 330;
        ClipToBounds = true;
        ToolTip = "移动指针读取最近网格点的坐标和相对量。";
    }

    public FieldResult? Field
    {
        get => (FieldResult?)GetValue(FieldProperty);
        set => SetValue(FieldProperty, value);
    }

    public Project? Project
    {
        get => (Project?)GetValue(ProjectProperty);
        set => SetValue(ProjectProperty, value);
    }

    protected override void OnRender(DrawingContext context)
    {
        base.OnRender(context);
        context.DrawRectangle(DrawingTools.Brush("#FFFFFF"), null, new Rect(0, 0, ActualWidth, ActualHeight));
        if (Field is null || Project is null || Field.Cells.Count == 0)
        {
            DrawingTools.Text(context, "剖面待更新", 24, 28, 18, bold: true);
            DrawingTools.Text(context, "完成计算后，选择剖面、频率及显示量，再点击“更新剖面与贡献”。", 24, 66, maximumWidth: ActualWidth - 48);
            return;
        }
        var f = Field;
        var model = Project.Model;
        var horizontal = f.Plane == "yz" ? model.Width : model.Length;
        var vertical = f.Plane == "xy" ? model.Width : model.Height;
        var availableWidth = Math.Max(1, ActualWidth - 110);
        var availableHeight = Math.Max(1, ActualHeight - 155);
        var scale = Math.Min(availableWidth / horizontal, availableHeight / vertical);
        var width = horizontal * scale;
        var height = vertical * scale;
        region = new Rect(65 + (availableWidth - width) / 2, 58 + (availableHeight - height) / 2, width, height);
        columns = f.Cells.Select(x => x.U).Distinct().Count();
        rows = f.Cells.Select(x => x.V).Distinct().Count();
        DrawingTools.Text(context, f.Quantity, 18, 12, 15, bold: true, maximumWidth: ActualWidth - 36);
        DrawingTools.Text(context, (Project.IsDemo ? "模拟数据 · " : "人工理想模型 · ") + f.Plane.ToUpperInvariant() + " 固定轴比例=" + DrawingTools.Number(f.Slice), 18, 36, 11);
        var shape = f.Quantity.Contains("振型", StringComparison.Ordinal);
        foreach (var cell in f.Cells)
        {
            var rectangle = new Rect(region.Left + (cell.U - 0.5 / columns) * region.Width,
                region.Bottom - (cell.V + 0.5 / rows) * region.Height, region.Width / columns + 0.5, region.Height / rows + 0.5);
            context.DrawRectangle(ColorBrush(cell.Value, f.Maximum, shape), null, rectangle);
        }
        context.DrawRectangle(null, new Pen(DrawingTools.Brush("#3A5D6A"), 1.2), region);
        for (var i = 0; i <= 4; i++)
        {
            var x = region.Left + region.Width * i / 4;
            var y = region.Bottom - region.Height * i / 4;
            DrawingTools.Line(context, x, region.Bottom, x, region.Bottom + 5, "#3A5D6A");
            DrawingTools.Line(context, region.Left - 5, y, region.Left, y, "#3A5D6A");
            DrawingTools.Text(context, DrawingTools.Number(horizontal * i / 4), x - 10, region.Bottom + 9, 10);
            DrawingTools.Text(context, DrawingTools.Number(vertical * i / 4), region.Left - 44, y - 7, 10);

        DrawingTools.Text(context, (f.Plane == "yz" ? "y" : "x") + " / m", region.Right + 8, region.Bottom + 9, 11);
        DrawingTools.Text(context, (f.Plane == "xy" ? "y" : "z") + " / m", region.Left - 42, region.Top - 25, 11);

    {

        {

    }

    private static Brush ColorBrush(double value, double maximum, bool shape)
    {
        var ratio = maximum > 1e-15 ? Math.Clamp(value / maximum, shape ? -1 : 0, 1) : 0;
        var start = Color.FromRgb(247, 246, 236);
        var end = shape && ratio < 0 ? Color.FromRgb(36, 110, 146) : Color.FromRgb(193, 92, 42);
        var amount = Math.Abs(ratio);
        var color = Color.FromRgb((byte)(start.R + (end.R - start.R) * amount),
            (byte)(start.G + (end.G - start.G) * amount), (byte)(start.B + (end.B - start.B) * amount));
        var brush = new SolidColorBrush(color);
        brush.Freeze();
        return brush;
    }

    protected override void OnMouseMove(MouseEventArgs e)
    {
        base.OnMouseMove(e);
        var position = e.GetPosition(this);
        if (Field is null || Project is null || !region.Contains(position))
            return;
        var u = (position.X - region.Left) / region.Width;
        var v = (region.Bottom - position.Y) / region.Height;
        var cell = Field.Cells.MinBy(x => Math.Pow(x.U - u, 2) + Math.Pow(x.V - v, 2))!;
        var point = Engine.PlanePoint(Project.Model, Field.Plane, cell.U, cell.V, Field.Slice);
        ToolTip = $"网格点 x={point.X:F3}m, y={point.Y:F3}m, z={point.Z:F3}m\n{Field.Quantity}：{DrawingTools.Number(cell.Value)}";
    }

    private Point Map(double u, double v)
    {
        return new Point(region.Left + u * region.Width, region.Bottom - v * region.Height);
    }
}

        maximumX = points.Max(x => x.X);
        minimumY = points.Min(x => x.Y);
        maximumY = points.Max(x => x.Y);
        if (minimumX == maximumX)
        {
            minimumX -= 0.5;
            maximumX += 0.5;

        if (minimumY == maximumY)

            DrawingTools.Line(context, plot.Left, y, plot.Right, y, "#E4EBEC");

            DrawingTools.Text(context, DrawingTools.Number(minimumY + (maximumY - minimumY) * i / 5), 10, y - 7, 10, maximumWidth: 62);

        context.DrawRectangle(null, new Pen(DrawingTools.Brush("#8EA8AE"), 1), plot);

    private void DrawSeries(DrawingContext context, PlotLine line)
    {
        if (line.Points.Count == 0)
            return;
        var geometry = new StreamGeometry();
        using (var drawing = geometry.Open())
        {
            var started = false;
            foreach (var point in line.Points)
            {
                if (!double.IsFinite(point.X) || !double.IsFinite(point.Y))
                {
                    started = false;
                    continue;
                }
                var mapped = new Point(MapX(point.X), MapY(point.Y));
                if (!started)
                {
                    drawing.BeginFigure(mapped, false, false);
                    started = true;
                }
                else
                    drawing.LineTo(mapped, true, false);
            }
        }
        geometry.Freeze();
        context.DrawGeometry(null, new Pen(DrawingTools.Brush(line.Color), 1.8), geometry);
        if (line.Points.Count <= 25)
        {
            foreach (var point in line.Points.Where(p => double.IsFinite(p.X) && double.IsFinite(p.Y)))
                context.DrawEllipse(DrawingTools.Brush(line.Color), null, new Point(MapX(point.X), MapY(point.Y)), 3, 3);
        }
    }

    protected override void OnMouseLeftButtonDown(MouseButtonEventArgs e)
    {
        base.OnMouseLeftButtonDown(e);
        var position = e.GetPosition(this);

    }

    protected override void OnMouseMove(MouseEventArgs e)
    {
        base.OnMouseMove(e);
        var position = e.GetPosition(this);
        if (Data is null || !plot.Contains(position))
            return;
        var x = minimumX + (position.X - plot.Left) / plot.Width * (maximumX - minimumX);
        var values = Data.Lines.Where(l => l.Points.Count > 0).Select(l =>
        {
            var point = l.Points.MinBy(p => Math.Abs(p.X - x))!;
            return l.Name + ": " + DrawingTools.Number(point.Y);
        });
        ToolTip = DrawingTools.Number(x) + " | 最近样点\n" + string.Join("\n", values);
    }

    private double MapX(double value) => plot.Left + (value - minimumX) / (maximumX - minimumX) * plot.Width;
    private double MapY(double value) => plot.Bottom - (value - minimumY) / (maximumY - minimumY) * plot.Height;
}

    }
}

public sealed class AsyncCommand(Func<Task> execute, Func<bool>? allowed = null) : ICommand
{
    private bool running;
    public bool CanExecute(object? parameter) => !running && (allowed?.Invoke() ?? true);

    public async void Execute(object? parameter)
    {
        if (!CanExecute(parameter))
            return;
        running = true;
        CommandManager.InvalidateRequerySuggested();
        try
        {
            await execute();
        }
        finally
        {
            running = false;
            CommandManager.InvalidateRequerySuggested();
        }
    }

    public event EventHandler? CanExecuteChanged
    {
        add => CommandManager.RequerySuggested += value;
        remove => CommandManager.RequerySuggested -= value;
    }
}

public abstract class ObservableObject : INotifyPropertyChanged

    protected bool Set<T>(ref T field, T value, [CallerMemberName] string? name = null)
    {

            return false;
        field = value;

    }

using System.Collections;
using System.ComponentModel;
using System.Globalization;
using Cabin.Desktop.Infrastructure;

namespace Cabin.Desktop.ViewModels;

public sealed class InputField : ObservableObject, INotifyDataErrorInfo
{
    private string text;
    private readonly Action changed;
    private readonly double? minimum;
    private readonly double? maximum;
    private readonly int maximumLength;
    private readonly bool required;
    private string? error;

    public InputField(string label, string value, Action changed, string hint = "",
        bool required = false, int maximumLength = 3000, double? minimum = null, double? maximum = null)
    {
        Label = label;
        text = value;
        this.changed = changed;
        Hint = hint;
        this.minimum = minimum;
        this.maximum = maximum;
        this.maximumLength = maximumLength;
        this.required = required;
        Validate();
    }

    public static InputField Number(string label, double value, double min, double max, Action changed, string hint)
    {
        return new(label, value.ToString("G12", CultureInfo.InvariantCulture), changed,
            hint, true, 40, min, max);
    }

    public string Label { get; }
    public string Hint { get; }
    public bool IsNumeric => minimum.HasValue;
    public bool HasErrors => error is not null;
    public string Error => error ?? "";
    public string Text
    {
        get => text;
        set
        {
            if (!Set(ref text, value))
                return;
            Validate();
            changed();
        }
    }

    public double Value
    {
        get
        {
            Validate();
            if (HasErrors)
                throw new ArgumentException(Label + "：" + error);
            return double.Parse(Text, NumberStyles.Float, CultureInfo.InvariantCulture);
        }
    }

    public event EventHandler<DataErrorsChangedEventArgs>? ErrorsChanged;

    public IEnumerable GetErrors(string? propertyName)
    {
        return error is null ? Array.Empty<string>() : new[] { error };
    }

    public void EnsureValid()
    {
        Validate();
        if (HasErrors)
            throw new ArgumentException(Label + "：" + error);
    }

    private void Validate()
    {
        string? next = null;
        if (required && string.IsNullOrWhiteSpace(text))
            next = "不能为空";
        else if (text.Length > maximumLength)
            next = $"不能超过{maximumLength}字符";
        else if (IsNumeric)
        {
            if (!double.TryParse(text, NumberStyles.Float, CultureInfo.InvariantCulture, out var number) || !double.IsFinite(number))
                next = "请输入有限数值，使用小数点";
            else if (number < minimum || number > maximum)
                next = $"范围为{minimum}至{maximum}";
        }
        if (error == next)
            return;
        error = next;
        Notify(nameof(HasErrors));
        Notify(nameof(Error));
        ErrorsChanged?.Invoke(this, new DataErrorsChangedEventArgs(nameof(Text)));
    }
}

    public Project Build(bool forCalculation)

            field.EnsureValid();
        var project = ProjectCodec.Clone(original);
        project.Name = Name.Text;
        project.Revision = Revision.Text;
        project.Owner = Owner.Text;
        project.DataSource = Source.Text;
        project.Notes = Notes.Text;
        project.Modified = DateTimeOffset.UtcNow;
        project.Model = new()
        {
            Length = Length.Value,
            Width = Width.Value,
            Height = Height.Value,
            SoundSpeed = SoundSpeed.Value,
            Damping = Damping.Value,
            MinimumFrequency = Minimum.Value,
            MaximumFrequency = Maximum.Value,
            FrequencyStep = Step.Value,
            Cutoff = Cutoff.Value
        };
        project.Sources = Sources.Select(x => x.Source()).ToList();
        project.Seats = Seats.Select(x => x.Seat()).ToList();
        var issues = Validation.Check(project, forCalculation);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join(Environment.NewLine, issues.Select(x => x.Path + "：" + x.Message)));
        return project;
    }

    {
        if (Sources.Count >= 8)
            throw new ArgumentException("最多8个声源");
        Sources.Add(new(new Emitter { Name = "声源 " + (Sources.Count + 1), Point = new(0, 0, 0) }, changed));
        changed();
    }

    {
        if (Seats.Count >= 12)
            throw new ArgumentException("最多12个座位耳点");
        Seats.Add(new(new EarPoint { Name = "耳点 " + (Seats.Count + 1), Point = new(0, 0, 0) }, changed));
        changed();
    }

    private bool Save(bool saveAs)
    {
        var project = Build(false);
        var target = path;
        if (saveAs || string.IsNullOrEmpty(target))
            target = services.SaveProjectPath(path, project.Name) ?? "";
        if (string.IsNullOrEmpty(target))
            return false;
        JournalEntry entry = new() { Action = "保存项目", Detail = "版本" + project.Revision + "；内嵌全部点位、基准与复核" };
        project.Journal.Add(entry);
        while (project.Journal.Count > 500)
            project.Journal.RemoveAt(0);
        NativeFiles.Save(target, project);
        _ = NativeFiles.Open(target);
        Journal.Clear();
        foreach (var item in project.Journal)
            Journal.Add(item);
        path = target;
        if (computedProject is not null)
        {
            computedProject.Journal = ProjectCodec.Clone(project.Journal);
            computedProject.Modified = project.Modified;
        }
        Dirty = false;
        Notify(nameof(FileLocation));
        Status = "已保存并重开校验：" + target;
        return true;
    }

    }

    private async Task Work(string message, Func<CancellationToken, Task> action)
    {
        Busy = true;
        Progress = 0;
        Status = message;
        cancellation = new();
        try
        {
            await action(cancellation.Token);
        }
        catch (OperationCanceledException)
        {
            Status = "计算已取消，未应用未完成的结果。";
        }
        catch (Exception error) when (Expected(error))
        {
            Status = "计算未完成：" + error.Message;
            services.Error(error.Message);
        }
        finally
        {
            cancellation.Dispose();
            cancellation = null;
            Busy = false;
        }
    }

    private async Task Calculate()
    {
        await Work("正在计算非零阶模态与座位复数响应…", async token =>
        {
            var project = Build(true);
            var calculated = await Task.Run(() => Engine.Calculate(project, token), token);
            token.ThrowIfCancellationRequested();
            computedProject = project;
            Notify(nameof(ComputedProject));
            Result = calculated;
            SelectedCurve = Result.Curves.FirstOrDefault();
            SelectedMode = Result.Modes.FirstOrDefault();
            Frequency = Math.Clamp(Frequency, project.Model.MinimumFrequency, project.Model.MaximumFrequency);
            Spectrum = BuildSpectrum(Result);
            Density = Diagnostics.Density(Result);
            Advice = Diagnostics.Advice(project, Result);
            Notify(nameof(ResultAvailable));
            Notify(nameof(ResultStatus));
            Section = "field";
            await RefreshField(token);
            Status = $"计算完成：{Result.Modes.Count}个模态，{Result.Frequencies.Count}个频率样点。结果为理想模型相对量。";
        });
    }

    private async Task UpdateField()
    {
        await Work("正在更新剖面及所选耳点贡献…", async token =>
        {
            await RefreshField(token);
            Status = "剖面与贡献已更新。";
        });
    }

    private async Task RefreshField(CancellationToken token)
    {
        if (computedProject is null || Result is null)
            return;
        var project = computedProject;
        var calculated = Result;
        var selectedFrequency = Frequency;
        var selectedPlane = Plane.Key;
        var selectedSlice = Slice;
        var shape = ShowShape;
        var mode = SelectedMode;
        if (shape && mode is null)
            throw new ArgumentException("截断内没有模态可显示振型，请使用响应幅值或提高截断频率");
        var seat = project.Seats.FirstOrDefault(x => x.Id == SelectedCurve?.Id);
        var next = await Task.Run(() =>
        {
            token.ThrowIfCancellationRequested();
            return Engine.Field(project, calculated, selectedPlane, selectedSlice, shape, mode, selectedFrequency);
        }, token);
        token.ThrowIfCancellationRequested();
        Field = next;
        Contributions = seat is null ? Array.Empty<Contribution>() : Engine.Contributions(project, calculated, seat, selectedFrequency);
        Coupling = seat is null ? Array.Empty<CouplingDiagnostic>() : Diagnostics.Coupling(project, calculated, seat);
    }

    private void CaptureBaseline()
    {
        if (Baselines.Count >= 12)
            throw new ArgumentException("最多保留12个基准，请先删除不需要的基准");
        if (string.IsNullOrWhiteSpace(BaselineName) || BaselineName.Length > 80)
            throw new ArgumentException("基准名称必须为1—80字符");
        var snapshot = Demo.Capture(computedProject!, BaselineName);
        Baselines.Add(snapshot);
        SelectedBaseline = snapshot;
        Dirty = true;
        Status = "已从当前计算输入捕获基准；请保存项目以保留。";
    }

    private async Task Compare()
    {
        await Work("正在复算基准并匹配耳点…", async token =>
        {
            var baseline = SelectedBaseline!;
            var next = await Task.Run(() => Cabin.Core.Analysis.Comparison.Compare(computedProject!, Result!, baseline), token);
            token.ThrowIfCancellationRequested();
            Comparison = next;
            Status = "基准比较完成。差值为相对响应比值，请同时阅读条件变化提示。";
        });
    }

    private void RestoreBaseline()
    {
        var project = Build(false);
        if (!services.Confirm("恢复基准输入", "以选中基准替换当前几何、来源、声源和耳点？此操作会清除当前计算结果。"))
            return;
        var baseline = SelectedBaseline!;
        project.Model = ProjectCodec.Clone(baseline.Model);
        project.Sources = ProjectCodec.Clone(baseline.Sources);
        project.Seats = ProjectCodec.Clone(baseline.Seats);
        project.IsDemo = baseline.IsDemo;
        project.DataSource = baseline.DataSource;
        Load(project, path, true);
        Status = "已恢复基准输入。请重新计算。";
    }

    private void DeleteBaseline()
    {
        if (!services.Confirm("删除基准", "删除选中基准？保存后将从项目文件中移除。"))
            return;
        Baselines.Remove(SelectedBaseline!);
        SelectedBaseline = Baselines.FirstOrDefault();
        Dirty = true;
    }

    private async Task RunScan()
    {
        await Work("正在扫描参数…", async token =>
        {
            var count = ScanCount.Value;
            if (count != Math.Truncate(count))
                throw new ArgumentException("扫描候选数必须为整数");
            var settings = new ScanSettings(ScanParameter.Key, ScanStart.Value, ScanEnd.Value, (int)count,
                SelectedSeat?.Id ?? Guid.Empty);
            var reporter = new Progress<int>(value => Progress = value);
            var next = await Task.Run(() => ParameterScan.Run(computedProject!, settings, token, reporter), token);
            token.ThrowIfCancellationRequested();
            Scan = next;
            ScanPlot = new("扫描耳点的采样RMS", ScanParameter.Label, "相对响应 / 无量纲", new[]
            {
                new PlotLine("有效候选", "#167D8D", next.Rows.Where(x => x.Error.Length == 0).Select(x => new PlotPoint(x.Parameter, x.Rms)).ToArray())
            });
            SelectedScanRow = next.Rows.Where(x => x.Error.Length == 0).MinBy(x => x.Rms);
            Status = $"扫描完成；{next.Rows.Count(x => x.Error.Length == 0)}个有效候选。最低值仅对当前选定耳点、频带和采样成立。";
        });
    }

    private void ApplyScan()
    {
        if (Scan is null || SelectedScanRow is null)
            return;
        if (!services.Confirm("应用扫描候选", "将所选参数值应用到当前项目？其他点位保持绝对坐标，需要重新计算。"))
            return;
        var project = Build(false);
        ParameterScan.Apply(project, Scan.Settings.Parameter, SelectedScanRow.Parameter);
        Load(project, path, true);
        Status = "扫描候选已应用。请重新计算并比较全部座位。";
    }

    private async Task ExtendTruncation()
    {
        await Work("正在增加100Hz截断并比较数值敏感性…", async token =>
        {
            var next = await Task.Run(() => Diagnostics.Extend(computedProject!, Result!), token);
            token.ThrowIfCancellationRequested();
            Truncation = next;
            Status = "截断敏感性计算完成；该检查不改变项目参数。";
        });
    }

    private void AddReview()
    {
        if (Reviews.Count >= 100)
            throw new ArgumentException("复核清单最多100项");
        var item = new ReviewEditor(new ReviewItem
        {
            Target = "待复核对象",
            Action = "填写实测或高保真仿真复核步骤",
            Result = "待验证"
        }, Changed);
        Reviews.Add(item);
        SelectedReview = item;
        Changed();
    }

    private void DeleteReview()
    {
        if (!services.Confirm("删除复核项", "删除选中的工程复核记录？"))
            return;
        Reviews.Remove(SelectedReview!);
        SelectedReview = Reviews.FirstOrDefault();
        Changed();
    }

    private void Export(string label, string extension, string text)
    {
        var target = services.ExportPath(label, extension);
        if (target is null)
            return;
        NativeFiles.WriteText(target, text);
        Status = "已导出：" + target;
    }

    private static PlotData BuildSpectrum(AnalysisResult calculated)
    {
        var colors = new[] { "#147D92", "#C06A32", "#7364AB", "#398056", "#AF426A", "#6A6A30" };
        var lines = calculated.Curves.Select((curve, i) => new PlotLine(curve.Name, colors[i % colors.Length],
            curve.Points.Select(x => new PlotPoint(x.Frequency, x.RelativeDb(calculated.Reference))).ToArray())).ToArray();
        return new("座位频响 · 公共全频带最大值为0dB参考", "频率 / Hz", "相对幅值 / dB（非dB SPL）", lines);
    }
}

using System.ComponentModel;
using System.Windows;
using System.Windows.Controls;
using Cabin.Desktop.Services;
using Cabin.Desktop.ViewModels;

namespace Cabin.Desktop.Views;

public partial class MainWindow : Window
{
    public MainWindow() : this(new DesktopServices())
    {
    }

    public MainWindow(IDesktopServices services)
    {
        InitializeComponent();
        var vm = new WorkspaceViewModel(services);
        DataContext = vm;
        vm.PropertyChanged += (_, args) =>
        {
            if (args.PropertyName == nameof(WorkspaceViewModel.Section))
            {
                SelectSection(Navigation.Items, vm.Section);
                WorkspaceScroll.ScrollToTop();
            }
        };
    }

    private static void SelectSection(ItemCollection items, string section)
    {
        foreach (var item in items.OfType<TreeViewItem>())
        {
            if (item.Tag is string tag && tag == section)
                item.IsSelected = true;
            SelectSection(item.Items, section);
        }
    }

    public FrameworkElement FieldVisual => FieldDrawing;

    public void ShowFieldSettings()
    {
        FieldSettings.IsExpanded = true;
        WorkspaceScroll.ScrollToTop();
    }

    public void ShowFullField()
    {
        FieldSettings.IsExpanded = false;
        WorkspaceScroll.UpdateLayout();
        WorkspaceScroll.ScrollToBottom();
    }

    private void Navigate(object sender, RoutedPropertyChangedEventArgs<object> e)
    {
        if (DataContext is WorkspaceViewModel vm && e.NewValue is TreeViewItem { Tag: string section })
            vm.Section = section;
    }

    private void OnClosing(object? sender, CancelEventArgs e)
    {
        if (DataContext is WorkspaceViewModel vm)
            e.Cancel = !vm.CanClose();
    }

    private void Exit(object sender, RoutedEventArgs e)
    {
        Close();
    }
}
