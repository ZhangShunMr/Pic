    private async Task Upload(InputFileChangeEventArgs args)
    {
        try
        {
            using var memory = new MemoryStream();
            await args.File.OpenReadStream(Software.MaximumBytes).CopyToAsync(memory);
            Vm.Import(memory.ToArray());
        }
        catch (Exception error) when (error is IOException or InvalidOperationException)

            Vm.Error("导入失败：" + error.Message);
        }
        finally
        {
            inputVersion++;

    private ElementReference plot;
    private static readonly string[] Colors = ["#236b72", "#c3663b", "#7974a6", "#788643", "#ab5776", "#456aa0"];
    private static string Points(SeatCurve curve, Cavity model, double reference)
    {
        return string.Join(" ", curve.Points.Select(p =>
            N(65 + (p.Frequency - model.MinimumFrequency) / (model.MaximumFrequency - model.MinimumFrequency) * 1005) + "," +

    }

<div class="help-copy">
    <h3>模型、单位和读取方法</h3>
    <p>模型是均匀介质、刚性矩形边界的理想声腔。长度、宽度、高度及坐标均用米；声速用m/s；频率用Hz。模态阻尼比为无量纲等效参数，默认0.03仅为研究假设。</p>
    <p>模态频率 f=c/2·√[(nx/Lx)²+(ny/Ly)²+(nz/Lz)²]。nx、ny、nz为非负整数，不同时为零。振型φ=cos(nxπx/Lx)cos(nyπy/Ly)cos(nzπz/Lz)。边界法向梯度为零；不同阶次频率相同仍各自保留。</p>
    <p>激励响应 H=Σ (2^d·1m³/V) φ(r) Σ[A exp(iθ)φ(s)]·100²/[fn²−f²+i2ζfnf]，其中d为非零阶次数量。逆范数2^d/V来自余弦模态体积积分，V为腔体体积；固定1m³参考体积用于跨尺寸统一标度。固定100Hz标度使此筛查指标无量纲，不包括实际源强、介质密度与绝对声压校准。</p>
    <p>模型不含000均匀压缩模态，不能用于密闭腔准静态压缩预测。低频完整响应和高阶尾项可能改变结果，需对截断频率、频率网格和阻尼做敏感性检查。源位于振型节点时该模态耦合减弱；测点位于节点时该模态接收减弱，多源相位还会产生抵消。</p>
    <p>振型色标固定为−1至+1。响应场颜色按当前剖面最大值归一化；各耳点曲线按全频带所有耳点最大幅值统一换算20log10(|H|/参考)，不是dB SPL。参考近零时显示下限−120dB并提示零响应。对比曲线使用当前与基准共用参考，表格显示峰值和频点RMS比值。</p>
    <p>峰值是离散频点最大值；RMS是频点等权的幅值平方均值开方，不是时间域或实测宽带声压。贡献表按各复数项幅值排序，不能把幅值百分比解释为能量贡献。</p>

public abstract class VmComponent : ComponentBase, IDisposable
{
    [Inject] public WorkspaceViewModel Vm { get; set; } = null!;

    protected override void OnInitialized()
    {
        Vm.PropertyChanged += Changed;
    }

    private void Changed(object? sender, PropertyChangedEventArgs args)

        _ = InvokeAsync(StateHasChanged);

}

namespace Cabin.App.ViewModels;

public sealed record DownloadItem(string Name, string Mime, byte[] Bytes, DateTimeOffset Created);

public sealed class DownloadVault
{
    private readonly Dictionary<Guid, DownloadItem> items = [];
    private readonly object gate = new();

    public string Add(string name, string mime, byte[] bytes)
    {
        lock (gate)
        {
            foreach (var old in items.Where(x => x.Value.Created < DateTimeOffset.UtcNow.AddMinutes(-20)).Select(x => x.Key).ToList())
                items.Remove(old);
            while (items.Count >= 30)
                items.Remove(items.OrderBy(x => x.Value.Created).First().Key);
            var id = Guid.NewGuid();
            items[id] = new(name, mime, bytes, DateTimeOffset.UtcNow);
            return "/download/" + id;
        }
    }

    public DownloadItem? Get(Guid id)
    {
        lock (gate)
        {
            return items.TryGetValue(id, out var value) && value.Created > DateTimeOffset.UtcNow.AddMinutes(-20) ? value : null;
        }
    }
}

    public void Export(string kind)
    {
        if (Current is null)
            return;
        Safe(() =>
        {
            if (kind == "project")
            {
                DownloadName = "座舱项目" + Software.Extension;
                DownloadUrl = vault.Add(DownloadName, "application/json", ProjectCodec.Encode(Current));
            }
            else
            {
                var text = kind switch
                {
                    "points" => PointCsv.Export(Current),
                    "reviews" => Exports.ReviewsCsv(Current),
                    "modes" when Result is not null => Exports.ModesCsv(Current, Result),
                    "responses" when Result is not null => Exports.ResponsesCsv(Current, Result),
                    "contributions" when Result is not null && SelectedSeat is not null => Exports.ContributionsCsv(Current, Result, SelectedSeat, CursorFrequency),
                    "comparison" when Difference is not null && Baseline is not null => Exports.ComparisonCsv(Current, Baseline, Difference),
                    "scan" when Scan is not null => Exports.ScanCsv(Current, Scan),
                    "report" when Result is not null => Exports.Html(Current, Result, Difference, Scan),
                    _ => throw new ArgumentException("此成果尚无有效结果，请先计算对应内容")
                };
                DownloadName = "座舱研究-" + kind + (kind == "report" ? ".html" : ".csv");
                var bytes = Encoding.UTF8.GetPreamble().Concat(Encoding.UTF8.GetBytes(text)).ToArray();
                DownloadUrl = vault.Add(DownloadName, kind == "report" ? "text/html; charset=utf-8" : "text/csv; charset=utf-8", bytes);
            }
            Status = "导出已准备：" + DownloadName + "。点击下载，保存位置由浏览器下载设置决定；链接20分钟有效。";
        });
    }

    public void Error(string message)
    {
        Status = message;
        Notify();
    }

    private void Safe(Action action)
    {
        try
        {
            action();
        }
        catch (Exception error) when (error is ArgumentException or IOException or UnauthorizedAccessException)
        {
            Status = "操作失败：" + error.Message;
        }
        Notify();
    }

    public static ComparisonResult Compare(Project current, AnalysisResult currentResult, Snapshot baseline)
    {
        var oldProject = Demo.FromSnapshot(baseline);
        var oldResult = Engine.Calculate(oldProject);
        var warnings = new List<string>();
        if (current.Model.MinimumFrequency != baseline.Model.MinimumFrequency ||
            current.Model.MaximumFrequency != baseline.Model.MaximumFrequency ||
            current.Model.FrequencyStep != baseline.Model.FrequencyStep)
            throw new ArgumentException("分析频带或步长不同，不能比较。请恢复基准频带与步长后重算。");
        if (current.Model.Cutoff != baseline.Model.Cutoff)
            throw new ArgumentException("模态截断不同，不能比较。请使用相同截断频率。");
        if (current.DataSource != baseline.DataSource || current.IsDemo != baseline.IsDemo)
            throw new ArgumentException("来源或模拟标记不同，不能比较。");
        if (current.Model.Damping != baseline.Model.Damping)
            warnings.Add("阻尼已变化；差值同时包含阻尼影响。");
        if (current.Model.SoundSpeed != baseline.Model.SoundSpeed)
            warnings.Add("声速已变化；差值同时包含介质条件影响。");
        if (current.Model.Length != baseline.Model.Length || current.Model.Width != baseline.Model.Width || current.Model.Height != baseline.Model.Height)
            warnings.Add("尺寸变化；模态按阶次匹配，耳点使用绝对米制坐标。");
        var differences = new List<SeatDifference>();
        foreach (var curve in currentResult.Curves)
        {
            var old = oldResult.Curves.FirstOrDefault(x => x.Id == curve.Id);
            if (old is null)
            {
                warnings.Add(curve.Name + "没有相同标识的基准耳点，跳过此耳点。");
                continue;
            }
            var oldSeat = baseline.Seats.First(x => x.Id == curve.Id);
            var seat = current.Seats.First(x => x.Id == curve.Id);
            if (oldSeat.Point.X != seat.Point.X || oldSeat.Point.Y != seat.Point.Y || oldSeat.Point.Z != seat.Point.Z)
                warnings.Add(curve.Name + "位置变化；差值包含测点变化。");
            if (old.Peak.Magnitude < 1e-12 || old.Rms < 1e-12)
            {
                warnings.Add(curve.Name + "基准近零，不能计算稳定比值。");
                continue;
            }
            differences.Add(new(curve.Name, old.Peak.Magnitude, curve.Peak.Magnitude,
                Engine.Decibels(curve.Peak.Magnitude, old.Peak.Magnitude), old.Peak.Frequency,
                curve.Peak.Frequency, Engine.Decibels(curve.Rms, old.Rms)));
        }
        if (differences.Count == 0)
            throw new ArgumentException("没有可匹配的非零基准耳点");
        var modes = new List<ModeDifference>();
        foreach (var mode in currentResult.Modes)
        {
            var old = oldResult.Modes.FirstOrDefault(x => x.Key == mode.Key);
            if (old is not null)
                modes.Add(new(mode.Key, old.Frequency, mode.Frequency, (mode.Frequency / old.Frequency - 1) * 100));
        }
        return new()
        {
            Warnings = warnings,
            Seats = differences,
            Modes = modes,
            Reference = Math.Max(oldResult.Reference, currentResult.Reference),

        };

    public static ScanResult Run(Project project, ScanSettings settings)
    {
        if (settings.Count is < 2 or > 21)
            throw new ArgumentException("扫描点数应为2至21");
        if (!double.IsFinite(settings.Start) || !double.IsFinite(settings.End) || settings.Start >= settings.End)
            throw new ArgumentException("扫描终值应大于初值，且均为有限数值");
        if (!project.Seats.Any(x => x.Id == settings.SeatId))
            throw new ArgumentException("请选择扫描耳点");
        if (settings.Parameter is not ("length" or "width" or "height" or "sourceX" or "sourceY" or "sourceZ" or "damping"))
            throw new ArgumentException("不支持的扫描参数");
        var rows = new List<ScanRow>();
        for (var i = 0; i < settings.Count; i++)
        {
            var value = settings.Start + (settings.End - settings.Start) * i / (settings.Count - 1);
            var candidate = ProjectCodec.Clone(project);
            candidate.Baselines.Clear();
            Apply(candidate, settings.Parameter, value);
            try
            {
                var result = Engine.Calculate(candidate);
                var curve = result.Curves.First(x => x.Id == settings.SeatId);
                rows.Add(new(value, curve.Peak.Magnitude, curve.Peak.Frequency, curve.Rms,
                    result.Modes.FirstOrDefault()?.Frequency ?? 0, ""));
            }
            catch (ArgumentException error)
            {
                rows.Add(new(value, 0, 0, 0, 0, error.Message));
            }
        }
        return new() { Settings = settings, Rows = rows };
    }

    public static void Apply(Project project, string parameter, double value)
    {
        switch (parameter)
        {
            case "length": project.Model.Length = value; break;
            case "width": project.Model.Width = value; break;
            case "height": project.Model.Height = value; break;
            case "damping": project.Model.Damping = value; break;
            default:
                var source = project.Sources.FirstOrDefault() ?? throw new ArgumentException("没有首个声源可扫描");
                switch (parameter)
                {
                    case "sourceX": source.Point.X = value; break;
                    case "sourceY": source.Point.Y = value; break;
                    case "sourceZ": source.Point.Z = value; break;
                    default: throw new ArgumentException("未知参数");
                }
                break;
        }
    }

using System.Numerics;
using Cabin.Core.Domain;

namespace Cabin.Core.Analysis;

public sealed record ResponsePoint(double Frequency, double Real, double Imaginary)
{
    public double Magnitude => Math.Sqrt(Real * Real + Imaginary * Imaginary);
    public double Phase => Math.Atan2(Imaginary, Real) * 180 / Math.PI;
    public double RelativeDb(double reference) => Engine.Decibels(Magnitude, reference);
}

public sealed record SeatCurve(Guid Id, string Name, List<ResponsePoint> Points)
{
    public ResponsePoint Peak => Points.OrderByDescending(x => x.Magnitude).First();
    public double Rms => Math.Sqrt(Points.Average(x => x.Magnitude * x.Magnitude));
}

public sealed class AnalysisResult
{
    public List<Mode> Modes { get; init; } = [];
    public List<SeatCurve> Curves { get; init; } = [];
    public List<double> Frequencies { get; init; } = [];
    public double Reference { get; init; }
    public DateTimeOffset Computed { get; init; } = DateTimeOffset.UtcNow;
    public int InBandModes(double minimum, double maximum) => Modes.Count(x => x.Frequency >= minimum && x.Frequency <= maximum);
}

public sealed record Contribution(string Key, double Frequency, double Magnitude, double Phase, double SourceMagnitude, double SeatShape);
public sealed record FieldCell(double U, double V, double Value);
public sealed record FieldResult(List<FieldCell> Cells, double Maximum, string Plane, double Slice, string Quantity);

public static class Engine
{
    public static List<double> FrequencyGrid(Cavity model)
    {
        var frequencies = new List<double>();
        var count = (int)Math.Floor((model.MaximumFrequency - model.MinimumFrequency) / model.FrequencyStep + 1e-8);
        for (var i = 0; i <= count; i++)
            frequencies.Add(model.MinimumFrequency + i * model.FrequencyStep);
        if (frequencies.Count == 0 || model.MaximumFrequency - frequencies[^1] > 1e-8)
            frequencies.Add(model.MaximumFrequency);
        return frequencies;
    }

    public static AnalysisResult Calculate(Project project)
    {
        var issues = Validation.Check(project);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
        var modes = Modes.Enumerate(project.Model);
        var frequencies = FrequencyGrid(project.Model);
        var couplings = modes.Select(x => Modes.SourceCoupling(x, project.Model, project.Sources)).ToList();
        var curves = new List<SeatCurve>();
        foreach (var seat in project.Seats)
        {
            var points = new List<ResponsePoint>();
            foreach (var frequency in frequencies)
            {
                var response = At(project.Model, modes, couplings, seat.Point, frequency);
                points.Add(new(frequency, response.Real, response.Imaginary));
            }
            curves.Add(new(seat.Id, seat.Name, points));
        }
        var reference = curves.SelectMany(x => x.Points).Select(x => x.Magnitude).DefaultIfEmpty(0).Max();
        return new()
        {
            Modes = modes,
            Frequencies = frequencies,
            Curves = curves,
            Reference = reference
        };
    }

    public static Complex At(Cavity model, IReadOnlyList<Mode> modes, IReadOnlyList<Complex> couplings, Position point, double frequency)
    {
        var sum = Complex.Zero;
        for (var i = 0; i < modes.Count; i++)
            sum += Modes.Term(modes[i], model, couplings[i], point, frequency);
        return sum;
    }

    public static Complex At(Project project, AnalysisResult result, Position point, double frequency)
    {
        var couplings = result.Modes.Select(x => Modes.SourceCoupling(x, project.Model, project.Sources)).ToList();
        return At(project.Model, result.Modes, couplings, point, frequency);
    }

    public static List<Contribution> Contributions(Project project, AnalysisResult result, EarPoint seat, double frequency)
    {
        return result.Modes.Select(mode =>
        {
            var source = Modes.SourceCoupling(mode, project.Model, project.Sources);
            var value = Modes.Term(mode, project.Model, source, seat.Point, frequency);
            return new Contribution(mode.Key, mode.Frequency, value.Magnitude,
                value.Phase * 180 / Math.PI, source.Magnitude, Modes.Shape(mode, project.Model, seat.Point));
        }).OrderByDescending(x => x.Magnitude).ToList();
    }

    public static FieldResult Field(Project project, AnalysisResult result, string plane, double fraction,
        bool shape, Mode? mode, double frequency, int columns = 48, int rows = 24)
    {
        if (plane is not ("xy" or "xz" or "yz"))
            throw new ArgumentException("剖面只能为xy、xz、yz");
        if (fraction < 0 || fraction > 1 || !double.IsFinite(fraction))
            throw new ArgumentException("剖面位置必须在0至1之间");
        var cells = new List<FieldCell>();
        var couplings = result.Modes.Select(x => Modes.SourceCoupling(x, project.Model, project.Sources)).ToList();
        for (var row = 0; row < rows; row++)
        {
            for (var column = 0; column < columns; column++)
            {
                var u = (column + 0.5) / columns;
                var v = (row + 0.5) / rows;
                var point = PlanePoint(project.Model, plane, u, v, fraction);
                var value = shape && mode is not null ? Modes.Shape(mode, project.Model, point) :
                    At(project.Model, result.Modes, couplings, point, frequency).Magnitude;
                cells.Add(new(u, v, value));
            }
        }
        var maximum = shape ? 1 : cells.Select(x => x.Value).DefaultIfEmpty(0).Max();
        return new(cells, maximum, plane, fraction, shape ? "有符号归一化振型" : "剖面局部归一化响应幅值");
    }

            "xz" => (point.X / model.Length, point.Z / model.Height),

            _ => throw new ArgumentException("未知剖面")
        };
    }

    public static double Decibels(double value, double reference)
    {
        if (reference <= 1e-15)
            return -120;
        return Math.Max(-120, 20 * Math.Log10(Math.Max(value, reference * 1e-6) / reference));
    }

    {

        if (shape)
        {
            var strength = Math.Abs(t);
            var target = t >= 0 ? (R: 204, G: 104, B: 56) : (R: 35, G: 117, B: 133);

            var g = (int)(241 + (target.G - 241) * strength);
            var b = (int)(225 + (target.B - 225) * strength);
            return $"rgb({r},{g},{b})";
        }
        return $"hsl({195 - 170 * t:F0},48%,{90 - 45 * t:F0}%)";
    }

public sealed record Mode(int Nx, int Ny, int Nz, double Frequency, int Degeneracy = 1)
{
    public string Key => $"{Nx},{Ny},{Nz}";
    public int Dimension => (Nx > 0 ? 1 : 0) + (Ny > 0 ? 1 : 0) + (Nz > 0 ? 1 : 0);

        Validation.Model(model, issues);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
        var result = new List<Mode>();
        var maxX = (int)Math.Floor(2 * model.Cutoff * model.Length / model.SoundSpeed);
        var maxY = (int)Math.Floor(2 * model.Cutoff * model.Width / model.SoundSpeed);
        var maxZ = (int)Math.Floor(2 * model.Cutoff * model.Height / model.SoundSpeed);
        for (var nx = 0; nx <= maxX; nx++)
        {
            for (var ny = 0; ny <= maxY; ny++)
            {
                for (var nz = 0; nz <= maxZ; nz++)
                {
                    if (nx + ny + nz == 0)
                        continue;
                    var f = Frequency(model, nx, ny, nz);
                    if (f <= model.Cutoff + 1e-9)
                        result.Add(new(nx, ny, nz, f));
                    if (result.Count > 10000)
                        throw new ArgumentException("模态数量超过10000，请降低截断频率");
                }
            }
        }
        result = result.OrderBy(x => x.Frequency).ThenBy(x => x.Nx).ThenBy(x => x.Ny).ThenBy(x => x.Nz).ToList();
        var start = 0;
        while (start < result.Count)
        {
            var end = start + 1;
            var tolerance = Math.Max(1e-8, result[start].Frequency * 1e-8);
            while (end < result.Count && result[end].Frequency - result[start].Frequency <= tolerance)
                end++;
            for (var i = start; i < end; i++)
                result[i] = result[i] with { Degeneracy = end - start };
            start = end;
        }
        return result;
    }

    public static double Frequency(Cavity model, int nx, int ny, int nz)
    {
        return model.SoundSpeed / 2 * Math.Sqrt(
            Math.Pow(nx / model.Length, 2) +
            Math.Pow(ny / model.Width, 2) +
            Math.Pow(nz / model.Height, 2));
    }

    public static double Shape(Mode mode, Cavity model, Position point)
    {
        return Math.Cos(mode.Nx * Math.PI * point.X / model.Length) *
            Math.Cos(mode.Ny * Math.PI * point.Y / model.Width) *
            Math.Cos(mode.Nz * Math.PI * point.Z / model.Height);
    }

    public static Complex SourceCoupling(Mode mode, Cavity model, IEnumerable<Emitter> sources)
    {
        var sum = Complex.Zero;
        foreach (var source in sources.Where(x => x.Enabled))
        {
            var phase = source.PhaseDegrees * Math.PI / 180;
            sum += Complex.FromPolarCoordinates(source.Amplitude, phase) * Shape(mode, model, source.Point);
        }
        return sum;
    }

    public static Complex Term(Mode mode, Cavity model, Complex coupling, Position point, double frequency)
    {
        var fn = mode.Frequency;
        var denominator = new Complex(fn * fn - frequency * frequency, 2 * model.Damping * fn * frequency);
        return mode.NormFactor * Software.ReferenceVolume / model.Volume * Shape(mode, model, point) * coupling *
            (Software.ReferenceFrequency * Software.ReferenceFrequency) / denominator;
    }
}

    public static CsvPreview Preview(string text, Cavity model)
    {

        if (text.Length > 100000)
        {
            preview.Issues.Add(new("CSV", "点位CSV最多100000字符"));

        }

            return preview;
        }

        {
            if (string.IsNullOrWhiteSpace(lines[i]))
                continue;
            var path = "第" + (i + 1) + "行";
            try
            {
                var cells = ParseLine(lines[i]);
                if (cells.Count != 8)
                    throw new ArgumentException("需要8列");
                if (cells[0] is not ("source" or "seat"))
                    throw new ArgumentException("kind必须为source或seat");
                var point = new Position(Number(cells[2]), Number(cells[3]), Number(cells[4]));
                var amplitude = cells[0] == "source" ? Number(cells[5]) : 0;
                var phase = cells[0] == "source" ? Number(cells[6]) : 0;
                var row = new PointRow(cells[0], cells[1], point, amplitude, phase, cells[7]);
                var issues = new List<Issue>();
                var sources = row.Kind == "source" ? new List<Emitter> { new() { Name = row.Name, Point = row.Point, Amplitude = amplitude, PhaseDegrees = phase } } : [];
                var seats = row.Kind == "seat" ? new List<EarPoint> { new() { Name = row.Name, Point = row.Point, Note = row.Note } } : [];
                Validation.Points(model, sources, seats, issues);
                if (issues.Count > 0)
                    throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
                preview.Rows.Add(row);
            }
            catch (ArgumentException error)
            {
                preview.Issues.Add(new(path, error.Message));
            }
        }
        if (preview.Rows.Count(x => x.Kind == "source") > 8 || preview.Rows.Count(x => x.Kind == "seat") > 12)
            preview.Issues.Add(new("CSV", "最多8个声源、12个耳点"));
        if (preview.Rows.Count == 0)
            preview.Issues.Add(new("CSV", "没有有效点位"));
        return preview;
    }

    public static void Apply(Project project, CsvPreview preview)
    {

            throw new ArgumentException("CSV预检未通过；项目未改变");
        var sources = new List<Emitter>();
        var seats = new List<EarPoint>();
        foreach (var row in preview.Rows)
        {
            if (row.Kind == "source")
                sources.Add(new() { Name = row.Name, Point = ProjectCodec.Clone(row.Point), Amplitude = row.Amplitude, PhaseDegrees = row.Phase });
            else
                seats.Add(new() { Name = row.Name, Point = ProjectCodec.Clone(row.Point), Note = row.Note });
        }
        project.Sources = sources;
        project.Seats = seats;
    }

    {

    public static string Join(params string[] fields) => string.Join(",", fields.Select(Quote));
    public static string N(double value) => value.ToString("G12", CultureInfo.InvariantCulture);
    public static string Quote(string value)
    {
        if (value.Contains(',') || value.Contains('"') || value.Contains('\n') || value.Contains('\r'))
            return "\"" + value.Replace("\"", "\"\"") + "\"";
        return value;
    }

    public static List<string> ParseLine(string line)
    {
        var cells = new List<string>();
        var value = new StringBuilder();
        var quoted = false;
        var closed = false;
        for (var i = 0; i < line.Length; i++)
        {
            var c = line[i];
            if (quoted)
            {
                if (c == '"')
                {
                    if (i + 1 < line.Length && line[i + 1] == '"')
                    {
                        value.Append('"');
                        i++;
                    }
                    else
                    {
                        quoted = false;
                        closed = true;
                    }
                }
                else
                    value.Append(c);
            }
            else if (c == ',')
            {
                cells.Add(value.ToString());
                value.Clear();
                closed = false;
            }
            else if (c == '"' && value.Length == 0 && !closed)
                quoted = true;
            else if (closed || c == '"')
                throw new ArgumentException("引号格式错误；不支持单元格内换行");
            else
                value.Append(c);
        }
        if (quoted)
            throw new ArgumentException("引号未闭合；不支持单元格内换行");
        cells.Add(value.ToString());
        return cells;
    }

    private static double Number(string value)
    {
        if (!double.TryParse(value, NumberStyles.Float, CultureInfo.InvariantCulture, out var number) || !double.IsFinite(number))
            throw new ArgumentException("数值使用小数点，不能为空或非有限值");
        return number;
    }

using System.Text.Json;

using Cabin.Core.Domain;

namespace Cabin.Core.IO;

public static class ProjectCodec
{
    public static readonly JsonSerializerOptions Options = new()
    {
        WriteIndented = true,
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow,
        MaxDepth = 24
    };

    public static byte[] Encode(Project project)
    {
        var issues = Validation.Check(project, false);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
        return JsonSerializer.SerializeToUtf8Bytes(project, Options);
    }

    public static Project Decode(byte[] bytes)
    {
        if (bytes.Length == 0 || bytes.Length > Software.MaximumBytes)
            throw new ArgumentException("项目应为非空JSON，且不超过2MiB");
        try
        {
            using var document = JsonDocument.Parse(bytes, new() { MaxDepth = 24 });
            var root = document.RootElement;
            Require(root, "schema", "id", "name", "revision", "dataSource", "isDemo", "created", "modified", "model", "sources", "seats", "baselines", "reviews", "journal");
            if (root.GetProperty("schema").GetInt32() != Software.Schema)
                throw new ArgumentException("只支持项目格式版本1；未执行迁移");
            Require(root.GetProperty("model"), "length", "width", "height", "soundSpeed", "damping", "minimumFrequency", "maximumFrequency", "frequencyStep", "cutoff");
            foreach (var source in root.GetProperty("sources").EnumerateArray())
            {
                Require(source, "id", "name", "point", "amplitude", "phaseDegrees", "enabled");
                Require(source.GetProperty("point"), "x", "y", "z");
            }
            foreach (var seat in root.GetProperty("seats").EnumerateArray())
            {
                Require(seat, "id", "name", "point");
                Require(seat.GetProperty("point"), "x", "y", "z");
            }
            foreach (var baseline in root.GetProperty("baselines").EnumerateArray())
            {
                Require(baseline, "id", "name", "model", "sources", "seats", "isDemo", "dataSource");
                Require(baseline.GetProperty("model"), "length", "width", "height", "soundSpeed", "damping", "minimumFrequency", "maximumFrequency", "frequencyStep", "cutoff");
                foreach (var source in baseline.GetProperty("sources").EnumerateArray())
                {
                    Require(source, "id", "name", "point", "amplitude", "phaseDegrees", "enabled");
                    Require(source.GetProperty("point"), "x", "y", "z");
                }
                foreach (var seat in baseline.GetProperty("seats").EnumerateArray())
                {
                    Require(seat, "id", "name", "point");
                    Require(seat.GetProperty("point"), "x", "y", "z");
                }
            }
            foreach (var item in root.GetProperty("reviews").EnumerateArray())
                Require(item, "id", "target", "action", "owner", "result");
            foreach (var entry in root.GetProperty("journal").EnumerateArray())
                Require(entry, "time", "action", "detail");
            var project = JsonSerializer.Deserialize<Project>(bytes, Options) ?? throw new ArgumentException("项目内容为null");
            var issues = Validation.Check(project, false);
            if (issues.Count > 0)
                throw new ArgumentException(string.Join("；", issues.Select(x => x.Path + "：" + x.Message)));
            return project;
        }
        catch (Exception error) when (error is JsonException or InvalidOperationException or FormatException or OverflowException)
        {
            throw new ArgumentException("JSON项目结构或数值格式错误：" + error.Message);
        }
    }

    public static T Clone<T>(T value)
    {
        return JsonSerializer.Deserialize<T>(JsonSerializer.SerializeToUtf8Bytes(value, Options), Options)!;
    }

    private static void Require(JsonElement element, params string[] fields)
    {
        if (element.ValueKind != JsonValueKind.Object)
            throw new ArgumentException("预期JSON对象");
        var names = element.EnumerateObject().Select(x => x.Name).ToList();
        if (names.Distinct(StringComparer.Ordinal).Count() != names.Count)
            throw new ArgumentException("JSON包含重复字段");
        foreach (var field in fields)
        {
            if (!element.TryGetProperty(field, out var value) || value.ValueKind == JsonValueKind.Null)
                throw new ArgumentException("缺少字段：" + field);
        }
    }

using Cabin.Core.Domain;

namespace Cabin.Core.IO;

public sealed record StoredProject(Guid Id, string Name, string Revision, DateTimeOffset Modified, bool IsDemo, string Error);

public sealed class ProjectStore(string directory)
{
    private readonly object gate = new();
    public string DirectoryPath { get; } = Path.GetFullPath(directory);

    public void Save(Project project)
    {
        var bytes = ProjectCodec.Encode(project);
        lock (gate)
        {
            Directory.CreateDirectory(DirectoryPath);
            var target = Path.Combine(DirectoryPath, project.Id + Software.Extension);
            var temporary = target + "." + Guid.NewGuid().ToString("N") + ".tmp";
            try
            {
                File.WriteAllBytes(temporary, bytes);
                File.Move(temporary, target, true);
            }
            finally
            {
                if (File.Exists(temporary))
                    File.Delete(temporary);
            }
        }
    }

    public Project Open(Guid id)
    {
        lock (gate)
        {
            var path = Path.Combine(DirectoryPath, id + Software.Extension);
            if (!File.Exists(path))
                throw new ArgumentException("本机项目已不存在，请从项目文件导入");
            var info = new FileInfo(path);
            if (info.Length > Software.MaximumBytes)
                throw new ArgumentException("项目超过2MiB，拒绝读取");
            return ProjectCodec.Decode(File.ReadAllBytes(path));
        }
    }

    public List<StoredProject> List()
    {
        var result = new List<StoredProject>();
        lock (gate)
        {
            if (!Directory.Exists(DirectoryPath))
                return result;
            foreach (var path in Directory.EnumerateFiles(DirectoryPath, "*" + Software.Extension))
            {
                var name = Path.GetFileName(path);
                if (!Guid.TryParse(name[..^Software.Extension.Length], out var id))
                    continue;
                try
                {
                    var project = Open(id);
                    result.Add(new(project.Id, project.Name, project.Revision, project.Modified, project.IsDemo, ""));
                }
                catch (Exception error) when (error is IOException or ArgumentException or UnauthorizedAccessException)
                {
                    result.Add(new(id, name, "?", File.GetLastWriteTimeUtc(path), false, error.Message));
                }
            }
        }
        return result.OrderByDescending(x => x.Modified).ToList();
    }

    public void Delete(Guid id)
    {
        lock (gate)
        {
            var path = Path.Combine(DirectoryPath, id + Software.Extension);
            if (!File.Exists(path))
                throw new ArgumentException("目标项目不存在");
            File.Delete(path);
        }
    }
}
