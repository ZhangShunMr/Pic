using System.Globalization;
using System.Text;
using Blower.Core.Domain;

namespace Blower.Core.Analysis;

public sealed record PlotLine(string Id, string Name, string Color, double X1, double Y1, double X2, double Y2);
public sealed record PlotBand(string Id, string Name, double Y, double Height, double LabelY);
public sealed record PlotGear(string Id, string Name, double X, double Rpm, bool Near);
public sealed record PlotTick(double Position, string Label);

public sealed class ChartModel
{
    public const double Width = 880;
    public const double Height = 420;
    public const double Left = 68;
    public const double Right = 844;
    public const double Top = 30;
    public const double Bottom = 358;
    public List<PlotLine> Lines { get; } = [];
    public List<PlotBand> Bands { get; } = [];
    public List<PlotGear> Gears { get; } = [];
    public List<PlotTick> XTicks { get; } = [];
    public List<PlotTick> YTicks { get; } = [];
    public double MaximumHz { get; private set; }
    public double MinimumRpm { get; private set; }
    public double MaximumRpm { get; private set; }

    public double X(double rpm)
    {
        return Left + (rpm - MinimumRpm) / (MaximumRpm - MinimumRpm) * (Right - Left);
    }

    public double Y(double frequency)
    {
        return Bottom - frequency / MaximumHz * (Bottom - Top);
    }

    public static string N(double number) => number.ToString("0.###", CultureInfo.InvariantCulture);

    public static ChartModel Create(Project project, AnalysisResult result, string planId, double? ceiling = null)
    {
        var orders = project.Orders.Where(x => x.Enabled).ToList();
        var maximum = Math.Max(100,
            Math.Max(result.Bands.Select(x => x.UpperHz).DefaultIfEmpty(0).Max(),
            orders.Select(x => Engine.Frequency(project.Settings.MaxRpm, x.EffectiveOrder(project.Settings.Blades)))
                .DefaultIfEmpty(0).Max()));
        var chart = new ChartModel
        {
            MaximumHz = ceiling is > 0 ? ceiling.Value : Math.Ceiling(maximum / 100) * 100,
            MinimumRpm = project.Settings.MinRpm,
            MaximumRpm = project.Settings.MaxRpm
        };
        for (var i = 0; i <= 5; i++)
        {
            var rpm = chart.MinimumRpm + (chart.MaximumRpm - chart.MinimumRpm) * i / 5;
            var hz = chart.MaximumHz * i / 5;
            chart.XTicks.Add(new(chart.X(rpm), rpm.ToString("0")));
            chart.YTicks.Add(new(chart.Y(hz), hz.ToString("0")));
        }
        var colors = new[] { "#cf5d18", "#197493", "#b87916", "#2a866b", "#5866ad", "#aa4a7a" };
        for (var i = 0; i < orders.Count; i++)
        {
            var order = orders[i];
            var q = order.EffectiveOrder(project.Settings.Blades);
            var endRpm = Math.Min(chart.MaximumRpm, chart.MaximumHz * 60 / q);
            if (endRpm < chart.MinimumRpm)
            {
                continue;
            }
            chart.Lines.Add(new(order.Id, $"{order.Name} · {q:0.##}阶", colors[i % colors.Length],
                chart.X(chart.MinimumRpm), chart.Y(Engine.Frequency(chart.MinimumRpm, q)),
                chart.X(endRpm), chart.Y(Engine.Frequency(endRpm, q))));
        }
        foreach (var band in result.Bands.Where(x => x.LowerHz <= chart.MaximumHz))
        {
            var top = chart.Y(Math.Min(chart.MaximumHz, band.UpperHz));
            var bottom = chart.Y(band.LowerHz);
            chart.Bands.Add(new(band.ModeId, $"{band.Name} {band.CenterHz:0.#}Hz", top,
                Math.Max(2, bottom - top), Math.Clamp(chart.Y(band.CenterHz) - 5, Top + 10, Bottom - 5)));
        }
        var plan = project.Plans.FirstOrDefault(x => x.Id == planId);
        if (plan != null)
        {
            foreach (var gear in plan.Gears)
            {
                chart.Gears.Add(new(gear.Id, gear.Name, chart.X(gear.Rpm), gear.Rpm,
                    result.Contacts.Any(x => x.PlanId == planId && x.GearId == gear.Id && x.Near)));
            }
        }
        return chart;
    }

    public string ToSvg(string title, string source)
    {
        static string E(string value) => System.Net.WebUtility.HtmlEncode(value);
        var b = new StringBuilder();
        b.Append($"<svg xmlns=\"http://www.w3.org/2000/svg\" viewBox=\"0 0 {Width} {Height + 60}\"><rect width=\"100%\" height=\"100%\" fill=\"#fffefa\"/>");
        b.Append("<g font-family=\"sans-serif\" font-size=\"11\" fill=\"#4c4c52\">");
        foreach (var tick in XTicks)
        {
            b.Append($"<path d=\"M{N(tick.Position)} {Top}V{Bottom}\" stroke=\"#deded9\"/><text x=\"{N(tick.Position)}\" y=\"380\" text-anchor=\"middle\">{E(tick.Label)}</text>");
        }
        foreach (var tick in YTicks)
        {
            b.Append($"<path d=\"M{Left} {N(tick.Position)}H{Right}\" stroke=\"#deded9\"/><text x=\"60\" y=\"{N(tick.Position + 4)}\" text-anchor=\"end\">{E(tick.Label)}</text>");
        }
        foreach (var band in Bands)
        {
            b.Append($"<rect x=\"{Left}\" y=\"{N(band.Y)}\" width=\"{Right - Left}\" height=\"{N(band.Height)}\" fill=\"#9b85bb\" opacity=\".25\"/><text x=\"{Right - 4}\" y=\"{N(band.LabelY)}\" text-anchor=\"end\">{E(band.Name)}</text>");
        }
        foreach (var line in Lines)
        {
            b.Append($"<path d=\"M{N(line.X1)} {N(line.Y1)}L{N(line.X2)} {N(line.Y2)}\" fill=\"none\" stroke=\"{line.Color}\" stroke-width=\"2\"/>");
        }
        foreach (var gear in Gears)
        {
            b.Append($"<path d=\"M{N(gear.X)} {Top}V{Bottom}\" stroke=\"{(gear.Near ? "#b44932" : "#267b67")}\" stroke-dasharray=\"5 5\"/><text x=\"{N(gear.X)}\" y=\"22\" text-anchor=\"middle\">{E(gear.Name)}</text>");
        }
        b.Append($"<text x=\"{Left}\" y=\"405\">转速 rpm →</text><text x=\"10\" y=\"16\">频率 Hz</text><text x=\"{Left}\" y=\"435\">{E(title)}</text><text x=\"{Left}\" y=\"455\">{E(source)} · {E(Software.Boundary)}</text></g></svg>");
        return b.ToString();
    }
}

public sealed class AnalysisResult
{
    public string InputHash { get; init; } = "";
    public DateTimeOffset Time { get; init; } = DateTimeOffset.UtcNow;
    public List<SensitiveBand> Bands { get; init; } = [];
    public List<Crossing> Crossings { get; init; } = [];
    public List<Contact> Contacts { get; init; } = [];
    public List<PlanSummary> Summaries { get; init; } = [];
}

public static class Engine
{
    public static double Frequency(double rpm, double order)
    {
        return rpm * order / 60.0;
    }

    public static SensitiveBand Band(Mode mode, DesignSettings settings)
    {
        var absolute = settings.AbsoluteBandHz;
        var relative = mode.FrequencyHz * settings.RelativeBandPercent / 100.0;
        var damping = settings.UseDamping ? (mode.DampingRatio ?? 0) * mode.FrequencyHz : 0;
        var width = Math.Max(Math.Max(absolute, relative), Math.Max(damping, mode.ExtraBandHz));
        var basis = $"max(绝对{absolute:0.###},百分比{relative:0.###},阻尼近似{damping:0.###},附加{mode.ExtraBandHz:0.###}) Hz";
        return new(mode.Id, mode.Name, mode.FrequencyHz, width,
            Math.Max(0, mode.FrequencyHz - width), mode.FrequencyHz + width, basis);
    }

    public static Contact Evaluate(Project project, SpeedPlan plan, Gear gear, Excitation order, Mode mode)
    {
        var q = order.EffectiveOrder(project.Settings.Blades);
        var band = Band(mode, project.Settings);
        var center = Frequency(gear.Rpm, q);
        var tolerance = Frequency(project.Settings.SpeedToleranceRpm, q);
        var lower = Math.Max(0, center - tolerance);
        var upper = center + tolerance;
        var clearance = Math.Max(band.LowerHz - upper, lower - band.UpperHz);
        return new(plan.Id, gear.Id, order.Id, mode.Id,
            plan.Name, gear.Name, order.Name, mode.Name,
            gear.Rpm, q, center, lower, upper, mode.FrequencyHz,
            band.HalfWidthHz, clearance, clearance <= 1e-8);
    }

    public static AnalysisResult Run(Project project)
    {
        Validation.RequireValid(project, true);
        var modes = project.Modes.Where(x => x.Enabled).ToList();
        var orders = project.Orders.Where(x => x.Enabled).ToList();
        var bands = modes.Select(x => Band(x, project.Settings)).ToList();
        var crossings = new List<Crossing>();
        foreach (var order in orders)
        {
            var q = order.EffectiveOrder(project.Settings.Blades);
            foreach (var band in bands)
            {
                var exact = 60 * band.CenterHz / q;
                var lower = 60 * band.LowerHz / q - project.Settings.SpeedToleranceRpm;
                var upper = 60 * band.UpperHz / q + project.Settings.SpeedToleranceRpm;
                crossings.Add(new(band.ModeId, order.Id, band.Name, order.Name, q,
                    exact, Math.Max(0, lower), upper,
                    upper >= project.Settings.MinRpm && lower <= project.Settings.MaxRpm));
            }
        }
        var contacts = new List<Contact>();
        var summaries = new List<PlanSummary>();
        foreach (var plan in project.Plans)
        {
            var gearSummaries = new List<GearSummary>();
            foreach (var gear in plan.Gears)
            {
                var current = new List<Contact>();
                foreach (var order in orders)
                {
                    foreach (var mode in modes)
                    {
                        current.Add(Evaluate(project, plan, gear, order, mode));
                    }
                }
                contacts.AddRange(current);
                gearSummaries.Add(new(gear.Id, gear.Name, gear.Rpm,
                    current.Count(x => x.Near), current.Count == 0 ? null : current.Min(x => x.ClearanceHz)));
            }
            var planContacts = contacts.Where(x => x.PlanId == plan.Id).ToList();
            summaries.Add(new(plan.Id, plan.Name, plan.Gears.Count,
                gearSummaries.Count(x => x.Contacts > 0), planContacts.Count(x => x.Near),
                planContacts.Count == 0 ? null : planContacts.Min(x => x.ClearanceHz), gearSummaries));
        }
        return new()
        {
            InputHash = IO.ProjectCodec.InputHash(project),
            Bands = bands,
            Crossings = crossings,
            Contacts = contacts,
            Summaries = summaries
        };
    }
}

using Blower.Core.Domain;

namespace Blower.Core.Analysis;

    string Baseline,
    string Target,
    int BaselineNear,
    int TargetNear,
    int BaselineContacts,
    int TargetContacts,
    bool EqualGearCount,
    IReadOnlyList<GearDifference> Rows);

public static class Comparisons
{
    public static PlanComparison Compare(Project project, AnalysisResult result, string baselineId, string targetId)
    {
        if (result.InputHash != IO.ProjectCodec.InputHash(project))
        {
            throw new InvalidOperationException("输入已变化，请重新计算后比较");
        }
        if (baselineId == targetId)
        {
            throw new InvalidOperationException("请选择两个不同方案");
        }
        var baseline = project.Plans.Single(x => x.Id == baselineId);
        var target = project.Plans.Single(x => x.Id == targetId);
        var a = result.Summaries.Single(x => x.PlanId == baselineId);
        var b = result.Summaries.Single(x => x.PlanId == targetId);
        var rows = new List<GearDifference>();
        for (var i = 0; i < Math.Max(baseline.Gears.Count, target.Gears.Count); i++)
        {
            var ga = baseline.Gears.ElementAtOrDefault(i);
            var gb = target.Gears.ElementAtOrDefault(i);
            rows.Add(new(i + 1, ga?.Name ?? "缺失", gb?.Name ?? "缺失",
                ga?.Rpm, gb?.Rpm,
                ga == null || gb == null ? null : gb.Rpm - ga.Rpm,
                ga == null ? null : a.Gears.Single(x => x.GearId == ga.Id).Contacts,
                gb == null ? null : b.Gears.Single(x => x.GearId == gb.Id).Contacts,
                ga == null || gb == null ? "缺挡位，不形成转速差" : "按方案内挡位序号比较"));
        }
        return new(baseline.Name, target.Name, a.NearGearCount, b.NearGearCount,
            a.ContactCount, b.ContactCount, baseline.Gears.Count == target.Gears.Count, rows);
    }
}

using Blower.Core.Domain;

namespace Blower.Core.Analysis;

public sealed record ReviewFinding(string Severity, string Object, string Finding, string Action);

public static class QualityReview
{
    public static IReadOnlyList<ReviewFinding> Inspect(Project project)
    {
        var findings = new List<ReviewFinding>();
        void Add(string severity, string obj, string finding, string action)
        {
            findings.Add(new(severity, obj, finding, action));
        }
        if (project.Origin == Origin.Simulation)
        {
            Add("来源", "项目", "当前为模拟设计数据", "只用于流程和公式核对，实际设计前替换来源与参数");
        }
        if (string.IsNullOrWhiteSpace(project.Owner))
        {
            Add("待补充", "项目", "负责人未填写", "选中项目，在右侧填写负责人并应用");
        }
        if (string.IsNullOrWhiteSpace(project.Vehicle))
        {
            Add("待补充", "项目", "车辆或平台为空", "补充平台及适用安装边界");
        }
        foreach (var mode in project.Modes.Where(x => x.Enabled))
        {
            if (mode.DampingRatio == null)
            {
                Add("未知", mode.Name, "阻尼比未知，阻尼半宽项未计入", "可保留未知；依据试验或仿真补充后重新计算");
            }
            if (mode.Origin != project.Origin)
            {
                Add("来源", mode.Name, "模态来源类别与项目不同", "核对模态来源；报告保留每个模态的独立标记");
            }
            if (string.IsNullOrWhiteSpace(mode.Location))
            {
                Add("待补充", mode.Name, "结构位置为空", "填写安装位置与可复现的模态边界条件");
            }
            if (mode.Source.Contains("待补充", StringComparison.Ordinal))
            {
                Add("待补充", mode.Name, "模态来源尚未完善", "补充文件编号、工况、安装状态及数据提供方");
            }
        }
        var enabled = project.Modes.Where(x => x.Enabled).OrderBy(x => x.FrequencyHz).ToList();
        for (var i = 1; i < enabled.Count; i++)
        {
            var a = Engine.Band(enabled[i - 1], project.Settings);
            var b = Engine.Band(enabled[i], project.Settings);
            if (a.UpperHz >= b.LowerHz)
            {
                Add("注意", a.Name + " / " + b.Name, "模态敏感区重叠", "阻尼带宽近似可能不适用；组合计数按模态对象分别计算，不代表独立能量");
            }
        }
        foreach (var plan in project.Plans)
        {
            if (plan.Gears.Count == 0)
            {
                Add("待补充", plan.Name, "没有挡位", "选中方案后添加挡位或导入CSV");
            }
            if (plan.Gears.Count > 0 && plan.Gears.All(x => x.Fixed))
            {
                Add("注意", plan.Name, "全部挡位固定", "扫描仅核对原始点，不会产生转速调整");
            }
        }
        foreach (var task in project.Checks.Where(x => x.State == CheckState.Recheck))
        {
            Add("需复核", task.Title, "输入变化后原验证证据需要复核", "在验证清单选中任务，核对新输入后更新证据与状态");
        }
        if (project.Settings.AbsoluteBandHz == 0 && project.Settings.RelativeBandPercent == 0)
        {
            Add("注意", "规则", "绝对和百分比敏感半宽均为零", "确认是否只使用阻尼或附加半宽，避免漏掉参数不确定性");
        }
        return findings;
    }
}

using System.Globalization;
using Blower.Core.Domain;
using Blower.Core.IO;

namespace Blower.Core.Analysis;

public sealed record RuleCase(
    double AbsoluteBandHz,
    double SpeedToleranceRpm,
    int NearGears,
    int Contacts,
    int ChangedStatuses,
    double MinimumClearanceHz,
    string GearNames);

public sealed class RuleStudyResult
{
    public string InputHash { get; init; } = "";
    public string PlanId { get; init; } = "";
    public string PlanName { get; init; } = "";
    public DateTimeOffset Time { get; init; } = DateTimeOffset.UtcNow;
    public List<RuleCase> Cases { get; init; } = [];
}

public static class RuleStudy
{
    public static IReadOnlyList<double> ParseValues(string text, double maximum, string field)
    {
        var values = new List<double>();
        foreach (var cell in text.Split(',', StringSplitOptions.TrimEntries))
        {
            if (!double.TryParse(cell, NumberStyles.Float, CultureInfo.InvariantCulture, out var number)
                || !double.IsFinite(number)
                || number < 0
                || number > maximum)
            {
                throw new InvalidDataException(field + $"只接受0～{maximum}的数值，用英文逗号分隔");
            }
            if (!values.Contains(number))
            {
                values.Add(number);
            }
        }
        if (values.Count is < 1 or > 5)
        {
            throw new InvalidDataException(field + "需要1～5个不同值");
        }
        return values.Order().ToList();
    }

    public static RuleStudyResult Run(Project project, string planId, string absoluteValues, string toleranceValues)
    {
        Validation.RequireValid(project, true);
        var plan = project.Plans.Single(x => x.Id == planId);
        if (plan.Gears.Count == 0)
        {
            throw new InvalidOperationException("当前方案没有挡位");
        }
        var baseline = Engine.Run(project);
        var original = baseline.Contacts.Where(x => x.PlanId == planId).ToDictionary(x => x.Key);
        var widths = ParseValues(absoluteValues, 5000, "绝对半宽");
        var tolerances = ParseValues(toleranceValues, 1000, "转速容差");
        var cases = new List<RuleCase>();
        foreach (var width in widths)
        {
            foreach (var tolerance in tolerances)
            {
                var candidate = ProjectCodec.Clone(project);
                candidate.Settings.AbsoluteBandHz = width;
                candidate.Settings.SpeedToleranceRpm = tolerance;
                var result = Engine.Run(candidate);
                var summary = result.Summaries.Single(x => x.PlanId == planId);
                var contacts = result.Contacts.Where(x => x.PlanId == planId).ToList();
                var changes = contacts.Count(x => x.Near != original[x.Key].Near);
                var gears = string.Join("、", summary.Gears.Where(x => x.Contacts > 0).Select(x => x.Name));
                cases.Add(new(width, tolerance, summary.NearGearCount, summary.ContactCount,
                    changes, summary.MinimumClearanceHz ?? 0, gears.Length == 0 ? "无接近挡位" : gears));
            }
        }
        return new()
        {
            InputHash = baseline.InputHash,
            PlanId = planId,
            PlanName = plan.Name,
            Cases = cases
        };
    }

    public static string ToCsv(Project project, RuleStudyResult study)
    {
        if (study.InputHash != ProjectCodec.InputHash(project))
        {
            throw new InvalidOperationException("规则敏感性试算已过期；请重新试算");
        }
        var rows = new List<string>
        {
            CsvExchange.Row("project", "source", "input_sha256", "plan", "absolute_half_width_hz",
                "speed_tolerance_rpm", "near_gears", "contacts", "changed_statuses", "minimum_clearance_hz", "gear_names")
        };
        foreach (var item in study.Cases)
        {
            rows.Add(CsvExchange.Row(project.Name, Labels.Source(project.Origin), study.InputHash,
                study.PlanName, CsvExchange.Number(item.AbsoluteBandHz), CsvExchange.Number(item.SpeedToleranceRpm),
                item.NearGears.ToString(), item.Contacts.ToString(), item.ChangedStatuses.ToString(),
                CsvExchange.Number(item.MinimumClearanceHz), item.GearNames));
        }
        return string.Join(Environment.NewLine, rows);
    }
}

using Blower.Core.Domain;

namespace Blower.Core.Analysis;

public sealed record ScanChoice(int Position, string Gear, double OriginalRpm, double CandidateRpm, int Contacts, double Cost);

public sealed record ScanResult(
    bool Feasible,
    string Message,
    string InputHash,
    string PlanId,
    IReadOnlyList<ScanChoice> Choices,
    int EvaluatedPoints,
    double TotalCost);

public static class SpeedScanner
{
    private sealed record Node(double Rpm, int Contacts, double LocalCost, double Cost, int Previous);

    public static ScanResult Scan(Project project, string planId)
    {
        Validation.RequireValid(project, true);
        var plan = project.Plans.Single(x => x.Id == planId);
        var hash = IO.ProjectCodec.InputHash(project);
        var s = project.Settings;
        var modes = project.Modes.Where(x => x.Enabled).ToList();
        var orders = project.Orders.Where(x => x.Enabled).ToList();
        if (plan.Gears.Count == 0)
        {
            return new(false, "当前方案没有挡位", hash, planId, [], 0, 0);
        }
        var pointLimit = (int)Math.Ceiling(2 * s.ScanWindowRpm / s.ScanStepRpm) + 3;
        if (pointLimit > 603)
        {
            return new(false, "扫描窗口与步长形成超过603个候选/挡位；请增大步长或缩小窗口", hash, planId, [], 0, 0);
        }
        var levels = new List<List<Node>>();
        var evaluated = 0;
        for (var i = 0; i < plan.Gears.Count; i++)
        {
            var gear = plan.Gears[i];
            var lower = Math.Max(s.MinRpm, gear.Rpm - s.ScanWindowRpm);
            var upper = Math.Min(s.MaxRpm, gear.Rpm + s.ScanWindowRpm);
            var candidates = new SortedSet<double> { gear.Rpm };
            if (!gear.Fixed)
            {
                for (var rpm = Math.Ceiling(lower / s.ScanStepRpm) * s.ScanStepRpm;
                    rpm <= upper + 1e-8; rpm += s.ScanStepRpm)
                {
                    candidates.Add(Math.Round(rpm, 6));
                }
            }
            var level = new List<Node>();
            foreach (var rpm in candidates)
            {
                var probe = new Gear { Id = gear.Id, Name = gear.Name, Rpm = rpm };
                var count = 0;
                foreach (var order in orders)
                {
                    foreach (var mode in modes)
                    {
                        if (Engine.Evaluate(project, plan, probe, order, mode).Near)
                        {
                            count++;
                        }
                    }
                }
                evaluated++;
                var shift = Math.Abs(rpm - gear.Rpm) / Math.Max(1, s.ScanWindowRpm);
                var local = count + s.ShiftWeight * shift;
                var cost = i == 0 ? local : double.PositiveInfinity;
                var previous = -1;
                if (i > 0)
                {
                    for (var j = 0; j < levels[i - 1].Count; j++)
                    {
                        var parent = levels[i - 1][j];
                        if (rpm - parent.Rpm < s.MinimumGapRpm - 1e-8)
                        {
                            continue;
                        }
                        if (parent.Cost + local < cost - 1e-10)
                        {
                            cost = parent.Cost + local;
                            previous = j;
                        }
                    }
                }
                level.Add(new(rpm, count, local, cost, previous));
            }
            levels.Add(level);
        }
        var last = levels[^1];
        var selected = Enumerable.Range(0, last.Count).MinBy(x => last[x].Cost);
        if (!double.IsFinite(last[selected].Cost))
        {
            return new(false, "当前窗口、固定挡位与最小间距无法同时满足；请检查约束", hash, planId, [], evaluated, 0);
        }
        var total = last[selected].Cost;
        var choices = new List<ScanChoice>();
        for (var i = levels.Count - 1; i >= 0; i--)
        {
            var node = levels[i][selected];
            var gear = plan.Gears[i];
            choices.Add(new(i + 1, gear.Name, gear.Rpm, node.Rpm, node.Contacts, node.LocalCost));
            selected = node.Previous;
        }
        choices.Reverse();
        return new(true, "约束内离散候选已生成；需验证风量、效率与实际响应", hash, planId, choices, evaluated, total);
    }
}

namespace Blower.Core.Domain;

public sealed record Issue(string Path, string Message, bool Blocks = true);

public static class Validation
{
    public static IReadOnlyList<Issue> Inspect(Project project, bool forAnalysis = false)
    {
        var issues = new List<Issue>();
        void Required(string value, string path, int max = 160)
        {
            if (string.IsNullOrWhiteSpace(value))
            {
                issues.Add(new(path, "不能为空"));
            }
            else if (value.Length > max)
            {
                issues.Add(new(path, $"不得超过{max}个字符"));
            }
        }
        void Range(double value, double min, double max, string path)
        {
            if (!double.IsFinite(value) || value < min || value > max)
            {
                issues.Add(new(path, $"必须在{min}至{max}之间，且为有限数值"));
            }
        }
        if (project.Schema != Software.Schema)
        {
            issues.Add(new("格式版本", "仅支持schema=1；请使用对应版本另存"));
        }
        Required(project.Id, "项目标识");
        if (project.Id.Length != 32 || project.Id.Any(x => !Uri.IsHexDigit(x)))
        {
            issues.Add(new("项目标识", "必须为32位十六进制"));
        }
        Required(project.Name, "项目名称");
        Required(project.Source, "项目来源", 2000);
        if (!Enum.IsDefined(project.Origin))
        {
            issues.Add(new("数据来源", "未知枚举值"));
        }
        var s = project.Settings;
        Range(s.Blades, 1, 200, "叶片数");
        Range(s.MinRpm, 0, 50000, "转速下限");
        Range(s.MaxRpm, 1, 50000, "转速上限");
        if (s.MaxRpm <= s.MinRpm)
        {
            issues.Add(new("转速范围", "上限必须大于下限"));
        }
        Range(s.AbsoluteBandHz, 0, 5000, "绝对半宽Hz");
        Range(s.RelativeBandPercent, 0, 30, "百分比半宽%");
        Range(s.SpeedToleranceRpm, 0, 1000, "转速容差rpm");
        Range(s.MinimumGapRpm, 1, 5000, "最小挡位间距rpm");
        Range(s.ScanStepRpm, 1, 500, "扫描步长rpm");
        Range(s.ScanWindowRpm, 0, 3000, "扫描窗口±rpm");
        Range(s.ShiftWeight, 0, 1, "偏移权重");
        if (project.Orders.Count > 24 || project.Modes.Count > 100 || project.Plans.Count > 20)
        {
            issues.Add(new("对象数量", "最多24个阶次、100个模态和20个方案"));
        }
        var ids = new HashSet<string>(StringComparer.Ordinal);
        void Unique(string id, string path)
        {
            if (string.IsNullOrWhiteSpace(id) || !ids.Add(id))
            {
                issues.Add(new(path, "标识为空或重复"));
            }
        }
        var effectiveOrders = new HashSet<double>();
        foreach (var order in project.Orders)
        {
            var path = $"阶次[{order.Name}]";
            Unique(order.Id, path);
            Required(order.Name, path + ".名称");
            Range(order.Multiplier, 0.1, 100, path + ".倍数");
            if (!Enum.IsDefined(order.Kind))
            {
                issues.Add(new(path + ".类别", "未知阶次类别"));
            }
            if (order.Enabled && !effectiveOrders.Add(order.EffectiveOrder(s.Blades)))
            {
                issues.Add(new(path, "存在重复的启用有效阶次；请合并或停用其中一项，避免组合重复计数"));
            }
        }
        foreach (var mode in project.Modes)
        {
            var path = $"模态[{mode.Name}]";
            Unique(mode.Id, path);
            Required(mode.Name, path + ".名称");
            Required(mode.Source, path + ".来源", 2000);
            Range(mode.FrequencyHz, 0.1, 50000, path + ".频率Hz");
            Range(mode.ExtraBandHz, 0, 5000, path + ".附加半宽Hz");
            if (mode.DampingRatio.HasValue)
            {
                Range(mode.DampingRatio.Value, 0, 0.1, path + ".阻尼比ζ");
            }
            if (!Enum.IsDefined(mode.Origin))
            {
                issues.Add(new(path + ".来源类别", "未知枚举值"));
            }
        }
        foreach (var plan in project.Plans)
        {
            var path = $"方案[{plan.Name}]";
            Unique(plan.Id, path);
            Required(plan.Name, path + ".名称");
            if (plan.Gears.Count > 24)
            {
                issues.Add(new(path, "最多24个挡位"));
            }
            var previous = double.NegativeInfinity;
            foreach (var gear in plan.Gears)
            {
                Unique(gear.Id, path + ".挡位");
                Required(gear.Name, path + ".挡位名称");
                Range(gear.Rpm, s.MinRpm, s.MaxRpm, path + $".{gear.Name}.rpm");
                if (gear.Rpm - previous < s.MinimumGapRpm - 1e-8)
                {
                    issues.Add(new(path + $".{gear.Name}", "挡位须按转速递增排列并满足最小间距"));
                }
                previous = gear.Rpm;
            }
        }
        if (project.Checks.Count > 2000 || project.Audit.Count > 5000)
        {
            issues.Add(new("记录数量", "验证记录最多2000条，审计最多5000条"));
        }
        var taskKeys = new HashSet<string>(StringComparer.Ordinal);
        foreach (var task in project.Checks)
        {
            Unique(task.Id, "验证记录");
            var expectedKey = string.Join("/", task.PlanId, task.GearId, task.OrderId, task.ModeId);
            if (task.Key != expectedKey || !taskKeys.Add(task.Key))
            {
                issues.Add(new("验证记录." + task.Title, "组合标识不一致或重复"));
            }
            var plan = project.Plans.FirstOrDefault(x => x.Id == task.PlanId);
            if (plan == null || !plan.Gears.Any(x => x.Id == task.GearId)
                || !project.Modes.Any(x => x.Id == task.ModeId)
                || !project.Orders.Any(x => x.Id == task.OrderId))
            {
                issues.Add(new("验证记录." + task.Title, "关联对象缺失；应先移除对应验证记录"));
            }
            if (!Enum.IsDefined(task.State))
            {
                issues.Add(new("验证状态", "未知枚举值"));
            }
            if (task.State == CheckState.Completed && string.IsNullOrWhiteSpace(task.Evidence))
            {
                issues.Add(new("验证记录." + task.Title, "已记录证据状态必须填写证据位置或结论"));
            }
        }
        if (forAnalysis)
        {
            if (!project.Orders.Any(x => x.Enabled))
            {
                issues.Add(new("阶次", "至少启用一个阶次"));
            }
            if (!project.Modes.Any(x => x.Enabled))
            {
                issues.Add(new("模态", "至少启用一个模态"));
            }
            if (!project.Plans.Any(x => x.Gears.Count > 0))
            {
                issues.Add(new("挡位方案", "至少建立一个含挡位的方案"));
            }
        }
        if (string.IsNullOrWhiteSpace(project.Owner))
        {
            issues.Add(new("负责人", "建议补充以便追溯", false));
        }
        return issues;
    }

    public static void RequireValid(Project project, bool forAnalysis = false)
    {
        var errors = Inspect(project, forAnalysis).Where(x => x.Blocks).ToList();
        if (errors.Count > 0)
        {
            throw new InvalidDataException(string.Join("；", errors.Select(x => x.Path + "：" + x.Message)));
        }
    }
}

using System.Globalization;
using System.Text;
using Blower.Core.Domain;

namespace Blower.Core.IO;

public sealed record CsvProblem(int Line, string Column, string Message);
public sealed record CsvPreview(string Kind, string InputHash, IReadOnlyList<CsvProblem> Problems,
    IReadOnlyList<Gear> Gears, IReadOnlyList<Mode> Modes)
{
    public bool Valid => Problems.Count == 0 && Gears.Count + Modes.Count > 0;
}

public static class CsvExchange
{
    public static string Cell(string? value)
    {
        value ??= "";
        var numeric = double.TryParse(value, NumberStyles.Float, CultureInfo.InvariantCulture, out var number)
            && double.IsFinite(number);
        if (!numeric && value.Length > 0 && "=+-@\t\r".Contains(value[0]))
        {
            value = "'" + value;
        }
        return "\"" + value.Replace("\"", "\"\"") + "\"";
    }

    public static string Number(double value)
    {
        return value.ToString("0.########", CultureInfo.InvariantCulture);
    }

    public static string Row(params string?[] values)
    {
        return string.Join(",", values.Select(Cell));
    }

    public static IReadOnlyList<string> ParseRow(string line)
    {
        var cells = new List<string>();
        var cell = new StringBuilder();
        var quoted = false;
        var closed = false;
        for (var i = 0; i < line.Length; i++)
        {
            var c = line[i];
            if (quoted)
            {
                if (c == '"' && i + 1 < line.Length && line[i + 1] == '"')
                {
                    cell.Append('"');
                    i++;
                }
                else if (c == '"')
                {
                    quoted = false;
                    closed = true;
                }
                else
                {
                    cell.Append(c);
                }
            }
            else if (c == ',')
            {
                cells.Add(cell.ToString());
                cell.Clear();
                closed = false;
            }
            else if (c == '"' && cell.Length == 0 && !closed)
            {
                quoted = true;
            }
            else if (closed || c == '"')
            {
                throw new InvalidDataException("引号必须包围整个单元格；不支持跨行单元格");
            }
            else
            {
                cell.Append(c);
            }
        }
        if (quoted)
        {
            throw new InvalidDataException("引号未闭合；不支持跨行单元格");
        }
        cells.Add(cell.ToString());
        return cells;
    }

    public static CsvPreview Preview(string text, string kind, Project project)
    {
        if (kind is not "gears" and not "modes")
        {
            throw new InvalidDataException("CSV类型仅支持gears或modes");
        }
        var problems = new List<CsvProblem>();
        var gears = new List<Gear>();
        var modes = new List<Mode>();
        var hash = ProjectCodec.InputHash(project);
        var rows = text.TrimStart('\uFEFF').Replace("\r\n", "\n").Split('\n');
        var expected = kind == "gears" ? "name,rpm,fixed" : "name,frequency_hz,damping_ratio,source";
        if (rows.Length == 0 || rows[0].TrimEnd('\r') != expected)
        {
            problems.Add(new(1, "表头", "必须为 " + expected));
            return new(kind, hash, problems, gears, modes);
        }
        var limit = kind == "gears" ? 24 : 100;
        var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        for (var i = 1; i < rows.Length; i++)
        {
            if (string.IsNullOrWhiteSpace(rows[i]))
            {
                continue;
            }
            if (gears.Count + modes.Count >= limit)
            {
                problems.Add(new(i + 1, "行数", "超过对象数量上限"));
                break;
            }
            try
            {
                var cells = ParseRow(rows[i].TrimEnd('\r'));
                if (cells.Count != (kind == "gears" ? 3 : 4))
                {
                    throw new InvalidDataException("列数不符；使用英文逗号和小数点");
                }
                if (string.IsNullOrWhiteSpace(cells[0]) || !names.Add(cells[0]))
                {
                    throw new InvalidDataException("名称为空或重复");
                }
                if (!double.TryParse(cells[1], NumberStyles.Float, CultureInfo.InvariantCulture, out var value)
                    || !double.IsFinite(value))
                {
                    throw new InvalidDataException("第二列须为有限数值");
                }
                if (kind == "gears")
                {
                    if (!bool.TryParse(cells[2], out var fixedGear))
                    {
                        throw new InvalidDataException("fixed只接受true或false");
                    }
                    gears.Add(new() { Name = cells[0], Rpm = value, Fixed = fixedGear });
                }
                else
                {
                    double? damping = null;
                    if (!string.IsNullOrWhiteSpace(cells[2]))
                    {
                        if (!double.TryParse(cells[2], NumberStyles.Float, CultureInfo.InvariantCulture, out var d)
                            || !double.IsFinite(d))
                        {
                            throw new InvalidDataException("阻尼比须为小数，未知留空");
                        }
                        damping = d;
                    }
                    modes.Add(new()
                    {
                        Name = cells[0], FrequencyHz = value, DampingRatio = damping,
                        Source = cells[3], Origin = project.Origin
                    });
                }
            }
            catch (InvalidDataException error)
            {
                problems.Add(new(i + 1, "内容", error.Message));
            }
        }
        if (gears.Count + modes.Count == 0 && problems.Count == 0)
        {
            problems.Add(new(2, "内容", "至少提供一个对象"));
        }
        var candidate = ProjectCodec.Clone(project);
        if (kind == "gears")
        {
            candidate.Plans = [new() { Name = "CSV预检", Gears = gears }];
            candidate.Checks = [];
        }
        else
        {
            candidate.Modes = modes;
            candidate.Checks = [];
        }
        foreach (var issue in Validation.Inspect(candidate).Where(x => x.Blocks))
        {
            problems.Add(new(0, issue.Path, issue.Message));
        }
        return new(kind, hash, problems, gears, modes);
    }
}

{

using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Text.Json.Serialization;
using Blower.Core.Domain;

namespace Blower.Core.IO;

public static class ProjectCodec
{
    public const int MaximumBytes = 4 * 1024 * 1024;
    public static readonly JsonSerializerOptions Options = new()
    {
        WriteIndented = true,
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow,
        Converters = { new JsonStringEnumConverter() },
        MaxDepth = 32
    };

    public static string Encode(Project project)
    {
        Validation.RequireValid(project);
        var content = JsonSerializer.Serialize(project, Options);
        if (Encoding.UTF8.GetByteCount(content) > MaximumBytes)
        {
            throw new InvalidDataException("项目超过4MiB上限；请拆分项目或精简备注");
        }
        return content;
    }

    public static Project Decode(string json)
    {
        if (Encoding.UTF8.GetByteCount(json) > MaximumBytes)
        {
            throw new InvalidDataException("项目超过4MiB上限");
        }
        try
        {
            using var document = JsonDocument.Parse(json, new() { MaxDepth = 32 });
            SchemaGuard.Inspect(document.RootElement);
            if (!document.RootElement.TryGetProperty("schema", out var schema)
                || schema.GetInt32() != Software.Schema)
            {
                throw new InvalidDataException("项目格式版本缺失或不支持；需要schema=1");
            }
            foreach (var name in new[] { "id", "name", "settings", "orders", "modes", "plans", "checks", "audit", "source" })
            {
                if (!document.RootElement.TryGetProperty(name, out var element)
                    || element.ValueKind == JsonValueKind.Null)
                {
                    throw new InvalidDataException("项目缺少必填字段：" + name);
                }
            }
            var project = JsonSerializer.Deserialize<Project>(json, Options)
                ?? throw new InvalidDataException("项目内容为空");
            CheckNullMembers(project);
            Validation.RequireValid(project);
            return project;
        }
        catch (Exception error) when (error is JsonException or InvalidOperationException or NullReferenceException or FormatException or OverflowException)
        {
            throw new InvalidDataException("项目JSON损坏或字段类型错误：" + error.Message, error);
        }
    }

    private static void CheckNullMembers(Project project)
    {
        if (project.Orders.Any(x => x is null)
            || project.Modes.Any(x => x is null)
            || project.Plans.Any(x => x is null || x.Gears is null || x.Gears.Any(g => g is null))
            || project.Checks.Any(x => x is null)
            || project.Audit.Any(x => x is null))
        {
            throw new InvalidDataException("对象列表含null，无法建立完整引用关系");
        }
    }

    public static Project Clone(Project project)
    {
        return JsonSerializer.Deserialize<Project>(JsonSerializer.Serialize(project, Options), Options)!;
    }

    public static string Hash(string content)
    {
        return Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(content))).ToLowerInvariant();
    }

    public static string InputHash(Project project)
    {
        var data = new { project.Settings, project.Orders, project.Modes, project.Plans };
        return Hash(JsonSerializer.Serialize(data, Options));
    }
}

using System.Text;
using Blower.Core.Domain;

namespace Blower.Core.IO;

public sealed record StoredProject(string Id, string Name, DateTimeOffset Modified, int Revision, bool Archived, string Status);
public sealed record OpenedProject(Project Project, string Token, string Path);
public sealed record SavedProject(string Token, string Path);

public sealed class ProjectStore
{
    private readonly string root;
    private static readonly object Gate = new();

    public ProjectStore(string root)
    {
        this.root = Path.GetFullPath(root);
        Directory.CreateDirectory(this.root);
    }

    private string PathFor(string id)
    {
        if (id.Length != 32 || id.Any(c => !Uri.IsHexDigit(c)))
        {
            throw new InvalidDataException("项目标识必须为32位十六进制；拒绝路径字符");
        }
        return Path.Combine(root, id + ".blowerproj");
    }

    public IReadOnlyList<StoredProject> List()
    {
        var result = new List<StoredProject>();
        lock (Gate)
        {
            foreach (var file in Directory.EnumerateFiles(root, "*.blowerproj"))
            {
                try
                {
                    var project = ProjectCodec.Decode(File.ReadAllText(file, Encoding.UTF8));
                    result.Add(new(project.Id, project.Name, project.Modified, project.Revision, project.Archived, "可打开"));
                }
                catch (Exception error) when (error is IOException or InvalidDataException or UnauthorizedAccessException)
                {
                    result.Add(new(Path.GetFileNameWithoutExtension(file), "损坏文件", File.GetLastWriteTimeUtc(file), 0, false, error.Message));
                }
            }
        }
        return result.OrderByDescending(x => x.Modified).ToList();
    }

    public OpenedProject Open(string id, bool backup = false)
    {
        lock (Gate)
        {
            var path = PathFor(id);
            var actual = backup ? path + ".bak" : path;
            var content = File.ReadAllText(actual, Encoding.UTF8);
            var project = ProjectCodec.Decode(content);
            if (project.Id != id)
            {
                throw new InvalidDataException("文件标识与内容不符；请通过项目导入检查");
            }
            var token = File.Exists(path) ? ProjectCodec.Hash(File.ReadAllText(path, Encoding.UTF8)) : "";
            return new(project, token, actual);
        }
    }

    public SavedProject Save(Project project, string expectedToken)
    {
        var content = ProjectCodec.Encode(project);
        var path = PathFor(project.Id);
        lock (Gate)
        {
            var exists = File.Exists(path);
            var current = exists ? ProjectCodec.Hash(File.ReadAllText(path, Encoding.UTF8)) : "";
            if (current != expectedToken)
            {
                throw new IOException("项目库中此项目已被其他会话修改；请先下载当前草稿，重新打开后比较或复制另存");
            }
            var temporary = path + "." + Software.Id() + ".tmp";
            try
            {
                using (var stream = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None))
                {
                    var bytes = Encoding.UTF8.GetBytes(content);
                    stream.Write(bytes);
                    stream.Flush(true);
                }
                if (exists)
                {
                    File.Copy(path, path + ".bak", true);
                }
                File.Move(temporary, path, true);
            }
            finally
            {
                if (File.Exists(temporary))
                {
                    File.Delete(temporary);
                }
            }
            return new(ProjectCodec.Hash(content), path);
        }
    }
}

using System.Text.Json;

namespace Blower.Core.IO;

public static class SchemaGuard
{
    private static void Require(JsonElement element, string path, params string[] names)
    {
        if (element.ValueKind != JsonValueKind.Object)
        {
            throw new InvalidDataException(path + "须为对象");
        }
        foreach (var name in names)
        {
            if (!element.TryGetProperty(name, out var value) || value.ValueKind == JsonValueKind.Null)
            {
                throw new InvalidDataException(path + "." + name + "缺失或为null");
            }
        }
        var seen = new HashSet<string>(StringComparer.Ordinal);
        foreach (var property in element.EnumerateObject())
        {
            if (!seen.Add(property.Name))
            {
                throw new InvalidDataException(path + "包含重复字段：" + property.Name);
            }
        }
    }

    private static void Each(JsonElement root, string name, Action<JsonElement, string> inspect)
    {
        var list = root.GetProperty(name);
        if (list.ValueKind != JsonValueKind.Array)
        {
            throw new InvalidDataException(name + "须为数组");
        }
        var index = 0;
        foreach (var element in list.EnumerateArray())
        {
            inspect(element, name + "[" + index + "]");
            index++;
        }
    }

    public static void Inspect(JsonElement root)
    {
        Require(root, "project", "schema", "id", "name", "origin", "source", "settings",
            "orders", "modes", "plans", "checks", "audit", "revision", "created", "modified", "archived");
        Require(root.GetProperty("settings"), "settings", "blades", "minRpm", "maxRpm",
            "absoluteBandHz", "relativeBandPercent", "useDamping", "speedToleranceRpm",
            "minimumGapRpm", "scanStepRpm", "scanWindowRpm", "shiftWeight");
        Each(root, "orders", (item, path) =>
        {
            Require(item, path, "id", "name", "kind", "multiplier", "enabled", "evidence");
        });
        Each(root, "modes", (item, path) =>
        {
            Require(item, path, "id", "name", "frequencyHz", "extraBandHz", "enabled", "origin", "source", "location");
            if (!item.TryGetProperty("dampingRatio", out _))
            {
                throw new InvalidDataException(path + ".dampingRatio缺失；未知阻尼请明确写null");
            }
        });
        Each(root, "plans", (item, path) =>
        {
            Require(item, path, "id", "name", "notes", "locked", "gears");
            Each(item, "gears", (gear, gearPath) =>
            {
                Require(gear, path + "." + gearPath, "id", "name", "rpm", "fixed", "notes");
            });
        });
        Each(root, "checks", (item, path) =>
        {
            Require(item, path, "id", "key", "planId", "gearId", "modeId", "orderId",
                "title", "method", "owner", "evidence", "state", "updated");
        });
        Each(root, "audit", (item, path) =>
        {
            Require(item, path, "time", "action", "detail", "revision");
        });
    }
}

namespace Blower.Core.Operations;

public static class ProjectCommands
{

        {
            Name = name,
            Notes = source.Notes,
            Gears = source.Gears.Select(x => new Gear
            {

                Rpm = x.Rpm,
                Fixed = x.Fixed,
                Notes = x.Notes
            }).ToList()
        };
    }

    public static Project Duplicate(Project source)
    {
        var copy = ProjectCodec.Clone(source);
        copy.Id = Software.Id();
        copy.Name += " · 副本";
        copy.Created = DateTimeOffset.UtcNow;
        copy.Modified = copy.Created;
        copy.Archived = false;
        copy.Revision = 0;
        copy.Audit = [new() { Action = "复制项目", Detail = source.Id + " / 修订" + source.Revision }];
        return copy;
    }

    public static void RemoveObject(Project project, string kind, string id)
    {
        var related = project.Checks.Where(x => kind switch
        {
            "mode" => x.ModeId == id,
            "order" => x.OrderId == id,
            "gear" => x.GearId == id,
            "plan" => x.PlanId == id,
            _ => false
        }).ToList();
        if (related.Count > 0)
        {
            throw new InvalidOperationException($"仍有{related.Count}条验证记录引用此对象；请先移除相关验证记录");
        }
        switch (kind)
        {
            case "mode":
                project.Modes.RemoveAll(x => x.Id == id);
                break;
            case "order":
                project.Orders.RemoveAll(x => x.Id == id);
                break;
            case "plan":
                if (project.Plans.Single(x => x.Id == id).Locked)
                {
                    throw new InvalidOperationException("先解除方案锁定");
                }
                project.Plans.RemoveAll(x => x.Id == id);
                break;
            case "gear":
                var plan = project.Plans.Single(x => x.Gears.Any(g => g.Id == id));
                if (plan.Locked)
                {
                    throw new InvalidOperationException("先解除方案锁定");
                }
                plan.Gears.RemoveAll(x => x.Id == id);
                break;
            case "check":
                project.Checks.RemoveAll(x => x.Id == id);
                break;
            default:
                throw new InvalidOperationException("此对象不可删除");
        }
    }

    public static SpeedPlan ApplyScan(Project project, ScanResult scan)
    {
        if (!scan.Feasible || scan.InputHash != ProjectCodec.InputHash(project))
        {
            throw new InvalidOperationException("扫描无可行解或输入已变化，请重新扫描");
        }
        var original = project.Plans.Single(x => x.Id == scan.PlanId);
        var candidate = CopyPlan(original, original.Name + " · 扫描候选");
        for (var i = 0; i < candidate.Gears.Count; i++)
        {
            candidate.Gears[i].Rpm = scan.Choices[i].CandidateRpm;
        }
        candidate.Notes = "离散扫描候选；目标=接近组合数+偏移权重×归一化偏移；风量及响应待验证";
        project.Plans.Add(candidate);
        return candidate;
    }

    public static int GenerateChecks(Project project, AnalysisResult result, string planId)
    {
        if (result.InputHash != ProjectCodec.InputHash(project))
        {
            throw new InvalidOperationException("计算已过期；先重新计算");
        }
        var count = 0;
        foreach (var contact in result.Contacts.Where(x => x.PlanId == planId && x.Near))
        {
            if (project.Checks.Any(x => x.Key == contact.Key))
            {
                continue;
            }
            project.Checks.Add(new()
            {
                Key = contact.Key,
                PlanId = contact.PlanId,
                GearId = contact.GearId,
                ModeId = contact.ModeId,
                OrderId = contact.OrderId,
                Title = $"{contact.Gear} / {contact.Order} / {contact.Mode}",
                Method = $"在{contact.Rpm:0.##}rpm附近复核转速与阶次；测量壳体/风道响应和车内声音；记录温度、安装、测点、仪器、频率分辨率及相干性。候选频率{contact.FrequencyHz:0.##}Hz。核对风量与效率。"
            });
            count++;
        }
        return count;
    }

    public static void MarkChecksForRecheck(Project project)
    {
        foreach (var task in project.Checks.Where(x => x.State == CheckState.Completed))
        {
            task.State = CheckState.Recheck;
            task.Updated = DateTimeOffset.UtcNow;
        }
    }
}

using Blower.Core.Domain;
using Blower.Core.IO;

namespace Blower.Core.Operations;

public sealed class ProjectSession
{
    private readonly Stack<Project> undo = new();
    private readonly Stack<Project> redo = new();
    public Project Current { get; private set; } = new();
    public string SavedHash { get; private set; } = "";
    public string StoreToken { get; private set; } = "";
    public bool CanUndo => undo.Count > 0 && !Current.Archived;
    public bool CanRedo => redo.Count > 0 && !Current.Archived;
    public bool Dirty => SavedHash != ProjectCodec.Hash(System.Text.Json.JsonSerializer.Serialize(Current, ProjectCodec.Options));

    public void Load(Project project, string storeToken = "", bool saved = false)
    {
        Validation.RequireValid(project);
        Current = ProjectCodec.Clone(project);
        undo.Clear();
        redo.Clear();
        StoreToken = storeToken;
        SavedHash = saved ? ProjectCodec.Hash(ProjectCodec.Encode(Current)) : "";
    }

    public void Change(string action, Action<Project> change, bool analysisChange = true, bool allowArchived = false)
    {
        if (Current.Archived && !allowArchived)
        {
            throw new InvalidOperationException("项目已归档；先解除归档");
        }
        var candidate = ProjectCodec.Clone(Current);
        change(candidate);
        if (analysisChange)
        {
            ProjectCommands.MarkChecksForRecheck(candidate);
        }
        candidate.Revision++;
        candidate.Modified = DateTimeOffset.UtcNow;
        candidate.Audit.Add(new()
        {
            Action = action,
            Detail = "已通过完整项目校验" + (analysisChange ? "；计算输入变化，已完成验证标为需复核" : ""),
            Revision = candidate.Revision
        });
        Validation.RequireValid(candidate);
        if (undo.Count >= 30)
        {
            var recent = undo.Take(29).Reverse().ToList();
            undo.Clear();
            foreach (var state in recent)
            {
                undo.Push(state);
            }
        }
        undo.Push(ProjectCodec.Clone(Current));
        redo.Clear();
        Current = candidate;
    }

    public void Undo()
    {
        if (!CanUndo)
        {
            throw new InvalidOperationException("没有可撤销操作或项目已归档");
        }
        redo.Push(ProjectCodec.Clone(Current));
        Restore(undo.Pop(), "撤销");
    }

    public void Redo()
    {
        if (!CanRedo)
        {
            throw new InvalidOperationException("没有可重做操作或项目已归档");
        }
        undo.Push(ProjectCodec.Clone(Current));
        Restore(redo.Pop(), "重做");
    }

    private void Restore(Project restored, string action)
    {
        var audit = Current.Audit.ToList();
        var revision = Current.Revision + 1;
        restored.Revision = revision;
        restored.Audit = audit;
        restored.Modified = DateTimeOffset.UtcNow;
        restored.Audit.Add(new() { Action = action, Detail = "恢复输入快照；保留累计审计", Revision = revision });
        ProjectCommands.MarkChecksForRecheck(restored);
        Current = restored;
    }

    public SavedProject Save(ProjectStore store)
    {
        var result = store.Save(Current, StoreToken);
        StoreToken = result.Token;
        SavedHash = ProjectCodec.Hash(ProjectCodec.Encode(Current));
        return result;
    }
}

using System.Globalization;
using System.Windows;

using Blower.Core.Analysis;

namespace Blower.Desktop.Controls;

public sealed class FrequencyCanvas : FrameworkElement
{
    public static readonly DependencyProperty ModelProperty = DependencyProperty.Register(
        nameof(Model), typeof(ChartModel), typeof(FrequencyCanvas),
        new FrameworkPropertyMetadata(null, FrameworkPropertyMetadataOptions.AffectsRender));
    public static readonly DependencyProperty SourceLabelProperty = DependencyProperty.Register(
        nameof(SourceLabel), typeof(string), typeof(FrequencyCanvas),
        new FrameworkPropertyMetadata("", FrameworkPropertyMetadataOptions.AffectsRender));
    public ChartModel? Model
    {
        get => (ChartModel?)GetValue(ModelProperty);
        set => SetValue(ModelProperty, value);
    }
    public string SourceLabel
    {
        get => (string)GetValue(SourceLabelProperty);
        set => SetValue(SourceLabelProperty, value);
    }
    protected override void OnRender(DrawingContext dc)
    {
        base.OnRender(dc);
        dc.DrawRectangle(Brushes.White, null, new Rect(RenderSize));
        var chart = Model;
        if (chart == null)
        {
            DrawText(dc, "尚无有效结果。填写输入后按 F5 计算。", 30, 40, 15, Brushes.SlateGray);
            return;
        }
        if (ActualWidth < 150 || ActualHeight < 150)
        {
            return;
        }
        var sx = ActualWidth / ChartModel.Width;
        var sy = ActualHeight / (ChartModel.Height + 60);
        double X(double x) => x * sx;
        double Y(double y) => y * sy;
        var gridPen = new Pen(new SolidColorBrush(Color.FromRgb(220, 230, 230)), 1);
        foreach (var tick in chart.XTicks)
        {
            dc.DrawLine(gridPen, new(X(tick.Position), Y(ChartModel.Top)), new(X(tick.Position), Y(ChartModel.Bottom)));
            DrawText(dc, tick.Label, X(tick.Position) - 16, Y(ChartModel.Bottom) + 8, 11, Brushes.DimGray);
        }
        foreach (var tick in chart.YTicks)
        {
            dc.DrawLine(gridPen, new(X(ChartModel.Left), Y(tick.Position)), new(X(ChartModel.Right), Y(tick.Position)));
            DrawText(dc, tick.Label, 5, Y(tick.Position) - 8, 11, Brushes.DimGray);
        }
        foreach (var band in chart.Bands)
        {
            var brush = new SolidColorBrush(Color.FromArgb(55, 125, 100, 174));
            dc.DrawRectangle(brush, null, new Rect(X(ChartModel.Left), Y(band.Y),
                X(ChartModel.Right - ChartModel.Left), Math.Max(2, Y(band.Height))));
            DrawText(dc, band.Name, X(ChartModel.Left) + 6, Y(band.LabelY) - 12, 11, Brushes.DarkSlateBlue);
        }
        foreach (var line in chart.Lines)
        {
            var brush = (Brush)new BrushConverter().ConvertFromString(line.Color)!;
            dc.DrawLine(new Pen(brush, 2), new(X(line.X1), Y(line.Y1)), new(X(line.X2), Y(line.Y2)));
        }
        foreach (var gear in chart.Gears)
        {
            var brush = gear.Near ? Brushes.Firebrick : Brushes.Teal;
            var pen = new Pen(brush, 1) { DashStyle = DashStyles.Dash };
            dc.DrawLine(pen, new(X(gear.X), Y(ChartModel.Top)), new(X(gear.X), Y(ChartModel.Bottom)));
            DrawText(dc, gear.Name, X(gear.X) - 15, 5, 11, brush);
        }
        DrawText(dc, "频率 Hz", 5, 5, 12, Brushes.DarkSlateGray);
        DrawText(dc, "转速 rpm →", X(ChartModel.Left), Y(390), 12, Brushes.DarkSlateGray);
        DrawText(dc, SourceLabel + " · 紫色：模态区间；红虚线：存在接近；绿虚线：未接近", X(ChartModel.Left), Y(413), 11, Brushes.DarkSlateGray);
        var legendX = X(ChartModel.Left);
        var legendY = Y(438);
        foreach (var line in chart.Lines)
        {
            var brush = (Brush)new BrushConverter().ConvertFromString(line.Color)!;
            var text = DrawText(dc, line.Name, legendX, legendY, 11, brush);
            legendX += text.Width + 22;
            if (legendX > ActualWidth - 170)
            {
                legendX = X(ChartModel.Left);
                legendY += 16;
            }
        }
    }
    private FormattedText DrawText(DrawingContext dc, string text, double x, double y, double size, Brush brush)
    {
        var formatted = new FormattedText(text, CultureInfo.GetCultureInfo("zh-CN"), FlowDirection.LeftToRight,
            new Typeface("Microsoft YaHei UI"), size, brush, VisualTreeHelper.GetDpi(this).PixelsPerDip);
        dc.DrawText(formatted, new Point(x, y));
        return formatted;
    }
}

using System.IO;
using System.Text;
using Blower.Core.Domain;
using Blower.Core.IO;

namespace Blower.Desktop.Services;

public sealed record DiskProject(Project Project, string Path, string Token);

public static class DesktopFiles
{
    public static DiskProject Open(string path)
    {
        var info = new FileInfo(path);
        if (info.Length > ProjectCodec.MaximumBytes)
        {
            throw new InvalidDataException("文件超过 4MiB 上限");
        }
        var content = File.ReadAllText(path, Encoding.UTF8);
        var project = ProjectCodec.Decode(content);
        return new(project, Path.GetFullPath(path), ProjectCodec.Hash(content));
    }

    public static string Save(Project project, string path, string? expectedToken)
    {
        var content = ProjectCodec.Encode(project);
        if (Encoding.UTF8.GetByteCount(content) > ProjectCodec.MaximumBytes)
        {
            throw new InvalidDataException("序列化项目超过 4MiB，请拆分项目或精简备注");
        }
        var current = File.Exists(path)
            ? ProjectCodec.Hash(File.ReadAllText(path, Encoding.UTF8)) : "";
        if (expectedToken != null && expectedToken != current)
        {
            throw new IOException("磁盘文件已被其他程序修改或删除。请另存为新文件，再打开原文件核对差异。");
        }
        WriteAtomic(path, content, false, true);
        var reopened = Open(path);
        if (ProjectCodec.InputHash(reopened.Project) != ProjectCodec.InputHash(project))
        {
            throw new IOException("保存后重开核对失败，请保留当前会话并另存");
        }
        return reopened.Token;
    }

    public static void WriteAtomic(string path, string content, bool bom, bool backup = false)
    {
        var full = Path.GetFullPath(path);
        var temporary = full + "." + Guid.NewGuid().ToString("N") + ".tmp";
        try
        {
            using (var stream = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None))
            {
                if (bom)
                {
                    stream.Write(Encoding.UTF8.GetPreamble());
                }
                stream.Write(Encoding.UTF8.GetBytes(content));
                stream.Flush(true);
            }
            if (backup && File.Exists(full))
            {
                File.Copy(full, full + ".bak", true);
            }
            File.Move(temporary, full, true);
        }
        finally
        {
            if (File.Exists(temporary))
            {
                File.Delete(temporary);
            }
        }
    }

    public static string ReadCsv(string path)
    {
        if (new FileInfo(path).Length > 1024 * 1024)
        {
            throw new InvalidDataException("CSV 超过 1MiB 预检上限");
        }
        return File.ReadAllText(path, new UTF8Encoding(false, true));
    }
}

using System.ComponentModel;
using System.Runtime.CompilerServices;
using System.Windows.Input;

namespace Blower.Desktop.ViewModels;

public abstract class ObservableObject : INotifyPropertyChanged
{
    public event PropertyChangedEventHandler? PropertyChanged;

    protected void Raise([CallerMemberName] string? property = null)
    {
        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(property));
    }

    protected bool Set<T>(ref T storage, T value, [CallerMemberName] string? property = null)
    {
        if (EqualityComparer<T>.Default.Equals(storage, value))
        {
            return false;
        }
        storage = value;
        Raise(property);
        return true;
    }
}

public sealed class RelayCommand(Action execute, Func<bool>? available = null) : ICommand
{
    public event EventHandler? CanExecuteChanged;

    public bool CanExecute(object? parameter)
    {
        return available?.Invoke() ?? true;
    }

    public void Execute(object? parameter)
    {
        execute();
    }

    public void Refresh()
    {
        CanExecuteChanged?.Invoke(this, EventArgs.Empty);
    }
}

        {

            if (hash != ProjectCodec.InputHash(Project))

            Raise(nameof(ComparisonStatus));

        catch (Exception e) when (e is InvalidOperationException or IOException or ArgumentException)
        {
            Message(title + "失败：" + e.Message, true);

        finally
        {
            busy = false;

            RefreshCommands();

    private void Add()
    {
        if (!LeaveEditor())
        {
            return;
        }
        switch (section)
        {
            case "orders":
                Change("新增阶次", p =>
                {
                    var used = p.Orders.Where(x => x.Enabled).Select(x => x.EffectiveOrder(p.Settings.Blades)).ToHashSet();
                    var q = Enumerable.Range(1, 100).FirstOrDefault(x => !used.Contains(x));
                    p.Orders.Add(new() { Multiplier = q == 0 ? 1 : q, Enabled = q != 0, Name = "轴阶次 " + q });
                });
                break;
            case "modes":
                Change("新增模态", p => p.Modes.Add(new() { Origin = p.Origin }));
                break;
            case "plans":
                if (ActivePlan == null || dialogs.Confirm("是：新增独立方案；否：向当前方案增加挡位。"))
                {
                    Change("新增方案", p => p.Plans.Add(new() { Name = "方案 " + (p.Plans.Count + 1) }));
                }
                else
                {
                    var id = ActivePlan.Id;
                    Change("新增挡位", p =>
                    {
                        var plan = p.Plans.Single(x => x.Id == id);
                        if (plan.Locked)
                        {
                            throw new InvalidOperationException("方案已锁定");
                        }
                        var rpm = plan.Gears.Count == 0 ? p.Settings.MinRpm : plan.Gears[^1].Rpm + p.Settings.MinimumGapRpm;
                        plan.Gears.Add(new() { Name = "挡位 " + (plan.Gears.Count + 1), Rpm = rpm });
                    });
                }
                break;
            default:
                Message("请在阶次、模态或挡位方案中新增对象。", true);
                break;
        }
    }

    private void Delete()
    {
        if (!LeaveEditor())
        {
            return;
        }
        var kind = selectedRow switch
        {
            Excitation => "order",
            Mode => "mode",
            Gear => "gear",
            VerificationTask => "check",
            _ => section == "plans" && ActivePlan != null ? "plan" : ""
        };
        var id = selectedRow switch
        {
            Excitation x => x.Id,
            Mode x => x.Id,
            Gear x => x.Id,
            VerificationTask x => x.Id,
            _ => ActivePlan?.Id ?? ""
        };
        if (kind.Length == 0)
        {
            Message("先选中需要删除的对象；删除方案时先取消挡位选择。", true);
            return;
        }
        if (dialogs.Confirm("删除所选对象？关联验证记录存在时会阻止删除，可撤销此操作。"))
        {
            Change("删除对象", p => ProjectCommands.RemoveObject(p, kind, id), kind != "check");
        }
    }

    private void CopyPlan()
    {
        if (LeaveEditor() && ActivePlan != null)
        {

            Change("复制方案", p =>
            {
                var source = p.Plans.Single(x => x.Id == id);
                p.Plans.Add(ProjectCommands.CopyPlan(source, source.Name + " · 副本"));
            });
        }
    }

f=q×n/60，n为rpm，q为有效阶次，f为Hz。精确交叉转速=60×模态频率/q。
B=max(绝对半宽,fn×相对百分比/100,启用阻尼时ζ×fn,附加半宽)。阻尼项为轻阻尼孤立模态近似，不用于估算幅值。
模态区间[max(0,fn−B),fn+B]；激励区间[max(0,q×(n−T)/60),q×(n+T)/60]。T为转速容差。
净距C=max(模态下界−激励上界,激励下界−模态上界)。C≤1e−8Hz表示潜在接近；负值为区间重叠，不是风险概率。
模拟核对：40叶片、1倍叶片频率、900rpm→600Hz。600Hz模态、阻尼0.02、默认5Hz及3%下半宽18Hz；敏感转速853～947rpm。947rpm触边接近，947.01rpm在外。
扫描局部代价=接近组合数+权重×abs(候选−原rpm)/max(1,窗口rpm)。动态规划满足最小间距；固定挡只有原点，不能保证连续最优。

输入与CSV
叶片1～200；转速0～50000rpm；倍数0.1～100；模态0.1～50000Hz；阻尼0～0.1或空；半宽0～5000Hz、相对0～30%；容差0～1000rpm；间距1～5000rpm；步长1～500rpm；窗口0～3000rpm；权重0～1。
最多24阶次、100模态、20方案、每方案24挡位，每挡最多603扫描候选。
CSV使用UTF-8、英文逗号及小数点。gears表头name,rpm,fixed，fixed为true/false；modes表头name,frequency_hz,damping_ratio,source，未知阻尼留空。先读取、预检，再整批应用。挡位导入新增方案，模态导入追加。行号0表示完整项目约束错误。
来源：Manual=人工设计；Simulation=模拟；DeclaredMeasurement=用户声明实测（未认证）。验证：Pending=待验证；Scheduled=已安排；Completed=已记录证据；Recheck=需复核。

错误恢复
属性错误：按提示修正对应字段再应用，或还原草稿。间距错误：检查全部方案相邻转速和最小间距。无结果：确认启用阶次、模态和含挡位方案并计算；关闭仅显示潜在接近可查看范围外组合。
扫描无解：核对窗口、步长、固定挡和间距。文件冲突：另存为新文件保留当前会话，再打开原文件核对。损坏文件可尝试同目录.bak。删除被阻止：先处理关联验证任务。归档项目先解除归档。

验证范围
本交付在Linux上进行Windows目标编译及跨平台核心测试。WPF实际启动、窗口缩放、绑定、文件对话框和真实截图须在Windows执行；不能用图形导出冒充桌面截图。工程效果须经试验复核。
""";
}

using System.IO;
using Blower.Core.Domain;
using Blower.Core.IO;
using Blower.Desktop.Services;

namespace Blower.Desktop.ViewModels;

public sealed partial class WorkspaceViewModel
{
    private ProjectStore? library;
    private string libraryToken = "";
    private string libraryId = "";
    public string LibraryPath => Path.Combine(
        Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
        "BlowerResonance", "Projects");
    private ProjectStore Library => library ??= new ProjectStore(LibraryPath);

    private void SaveLibrary()
    {
        if (!LeaveEditor())
        {
            return;
        }
        var saved = Library.Save(Project, libraryId == Project.Id ? libraryToken : "");
        libraryToken = saved.Token;
        libraryId = Project.Id;
        filePath = saved.Path;
        diskToken = saved.Token;
        savedContent = ProjectCodec.Encode(Project);
        Rebuild();
        Message("项目库保存成功：" + saved.Path);
    }

    private void OpenLibrary(bool backup)
    {
        if (SelectedRow is not StoredProject selected)
        {
            throw new InvalidOperationException("先在本地项目库选中项目");
        }
        if (!CanClose())
        {
            return;
        }
        var opened = Library.Open(selected.Id, backup);
        if (backup)
        {
            Load(opened.Project);
            libraryId = selected.Id;
            libraryToken = opened.Token;
            Message("已从真实备份恢复到未保存草稿；原项目仍在磁盘，核对后另存或保存到项目库。");
        }
        else
        {
            Load(opened.Project, opened.Path, opened.Token, true);
            libraryId = selected.Id;
            libraryToken = opened.Token;
            Message("已打开本地项目库项目：" + opened.Path);
        }
    }

    private void AddPlan()
    {
        if (!LeaveEditor())
        {
            return;
        }
        var newId = Software.Id();
        Change("新增方案", p => p.Plans.Add(new() { Id = newId, Name = "方案 " + (p.Plans.Count + 1) }));
        ActivePlan = Project.Plans.Single(x => x.Id == newId);
    }

    private void AddGear()
    {
        if (!LeaveEditor())
        {
            return;
        }
        var id = ActivePlan?.Id ?? throw new InvalidOperationException("先新增或选择方案");
        Change("新增挡位", p =>
        {
            var plan = p.Plans.Single(x => x.Id == id);
            if (plan.Locked)
            {
                throw new InvalidOperationException("方案已锁定，请在方案属性解除锁定");
            }
            var rpm = plan.Gears.Count == 0 ? p.Settings.MinRpm : plan.Gears[^1].Rpm + p.Settings.MinimumGapRpm;
            plan.Gears.Add(new() { Name = "挡位 " + (plan.Gears.Count + 1), Rpm = rpm });
        });
    }
}

    {
        try
        {
            action();
        }
        catch (Exception e) when (e is InvalidDataException or InvalidOperationException
            or IOException or UnauthorizedAccessException or ArgumentException)

            Message(e.Message, true);
        }
