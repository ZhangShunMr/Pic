0001  @inherits VmView
0002  <section class="page-heading"><div><span class="eyebrow">MODEL & LIMITS</span><h2>模型、参数与结果解释</h2><p>先核实输入与简化假设，再将筛查结果用于实物验证计划。</p></div></section>
0003  <div class="two-column">
0004  <section class="card"><h3>法向接触与预紧</h3>
0005  <p>饰板作为刚性平面，具有法向平移和两个小转角自由度。卡扣是双向弹簧，可提供正、负法向反力；三个以上非共线卡扣才能稳定约束三个自由度。</p>
0006  <p class="formula">压缩 c = max(0, 预紧量 − 间隙 − 公差偏移 + 局部位移)<br />接触法向力 N = 法向刚度 × c</p>
0007  <p>预紧量表示装配后的等效干涉压缩，不是外加预紧力。软件将所有接触预紧与卡扣反力一起求平衡。因此预紧0.2mm不一定产生0.2mm的最终压缩。</p>
0008  <h3>切向摩擦与滑移</h3>
0009  <p>法向平衡后求解一个切向平移自由度。卡扣切向力为刚度乘位移，已接触点摩擦力限制为μN。点位的弹性切向需求超过上限时，显示“滑移余量”。</p>
0010  <p class="formula">接触切向力 = min(切向刚度 × |u|, μN)<br />滑移余量 = max(0, |u| − μN / 切向刚度)</p>
0011  <p>这是当前相位的库仑限幅筛查，没有摩擦记忆、粘滑振荡或声学辐射。完全脱离点的摩擦与滑移余量均为0，不等于不存在碰撞问题。</p>
0012  <h3>公差与相位</h3>
0013  <p>每个非零公差点分别枚举正负间隙偏移，并叠加名义工况；接触列表顺序就是角点向量顺序。相位在0至360°间均匀取点，不重复360°。离散包络可能遗漏采样间峰值。</p>
0014  </section>
0015  <section class="card"><h3>可独立核对案例</h3>
0016  <ol>
0017      <li>建立400×300mm矩形，质量1kg，质心(200,150)mm。</li>
0018      <li>在(40,40)、(360,40)、(360,260)、(40,260)mm放四个卡扣，各法向刚度100N/mm、切向刚度50N/mm。</li>
0019      <li>不放接触点。设置法向偏置1g，法向、切向幅值和切向偏置均为0。</li>
0020      <li>计算后各卡扣法向力2.4516625N，质心位移0.024516625mm，四卡扣总反力9.80665N。</li>
0021      <li>再在中心增加法向刚度100N/mm、预紧0.2mm、间隙0、公差0的接触点；载荷全部设0。</li>
0022      <li>重新计算，中心点压缩0.16mm、法向力16N；质心位移−0.04mm，四卡扣各−4N。</li>

0001  using System.Collections.Concurrent;
0002  using System.Text;
0003
0004  namespace TrimContact.App;
0005
0006  public sealed record DownloadFile(byte[] Bytes, string ContentType, string Name, DateTimeOffset Expiry);
0007
0008  public sealed class DownloadVault
0009  {
0010      private readonly ConcurrentDictionary<string, DownloadFile> files = new();
0011
0012      public string Add(string text, string contentType, string name)
0013      {
0014          foreach (var stale in files.Where(p => p.Value.Expiry < DateTimeOffset.UtcNow))
0015              files.TryRemove(stale.Key, out _);
0016          if (files.Count >= 100)
0017              foreach (var old in files.OrderBy(p => p.Value.Expiry).Take(20))
0018                  files.TryRemove(old.Key, out _);
0019          var id = Guid.NewGuid().ToString("N");
0020          var bytes = Encoding.UTF8.GetBytes(text);
0021          if (contentType.StartsWith("text/csv", StringComparison.Ordinal))
0022              bytes = Encoding.UTF8.GetPreamble().Concat(bytes).ToArray();
0023          files[id] = new DownloadFile(bytes, contentType, name, DateTimeOffset.UtcNow.AddMinutes(30));
0024          return "/download/" + id;
0025      }
0026
0027      public DownloadFile? Get(string id) => files.TryGetValue(id, out var file)
0028          && file.Expiry >= DateTimeOffset.UtcNow ? file : null;
0029  }

0033          CancellationToken cancellation = default)
0034      {
0035          var solver = new ContactSolver(project.Panel, variant);
0036          var result = new VariantResult
0037          {
0038              VariantId = variant.Id,
0039              Name = variant.Name,
0040              Contacts = variant.Contacts.Select(c => new PointEnvelope
0041                  { PointId = c.Id, Name = c.Name, Kind = "接触点" }).ToList(),
0042              Clips = variant.Clips.Select(c => new PointEnvelope
0043                  { PointId = c.Id, Name = c.Name, Kind = "卡扣" }).ToList()

0052                  cancellation.ThrowIfCancellationRequested();

0054                  for (int phaseIndex = 0; phaseIndex < project.Settings.Phases; phaseIndex++)
0055                  {
0056                      var degrees = 360.0 * phaseIndex / project.Settings.Phases;
0057                      var radians = degrees * Math.PI / 180;
0058                      var normal = load.NormalBiasG + load.NormalAmplitudeG * Math.Sin(radians);
0059                      var shear = load.TangentialBiasG + load.TangentialAmplitudeG
0060                          * Math.Sin(radians + load.PhaseDegrees * Math.PI / 180);
0061                      var state = solver.Solve(degrees, normal, shear, tolerance);
0062                      var corner = cornerIndex == 0 ? "名义" : $"角点{cornerIndex}[{string.Join(",", tolerance.Select(t => t.ToString("G4", System.Globalization.CultureInfo.InvariantCulture)))}]mm";
0063                      var condition = $"{load.Name} / {corner} / {degrees:F1}°";
0064                      for (int i = 0; i < result.Contacts.Count; i++)
0065                          result.Contacts[i].Accumulate(state.Contacts[i], condition, false);
0066                      for (int i = 0; i < result.Clips.Count; i++)
0067                          result.Clips[i].Accumulate(state.Clips[i], condition, true);
0068                      result.MaxResidualN = Math.Max(result.MaxResidualN, state.ResidualN);
0069                      result.EvaluatedStates++;

0071                          trace.Samples.Add(state);

0080      public static IEnumerable<double[]> ToleranceCases(DesignVariant variant, bool enabled)
0081      {
0082          yield return new double[variant.Contacts.Count];
0083          if (!enabled)
0084              yield break;
0085          var uncertain = variant.Contacts.Select((c, i) => (c, i)).Where(p => p.c.ToleranceMm > 0).ToArray();
0086          if (uncertain.Length == 0)
0087              yield break;
0088          for (int mask = 0; mask < (1 << uncertain.Length); mask++)
0089          {
0090              var values = new double[variant.Contacts.Count];
0091              for (int bit = 0; bit < uncertain.Length; bit++)
0092                  values[uncertain[bit].i] = ((mask & (1 << bit)) == 0 ? -1 : 1) * uncertain[bit].c.ToleranceMm;
0093              yield return values;
0094          }
0095      }

0123                  result.Clips.Max(c => c.MaxUtilization), result.Contacts.Max(c => c.MaxCompressionMm),

0001  using TrimContact.Core.Domain;

0003  namespace TrimContact.Core.Analysis;

0005  public sealed record PointState(Guid Id, string Name, double CompressionMm, double NormalN,
0006      double ShearN, double SlipMm, double Utilization);
0007
0008  public sealed record PhaseState(double PhaseDegrees, double NormalForceN, double TangentialForceN,
0009      Vector3 Displacement, double TangentialMm, double ResidualN,
0010      IReadOnlyList<PointState> Contacts, IReadOnlyList<PointState> Clips);
0011
0012  public sealed class ContactSolver
0013  {
0014      private readonly DesignVariant variant;
0015      private readonly Panel panel;
0016      private readonly Matrix3 clipMatrix = new();
0017      private readonly Vector3[] clipBasis;
0018      private readonly Vector3[] contactBasis;
0019
0020      public ContactSolver(Panel panel, DesignVariant variant)
0021      {
0022          this.panel = panel;
0023          this.variant = variant;
0024          var scale = Geometry.BoundsOf(panel.Outline).Scale;
0025          Vector3 Basis(double x, double y) => new(1, (x - panel.CenterXmm) / scale, (y - panel.CenterYmm) / scale);
0026          clipBasis = variant.Clips.Select(c => Basis(c.Xmm, c.Ymm)).ToArray();
0027          contactBasis = variant.Contacts.Select(c => Basis(c.Xmm, c.Ymm)).ToArray();
0028          for (int i = 0; i < variant.Clips.Count; i++)
0029              clipMatrix.AddSpring(variant.Clips[i].NormalStiffness, clipBasis[i]);
0030          clipMatrix.Solve(new Vector3(1, 0, 0));
0031      }
0032
0033      private double Energy(Vector3 q, Vector3 force, IReadOnlyList<double> offsets)
0034      {
0035          var value = 0.5 * q.Dot(clipMatrix.Multiply(q)) - force.Dot(q);
0036          for (int i = 0; i < variant.Contacts.Count; i++)
0037          {
0038              var compression = Math.Max(0, offsets[i] + contactBasis[i].Dot(q));
0039              value += 0.5 * variant.Contacts[i].NormalStiffness * compression * compression;
0040          }
0041          return value;
0042      }
0043
0044      private (Vector3 Q, double Residual) SolveNormal(double forceN, IReadOnlyList<double> tolerance)
0045      {
0046          var force = new Vector3(forceN, 0, 0);
0047          var offsets = variant.Contacts.Select((c, i) => c.PreloadMm - c.GapMm - tolerance[i]).ToArray();
0048          var q = clipMatrix.Solve(force);
0049          var threshold = 1e-8 * Math.Max(1, Math.Abs(forceN));
0050          for (int iteration = 0; iteration < 100; iteration++)
0051          {
0052              var gradient = clipMatrix.Multiply(q) - force;
0053              var hessian = clipMatrix.Copy();
0054              for (int i = 0; i < offsets.Length; i++)
0055              {
0056                  var compression = offsets[i] + contactBasis[i].Dot(q);
0057                  if (compression <= 0)
0058                      continue;
0059                  var stiffness = variant.Contacts[i].NormalStiffness;
0060                  gradient += stiffness * compression * contactBasis[i];
0061                  hessian.AddSpring(stiffness, contactBasis[i]);
0062              }
0063              if (gradient.Norm <= threshold)
0064                  return (q, gradient.Norm);
0065              var step = hessian.Solve(-1 * gradient);
0066              var oldEnergy = Energy(q, force, offsets);
0067              var slope = gradient.Dot(step);
0068              var alpha = 1.0;
0069              while (alpha > 1e-10)
0070              {
0071                  var candidate = q + alpha * step;
0072                  if (Energy(candidate, force, offsets) <= oldEnergy + 1e-4 * alpha * slope + 1e-12)
0073                  {
0074                      q = candidate;
0075                      break;
0076                  }
0077                  alpha /= 2;
0078              }
0079              if (alpha <= 1e-10)
0080                  throw new InvalidDataException("法向求解线搜索失败，请检查过大的刚度差和预紧量");
0081          }
0082          throw new InvalidDataException("法向求解100次未收敛，结果取消");
0083      }
0084
0085      public PhaseState Solve(double phaseDegrees, double normalG, double tangentialG, IReadOnlyList<double> tolerance)
0086      {
0087          if (tolerance.Count != variant.Contacts.Count)
0088              throw new ArgumentException("公差维度与接触点不一致");
0089          var normalForce = panel.MassKg * Product.Gravity * normalG;
0090          var shearForce = panel.MassKg * Product.Gravity * tangentialG;
0091          var (q, normalResidual) = SolveNormal(normalForce, tolerance);
0092          var compression = variant.Contacts.Select((c, i) => Math.Max(0,
0093              c.PreloadMm - c.GapMm - tolerance[i] + contactBasis[i].Dot(q))).ToArray();
0094          var normal = variant.Contacts.Select((c, i) => c.NormalStiffness * compression[i]).ToArray();
0095          var clipKt = variant.Clips.Sum(c => c.TangentialStiffness);
0096          double ShearAt(double u)
0097          {
0098              double force = clipKt * u;
0099              for (int i = 0; i < variant.Contacts.Count; i++)
0100                  force += Math.Min(variant.Contacts[i].TangentialStiffness * u, variant.Contacts[i].Friction * normal[i]);
0101              return force;
0102          }
0103          double low = 0;
0104          var high = Math.Abs(shearForce) / clipKt;
0105          for (int iteration = 0; iteration < 70; iteration++)
0106          {
0107              var middle = (low + high) / 2;
0108              if (ShearAt(middle) < Math.Abs(shearForce))
0109                  low = middle;
0110              else
0111                  high = middle;
0112          }
0113          var magnitude = (low + high) / 2;
0114          var sign = Math.Sign(shearForce);
0115          var contacts = new List<PointState>();
0116          for (int i = 0; i < variant.Contacts.Count; i++)
0117          {
0118              var c = variant.Contacts[i];
0119              var cap = c.Friction * normal[i];
0120              var shear = Math.Min(c.TangentialStiffness * magnitude, cap);
0121              var slip = compression[i] > Product.ContactEpsilon
0122                  ? Math.Max(0, magnitude - cap / c.TangentialStiffness) : 0;
0123              contacts.Add(new(c.Id, c.Name, compression[i], normal[i], sign * shear,
0124                  slip, compression[i] / c.CompressionLimitMm));
0125          }
0126          var clips = variant.Clips.Select((c, i) =>
0127          {
0128              var n = c.NormalStiffness * clipBasis[i].Dot(q);
0129              var shear = c.TangentialStiffness * magnitude;
0130              return new PointState(c.Id, c.Name, 0, n, sign * shear, 0,
0131                  Math.Sqrt(n * n + shear * shear) / c.ForceLimitN);
0132          }).ToArray();
0133          var residual = Math.Max(normalResidual, Math.Abs(ShearAt(magnitude) - Math.Abs(shearForce)));
0134          return new PhaseState(phaseDegrees, normalForce, shearForce, q, sign * magnitude,
0135              residual, contacts, clips);
0136      }
0137  }

0001  namespace TrimContact.Core.Analysis;
0002
0003  public readonly record struct Vector3(double Z, double X, double Y)
0004  {
0005      public static Vector3 operator +(Vector3 a, Vector3 b) => new(a.Z + b.Z, a.X + b.X, a.Y + b.Y);
0006      public static Vector3 operator -(Vector3 a, Vector3 b) => new(a.Z - b.Z, a.X - b.X, a.Y - b.Y);
0007      public static Vector3 operator *(double s, Vector3 v) => new(s * v.Z, s * v.X, s * v.Y);
0008      public double Dot(Vector3 v) => Z * v.Z + X * v.X + Y * v.Y;
0009      public double Norm => Math.Sqrt(Dot(this));
0010  }
0011
0012  public sealed class Matrix3
0013  {
0014      private readonly double[,] values = new double[3, 3];
0015
0016      public void AddSpring(double stiffness, Vector3 b)
0017      {
0018          double[] v = [b.Z, b.X, b.Y];
0019          for (int i = 0; i < 3; i++)
0020              for (int j = 0; j < 3; j++)
0021                  values[i, j] += stiffness * v[i] * v[j];
0022      }
0023
0024      public Matrix3 Copy()
0025      {
0026          var copy = new Matrix3();
0027          Array.Copy(values, copy.values, values.Length);
0028          return copy;
0029      }
0030
0031      public Vector3 Multiply(Vector3 q) => new(
0032          values[0, 0] * q.Z + values[0, 1] * q.X + values[0, 2] * q.Y,
0033          values[1, 0] * q.Z + values[1, 1] * q.X + values[1, 2] * q.Y,
0034          values[2, 0] * q.Z + values[2, 1] * q.X + values[2, 2] * q.Y);
0035
0036      public Vector3 Solve(Vector3 rhs)
0037      {
0038          var a = new double[3, 4];
0039          double[] b = [rhs.Z, rhs.X, rhs.Y];
0040          var scale = 0.0;
0041          for (int i = 0; i < 3; i++)
0042          {
0043              for (int j = 0; j < 3; j++)
0044              {
0045                  a[i, j] = values[i, j];
0046                  scale = Math.Max(scale, Math.Abs(a[i, j]));
0047              }
0048              a[i, 3] = b[i];
0049          }
0050          for (int col = 0; col < 3; col++)
0051          {
0052              int pivot = col;
0053              for (int row = col + 1; row < 3; row++)
0054                  if (Math.Abs(a[row, col]) > Math.Abs(a[pivot, col]))
0055                      pivot = row;
0056              if (Math.Abs(a[pivot, col]) < Math.Max(1e-12, scale * 1e-10))
0057                  throw new InvalidDataException("卡扣布置或刚度导致矩阵近奇异，请分散卡扣并检查刚度比");
0058              for (int j = col; j < 4; j++)
0059                  (a[col, j], a[pivot, j]) = (a[pivot, j], a[col, j]);
0060              var diagonal = a[col, col];
0061              for (int j = col; j < 4; j++)
0062                  a[col, j] /= diagonal;
0063              for (int row = 0; row < 3; row++)
0064              {
0065                  if (row == col)
0066                      continue;
0067                  var factor = a[row, col];
0068                  for (int j = col; j < 4; j++)
0069                      a[row, j] -= factor * a[col, j];
0070              }
0071          }
0072          var result = new Vector3(a[0, 3], a[1, 3], a[2, 3]);
0073          if (!double.IsFinite(result.Norm))
0074              throw new InvalidDataException("求解结果非有限数值");
0075          return result;
0076      }
0077  }

0003  public readonly record struct Bounds(double MinX, double MinY, double MaxX, double MaxY)
0004  {
0005      public double Width => MaxX - MinX;
0006      public double Height => MaxY - MinY;

0010  public static class Geometry
0011  {
0012      public static Bounds BoundsOf(IReadOnlyList<Vertex> points)
0013      {
0014          if (points.Count == 0)
0015              return new Bounds(0, 0, 400, 300);
0016          return new Bounds(points.Min(p => p.Xmm), points.Min(p => p.Ymm),
0017              points.Max(p => p.Xmm), points.Max(p => p.Ymm));
0018      }
0019
0020      public static double SignedArea(IReadOnlyList<Vertex> points)
0021      {
0022          double area = 0;
0023          for (int i = 0; i < points.Count; i++)
0024          {
0025              var a = points[i];
0026              var b = points[(i + 1) % points.Count];
0027              area += a.Xmm * b.Ymm - b.Xmm * a.Ymm;
0028          }
0029          return area / 2;
0030      }
0031
0032      private static double Cross(double ax, double ay, double bx, double by) => ax * by - ay * bx;
0033
0034      private static bool OnSegment(Vertex a, Vertex b, double x, double y)
0035      {
0036          const double eps = 1e-8;
0037          return Math.Abs(Cross(b.Xmm - a.Xmm, b.Ymm - a.Ymm, x - a.Xmm, y - a.Ymm)) < eps
0038              && x >= Math.Min(a.Xmm, b.Xmm) - eps && x <= Math.Max(a.Xmm, b.Xmm) + eps
0039              && y >= Math.Min(a.Ymm, b.Ymm) - eps && y <= Math.Max(a.Ymm, b.Ymm) + eps;
0040      }
0041
0042      public static bool Contains(IReadOnlyList<Vertex> points, double x, double y)
0043      {
0044          bool inside = false;
0045          for (int i = 0, j = points.Count - 1; i < points.Count; j = i++)
0046          {
0047              var a = points[j];
0048              var b = points[i];
0049              if (OnSegment(a, b, x, y))
0050                  return true;
0051              if ((a.Ymm > y) != (b.Ymm > y)
0052                  && x < (b.Xmm - a.Xmm) * (y - a.Ymm) / (b.Ymm - a.Ymm) + a.Xmm)
0053                  inside = !inside;
0054          }
0055          return inside;
0056      }
0057
0058      private static bool Intersects(Vertex a, Vertex b, Vertex c, Vertex d)
0059      {
0060          var abC = Cross(b.Xmm - a.Xmm, b.Ymm - a.Ymm, c.Xmm - a.Xmm, c.Ymm - a.Ymm);
0061          var abD = Cross(b.Xmm - a.Xmm, b.Ymm - a.Ymm, d.Xmm - a.Xmm, d.Ymm - a.Ymm);
0062          var cdA = Cross(d.Xmm - c.Xmm, d.Ymm - c.Ymm, a.Xmm - c.Xmm, a.Ymm - c.Ymm);
0063          var cdB = Cross(d.Xmm - c.Xmm, d.Ymm - c.Ymm, b.Xmm - c.Xmm, b.Ymm - c.Ymm);
0064          if (abC * abD < 0 && cdA * cdB < 0)
0065              return true;
0066          return OnSegment(a, b, c.Xmm, c.Ymm) || OnSegment(a, b, d.Xmm, d.Ymm)
0067              || OnSegment(c, d, a.Xmm, a.Ymm) || OnSegment(c, d, b.Xmm, b.Ymm);
0068      }

0070      public static bool IsSimple(IReadOnlyList<Vertex> points)
0071      {
0072          if (points.Count < 3 || Math.Abs(SignedArea(points)) < 1)
0073              return false;
0074          for (int i = 0; i < points.Count; i++)
0075          {
0076              var next = (i + 1) % points.Count;
0077              if (Math.Abs(points[i].Xmm - points[next].Xmm) + Math.Abs(points[i].Ymm - points[next].Ymm) < 1e-6)
0078                  return false;
0079              for (int j = i + 1; j < points.Count; j++)
0080              {
0081                  if (j == next || (j + 1) % points.Count == i)
0082                      continue;
0083                  if (Intersects(points[i], points[next], points[j], points[(j + 1) % points.Count]))
0084                      return false;
0085              }
0086          }
0087          return true;
0088      }
0089
0090      public static double TriangleSpread(IReadOnlyList<Clip> clips, double scale)
0091      {
0092          double max = 0;
0093          for (int i = 0; i < clips.Count; i++)
0094              for (int j = i + 1; j < clips.Count; j++)
0095                  for (int k = j + 1; k < clips.Count; k++)
0096                  {
0097                      var a = clips[i];
0098                      var b = clips[j];
0099                      var c = clips[k];
0100                      max = Math.Max(max, Math.Abs(Cross(b.Xmm - a.Xmm, b.Ymm - a.Ymm,
0101                          c.Xmm - a.Xmm, c.Ymm - a.Ymm)) / (scale * scale));
0102                  }
0103          return max;
0104      }

0009      public static IReadOnlyList<CsvRecord> Read(string text)
0010      {
0011          if (Encoding.UTF8.GetByteCount(text) > 128 * 1024)
0012              throw new InvalidDataException("CSV最多128KB");
0013          text = text.TrimStart('\uFEFF');
0014          var rows = new List<CsvRecord>();
0015          var cells = new List<string>();
0016          var field = new StringBuilder();
0017          bool quoted = false;
0018          bool closedQuote = false;
0019          bool fieldStarted = false;
0020          int line = 1;
0021          int recordLine = 1;
0022          void EndField()
0023          {
0024              cells.Add(field.ToString());
0025              field.Clear();
0026              closedQuote = false;
0027              fieldStarted = false;
0028          }
0029          void EndRecord()
0030          {
0031              EndField();
0032              rows.Add(new CsvRecord(recordLine, cells.ToArray()));
0033              cells.Clear();
0034              if (rows.Count > 101)
0035                  throw new InvalidDataException("CSV最多100条数据记录，加一行表头");
0036              recordLine = line + 1;
0037          }
0038          for (int i = 0; i < text.Length; i++)
0039          {
0040              var c = text[i];
0041              if (quoted)
0042              {
0043                  if (c == '"')
0044                  {
0045                      if (i + 1 < text.Length && text[i + 1] == '"')
0046                      {
0047                          field.Append('"');
0048                          i++;
0049                      }
0050                      else
0051                      {
0052                          quoted = false;
0053                          closedQuote = true;
0054                      }
0055                  }
0056                  else
0057                  {
0058                      field.Append(c);
0059                      if (c == '\n')
0060                          line++;
0061                  }
0062                  continue;
0063              }
0064              if (c == ',')
0065              {
0066                  EndField();
0067                  continue;
0068              }
0069              if (c is '\r' or '\n')
0070              {
0071                  EndRecord();
0072                  if (c == '\r' && i + 1 < text.Length && text[i + 1] == '\n')
0073                      i++;
0074                  line++;
0075                  recordLine = line;
0076                  continue;
0077              }
0078              if (closedQuote)
0079                  throw new InvalidDataException($"CSV第{line}行：闭引号后只能是分隔符或换行");
0080              if (c == '"')
0081              {
0082                  if (fieldStarted)
0083                      throw new InvalidDataException($"CSV第{line}行：未引用字段中含引号");
0084                  quoted = true;
0085                  fieldStarted = true;
0086                  continue;
0087              }
0088              field.Append(c);
0089              fieldStarted = true;
0090          }
0091          if (quoted)
0092              throw new InvalidDataException($"CSV第{recordLine}行：引号未闭合");
0093          if (field.Length > 0 || cells.Count > 0 || closedQuote)
0094              EndRecord();
0095          if (rows.Count < 2)
0096              throw new InvalidDataException("CSV至少包含表头和一条数据");
0097          return rows;
0098      }

0047      private static void RequireMembers(JsonElement element, Type type, string path)
0048      {
0049          if (element.ValueKind == JsonValueKind.Null)
0050          {
0051              if (Nullable.GetUnderlyingType(type) is not null)
0052                  return;
0053              throw new InvalidDataException(path + "不能为null");

0056          {

0061                  RequireMembers(child, type.GenericTypeArguments[0], path + $"[{index++}]");
0062              return;
0063          }
0064          if (type.Namespace != typeof(DesignProject).Namespace || type.IsEnum)
0065              return;
0066          if (element.ValueKind != JsonValueKind.Object)
0067              throw new InvalidDataException(path + "应为对象");
0068          foreach (var property in type.GetProperties(BindingFlags.Instance | BindingFlags.Public))
0069          {
0070              var name = JsonNamingPolicy.CamelCase.ConvertName(property.Name);
0071              if (!element.TryGetProperty(name, out var child))
0072                  throw new InvalidDataException(path + "/" + name + "缺失；不自动填充未知旧版本");
0073              RequireMembers(child, property.PropertyType, path + "/" + name);
0074          }
0075          var names = new HashSet<string>();
0076          foreach (var property in element.EnumerateObject())
0077              if (!names.Add(property.Name))
0078                  throw new InvalidDataException(path + "/" + property.Name + "重复字段");
0079      }

0026      public static BatchPreview Preview(DesignProject project, Guid variantId, BatchLayoutRequest request)
0027      {
0028          if (project.Archived)
0029              throw new InvalidOperationException("归档项目不能批量修改");
0030          if (request.Target is not ("接触点" or "卡扣" or "全部点位"))
0031              throw new InvalidDataException("未知批量修改范围");
0032          double[] values = [request.ShiftXmm, request.ShiftYmm, request.NormalStiffnessFactor,
0033              request.TangentialStiffnessFactor, request.GapDeltaMm, request.PreloadDeltaMm];
0034          if (values.Any(v => !double.IsFinite(v)))
0035              throw new InvalidDataException("批量参数须为有限数值");
0036          if (Math.Abs(request.ShiftXmm) > 1000 || Math.Abs(request.ShiftYmm) > 1000)
0037              throw new InvalidDataException("单次点位平移不超过±1000mm");
0038          if (request.NormalStiffnessFactor is <= 0 or > 100 || request.TangentialStiffnessFactor is <= 0 or > 100)
0039              throw new InvalidDataException("刚度缩放系数须大于0且不超过100");
0040          if (Math.Abs(request.GapDeltaMm) > 20 || Math.Abs(request.PreloadDeltaMm) > 20)
0041              throw new InvalidDataException("单次间隙和预紧增量不超过±20mm");
0042          if (request.Target == "卡扣" && (request.GapDeltaMm != 0 || request.PreloadDeltaMm != 0))
0043              throw new InvalidDataException("卡扣不使用间隙和预紧量，增量须为0");
0044          var clone = ProjectCodec.Clone(project);
0045          var variant = clone.Variants.SingleOrDefault(v => v.Id == variantId)
0046              ?? throw new InvalidDataException("目标方案不存在");
0047          var changes = new List<BatchChange>();
0048          double Record(string name, string field, string unit, double before, double after)
0049          {
0050              if (before != after)
0051                  changes.Add(new(name, field, unit, before, after));
0052              return after;
0053          }
0054          if (request.Target is "卡扣" or "全部点位")
0055          {
0056              foreach (var clip in variant.Clips)
0057              {
0058                  clip.Xmm = Record(clip.Name, "X", "mm", clip.Xmm, clip.Xmm + request.ShiftXmm);
0059                  clip.Ymm = Record(clip.Name, "Y", "mm", clip.Ymm, clip.Ymm + request.ShiftYmm);
0060                  clip.NormalStiffness = Record(clip.Name, "法向刚度", "N/mm", clip.NormalStiffness,
0061                      clip.NormalStiffness * request.NormalStiffnessFactor);
0062                  clip.TangentialStiffness = Record(clip.Name, "切向刚度", "N/mm", clip.TangentialStiffness,
0063                      clip.TangentialStiffness * request.TangentialStiffnessFactor);
0064              }
0065          }
0066          if (request.Target is "接触点" or "全部点位")
0067          {
0068              foreach (var contact in variant.Contacts)
0069              {
0070                  contact.Xmm = Record(contact.Name, "X", "mm", contact.Xmm, contact.Xmm + request.ShiftXmm);
0071                  contact.Ymm = Record(contact.Name, "Y", "mm", contact.Ymm, contact.Ymm + request.ShiftYmm);
0072                  contact.NormalStiffness = Record(contact.Name, "法向刚度", "N/mm", contact.NormalStiffness,
0073                      contact.NormalStiffness * request.NormalStiffnessFactor);
0074                  contact.TangentialStiffness = Record(contact.Name, "切向刚度", "N/mm", contact.TangentialStiffness,
0075                      contact.TangentialStiffness * request.TangentialStiffnessFactor);
0076                  contact.GapMm = Record(contact.Name, "名义间隙", "mm", contact.GapMm,
0077                      contact.GapMm + request.GapDeltaMm);

0079                      contact.PreloadMm + request.PreloadDeltaMm);

0084          ProjectValidation.Require(clone);

0017

0026      {
0027          ProjectValidation.Require(project);
0028          Current = ProjectCodec.Clone(project);

0030          SavedPath = path;

0032          redo.Clear();

0036      public void Edit(string action, Action<DesignProject> change)
0037      {
0038          if (Current.Archived)
0039              throw new InvalidOperationException("归档项目为只读，请复制项目后继续设计");
0040          var next = ProjectCodec.Clone(Current);
0041          change(next);
0042          ProjectValidation.Require(next);
0043          next.ModifiedUtc = DateTimeOffset.UtcNow;
0044          next.Audit.Add(new AuditEntry { Action = action });
0045          if (next.Audit.Count > 500)
0046              next.Audit.RemoveRange(0, next.Audit.Count - 500);
0047          undo.Push(ProjectCodec.Clone(Current));
0048          if (undo.Count > 20)
0049          {
0050              var keep = undo.Take(20).Reverse().ToArray();
0051              undo.Clear();
0052              foreach (var item in keep)
0053                  undo.Push(item);
0054          }
0055          Current = next;
0056          redo.Clear();
0057      }

0060      {
0061          if (!CanUndo)
0062              return;
0063          redo.Push(Current);
0064          Current = undo.Pop();
0065      }

0068      {
0069          if (!CanRedo)
0070              return;
0071          undo.Push(Current);
0072          Current = redo.Pop();
0073      }

0078          SavedPath = result.Path;
