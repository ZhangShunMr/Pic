// ===== FILE: src/Avas.Core/Analysis/Acoustics.cs =====
using Avas.Core.Domain;

namespace Avas.Core.Analysis;

public static class Acoustics
{
    public static double Sum(IEnumerable<double> levels)
    {
        var values = levels.ToArray();
        if (values.Length == 0)
            return double.NegativeInfinity;
        var max = values.Max();
        if (double.IsNegativeInfinity(max))
            return max;
        return max + 10 * Math.Log10(values.Sum(x => Math.Pow(10, (x - max) / 10)));
    }

    public static double DirectionalLoss(Speaker speaker, double x, double y)
    {
        if (speaker.Exponent == 0)
            return 0;
        var angle = Math.Atan2(y - speaker.Y, x - speaker.X) - speaker.Heading * Math.PI / 180;
        var forward = Math.Max(0, Math.Cos(angle));
        var amplitude = speaker.RearFloor + (1 - speaker.RearFloor) * Math.Pow(forward, speaker.Exponent);
        return 20 * Math.Log10(amplitude);
    }

    public static double Distance(Speaker speaker, double x, double y, double height)
    {
        return Math.Sqrt(Math.Pow(x - speaker.X, 2) + Math.Pow(y - speaker.Y, 2)
            + Math.Pow(height - speaker.Height, 2));
    }

    public static double Propagation(Speaker speaker, Settings settings, double x, double y)
    {
        var distance = Math.Max(settings.MinimumDistance, Distance(speaker, x, y, settings.ReceiverHeight));
        return -20 * Math.Log10(distance / speaker.ReferenceDistance) + DirectionalLoss(speaker, x, y);
    }

    public static double Select(IReadOnlyList<double> bands, Settings settings)
    {
        if (settings.SelectedHz != 0)
            return bands[Array.IndexOf(Software.Frequencies, settings.SelectedHz)];
        return Sum(bands.Select((level, index) => level + (settings.AWeighted ? Software.AWeights[index] : 0)));
    }

    public static bool InsideVehicle(Settings settings, double x, double y)
    {
        return Math.Abs(x) <= settings.VehicleLength / 2 && Math.Abs(y) <= settings.VehicleWidth / 2;
    }

    public static string Metric(Settings settings)
    {
        if (settings.SelectedHz != 0)
            return $"{settings.SelectedHz}Hz频带声级 dB re 20µPa";
        return settings.AWeighted ? "七带A计权合成 dB(A)" : "七带非计权合成 dB re 20µPa";
    }
}

public sealed record SourceContribution(Guid Id, string Name, double Distance, double Loss, double[] Bands, string Origin);

public sealed class PointResult
{
    public Guid? ListenerId { get; init; }
    public string Name { get; init; } = "";
    public double X { get; init; }
    public double Y { get; init; }
    public double[] Bands { get; init; } = [];
    public double Signal { get; init; }
    public double Background { get; init; }
    public double Combined { get; init; }
    public double Margin => Signal - Background;
    public bool Covered { get; init; }
    public bool NearFieldClamped { get; init; }
    public bool InVehicle { get; init; }
    public List<SourceContribution> Contributions { get; init; } = [];
}

public sealed class Calculation
{
    public DateTimeOffset At { get; init; } = DateTimeOffset.UtcNow;
    public string InputHash { get; init; } = "";
    public string Metric { get; init; } = "";
    public List<PointResult> Grid { get; init; } = [];
    public List<PointResult> Listeners { get; init; } = [];
    public int CoveredCount => Grid.Count(x => x.Covered);
    public double Coverage => Grid.Count == 0 ? 0 : 100.0 * CoveredCount / Grid.Count;
    public int ClampedCount => Grid.Count(x => x.NearFieldClamped);
    public double Minimum => Grid.Count == 0 ? 0 : Grid.Min(x => x.Signal);
    public double Maximum => Grid.Count == 0 ? 0 : Grid.Max(x => x.Signal);
    public double MeanEnergy => Acoustics.Sum(Grid.Select(x => x.Signal)) - 10 * Math.Log10(Grid.Count);
    public List<string> Warnings { get; init; } = [];
}

public static class FieldEngine
{
    public static Calculation Calculate(Project project)
    {
        var issues = Validation.Check(project, true);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("\n", issues));
        var settings = project.Settings;
        var nx = (int)Math.Floor((settings.MaxX - settings.MinX) / settings.Step + 1e-9);
        var ny = (int)Math.Floor((settings.MaxY - settings.MinY) / settings.Step + 1e-9);
        var grid = new List<PointResult>();
        for (var ix = 0; ix <= nx; ix++)
        {
            for (var iy = 0; iy <= ny; iy++)
            {
                var x = settings.MinX + ix * settings.Step;
                var y = settings.MinY + iy * settings.Step;
                if (settings.ExcludeVehicle && Acoustics.InsideVehicle(settings, x, y))
                    continue;
                grid.Add(Evaluate(project, x, y));
            }
        }
        if (grid.Count == 0)
            throw new ArgumentException("所有网格均位于排除的车辆内部；请扩大区域或关闭内部排除");
        var warnings = new List<string> { Software.Boundary };
        if (grid.Any(x => x.NearFieldClamped))
            warnings.Add("部分节点距离小于距离下限，使用截断距离；近源值不具备近场物理意义。");
        if (project.Speakers.Any(x => x.Origin == Origin.Simulation) || settings.BackgroundOrigin == Origin.Simulation)
            warnings.Add("包含模拟数据，不能证明真实应用效果。");
        if (project.Speakers.Count(x => x.Enabled) > 1)
            warnings.Add("多声源采用非相干能量叠加；相同信号的相位干涉未计算。");
        if (project.Listeners.Any(x => x.X < settings.MinX || x.X > settings.MaxX || x.Y < settings.MinY || x.Y > settings.MaxY))
            warnings.Add("部分观察点位于网格范围外，点值仍独立计算，不计入网格覆盖率。");
        return new Calculation
        {
            InputHash = IO.ProjectCodec.InputHash(project),
            Metric = Acoustics.Metric(settings),
            Grid = grid,
            Listeners = project.Listeners.Select(x => Evaluate(project, x.X, x.Y, x.Name, x.Id)).ToList(),
            Warnings = warnings
        };
    }

    public static PointResult Evaluate(Project project, double x, double y, string name = "网格节点", Guid? id = null)
    {
        var settings = project.Settings;
        var sources = new List<SourceContribution>();
        var clamped = false;
        foreach (var speaker in project.Speakers.Where(s => s.Enabled))
        {
            var distance = Acoustics.Distance(speaker, x, y, settings.ReceiverHeight);
            clamped |= distance < settings.MinimumDistance;
            var loss = Acoustics.Propagation(speaker, settings, x, y);
            var bands = Software.Frequencies.Select(hz =>
            {
                var band = speaker.Bands.Single(b => b.Hz == hz);
                return band.Level + band.Correction + speaker.Gain + loss;
            }).ToArray();
            sources.Add(new(speaker.Id, speaker.Name, distance, loss, bands, speaker.Origin.Label() + " / " + speaker.Source));
        }
        var combinedBands = Software.Frequencies.Select((_, i) => Acoustics.Sum(sources.Select(s => s.Bands[i]))).ToArray();
        var backgroundBands = Software.Frequencies.Select(hz => settings.Background.Single(b => b.Hz == hz).Level).ToArray();
        var signal = Acoustics.Select(combinedBands, settings);
        var background = Acoustics.Select(backgroundBands, settings);
        return new PointResult
        {
            ListenerId = id,
            Name = name,
            X = x,
            Y = y,
            Bands = combinedBands,
            Signal = signal,
            Background = background,
            Combined = Acoustics.Sum([signal, background]),
            Covered = signal >= settings.LevelTarget && signal - background >= settings.MarginTarget,
            NearFieldClamped = clamped,
            InVehicle = Acoustics.InsideVehicle(settings, x, y),
            Contributions = sources
        };
    }
}

using System.Globalization;
using System.Net;
using System.Text;
using Avas.Core.Domain;

public static class ChartRenderer
{
    public static string Number(double value) => value.ToString("0.###", CultureInfo.InvariantCulture);
    public static string Escape(string value) => WebUtility.HtmlEncode(value);

    public static string Color(double value, double min, double max)
    {
        var ratio = Math.Clamp((value - min) / Math.Max(1, max - min), 0, 1);
        var stops = new (int R, int G, int B)[] { (31, 61, 115), (28, 127, 150), (83, 183, 140), (235, 208, 100), (220, 93, 71) };
        var position = ratio * (stops.Length - 1);
        var left = Math.Min(stops.Length - 2, (int)position);
        var fraction = position - left;
        var a = stops[left];
        var b = stops[left + 1];
        return $"#{(int)(a.R + (b.R - a.R) * fraction):x2}{(int)(a.G + (b.G - a.G) * fraction):x2}{(int)(a.B + (b.B - a.B) * fraction):x2}";
    }

    public static string Map(Project project, Calculation? result, bool coverageOnly = false)
    {
        var settings = project.Settings;
        if (settings.MaxX <= settings.MinX || settings.MaxY <= settings.MinY)
            return "<svg viewBox='0 0 900 620'><text x='30' y='40'>请修正网格坐标范围</text></svg>";
        const double width = 900;
        const double height = 620;
        const double margin = 65;
        var scale = Math.Min((width - margin * 2) / (settings.MaxX - settings.MinX), (height - margin * 2) / (settings.MaxY - settings.MinY));
        double X(double x) => margin + (x - settings.MinX) * scale;
        double Y(double y) => height - margin - (y - settings.MinY) * scale;
        var svg = new StringBuilder("<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 900 620' role='img' aria-label='车辆周围声场布局，X正向为车前，Y正向为车左'>");
        svg.Append("<rect width='900' height='620' fill='#f5f7fa'/>");
        svg.Append($"<metadata>{Escape(IO.ProjectCodec.Json(new { Software.Model, Software.Version, Boundary = Software.Boundary, project.Name, Source = IO.ProjectCodec.SourceSummary(project), Conditions = IO.ProjectCodec.Conditions(project.Settings), InputHash = IO.ProjectCodec.InputHash(project) }))}</metadata>");
        svg.Append($"<text x='25' y='27' font-size='15' fill='#24344d'>{Escape(result?.Metric ?? "布局预览，尚无有效计算结果")}</text>");
        if (result is not null)
        {
            var min = Math.Floor(result.Minimum / 5) * 5;
            var max = Math.Max(min + 5, Math.Ceiling(result.Maximum / 5) * 5);
            var half = settings.Step * scale / 2;
            foreach (var point in result.Grid)
            {
                var color = coverageOnly ? point.Covered ? "#4aa47b" : "#e7b66a" : Color(point.Signal, min, max);
                svg.Append($"<rect x='{Number(X(point.X) - half)}' y='{Number(Y(point.Y) - half)}' width='{Number(half * 2 + 0.2)}' height='{Number(half * 2 + 0.2)}' fill='{color}'><title>{Number(point.X)},{Number(point.Y)}m：{Number(point.Signal)}dB；背景差{Number(point.Margin)}dB；{(point.Covered ? "满足工程目标" : "未满足工程目标")}</title></rect>");
            }
            if (!coverageOnly)
            {
                for (var i = 0; i < 100; i++)
                    svg.Append($"<rect x='{730 + i}' y='580' width='1.5' height='12' fill='{Color(min + (max - min) * i / 99, min, max)}'/>");
                svg.Append($"<text x='730' y='610' font-size='11'>{Number(min)}—{Number(max)}dB</text>");
            }
        }
        for (var x = Math.Ceiling(settings.MinX / 2) * 2; x <= settings.MaxX; x += 2)
        {
            svg.Append($"<line x1='{Number(X(x))}' y1='{Number(Y(settings.MinY))}' x2='{Number(X(x))}' y2='{Number(Y(settings.MaxY))}' stroke='#ffffff' stroke-opacity='.35'/>");
            svg.Append($"<text x='{Number(X(x))}' y='{Number(Y(settings.MinY) + 24)}' text-anchor='middle' font-size='12'>{Number(x)}</text>");
        }
        for (var y = Math.Ceiling(settings.MinY / 2) * 2; y <= settings.MaxY; y += 2)
        {
            svg.Append($"<line x1='{Number(X(settings.MinX))}' y1='{Number(Y(y))}' x2='{Number(X(settings.MaxX))}' y2='{Number(Y(y))}' stroke='#ffffff' stroke-opacity='.35'/>");
            svg.Append($"<text x='{Number(X(settings.MinX) - 14)}' y='{Number(Y(y) + 4)}' text-anchor='end' font-size='12'>{Number(y)}</text>");
        }
        svg.Append($"<rect x='{Number(X(-settings.VehicleLength / 2))}' y='{Number(Y(settings.VehicleWidth / 2))}' width='{Number(settings.VehicleLength * scale)}' height='{Number(settings.VehicleWidth * scale)}' rx='12' fill='#e9edf4' stroke='#33465e' stroke-width='2'/>");
        svg.Append($"<text x='{Number(X(0))}' y='{Number(Y(0) + 5)}' text-anchor='middle' font-size='13' fill='#253b56'>车辆 →</text>");
        foreach (var speaker in project.Speakers)
        {
            var x = X(speaker.X);
            var y = Y(speaker.Y);
            var angle = speaker.Heading * Math.PI / 180;
            var x2 = x + Math.Cos(angle) * 27;
            var y2 = y - Math.Sin(angle) * 27;
            svg.Append($"<line x1='{Number(x)}' y1='{Number(y)}' x2='{Number(x2)}' y2='{Number(y2)}' stroke='#172e4d' stroke-width='3'/>");
            svg.Append($"<circle cx='{Number(x)}' cy='{Number(y)}' r='7' fill='{(speaker.Enabled ? "#fdfdfd" : "#aaa")}' stroke='#172e4d' stroke-width='3'><title>{Escape(speaker.Name)}，朝向{Number(speaker.Heading)}°</title></circle>");
        }
        foreach (var point in project.Listeners)
        {
            svg.Append($"<circle cx='{Number(X(point.X))}' cy='{Number(Y(point.Y))}' r='5' fill='#fff' stroke='#111' stroke-width='2'/>");
            svg.Append($"<text x='{Number(X(point.X) + 9)}' y='{Number(Y(point.Y) - 8)}' font-size='12' fill='#16273f' paint-order='stroke' stroke='#fff' stroke-width='3'>{Escape(point.Name)}</text>");
        }
        svg.Append("<text x='65' y='607' font-size='12' fill='#24344d'>X / m →车前　Y / m ↑车左　圆圈为扬声器与观察点；轮廓不参与遮挡计算</text>");
        svg.Append("</svg>");
        return svg.ToString();
    }

    public static string Spectrum(PointResult point, Settings settings)
    {
        var background = Software.Frequencies.Select(hz => settings.Background.Single(x => x.Hz == hz).Level).ToArray();
        var all = point.Bands.Concat(background).ToArray();
        var min = Math.Floor(all.Min() / 10) * 10 - 10;
        var max = Math.Ceiling(all.Max() / 10) * 10 + 10;
        double Y(double value) => 240 - (value - min) / (max - min) * 200;
        var svg = new StringBuilder("<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 760 300' role='img' aria-label='观察点七频带声压级与背景声对比'>");
        svg.Append("<rect width='760' height='300' fill='#fff'/>");
        for (var value = min; value <= max; value += 10)
        {
            svg.Append($"<line x1='55' y1='{Number(Y(value))}' x2='730' y2='{Number(Y(value))}' stroke='#e1e6ed'/>");
            svg.Append($"<text x='46' y='{Number(Y(value) + 4)}' text-anchor='end' font-size='12'>{Number(value)}</text>");
        }
        var signalPoints = new List<string>();
        var backgroundPoints = new List<string>();
        for (var i = 0; i < 7; i++)
        {
            var x = 85 + i * 100;
            signalPoints.Add($"{x},{Number(Y(point.Bands[i]))}");
            backgroundPoints.Add($"{x},{Number(Y(background[i]))}");
            svg.Append($"<circle cx='{x}' cy='{Number(Y(point.Bands[i]))}' r='4' fill='#146d81'/>");
            svg.Append($"<text x='{x}' y='263' text-anchor='middle' font-size='12'>{Software.Frequencies[i]}</text>");
        }
        svg.Append($"<polyline points='{string.Join(" ", signalPoints)}' fill='none' stroke='#146d81' stroke-width='3'/>");
        svg.Append($"<polyline points='{string.Join(" ", backgroundPoints)}' fill='none' stroke='#b47936' stroke-width='2' stroke-dasharray='6 4'/>");
        svg.Append("<text x='55' y='20' font-size='12'>未计权频带声压级 / dB re 20µPa</text><text x='400' y='290' font-size='12'>中心频率 / Hz　蓝实线：警示声　棕虚线：背景</text></svg>");
        return svg.ToString();
    }
}

using System.Text;
using Avas.Core.Domain;

namespace Avas.Core.Analysis;

public sealed record ReviewFinding(string Severity, string Object, string Finding, string Action);

public static class ReviewEngine
{
    public static List<ReviewFinding> Inspect(Project project)
    {
        var findings = new List<ReviewFinding>();
        var settings = project.Settings;
        foreach (var source in project.Speakers)
        {
            if (!source.Enabled)
            {
                findings.Add(new("信息", source.Name, "当前源未启用，不参与声场和试听", "需要参与时到扬声器页启用并应用"));
                continue;
            }
            if (source.X < settings.MinX || source.X > settings.MaxX || source.Y < settings.MinY || source.Y > settings.MaxY)
                findings.Add(new("提示", source.Name, "声源位于显示网格外，仍参与计算", "核对坐标，或在背景与网格页扩大显示范围"));
            if (Acoustics.InsideVehicle(settings, source.X, source.Y))
                findings.Add(new("边界", source.Name, "声源位于车辆轮廓内，本模型仍向外自由传播", "不要将该结果解释为已包含车身遮挡；用实测修正设计"));
            if (source.Origin == Origin.Simulation)
                findings.Add(new("来源", source.Name, "使用模拟输入", "正式设计时替换为有来源的输入，保留模拟标识直至替换完成"));
            if (source.Origin == Origin.MeasurementDeclared)
                findings.Add(new("来源", source.Name, "实测来源仅为用户声明，软件未认证校准", "核对轴向参考距离、仪器和测量记录是否与频带一致"));
            if (source.Source.Contains("请填写", StringComparison.Ordinal))
                findings.Add(new("待补", source.Name, "来源仍是默认占位说明", "到扬声器页填写实际来源与轴向参考条件"));
            if (source.ReferenceDistance < settings.MinimumDistance)
                findings.Add(new("提示", source.Name, "参考距离小于传播距离下限", "核对近源截断设置，参考距离仍用于声级换算"));
            if (source.Bands.Any(x => Math.Abs(x.Correction) > 15))
                findings.Add(new("复核", source.Name, "至少一个频带修正幅度超过15dB", "核查是否重复叠加频响；15dB只是工程复核提示，不是判定限值"));
            if (source.Audio is not null)
                findings.Add(new("音频", source.Name, "已内嵌数字音频，物理频带仍独立维护", "核查录音内容与参考频谱是否对应；试听不证明标定关系"));
        }
        foreach (var group in project.Speakers.Where(x => x.Enabled).GroupBy(x => (x.X, x.Y, x.Height)).Where(x => x.Count() > 1))
            findings.Add(new("复核", string.Join("、", group.Select(x => x.Name)), "多个启用源位置完全相同", "确认是否有意叠加；复制源后常需调整坐标，相干效应未计算"));
        foreach (var point in project.Listeners)
        {
            if (Acoustics.InsideVehicle(settings, point.X, point.Y))
                findings.Add(new("边界", point.Name, "观察点位于车辆轮廓内", "本工具只给自由场点值，不能解释为车内声级"));
            if (point.X < settings.MinX || point.X > settings.MaxX || point.Y < settings.MinY || point.Y > settings.MaxY)
                findings.Add(new("信息", point.Name, "点位于网格外，点值仍计算但不计入网格覆盖", "需要计入区域时扩大网格，或保持其为独立观察点"));
            if (project.Speakers.Any(x => x.Enabled && Acoustics.Distance(x, point.X, point.Y, settings.ReceiverHeight) < settings.MinimumDistance))
                findings.Add(new("边界", point.Name, "接近至少一个声源，距离将被截断", "增大接收距离；不能将截断值视为近场解"));
        }
        if (settings.BackgroundSource.Contains("请填写", StringComparison.Ordinal))
            findings.Add(new("待补", "背景", "背景来源仍为默认占位说明", "在背景与网格页填写环境及频谱来源"));
        if (settings.SelectedHz == 0 && settings.AWeighted)
            findings.Add(new("口径", "合成", "使用中心频率A计权近似", "仅解释七个输入频带范围，不称全频积分声级"));
        if (project.Listeners.Count == 0)
            findings.Add(new("流程", "观察点", "没有独立观察点，无法查看点频谱或制作点试听", "在布局页添加点或追加等角点集"));
        findings.Add(new("固定假设", "背景", "背景在整个网格中均匀，多源按非相干能量叠加", "有空间非均匀、相干或真实车身影响时须使用其他模型或试验"));
        return findings;
    }
}

public sealed class TransectRequest
{
    public double StartX { get; set; }
    public double StartY { get; set; }
    public double Heading { get; set; }
    public double Length { get; set; } = 10;
    public double Step { get; set; } = 0.5;
}

public sealed record TransectPoint(double Distance, PointResult Result);

public sealed class TransectResult
{
    public string InputHash { get; init; } = "";
    public string Metric { get; init; } = "";
    public string Source { get; init; } = "";
    public string Conditions { get; init; } = "";
    public List<TransectPoint> Points { get; init; } = [];
    public double? LastCovered => Points.Where(x => x.Result.Covered).Select(x => (double?)x.Distance).LastOrDefault();
    public int Transitions => Points.Zip(Points.Skip(1)).Count(x => x.First.Result.Covered != x.Second.Result.Covered);
}

public static class TransectEngine
{
    public static TransectResult Run(Project project, TransectRequest request)
    {
        var issues = Validation.Check(project, true);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("\n", issues));
        if (!double.IsFinite(request.StartX) || !double.IsFinite(request.StartY)
            || Math.Abs(request.StartX) > 100 || Math.Abs(request.StartY) > 100)
            throw new ArgumentException("剖面起点坐标必须在-100至100m范围");
        if (!double.IsFinite(request.Heading) || request.Heading is < -360 or > 360)
            throw new ArgumentException("剖面角度范围-360至360°");
        if (!double.IsFinite(request.Length) || request.Length is <= 0 or > 100)
            throw new ArgumentException("剖面长度必须大于0且不超过100m");
        if (!double.IsFinite(request.Step) || request.Step is < 0.1 or > 10)
            throw new ArgumentException("剖面步长范围0.1至10m");
        var count = (int)Math.Floor(request.Length / request.Step + 1e-9) + 1;
        if (count > 501)
            throw new ArgumentException("剖面最多501点，请增大步长");
        var angle = request.Heading * Math.PI / 180;
        var points = new List<TransectPoint>();
        for (var i = 0; i < count; i++)
        {
            var distance = i * request.Step;
            var x = request.StartX + distance * Math.Cos(angle);
            var y = request.StartY + distance * Math.Sin(angle);
            if (Math.Abs(x) > 100 || Math.Abs(y) > 100)
                throw new ArgumentException("剖面终点超出坐标范围，请缩短长度或调整起点");
            var point = FieldEngine.Evaluate(project, x, y, $"剖面{i + 1}");
            points.Add(new TransectPoint(distance, point));
        }
        return new TransectResult
        {
            InputHash = IO.ProjectCodec.InputHash(project),
            Metric = Acoustics.Metric(project.Settings),
            Source = IO.ProjectCodec.SourceSummary(project),
            Conditions = IO.ProjectCodec.Conditions(project.Settings),
            Points = points
        };
    }

    public static byte[] Csv(TransectResult result)
    {
        var text = new StringBuilder("\uFEFFModel,Version,Source,Conditions,InputHash,Metric,Distance_m,X_m,Y_m,Signal_dB,Margin_dB,EngineeringTargetMet,DistanceClamped,InVehicle\r\n");
        foreach (var point in result.Points)
        {
            var p = point.Result;
            text.AppendLine(string.Join(",", Software.Model, Software.Version, IO.SpectrumCsv.Cell(result.Source), IO.SpectrumCsv.Cell(result.Conditions), result.InputHash, IO.SpectrumCsv.Cell(result.Metric),
                ChartRenderer.Number(point.Distance), ChartRenderer.Number(p.X), ChartRenderer.Number(p.Y),
                ChartRenderer.Number(p.Signal), ChartRenderer.Number(p.Margin), p.Covered, p.NearFieldClamped, p.InVehicle));
        }
        return Encoding.UTF8.GetBytes(text.ToString());
    }

public static class ScenarioEngine
{
    public static Snapshot Capture(Project project, string name)
    {
        if (string.IsNullOrWhiteSpace(name) || name.Length > 100)
            throw new ArgumentException("方案名称必填，最多100字符");
        var copy = ProjectCodec.Clone(project);
        return new Snapshot
        {
            Name = name.Trim(),
            InputHash = ProjectCodec.InputHash(project),
            Settings = copy.Settings,
            Speakers = copy.Speakers,
            Listeners = copy.Listeners
        };
    }

    public static Project Restore(Snapshot snapshot)
    {
        return ProjectCodec.Clone(new Project
        {
            Name = snapshot.Name,
            Settings = snapshot.Settings,
            Speakers = snapshot.Speakers,
            Listeners = snapshot.Listeners
        });
    }

    public static Comparison Compare(Snapshot baseline, Snapshot target)
    {
        var a = Restore(baseline);
        var b = Restore(target);
        var reasons = Conditions(a, b);
        if (reasons.Count > 0)
            return new Comparison { BaselineName = baseline.Name, TargetName = target.Name, Reasons = reasons };
        var resultA = FieldEngine.Calculate(a);
        var resultB = FieldEngine.Calculate(b);
        var points = resultA.Listeners.Select(p =>
        {
            var q = resultB.Listeners.Single(x => x.ListenerId == p.ListenerId);
            return new PointDelta(p.Name, p.X, p.Y, p.Signal, q.Signal, q.Signal - p.Signal);
        }).ToList();
        return new Comparison
        {
            BaselineName = baseline.Name,
            TargetName = target.Name,
            Baseline = resultA,
            Target = resultB,
            BaselineSource = ProjectCodec.SourceSummary(a),
            TargetSource = ProjectCodec.SourceSummary(b),
            Conditions = ProjectCodec.Conditions(a.Settings),
            Points = points
        };
    }

    public static List<string> Conditions(Project a, Project b)
    {
        var reasons = new List<string>();
        var settingsA = ProjectCodec.Json(a.Settings);
        var settingsB = ProjectCodec.Json(b.Settings);
        if (settingsA != settingsB)
            reasons.Add("计算区域、步长、接收高度、车辆轮廓、背景、计权、频带或工程目标不同；请在相同设置下重存快照。");
        var pointsA = a.Listeners.OrderBy(x => x.Id).Select(x => new { x.Id, x.X, x.Y });
        var pointsB = b.Listeners.OrderBy(x => x.Id).Select(x => new { x.Id, x.X, x.Y });
        if (ProjectCodec.Json(pointsA) != ProjectCodec.Json(pointsB))
            reasons.Add("观察点标识或坐标不同；不能按相似名称猜测对应关系。");
        var originsA = a.Speakers.Where(x => x.Enabled).Select(x => x.Origin).Distinct().Order().ToArray();
        var originsB = b.Speakers.Where(x => x.Enabled).Select(x => x.Origin).Distinct().Order().ToArray();
        if (!originsA.SequenceEqual(originsB))
            reasons.Add("两方案的数据来源类别不同，禁止形成无标记差值。");
        if (!a.Speakers.Any(x => x.Enabled) || !b.Speakers.Any(x => x.Enabled))
            reasons.Add("至少一个方案没有启用的扬声器。");
        return reasons;
    }

public static class ScanEngine
{
    public static ScanResult Run(Project original, ScanRequest request)
    {
        if (!double.IsFinite(request.Start) || !double.IsFinite(request.End) || !double.IsFinite(request.Step)
            || request.Step <= 0 || request.End < request.Start)
            throw new ArgumentException("扫描起止必须为有限数，终值不小于初值，步长必须为正");
        var count = (int)Math.Min(1000, Math.Floor((request.End - request.Start) / request.Step) + 1);
        if (count is < 1 or > 25)
            throw new ArgumentException("每次扫描最多25个取值，请增大步长");
        if (!Enum.IsDefined(request.Parameter))
            throw new ArgumentException("扫描参数无效");
        var project = ProjectCodec.Clone(original);
        var speaker = project.Speakers.SingleOrDefault(x => x.Id == request.SpeakerId)
            ?? throw new ArgumentException("请先选择需要扫描的扬声器");
        if (!speaker.Enabled)
            throw new ArgumentException("扫描对象必须启用");
        var rows = new List<ScanRow>();
        for (var i = 0; i < count; i++)
        {
            var value = request.Start + i * request.Step;
            Apply(speaker, request.Parameter, value);
            var result = FieldEngine.Calculate(project);
            rows.Add(new ScanRow(value, result.Coverage, result.Minimum, result.Maximum,
                result.Grid.Min(x => x.Margin), result.Listeners.Count(x => x.Covered)));
        }
        return new ScanResult
        {
            InputHash = ProjectCodec.InputHash(original),
            SpeakerId = speaker.Id,
            SpeakerName = speaker.Name,
            Parameter = request.Parameter,
            Metric = Acoustics.Metric(original.Settings),
            Source = ProjectCodec.SourceSummary(original),
            Conditions = ProjectCodec.Conditions(original.Settings),
            Rows = rows
        };
    }

    public static void Apply(Speaker speaker, ScanParameter parameter, double value)
    {
        switch (parameter)
        {
            case ScanParameter.Heading:
                speaker.Heading = value;
                break;
            case ScanParameter.Gain:
                speaker.Gain = value;
                break;
            case ScanParameter.PositionX:
                speaker.X = value;
                break;
            case ScanParameter.PositionY:
                speaker.Y = value;
                break;
            default:
                throw new ArgumentException("不支持的扫描参数");
        }
    }

    public static string Label(ScanParameter parameter) => parameter switch
    {
        ScanParameter.Heading => "朝向 °",
        ScanParameter.Gain => "增益 dB",
        ScanParameter.PositionX => "X坐标 m",
        ScanParameter.PositionY => "Y坐标 m",
        _ => "未知"
    };
}

public static class AuditionEngine
{
    public static Audition Render(Project project, Listener point, double seconds, double playbackReference, bool includeBackground)
    {
        var errors = Validation.Check(project, true);
        if (errors.Count > 0)
            throw new ArgumentException(string.Join("\n", errors));
        if (!double.IsFinite(seconds) || seconds is < 1 or > 8)
            throw new ArgumentException("试听长度范围1至8秒");
        if (!double.IsFinite(playbackReference) || playbackReference is < 60 or > 140)
            throw new ArgumentException("数字播放参考范围60至140dB；不是设备声压标定");
        const int rate = 24000;
        var count = (int)(seconds * rate);
        var mix = new double[count];
        var notes = new List<string>();
        var sourceIndex = 0;
        foreach (var speaker in project.Speakers.Where(x => x.Enabled))
        {
            double[] input;
            if (speaker.Audio is not null)
            {
                var wave = WaveCodec.Read(Convert.FromBase64String(speaker.Audio.Data));
                input = SpectralAudio.Resample(wave, rate, count);
                notes.Add($"{speaker.Name}：导入音频{speaker.Audio.Name}，短记录尾部补静音");
            }
            else
            {
                input = Noise(count, 420 + sourceIndex);
                notes.Add($"{speaker.Name}：无音频，使用固定种子模拟频带噪声");
            }
            var propagation = Acoustics.Propagation(speaker, project.Settings, point.X, point.Y);
            var target = Software.Frequencies.Select(hz =>
            {
                var band = speaker.Bands.Single(x => x.Hz == hz);
                return Math.Pow(10, (band.Level + band.Correction + speaker.Gain + propagation - playbackReference) / 20);
            }).ToArray();
            var shaped = SpectralAudio.Shape(input, rate, target);
            for (var i = 0; i < count; i++)
                mix[i] += shaped[i];
            sourceIndex++;
        }
        if (includeBackground)
        {
            var asset = project.Settings.BackgroundAudio;
            var input = asset is null ? Noise(count, 802) : SpectralAudio.Resample(WaveCodec.Read(Convert.FromBase64String(asset.Data)), rate, count);
            var target = Software.Frequencies.Select(hz =>
                Math.Pow(10, (project.Settings.Background.Single(x => x.Hz == hz).Level - playbackReference) / 20)).ToArray();
            var background = SpectralAudio.Shape(input, rate, target);
            for (var i = 0; i < count; i++)
                mix[i] += background[i];
            notes.Add(asset is null ? "背景为固定种子频带噪声模拟" : "背景采用导入音频的频谱整形");
        }
        var peak = mix.Max(x => Math.Abs(x));
        if (peak > 0.98)
            throw new ArgumentException($"预计峰值{peak:F3}超过0.98，请将数字播放参考提高至少{Math.Ceiling(20 * Math.Log10(peak / 0.98))}dB后重试；不自动归一化");
        var fade = Math.Min(1200, count / 2);
        for (var i = 0; i < fade; i++)
        {
            var factor = 0.5 - 0.5 * Math.Cos(Math.PI * i / fade);
            mix[i] *= factor;
            mix[count - 1 - i] *= factor;
        }
        notes.Add($"24kHz单声道、七频带整形，50ms淡入淡出；数字幅值1对应参数{playbackReference}dB，仅定义相对音量。");
        notes.Add("未模拟到达时延、双耳线索和车身；混合波形相位与非相干能量模型不等价；不得据此判定真实可听性。");
        notes.Add("设备音量、声卡及耳机未校准，改变系统音量会改变实际听感。");
        var metadata = IO.ProjectCodec.Json(new
        {
            Software.Name,
            Software.Version,
            Software.Model,
            Boundary = Software.Boundary,
            Project = project.Name,
            Source = IO.ProjectCodec.SourceSummary(project),
            Conditions = IO.ProjectCodec.Conditions(project.Settings),
            Point = new { point.Id, point.Name, point.X, point.Y },
            Seconds = seconds,
            DigitalReferenceDb = playbackReference,
            IncludeBackground = includeBackground,
            Description = notes,
            InputHash = IO.ProjectCodec.InputHash(project)
        });
        var waveBytes = WaveCodec.Write(mix, rate, metadata);
        return new Audition(waveBytes, string.Join("\n", notes), peak, false, IO.ProjectCodec.InputHash(project));
    }

    private static double[] Noise(int count, int seed)
    {
        var random = new Random(seed);
        return Enumerable.Range(0, count).Select(_ => random.NextDouble() * 2 - 1).ToArray();
    }
}

using System.Numerics;
using Avas.Core.Domain;

namespace Avas.Core.Audio;

public static class Fourier
{
    public static void Transform(Complex[] data, bool inverse = false)
    {
        var count = data.Length;
        if (count == 0 || (count & (count - 1)) != 0)
            throw new ArgumentException("FFT长度必须是2的幂");
        for (int i = 1, j = 0; i < count; i++)
        {
            var bit = count >> 1;
            for (; (j & bit) != 0; bit >>= 1)
                j ^= bit;
            j ^= bit;
            if (i < j)
                (data[i], data[j]) = (data[j], data[i]);
        }
        for (var length = 2; length <= count; length <<= 1)
        {
            var angle = (inverse ? 2 : -2) * Math.PI / length;
            var root = new Complex(Math.Cos(angle), Math.Sin(angle));
            for (var offset = 0; offset < count; offset += length)
            {
                var factor = Complex.One;
                for (var i = 0; i < length / 2; i++)
                {
                    var a = data[offset + i];
                    var b = data[offset + i + length / 2] * factor;
                    data[offset + i] = a + b;
                    data[offset + i + length / 2] = a - b;
                    factor *= root;
                }
            }
        }
        if (inverse)
            for (var i = 0; i < count; i++)
                data[i] /= count;
    }

    public static int BandIndex(double frequency)
    {
        for (var i = 0; i < Software.Frequencies.Length; i++)
            if (frequency >= Software.Frequencies[i] / Math.Sqrt(2) && frequency < Software.Frequencies[i] * Math.Sqrt(2))
                return i;
        return -1;
    }
}

public sealed record DigitalSpectrum(double[] Levels, double CapturedRms, double FullRms, int Frames, string Note);

public static class SpectralAudio
{
    public static DigitalSpectrum Analyze(WaveData audio)
    {
        const int blockSize = 4096;
        var energy = new double[7];
        var frames = 0;
        var fullEnergy = audio.Samples.Sum(x => x * x) / audio.Samples.Length;
        for (var start = 0; start < audio.Samples.Length; start += blockSize)
        {
            var count = Math.Min(blockSize, audio.Samples.Length - start);
            var block = new Complex[blockSize];
            for (var i = 0; i < count; i++)
                block[i] = audio.Samples[start + i];
            Fourier.Transform(block);
            for (var bin = 1; bin <= blockSize / 2; bin++)
            {
                var index = Fourier.BandIndex((double)bin * audio.Rate / blockSize);
                if (index < 0)
                    continue;
                var factor = bin == blockSize / 2 ? 1 : 2;
                energy[index] += factor * block[bin].Magnitude * block[bin].Magnitude / blockSize;
            }
            frames += count;
        }
        for (var i = 0; i < 7; i++)
            energy[i] /= frames;
        var levels = energy.Select(x => x > 1e-16 ? 10 * Math.Log10(x) : -160).ToArray();
        return new DigitalSpectrum(levels, Math.Sqrt(energy.Sum()), Math.Sqrt(fullEnergy), frames,
            "矩形窗4096点FFT能量分带；双声道先取平均；dBFS(RMS)参考幅值1，非声压级；有限记录会有频谱泄漏。");
    }

    public static List<BandInput> RelativeBands(DigitalSpectrum spectrum, double declaredTotal)
    {
        if (!double.IsFinite(declaredTotal) || declaredTotal is < 0 or > 120)
            throw new ArgumentException("用户声明的七带合成参考声级必须为0至120dB");
        if (spectrum.CapturedRms < 1e-7)
            throw new ArgumentException("七频带内音频能量过低，不能推导相对频谱");
        var total = Analysis.Acoustics.Sum(spectrum.Levels);
        var levels = spectrum.Levels.Select(x => declaredTotal + x - total).ToArray();
        if (levels.Any(x => x < -60 || x > 140))
            throw new ArgumentException("按相对谱分配后存在超出-60至140dB的频带；请手工填写有依据的七频带，不自动截断或补底噪");
        return levels.Select((x, i) => new BandInput
        {
            Hz = Software.Frequencies[i],
            Level = x
        }).ToList();
    }

    public static double[] Resample(WaveData wave, int rate, int count)
    {
        var output = new double[count];
        for (var i = 0; i < count; i++)
        {
            var position = (double)i * wave.Rate / rate;
            if (position >= wave.Samples.Length - 1)
                break;
            var left = (int)position;
            var fraction = position - left;
            output[i] = wave.Samples[left] * (1 - fraction) + wave.Samples[left + 1] * fraction;
        }
        return output;
    }

    public static double[] Shape(double[] input, int rate, double[] targetRms)
    {
        var size = 1;
        while (size < input.Length)
            size <<= 1;
        var spectrum = new Complex[size];
        for (var i = 0; i < input.Length; i++)
            spectrum[i] = input[i];
        Fourier.Transform(spectrum);
        var energy = new double[7];
        for (var bin = 0; bin < size; bin++)
        {
            var hz = (double)Math.Min(bin, size - bin) * rate / size;
            var index = Fourier.BandIndex(hz);
            if (index >= 0)
                energy[index] += spectrum[bin].Magnitude * spectrum[bin].Magnitude / size / input.Length;
        }
        for (var bin = 0; bin < size; bin++)
        {
            var hz = (double)Math.Min(bin, size - bin) * rate / size;
            var index = Fourier.BandIndex(hz);
            var gain = index >= 0 && energy[index] > 1e-14 ? targetRms[index] / Math.Sqrt(energy[index]) : 0;
            spectrum[bin] *= gain;
        }
        Fourier.Transform(spectrum, true);
        return spectrum.Take(input.Length).Select(x => x.Real).ToArray();
    }
}

// ===== FILE: src/Avas.Core/Audio/WaveCodec.cs =====
using System.Buffers.Binary;
using System.Text;
using Avas.Core.Domain;
using Avas.Core.IO;

namespace Avas.Core.Audio;

public sealed record WaveData(int Rate, int Channels, double[] Samples)
{
    public double Duration => (double)Samples.Length / Rate;
}

public static class WaveCodec
{
    public static WaveData Read(byte[] bytes)
    {
        if (bytes.Length is < 44 or > 10485760)
            throw new ArgumentException("WAV文件必须在44字节至10MiB之间");
        if (Encoding.ASCII.GetString(bytes, 0, 4) != "RIFF" || Encoding.ASCII.GetString(bytes, 8, 4) != "WAVE")
            throw new ArgumentException("只支持RIFF WAVE；请先转换为16位PCM WAV");
        var declared = BinaryPrimitives.ReadUInt32LittleEndian(bytes.AsSpan(4, 4));
        if (declared + 8L != bytes.Length)
            throw new ArgumentException("WAV RIFF声明长度与文件长度不一致");
        var format = 0;
        var channels = 0;
        var rate = 0;
        var bits = 0;
        var align = 0;
        var byteRate = 0;
        byte[]? data = null;
        var position = 12;
        var seenFormat = false;
        while (position + 8 <= bytes.Length)
        {
            var name = Encoding.ASCII.GetString(bytes, position, 4);
            var size = BinaryPrimitives.ReadUInt32LittleEndian(bytes.AsSpan(position + 4, 4));
            position += 8;
            if (size > int.MaxValue || position + (long)size > bytes.Length)
                throw new ArgumentException("WAV数据块截断");
            if (name == "fmt ")
            {
                if (seenFormat || size < 16)
                    throw new ArgumentException("WAV格式块重复或不完整");
                seenFormat = true;
                format = BinaryPrimitives.ReadUInt16LittleEndian(bytes.AsSpan(position, 2));
                channels = BinaryPrimitives.ReadUInt16LittleEndian(bytes.AsSpan(position + 2, 2));
                rate = BinaryPrimitives.ReadInt32LittleEndian(bytes.AsSpan(position + 4, 4));
                byteRate = BinaryPrimitives.ReadInt32LittleEndian(bytes.AsSpan(position + 8, 4));
                align = BinaryPrimitives.ReadUInt16LittleEndian(bytes.AsSpan(position + 12, 2));
                bits = BinaryPrimitives.ReadUInt16LittleEndian(bytes.AsSpan(position + 14, 2));
            }
            if (name == "data")
            {
                if (data is not null)
                    throw new ArgumentException("WAV存在多个data块，请转换后重试");
                data = bytes.AsSpan(position, (int)size).ToArray();
            }
            position += checked((int)size + (int)(size % 2));
        }
        if (format != 1 || bits != 16 || channels is < 1 or > 2 || rate is < 16000 or > 96000)
            throw new ArgumentException("仅支持16位PCM、单/双声道、16000至96000Hz WAV");
        if (align != channels * 2 || byteRate != rate * align || data is null || data.Length == 0 || data.Length % align != 0)
            throw new ArgumentException("WAV块对齐、字节率或采样长度无效");
        var frames = data.Length / align;
        if ((double)frames / rate > 30)
            throw new ArgumentException("音频最多30秒，请先裁剪");
        var samples = new double[frames];
        for (var frame = 0; frame < frames; frame++)
        {
            double sum = 0;
            for (var channel = 0; channel < channels; channel++)
            {
                var sample = BinaryPrimitives.ReadInt16LittleEndian(data.AsSpan(frame * align + channel * 2, 2));
                sum += sample / 32768.0;
            }
            samples[frame] = sum / channels;
        }
        return new WaveData(rate, channels, samples);
    }

    public static AudioAsset CreateAsset(string fileName, byte[] bytes)
    {
        var data = Read(bytes);
        return new AudioAsset
        {
            Name = Path.GetFileName(fileName),
            Sha256 = ProjectCodec.Hash(bytes),
            Data = Convert.ToBase64String(bytes),
            Rate = data.Rate,
            Channels = data.Channels,
            Duration = data.Duration
        };
    }

    public static byte[] Write(double[] samples, int rate = 24000, string? metadata = null)
    {
        using var stream = new MemoryStream();
        using var writer = new BinaryWriter(stream);
        writer.Write(Encoding.ASCII.GetBytes("RIFF"));
        writer.Write(36 + samples.Length * 2);
        writer.Write(Encoding.ASCII.GetBytes("WAVEfmt "));
        writer.Write(16);
        writer.Write((short)1);
        writer.Write((short)1);
        writer.Write(rate);
        writer.Write(rate * 2);
        writer.Write((short)2);
        writer.Write((short)16);
        writer.Write(Encoding.ASCII.GetBytes("data"));
        writer.Write(samples.Length * 2);
        foreach (var sample in samples)
            writer.Write((short)Math.Round(Math.Clamp(sample, -1, 0.999969) * 32768));
        if (metadata is not null)
        {
            var comment = Encoding.UTF8.GetBytes(metadata + "\0");
            writer.Write(Encoding.ASCII.GetBytes("LIST"));
            writer.Write(12 + comment.Length + comment.Length % 2);
            writer.Write(Encoding.ASCII.GetBytes("INFOICMT"));
            writer.Write(comment.Length);
            writer.Write(comment);
            if (comment.Length % 2 != 0)
                writer.Write((byte)0);
            var end = stream.Position;
            stream.Position = 4;
            writer.Write(checked((int)end - 8));
            stream.Position = end;
        }
        return stream.ToArray();
    }
}

public static class Validation
{
    public static List<Issue> Check(Project project, bool forCalculation = false)
    {
        var issues = new List<Issue>();
        if (project.Schema != 1)
            issues.Add(new("Schema", "仅支持项目格式版本1，不能静默迁移"));
        Text(issues, "项目名称", project.Name, 100, true);
        Text(issues, "负责人", project.Owner, 100);
        Text(issues, "车辆", project.Vehicle, 100);
        Text(issues, "目的", project.Purpose, 2000);
        Text(issues, "备注", project.Note, 4000);
        if (project.Id == Guid.Empty)
            issues.Add(new("项目标识", "不能是空GUID"));
        if (project.Speakers is null || project.Listeners is null || project.Settings is null || project.Snapshots is null || project.Audit is null)
        {
            issues.Add(new("项目结构", "集合与设置不能为null"));
            return issues;
        }
        if (project.Speakers.Count > 8)
            issues.Add(new("扬声器", "最多8个"));
        if (project.Listeners.Count > 100)
            issues.Add(new("观察点", "最多100个"));
        if (project.Snapshots.Count > 20)
            issues.Add(new("方案快照", "最多20个"));
        if (project.Audit.Count > 1000)
            issues.Add(new("审计", "最多1000条"));
        Unique(issues, "扬声器", project.Speakers.Select(x => x.Id));
        Unique(issues, "观察点", project.Listeners.Select(x => x.Id));
        Unique(issues, "方案", project.Snapshots.Select(x => x.Id));
        foreach (var source in project.Speakers)
            CheckSpeaker(issues, source);
        foreach (var point in project.Listeners)
        {
            Text(issues, $"观察点 {point.Id} 名称", point.Name, 100, true);
            Range(issues, $"{point.Name}.X", point.X, -100, 100);
            Range(issues, $"{point.Name}.Y", point.Y, -100, 100);
            Text(issues, $"{point.Name}.备注", point.Note, 2000);
        }
        CheckSettings(issues, project.Settings);
        foreach (var snapshot in project.Snapshots)
        {
            Text(issues, "快照名称", snapshot.Name, 100, true);
            if (snapshot.Settings is null || snapshot.Speakers is null || snapshot.Listeners is null)
            {
                issues.Add(new("快照", "缺少设置、扬声器或观察点"));
                continue;
            }
            var nested = new Project
            {
                Name = snapshot.Name,
                Settings = snapshot.Settings,
                Speakers = snapshot.Speakers,
                Listeners = snapshot.Listeners
            };
            issues.AddRange(Check(nested).Select(x => new Issue("快照/" + x.Path, x.Message)));
        }
        if (forCalculation && !project.Speakers.Any(x => x.Enabled))
            issues.Add(new("扬声器", "请添加并启用至少一个扬声器"));
        return issues;
    }

    {
        Range(issues, "网格.MinX", settings.MinX, -100, 100);
        Range(issues, "网格.MaxX", settings.MaxX, -100, 100);
        Range(issues, "网格.MinY", settings.MinY, -100, 100);
        Range(issues, "网格.MaxY", settings.MaxY, -100, 100);
        Range(issues, "网格步长", settings.Step, 0.25, 5);
        Range(issues, "接收高度", settings.ReceiverHeight, 0, 10);
        Range(issues, "距离下限", settings.MinimumDistance, 0.1, 10);
        Range(issues, "车辆长度", settings.VehicleLength, 1, 20);
        Range(issues, "车辆宽度", settings.VehicleWidth, 0.5, 5);
        Range(issues, "工程声级目标", settings.LevelTarget, -60, 140);
        Range(issues, "工程背景差目标", settings.MarginTarget, -40, 60);
        if (settings.MaxX <= settings.MinX || settings.MaxY <= settings.MinY)
            issues.Add(new("网格", "最大坐标必须大于最小坐标"));
        if (settings.Step > 0 && double.IsFinite(settings.Step))
        {
            var count = (Math.Floor((settings.MaxX - settings.MinX) / settings.Step) + 1)
                * (Math.Floor((settings.MaxY - settings.MinY) / settings.Step) + 1);
            if (!double.IsFinite(count) || count > 20000)
                issues.Add(new("网格", "节点上限20000，请增大步长或缩小范围"));
        }
        if (settings.SelectedHz != 0 && !Software.Frequencies.Contains(settings.SelectedHz))
            issues.Add(new("显示频带", "请选择0（合成）或七个支持频带"));
        Text(issues, "背景来源", settings.BackgroundSource, 2000, true);
        if (!Enum.IsDefined(settings.BackgroundOrigin))
            issues.Add(new("背景来源类别", "无效枚举"));
        Bands(issues, "背景频带", settings.Background, false);
        Asset(issues, "背景音频", settings.BackgroundAudio);
    }

    public static void Bands(List<Issue> issues, string path, List<BandInput>? bands, bool correction)
    {
        if (bands is null || bands.Count != Software.Frequencies.Length)
        {
            issues.Add(new(path, "必须完整提供七个频带，缺失不能补零"));
            return;
        }
        if (!bands.Select(x => x.Hz).Order().SequenceEqual(Software.Frequencies))
            issues.Add(new(path, "频率必须各出现一次：125,250,500,1000,2000,4000,8000"));
        foreach (var band in bands)
        {
            Range(issues, $"{path}.{band.Hz}Hz声级", band.Level, -60, 140);
            Range(issues, $"{path}.{band.Hz}Hz修正", band.Correction, correction ? -40 : 0, correction ? 40 : 0);
        }
    }

    private static void Asset(List<Issue> issues, string path, AudioAsset? asset)
    {
        if (asset is null)
            return;
        Text(issues, path + ".名称", asset.Name, 200, true);
        Text(issues, path + ".来源", asset.Source, 2000, true);
        if (asset.Data is null || asset.Data.Length > 14000000)
            issues.Add(new(path, "单音频最大10MiB"));
        Range(issues, path + ".时长", asset.Duration, 0.001, 30);
        if (asset.Rate is < 16000 or > 96000 || asset.Channels is < 1 or > 2)
            issues.Add(new(path, "采样率16000至96000Hz，单声道或双声道"));
    }

    private static void Unique(List<Issue> issues, string path, IEnumerable<Guid> ids)
    {
        var list = ids.ToList();
        if (list.Contains(Guid.Empty) || list.Distinct().Count() != list.Count)
            issues.Add(new(path, "对象标识为空或重复"));
    }

    private static void Text(List<Issue> issues, string path, string? value, int max, bool required = false)
    {
        if (value is null || value.Length > max || (required && string.IsNullOrWhiteSpace(value)))
            issues.Add(new(path, $"{(required ? "必填；" : "")}最多{max}字符"));
    }

    private static void Range(List<Issue> issues, string path, double value, double min, double max)
    {
        if (!double.IsFinite(value) || value < min || value > max)
            issues.Add(new(path, $"必须为{min}至{max}的有限数值"));
    }
}

using System.Globalization;
using System.Text;
using Avas.Core.Domain;

namespace Avas.Core.IO;

public sealed record ListenerPreview(List<Listener> Points, List<string> Errors)
{
    public bool Valid => Errors.Count == 0 && Points.Count > 0;
}

public static class ListenerCsv
{
    public const string Header = "Name,X_m,Y_m,Note";

    public static string Template() => Header + "\n车前5m,7.3,0,人工定义观察点\n左前点,5,4,人工定义观察点";

    public static ListenerPreview Parse(string text)
    {
        var points = new List<Listener>();
        var errors = new List<string>();
        if (text.Length > 100000)
            return new(points, ["观察点CSV最多100000字符"]);
        var lines = text.TrimStart('\uFEFF').Replace("\r", "").Split('\n');
        if (lines.Length == 0 || lines[0] != Header)
            errors.Add("第1行表头必须为 " + Header);
        for (var i = 1; i < lines.Length; i++)
        {
            if (string.IsNullOrWhiteSpace(lines[i]))
                continue;
            List<string> fields;
            try
            {
                fields = SplitLine(lines[i]);
            }
            catch (ArgumentException error)
            {
                errors.Add($"第{i + 1}行：{error.Message}");
                continue;
            }
            if (fields.Count != 4)
            {
                errors.Add($"第{i + 1}行：应有名称、X、Y、备注四列");
                continue;
            }
            if (string.IsNullOrWhiteSpace(fields[0]) || fields[0].Length > 100 || fields[3].Length > 2000)
            {
                errors.Add($"第{i + 1}行：名称必填且不超过100字符，备注不超过2000字符");
                continue;
            }
            if (!double.TryParse(fields[1], NumberStyles.Float, CultureInfo.InvariantCulture, out var x)
                || !double.TryParse(fields[2], NumberStyles.Float, CultureInfo.InvariantCulture, out var y)
                || !double.IsFinite(x) || !double.IsFinite(y) || Math.Abs(x) > 100 || Math.Abs(y) > 100)
            {
                errors.Add($"第{i + 1}行：X/Y必须是-100至100m的有限数值");
                continue;
            }
            if (points.Any(p => p.Name == fields[0]))
            {
                errors.Add($"第{i + 1}行：名称与前行重复，无法清楚区分观察点");
                continue;
            }
            points.Add(new Listener { Name = fields[0], X = x, Y = y, Note = fields[3] });
        }
        if (points.Count is < 1 or > 100)
            errors.Add("有效观察点必须为1至100个");
        return new(points, errors);
    }

    public static List<string> SplitLine(string line)
    {
        var fields = new List<string>();
        var current = new StringBuilder();
        var quoted = false;
        var closed = false;
        for (var i = 0; i < line.Length; i++)
        {
            var c = line[i];
            if (quoted)
            {
                if (c == '"')
                {
                    if (i + 1 < line.Length && line[i + 1] == '"')
                    {
                        current.Append('"');
                        i++;
                    }
                    else
                    {
                        quoted = false;
                        closed = true;
                    }
                }
                else
                    current.Append(c);
            }
            else if (c == ',')
            {
                fields.Add(current.ToString());
                current.Clear();
                closed = false;
            }
            else if (closed)
                throw new ArgumentException("结束引号之后只能接逗号或行尾");
            else if (c == '"')
            {
                if (current.Length != 0)
                    throw new ArgumentException("引号只能出现在字段开头");
                quoted = true;
            }
            else
                current.Append(c);
        }
        if (quoted)
            throw new ArgumentException("引号未闭合；不支持字段内换行");
        fields.Add(current.ToString());
        return fields;
    }

    public static byte[] Write(IEnumerable<Listener> points)
    {
        var text = new StringBuilder("\uFEFF" + Header + "\r\n");
        foreach (var point in points)
        {
            var name = point.Name.Replace("\r", " ").Replace("\n", " ");
            var note = point.Note.Replace("\r", " ").Replace("\n", " ");
            text.AppendLine(string.Join(",", Quote(name), point.X.ToString("R", CultureInfo.InvariantCulture),
                point.Y.ToString("R", CultureInfo.InvariantCulture), Quote(note)));
        }
        return Encoding.UTF8.GetBytes(text.ToString());
    }

    private static string Quote(string text) => "\"" + text.Replace("\"", "\"\"") + "\"";
}

using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Text.Json.Serialization;
using Avas.Core.Audio;
using Avas.Core.Domain;

namespace Avas.Core.IO;

public static class ProjectCodec
{
    public const int MaximumBytes = 80 * 1024 * 1024;
    private static readonly JsonSerializerOptions Options = new()
    {
        WriteIndented = true,
        PropertyNameCaseInsensitive = false,
        UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow,
        MaxDepth = 32,
        Converters = { new JsonStringEnumConverter() }
    };

    public static string Json<T>(T value) => JsonSerializer.Serialize(value, Options);

    public static T Clone<T>(T value)
    {
        return JsonSerializer.Deserialize<T>(Json(value), Options)
            ?? throw new ArgumentException("不能复制空对象");
    }

    public static byte[] Encode(Project project)
    {
        var issues = Validation.Check(project);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("\n", issues));
        var bytes = Encoding.UTF8.GetBytes(Json(project));
        if (bytes.Length > MaximumBytes)
            throw new ArgumentException("项目超过80MiB，请移除不需要的音频或快照");
        return bytes;
    }

    public static Project Decode(byte[] bytes)
    {
        if (bytes.Length > MaximumBytes)
            throw new ArgumentException("项目超过80MiB限制");
        Project project;
        try
        {
            using var document = JsonDocument.Parse(bytes);
            if (document.RootElement.ValueKind != JsonValueKind.Object
                || !document.RootElement.TryGetProperty("Schema", out var schema)
                || schema.ValueKind != JsonValueKind.Number || !schema.TryGetInt32(out var format) || format != 1)
                throw new ArgumentException("缺少Schema或版本不受支持；本软件仅支持格式1");
            ValidateShape(document.RootElement, typeof(Project), "项目");
            project = JsonSerializer.Deserialize<Project>(bytes, Options)
                ?? throw new ArgumentException("项目为空");
        }
        catch (JsonException error)
        {
            throw new ArgumentException($"项目JSON损坏，位置{error.Path}：{error.Message}");
        }
        var issues = Validation.Check(project);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("\n", issues));
        ValidateAssets(project);
        foreach (var snapshot in project.Snapshots)
        {
            var captured = new Project { Settings = snapshot.Settings, Speakers = snapshot.Speakers, Listeners = snapshot.Listeners };
            if (InputHash(captured) != snapshot.InputHash)
                throw new ArgumentException($"快照 {snapshot.Name} 的输入哈希不一致，不能作为可追溯方案打开");
        }
        return project;
    }

    {
        var input = new
        {
            Model = Software.Model,
            project.Settings,
            Speakers = project.Speakers.OrderBy(x => x.Id).ToArray(),
            Listeners = project.Listeners.OrderBy(x => x.Id).ToArray()
        };
        return Hash(Encoding.UTF8.GetBytes(Json(input)));
    }

            settings.SelectedHz,
            settings.AWeighted,

            settings.BackgroundOrigin,

            BackgroundAudioHash = settings.BackgroundAudio?.Sha256

    private static void ValidateShape(JsonElement element, Type type, string path)
    {
        if (element.ValueKind != JsonValueKind.Object)
            throw new ArgumentException(path + "必须为对象，不能为null");
        foreach (var property in type.GetProperties().Where(x => x.CanWrite))
        {
            var fieldPath = path + "." + property.Name;
            if (!element.TryGetProperty(property.Name, out var value))
                throw new ArgumentException(fieldPath + "缺失；格式1不自动补默认值");
            if (value.ValueKind == JsonValueKind.Null)
            {
                if (property.PropertyType == typeof(AudioAsset))
                    continue;
                throw new ArgumentException(fieldPath + "不能为null");
            }
            var fieldType = property.PropertyType;
            if (fieldType.IsGenericType && fieldType.GetGenericTypeDefinition() == typeof(List<>))
            {
                if (value.ValueKind != JsonValueKind.Array)
                    throw new ArgumentException(fieldPath + "必须为数组");
                var itemType = fieldType.GetGenericArguments()[0];
                var index = 0;
                foreach (var item in value.EnumerateArray())
                    ValidateShape(item, itemType, fieldPath + "[" + index++ + "]");
            }
            else if (fieldType == typeof(Settings) || fieldType == typeof(AudioAsset))
                ValidateShape(value, fieldType, fieldPath);
        }
    }

    private static void ValidateAssets(Project project)
    {
        var assets = project.Speakers.Select(x => x.Audio).Append(project.Settings.BackgroundAudio)
            .Concat(project.Snapshots.SelectMany(x => x.Speakers.Select(s => s.Audio).Append(x.Settings.BackgroundAudio)));
        foreach (var asset in assets.Where(x => x is not null).Cast<AudioAsset>())
        {
            byte[] data;
            try
            {
                data = Convert.FromBase64String(asset.Data);
            }
            catch (FormatException)
            {
                throw new ArgumentException($"音频 {asset.Name} 的内嵌数据不是有效Base64");
            }
            if (Hash(data) != asset.Sha256)
                throw new ArgumentException($"音频 {asset.Name} 校验和不一致，文件可能损坏");
            var decoded = WaveCodec.Read(data);
            if (decoded.Rate != asset.Rate || decoded.Channels != asset.Channels
                || Math.Abs(decoded.Duration - asset.Duration) > 0.001)
                throw new ArgumentException($"音频 {asset.Name} 的采样信息与数据不一致");
        }
    }
}

using Avas.Core.Domain;

namespace Avas.Core.IO;

public sealed record StoredProject(string Id, string Name, string FileName, DateTimeOffset Modified, long Size, string Hash, bool HasBackup, string? Error);
public sealed record SaveReceipt(string Path, string Hash, bool BackupCreated);

public sealed class ProjectStore
{
    private static readonly object Gate = new();
    public string DirectoryPath { get; }

    public ProjectStore(string path)
    {
        DirectoryPath = Path.GetFullPath(path);
        Directory.CreateDirectory(DirectoryPath);
    }

    public List<StoredProject> List()
    {
        lock (Gate)
        {
            return Directory.EnumerateFiles(DirectoryPath, "*.avasproj").Select(path =>
            {
                var info = new FileInfo(path);
                try
                {
                    var data = File.ReadAllBytes(path);
                    var project = ProjectCodec.Decode(data);
                    return new StoredProject(project.Id.ToString(), project.Name, info.Name,
                        info.LastWriteTimeUtc, info.Length, ProjectCodec.Hash(data), File.Exists(path + ".bak"), null);
                }
                catch (Exception error) when (error is ArgumentException or IOException or FormatException)
                {
                    return new StoredProject(Path.GetFileNameWithoutExtension(path), info.Name, info.Name,
                        info.LastWriteTimeUtc, info.Length, "", File.Exists(path + ".bak"), error.Message);
                }
            }).OrderByDescending(x => x.Modified).ToList();
        }
    }

    public (Project Project, string Hash) Open(Guid id, bool backup = false)
    {
        lock (Gate)
        {
            var path = FilePath(id) + (backup ? ".bak" : "");
            var data = File.ReadAllBytes(path);
            return (ProjectCodec.Decode(data), ProjectCodec.Hash(data));
        }
    }

    public SaveReceipt Save(Project project, string? expectedHash)
    {
        var bytes = ProjectCodec.Encode(project);
        lock (Gate)
        {
            var path = FilePath(project.Id);
            var exists = File.Exists(path);
            if (exists)
            {
                var currentHash = ProjectCodec.Hash(File.ReadAllBytes(path));
                if (expectedHash is null || currentHash != expectedHash)
                    throw new IOException("同标识项目已存在或被其他会话修改；请先打开磁盘版，或复制为独立项目后保存。");
            }
            var temporary = path + ".tmp-" + Guid.NewGuid().ToString("N");
            try
            {
                using (var stream = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None))
                {
                    stream.Write(bytes);
                    stream.Flush(true);
                }
                if (exists)
                    File.Copy(path, path + ".bak", true);
                File.Move(temporary, path, true);
            }
            finally
            {
                if (File.Exists(temporary))
                    File.Delete(temporary);
            }
            return new SaveReceipt(path, ProjectCodec.Hash(bytes), exists);
        }
    }

    private string FilePath(Guid id) => Path.Combine(DirectoryPath, id.ToString("N") + ".avasproj");
}

using System.Globalization;
using System.Text;
using Avas.Core.Domain;

namespace Avas.Core.IO;

public sealed class SpectrumPreview
{
    public List<BandInput> Bands { get; init; } = [];
    public List<string> Errors { get; init; } = [];
    public bool Valid => Errors.Count == 0 && Bands.Count == 7;
}

public static class SpectrumCsv
{
    public const string Header = "FrequencyHz,LevelDb,CorrectionDb";

    public static string Template(bool background = false)
    {
        return Header + "\n" + string.Join("\n", Software.Frequencies.Select(hz => $"{hz},{(background ? 35 : 60)},0"));
    }

    public static SpectrumPreview Parse(string text, bool background = false)
    {
        var errors = new List<string>();
        var bands = new List<BandInput>();
        if (text.Length > 20000)
            return new SpectrumPreview { Errors = ["文件超过20000字符限制"] };
        var lines = text.TrimStart('\uFEFF').Replace("\r", "").Split('\n');
        if (lines.Length == 0 || lines[0] != Header)
            errors.Add("第1行表头必须为 " + Header + "；使用UTF-8与英文逗号");
        for (var i = 1; i < lines.Length; i++)
        {
            if (string.IsNullOrWhiteSpace(lines[i]))
                continue;
            var fields = lines[i].Split(',');
            if (fields.Length != 3)
            {
                errors.Add($"第{i + 1}行必须恰好三列，不支持千分位或引号");
                continue;
            }
            if (!int.TryParse(fields[0], NumberStyles.Integer, CultureInfo.InvariantCulture, out var hz)
                || !Software.Frequencies.Contains(hz))
            {
                errors.Add($"第{i + 1}行频率无效，仅支持七个倍频程中心频率");
                continue;
            }
            if (!double.TryParse(fields[1], NumberStyles.Float, CultureInfo.InvariantCulture, out var level)
                || !double.IsFinite(level) || level is < -60 or > 140)
            {
                errors.Add($"第{i + 1}行LevelDb必须为-60至140的有限数值");
                continue;
            }
            if (!double.TryParse(fields[2], NumberStyles.Float, CultureInfo.InvariantCulture, out var correction)
                || !double.IsFinite(correction) || correction is < -40 or > 40 || (background && correction != 0))
            {
                errors.Add($"第{i + 1}行CorrectionDb无效；声源范围-40至40，背景必须为0");
                continue;
            }
            if (bands.Any(x => x.Hz == hz))
            {
                errors.Add($"第{i + 1}行频率{hz}重复");
                continue;
            }
            bands.Add(new BandInput { Hz = hz, Level = level, Correction = correction });
        }
        foreach (var hz in Software.Frequencies.Except(bands.Select(x => x.Hz)))
            errors.Add($"缺少{hz}Hz，不能自动补零");
        return new SpectrumPreview { Bands = bands.OrderBy(x => x.Hz).ToList(), Errors = errors };
    }

    public static byte[] Write(IEnumerable<BandInput> bands)
    {
        var rows = bands.OrderBy(x => x.Hz).Select(x =>
            FormattableString.Invariant($"{x.Hz},{x.Level:R},{x.Correction:R}"));
        return Encoding.UTF8.GetBytes("\uFEFF" + Header + "\r\n" + string.Join("\r\n", rows));
    }

    public static string Cell(string? value)
    {
        var safe = value ?? "";
        if (safe.Length > 0 && "=+-@\t\r".Contains(safe[0]))
            safe = "'" + safe;
        return "\"" + safe.Replace("\"", "\"\"") + "\"";
    }
}

    public FieldView()
    {
        MinWidth = 250;
        MinHeight = 220;
        MouseMove += Hover;
        MouseLeftButtonDown += Pick;
        AutomationPropertiesName();
    }

    private void AutomationPropertiesName() => System.Windows.Automation.AutomationProperties.SetName(this, "车辆周围自由场计算图，X正向为车前，Y正向为车左");
    private static SolidColorBrush Brush(string color)
    {
        var value = new SolidColorBrush((Color)ColorConverter.ConvertFromString(color));
        value.Freeze();
        return value;
    }
    private Point Screen(double x, double y)
    {
        var settings = Project!.Settings;
        return new Point(plot.Left + (x - settings.MinX) * scale, plot.Bottom - (y - settings.MinY) * scale);
    }
    private Point World(Point point)
    {
        var settings = Project!.Settings;
        return new Point(settings.MinX + (point.X - plot.Left) / scale, settings.MinY + (plot.Bottom - point.Y) / scale);
    }
    private void Text(DrawingContext dc, string text, Point at, double size = 12, Brush? color = null)
    {
        var formatted = new FormattedText(text, CultureInfo.CurrentCulture, FlowDirection.LeftToRight, typeface, size, color ?? Ink, VisualTreeHelper.GetDpi(this).PixelsPerDip);
        dc.DrawText(formatted, at);
    }

    protected override void OnRender(DrawingContext dc)
    {
        base.OnRender(dc);
        dc.DrawRectangle(Brushes.White, null, new Rect(RenderSize));
        if (Project is null || ActualWidth < 100 || ActualHeight < 100)
            return;
        var s = Project.Settings;
        scale = Math.Min((ActualWidth - 90) / (s.MaxX - s.MinX), (ActualHeight - 145) / (s.MaxY - s.MinY));
        if (!double.IsFinite(scale) || scale <= 0)
            return;
        var width = (s.MaxX - s.MinX) * scale;
        var height = (s.MaxY - s.MinY) * scale;
        plot = new Rect((ActualWidth - width) / 2 + 10, 62, width, height);
        Text(dc, Result?.Metric ?? "已应用布局 · 无当前有效计算结果", new Point(14, 12), 12);
        dc.DrawRectangle(Brush("#F4F7F1"), Outline, plot);
        dc.PushClip(new RectangleGeometry(plot));
        DrawField(dc);
        DrawGrid(dc);
        DrawObjects(dc);
        dc.Pop();
        DrawAxes(dc);
        DrawLegend(dc);
        if (Result is not null)
            Text(dc, $"有效节点 {Result.Grid.Count} · 工程目标覆盖 {Result.Coverage:F1}%", new Point(14, 33), 10);
    }

    private void DrawField(DrawingContext dc)
    {
        if (Result is null || Project is null)
            return;
        var low = Math.Floor(Result.Minimum / 5) * 5;
        var high = Math.Max(low + 5, Math.Ceiling(Result.Maximum / 5) * 5);
        var palette = Enumerable.Range(0, 128).Select(i => Brush(ChartRenderer.Color(low + (high - low) * i / 127, low, high))).ToArray();
        var yes = Brush("#4E9F76");
        var no = Brush("#E4BD75");
        var half = Project.Settings.Step * scale / 2;
        foreach (var point in Result.Grid)
        {
            var screen = Screen(point.X, point.Y);
            var index = (int)Math.Clamp((point.Signal - low) / (high - low) * 127, 0, 127);
            var color = CoverageOnly ? point.Covered ? yes : no : palette[index];
            dc.DrawRectangle(color, null, new Rect(screen.X - half, screen.Y - half, half * 2 + .5, half * 2 + .5));
        }
    }

    private void DrawGrid(DrawingContext dc)
    {
        var s = Project!.Settings;
        var tick = TickSize(Math.Max(s.MaxX - s.MinX, s.MaxY - s.MinY));
        for (var x = Math.Ceiling(s.MinX / tick) * tick; x <= s.MaxX; x += tick)
            dc.DrawLine(GridPen, Screen(x, s.MinY), Screen(x, s.MaxY));
        for (var y = Math.Ceiling(s.MinY / tick) * tick; y <= s.MaxY; y += tick)
            dc.DrawLine(GridPen, Screen(s.MinX, y), Screen(s.MaxX, y));
    }
    private static double TickSize(double extent)
    {
        var raw = extent / 7;
        var exponent = Math.Pow(10, Math.Floor(Math.Log10(raw)));
        return Math.Ceiling(raw / exponent) * exponent;
    }
    private void DrawObjects(DrawingContext dc)
    {
        var project = Project!;
        var s = project.Settings;
        var corner = Screen(-s.VehicleLength / 2, s.VehicleWidth / 2);
        var vehicle = new Rect(corner, new Size(s.VehicleLength * scale, s.VehicleWidth * scale));
        dc.DrawRoundedRectangle(Brush("#E8EBE7"), Outline, vehicle, 9, 9);
        Text(dc, "车前 →", new Point(vehicle.Left + 8, vehicle.Top + Math.Max(0, vehicle.Height / 2 - 8)), 11);
        foreach (var speaker in project.Speakers)
        {
            var point = Screen(speaker.X, speaker.Y);
            var angle = speaker.Heading * Math.PI / 180;
            var end = new Point(point.X + 28 * Math.Cos(angle), point.Y - 28 * Math.Sin(angle));
            dc.DrawLine(new Pen(Ink, 3), point, end);
            dc.DrawEllipse(speaker.Enabled ? Brushes.White : Brushes.Gray, Outline, point, 6, 6);
        }
        foreach (var listener in project.Listeners)
        {
            var point = Screen(listener.X, listener.Y);
            var selected = listener.Id == SelectedListenerId;
            dc.DrawEllipse(selected ? Brush("#F2DA58") : Brushes.White, new Pen(Ink, selected ? 3 : 1.5), point, selected ? 7 : 4, selected ? 7 : 4);
            Text(dc, listener.Name, new Point(point.X + 9, point.Y - 20), 11);
        }
    }
    private void DrawAxes(DrawingContext dc)
    {
        var s = Project!.Settings;
        var tick = TickSize(Math.Max(s.MaxX - s.MinX, s.MaxY - s.MinY));
        for (var x = Math.Ceiling(s.MinX / tick) * tick; x <= s.MaxX; x += tick)
            Text(dc, x.ToString("0.#", CultureInfo.InvariantCulture), new Point(Screen(x, s.MinY).X - 7, plot.Bottom + 6), 10);
        for (var y = Math.Ceiling(s.MinY / tick) * tick; y <= s.MaxY; y += tick)
            Text(dc, y.ToString("0.#", CultureInfo.InvariantCulture), new Point(plot.Left - 34, Screen(s.MinX, y).Y - 7), 10);
        Text(dc, "X / m → 车前     Y / m ↑ 车左", new Point(plot.Left, plot.Bottom + 25), 11);
        Text(dc, "轮廓仅示意，不模拟遮挡；网格节点覆盖不代表人群可听率。", new Point(14, ActualHeight - 22), 10);
    }
    private void DrawLegend(DrawingContext dc)
    {
        if (Result is null)
            return;
        var y = plot.Bottom + 45;
        if (CoverageOnly)
        {
            dc.DrawRectangle(Brush("#4E9F76"), null, new Rect(14, y, 14, 12));
            Text(dc, "满足工程目标", new Point(34, y - 2), 10);
            dc.DrawRectangle(Brush("#E4BD75"), null, new Rect(155, y, 14, 12));
            Text(dc, "未满足", new Point(175, y - 2), 10);
        }
        else
        {
            var low = Math.Floor(Result.Minimum / 5) * 5;
            var high = Math.Max(low + 5, Math.Ceiling(Result.Maximum / 5) * 5);
            for (var i = 0; i < 100; i++)
                dc.DrawRectangle(Brush(ChartRenderer.Color(low + (high - low) * i / 99, low, high)), null, new Rect(14 + i, y, 1.5, 10));
            Text(dc, $"{low:F0}—{high:F0} dB", new Point(124, y - 2), 10);
        }
    }
    private void Pick(object sender, MouseButtonEventArgs e)
    {
        if (Project is null || scale <= 0)
            return;
        var mouse = e.GetPosition(this);
        var nearest = Project.Listeners.OrderBy(p => (Screen(p.X, p.Y) - mouse).LengthSquared).FirstOrDefault();
        if (nearest is not null && (Screen(nearest.X, nearest.Y) - mouse).Length <= 16)
            PointPicked?.Invoke(this, nearest.Id);
    }
    private void Hover(object sender, MouseEventArgs e)
    {
        if (Project is null || scale <= 0 || !plot.Contains(e.GetPosition(this)))
        {
            ToolTip = null;
            return;
        }
        var p = World(e.GetPosition(this));
        var result = Result?.Grid.OrderBy(q => Math.Pow(q.X - p.X, 2) + Math.Pow(q.Y - p.Y, 2)).FirstOrDefault();
        ToolTip = result is null ? $"X={p.X:F2} m，Y={p.Y:F2} m；当前无有效结果"
            : $"最近有效节点 ({result.X:F2}, {result.Y:F2}) m\n警示声 {result.Signal:F2} dB，背景差 {result.Margin:F2} dB\n{(result.Covered ? "满足工程目标" : "未满足工程目标")}；距离截断 {result.NearFieldClamped}";
    }
}

using System.IO;
using System.Media;
using System.Windows;
using Microsoft.Win32;
using Avas.Workspace;

namespace Avas.Desktop;

public sealed class DesktopServices : IDesktopServices, IDisposable
{
    private readonly string temporaryDirectory = Path.Combine(Path.GetTempPath(), "AvasCoverage-" + Guid.NewGuid().ToString("N"));
    private SoundPlayer? player;
    private int audioRevision;
    public string LastReadName { get; private set; } = "";

    public bool Confirm(string title, string message) => MessageBox.Show(Application.Current.MainWindow, message, title,
        MessageBoxButton.OKCancel, MessageBoxImage.Question) == MessageBoxResult.OK;

    public byte[]? ReadFile(string kind)
    {
        var dialog = new OpenFileDialog
        {
            Title = "选择本机输入文件",
            CheckFileExists = true,
            Multiselect = false,
            Filter = kind switch
            {
                "project" => "AVAS 项目 (*.avasproj)|*.avasproj|项目备份 (*.bak)|*.bak",
                "wave" => "PCM16 WAV (*.wav)|*.wav",
                _ => "UTF-8 CSV (*.csv)|*.csv"
            }
        };
        if (dialog.ShowDialog(Application.Current.MainWindow) != true)
            return null;
        var size = new FileInfo(dialog.FileName).Length;
        var maximum = kind switch { "project" => 80L * 1024 * 1024, "wave" => 10L * 1024 * 1024, _ => 1024L * 1024 };
        if (size > maximum)
            throw new ArgumentException($"所选文件超过 {maximum / (1024 * 1024)} MiB 限制。");
        LastReadName = Path.GetFileName(dialog.FileName);
        return File.ReadAllBytes(dialog.FileName);
    }

    public string WriteExport(byte[] bytes, string suggestedName)
    {
        var extension = Path.GetExtension(suggestedName);
        var dialog = new SaveFileDialog
        {
            Title = "选择成果保存位置",
            FileName = suggestedName,
            AddExtension = true,
            DefaultExt = extension,
            OverwritePrompt = true,
            Filter = $"{extension} 文件|*{extension}"
        };
        if (dialog.ShowDialog(Application.Current.MainWindow) != true)
            throw new OperationCanceledException();
        var path = dialog.FileName;
        var temporary = path + ".tmp-" + Guid.NewGuid().ToString("N");
        try
        {
            using (var output = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None))
            {
                output.Write(bytes);
                output.Flush(true);
            }
            File.Move(temporary, path, true);
        }
        finally
        {
            if (File.Exists(temporary))
                File.Delete(temporary);
        }
        return path;
    }

    public string WriteTemporaryAudio(byte[] bytes)
    {
        StopAudio();
        Directory.CreateDirectory(temporaryDirectory);
        foreach (var prior in Directory.EnumerateFiles(temporaryDirectory, "*.wav"))
            File.Delete(prior);
        var path = Path.Combine(temporaryDirectory, $"audition-{++audioRevision}.wav");
        File.WriteAllBytes(path, bytes);
        return path;
    }

    public void PlayAudio(string path)
    {
        StopAudio();
        player = new SoundPlayer(path);
        player.Load();
        player.Play();
    }

    public void StopAudio()
    {
        player?.Stop();
        player?.Dispose();
        player = null;
    }

    public void Dispose()
    {
        StopAudio();
        if (Directory.Exists(temporaryDirectory))
        {
            try { Directory.Delete(temporaryDirectory, true); }
            catch (IOException) { }
            catch (UnauthorizedAccessException) { }
        }
    }
}

using System.ComponentModel;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Data;
using System.Windows.Input;
using Avas.Core.Analysis;
using Avas.Workspace;

namespace Avas.Desktop;

public partial class MainWindow : Window
{
    private readonly WorkspaceViewModel workspace;
    private readonly HashSet<ValidationError> bindingErrors = [];

    public MainWindow(WorkspaceViewModel workspace)
    {
        this.workspace = workspace;
        InitializeComponent();
        DataContext = workspace;
        AddHandler(Validation.ErrorEvent, new EventHandler<ValidationErrorEventArgs>(BindingError));
        AddHandler(Binding.SourceUpdatedEvent, new EventHandler<DataTransferEventArgs>(EditorSourceUpdated));
        CommandManager.AddPreviewExecutedHandler(this, BeforeCommand);
        PreviewKeyDown += BeforeShortcut;
    }

    private void EditorSourceUpdated(object? sender, DataTransferEventArgs e)
    {
        workspace.MarkDraft();
    }

    private void BindingError(object? sender, ValidationErrorEventArgs e)
    {
        if (e.Action == ValidationErrorEventAction.Added)
            bindingErrors.Add(e.Error);
        else
            bindingErrors.Remove(e.Error);
        workspace.BindingErrorCount = bindingErrors.Count;
        workspace.NotifyEditorState();
    }

    private void BeforeShortcut(object sender, KeyEventArgs e)
    {
        if (e.Key == Key.F5 || (Keyboard.Modifiers == ModifierKeys.Control && e.Key is Key.S or Key.O or Key.N))
            FlushEditor();
    }

    private void BeforeCommand(object sender, ExecutedRoutedEventArgs e) => FlushEditor();

    private void FlushEditor()
    {
        if (Keyboard.FocusedElement is TextBox text)
            text.GetBindingExpression(TextBox.TextProperty)?.UpdateSource();
        if (Keyboard.FocusedElement is FrameworkElement element)
        {
            var parent = System.Windows.Media.VisualTreeHelper.GetParent(element);
            while (parent is not null && parent is not DataGrid)
                parent = System.Windows.Media.VisualTreeHelper.GetParent(parent);
            if (parent is DataGrid grid)
            {
                grid.CommitEdit(DataGridEditingUnit.Cell, true);
                grid.CommitEdit(DataGridEditingUnit.Row, true);
            }
        }
    }

    private void WindowClosing(object? sender, CancelEventArgs e)
    {
        FlushEditor();
        e.Cancel = !workspace.CanClose();
        if (!e.Cancel)
            workspace.DisposeServices();
    }

    private void CoverageChanged(object sender, RoutedEventArgs e) => Field?.InvalidateVisual();

    private void PointPicked(object? sender, Guid id)
    {
        workspace.Execute(() =>
        {
            if (workspace.HasDraft || workspace.BindingErrorCount > 0)
                throw new ArgumentException("请先应用或放弃字段编辑，再选择观察点。");
            workspace.SelectListener(id);
        });
    }

    private void ResultSelectionChanged(object sender, SelectionChangedEventArgs e)
    {
        if (((DataGrid)sender).SelectedItem is PointResult { ListenerId: { } id })
            PointPicked(sender, id);
    }

    private void ShowHelp(object sender, RoutedEventArgs e)
    {
        var window = new HelpWindow { Owner = this };
        window.ShowDialog();
    }
}

using System.ComponentModel;

public interface IDesktopServices
{
    bool Confirm(string title, string message);
    byte[]? ReadFile(string kind);
    string LastReadName { get; }
    string WriteExport(byte[] bytes, string suggestedName);
    string WriteTemporaryAudio(byte[] bytes);
    void PlayAudio(string path);
    void StopAudio();
}

public sealed class ActionCommand : ICommand
{
    private readonly Func<object?, Task> action;
    private readonly Func<bool> available;
    public event EventHandler? CanExecuteChanged;

    public ActionCommand(Func<object?, Task> action, Func<bool> available)
    {
        this.action = action;
        this.available = available;
    }

    public bool CanExecute(object? parameter) => available();
    public async void Execute(object? parameter)
    {
        if (available())
            await action(parameter);
    }
    public void Refresh() => CanExecuteChanged?.Invoke(this, EventArgs.Empty);
}

public sealed partial class WorkspaceViewModel
{
    private readonly Dictionary<string, ActionCommand> commands = [];
    public IReadOnlyDictionary<string, ActionCommand> Commands => commands;
    public string[] Sections { get; } = ["项目", "扬声器", "背景与网格", "观察点", "方案比较", "扫描与剖面", "音频试听", "成果与复核"];
    public Origin[] Origins { get; } = Enum.GetValues<Origin>();
    public ScanParameter[] ScanParameters { get; } = Enum.GetValues<ScanParameter>();
    public int[] MetricBands { get; } = [0, .. Software.Frequencies];
    public string SelectedSection
    {
        get => Tab;
        set
        {
            if (value == Tab)
                return;
            Execute(() =>
            {
                RequireApplied();
                Tab = value;
            });
        }
    }
    public Speaker? SelectedSpeaker
    {
        get => SpeakerDraft;
        set
        {
            if (value is null || value.Id == SelectedSpeakerId)
                return;
            Execute(() =>
            {
                RequireApplied();
                SelectSpeaker(value.Id);
            });
        }
    }
    public Listener? SelectedListener
    {
        get => ListenerDraft;
        set
        {
            if (value is null || value.Id == SelectedListenerId)
                return;
            Execute(() =>
            {
                RequireApplied();
                SelectListener(value.Id);
            });

    private void ReadText(bool listeners)
    {
        RequireApplied();
        var bytes = files.ReadFile("csv");
        if (bytes is null)
            return;
        if (bytes.Length > 1024 * 1024)
            throw new ArgumentException("CSV超过1MiB，请缩小输入文件");
        var text = new System.Text.UTF8Encoding(false, true).GetString(bytes);
        if (listeners)
            ListenerText = text;
        else
            CsvText = text;
        Status = "已读取UTF-8 CSV文本，尚未应用；请预检后确认。";
    }

    private void Add(string name, Action<object?> action)
    {
        commands.Add(name, new ActionCommand(p =>
        {
            Execute(() => action(p));
            return Task.CompletedTask;
        }, () => !Busy));
    }

    private void AddAsync(string name, Func<Task> action)
    {
        commands.Add(name, new ActionCommand(async _ =>
        {
            try
            {
                RequireValidBindings();
                await action();
            }
            catch (Exception error) when (error is ArgumentException or IOException or UnauthorizedAccessException or InvalidOperationException or FormatException or OverflowException)
            {
                Failure(error.Message);
            }
        }, () => !Busy));
    }

    private async Task RunWorkAsync<T>(Func<T> compute, Action<T> accept)
    {
        if (Busy)
            return;
        Busy = true;
        Status = "正在处理，请等待…";
        Errors = [];
        Notify();
        try
        {
            var result = await Task.Run(compute);
            accept(result);
        }
        catch (Exception error) when (error is ArgumentException or IOException or UnauthorizedAccessException or InvalidOperationException or FormatException or OverflowException)
        {
            Failure(error.Message);
        }
        finally
        {
            Busy = false;
            Notify();
        }
    }

    private void RequireValidBindings()
    {
        if (BindingErrorCount > 0)
            throw new ArgumentException("存在格式错误，请先修正红框字段；数字不能填入文字或留空。");
    }

    public bool CanClose()
    {
        if (Busy)
        {
            Status = "任务进行中，请等待结束再关闭。";
            Notify();
            return false;
        }
        return !(Dirty || HasDraft || BindingErrorCount > 0) || files.Confirm("未保存内容", "存在未保存内容。放弃并关闭？如需保留，请取消，应用编辑后保存到项目库。");
    }

    private bool ConfirmReplacement()
    {
        return !(Dirty || HasDraft || BindingErrorCount > 0) || files.Confirm("替换当前项目", "存在未保存内容。放弃并替换当前项目？如需保留，请先取消并保存。");
    }

    private void ReplaceSession(bool demo)
    {
        if (ConfirmReplacement())
        {
            BindingErrorCount = 0;
            New(demo);
        }
    }

    private void OpenStored(bool backup)
    {
        var item = SelectedStored ?? throw new ArgumentException("请先选择项目库条目");
        if (!Guid.TryParse(item.Id, out var id))
            throw new ArgumentException("项目库条目标识无效，请导入完整备份文件");
        if (ConfirmReplacement())
        {
            BindingErrorCount = 0;
            Open(id, backup);
        }
    }
}

            source.Name = source.Name.Length < 96 ? source.Name + "副本" : "扬声器副本";
            candidate.Speakers.Add(source);
            SelectedSpeakerId = source.Id;
            Commit(candidate, "复制扬声器 " + source.Name);
            Status = "扬声器已复制，位置暂时相同；请调整后重新计算。";
        });
    }

    public void AddRing(double radius, int count)
    {
        Execute(() =>
        {
            RequireApplied();
            RequireEditable();
            if (!double.IsFinite(radius) || radius is < 1 or > 50 || count is < 4 or > 36)
                throw new ArgumentException("环形观察点半径1至50m，数量4至36");
            if (Project.Listeners.Count + count > 100)
                throw new ArgumentException("添加后观察点数量超过100");
            var candidate = ProjectCodec.Clone(Project);
            for (var i = 0; i < count; i++)
            {
                var angle = 2 * Math.PI * i / count;
                candidate.Listeners.Add(new Listener
                {
                    Name = $"环形{radius:g}m-{i + 1}",
                    X = Math.Round(radius * Math.Cos(angle), 5),
                    Y = Math.Round(radius * Math.Sin(angle), 5),
                    Note = "以车辆中心为圆心等角生成；非标准测点布置"
                });
            }
            Commit(candidate, $"生成{count}个半径{radius}m环形观察点");
            Status = "环形观察点已生成；这是自定义几何点集，不是标准测点。";
        });
    }

    public void CaptureSnapshot()
    {
        Execute(() =>
        {
            RequireApplied();
            RequireEditable();
            if (Stale)
                throw new ArgumentException("请先计算当前输入，再保存方案快照");
            if (Project.Snapshots.Count >= 20)
                throw new ArgumentException("最多20个快照，请删除不需要的快照");
            var candidate = ProjectCodec.Clone(Project);
            var snapshot = ScenarioEngine.Capture(Project, SnapshotName);
            candidate.Snapshots.Add(snapshot);
            Commit(candidate, "保存方案快照 " + snapshot.Name);
            if (string.IsNullOrEmpty(BaselineId))
                BaselineId = snapshot.Id.ToString();
            else
                TargetId = snapshot.Id.ToString();
            Status = "已保存独立方案快照；快照随项目文件内嵌保存。";
        });
    }

    public void RestoreSnapshot(Guid id)
    {
        Execute(() =>
        {
            RequireApplied();
            RequireEditable();
            var snapshot = Project.Snapshots.Single(x => x.Id == id);
            var restored = ScenarioEngine.Restore(snapshot);
            var candidate = ProjectCodec.Clone(Project);
            candidate.Settings = restored.Settings;
            candidate.Speakers = restored.Speakers;
            candidate.Listeners = restored.Listeners;
            Commit(candidate, "恢复快照输入 " + snapshot.Name);
            Status = "已恢复快照的布局、频谱、背景与观察点，请重新计算。";
        });
    }

    public void DeleteSnapshot(Guid id)
    {
        Execute(() =>
        {
            RequireApplied();
            RequireEditable();
            var candidate = ProjectCodec.Clone(Project);
            var snapshot = candidate.Snapshots.Single(x => x.Id == id);
            candidate.Snapshots.Remove(snapshot);
            Commit(candidate, "删除快照 " + snapshot.Name);
            if (BaselineId == id.ToString())
                BaselineId = "";
            if (TargetId == id.ToString())
                TargetId = "";
            Status = "快照已删除。";
        });
    }

    public void Compare()
    {
        Execute(() =>
        {
            RequireApplied();
            if (!Guid.TryParse(BaselineId, out var baselineId) || !Guid.TryParse(TargetId, out var targetId))
                throw new ArgumentException("请选择基线与目标两个快照");
            if (baselineId == targetId)
                throw new ArgumentException("基线与目标应为两个不同快照");
            Comparison = ScenarioEngine.Compare(Project.Snapshots.Single(x => x.Id == baselineId), Project.Snapshots.Single(x => x.Id == targetId));
            Status = Comparison.Comparable ? "比较完成；差值=目标减基线，覆盖差为百分点。" : "比较已阻断，请查看不一致条件。";
        });
    }

    public async Task RunScanAsync()
    {
        RequireApplied();
        var input = ProjectCodec.Clone(Project);
        var request = ProjectCodec.Clone(ScanRequest);
        await RunWorkAsync(() => ScanEngine.Run(input, request), result =>
        {
            Scan = result;
            Status = $"扫描完成，共{result.Rows.Count}组；选择一行后可应用该值。";
        });
    }

    public void ApplyScan(double value)
    {
        Execute(() =>
        {
            RequireApplied();
            RequireEditable();
            if (Scan is null || Scan.InputHash != ProjectCodec.InputHash(Project))
                throw new ArgumentException("扫描结果已过期，请重新扫描");
            var candidate = ProjectCodec.Clone(Project);
            var source = candidate.Speakers.Single(x => x.Id == Scan.SpeakerId);
            ScanEngine.Apply(source, Scan.Parameter, value);
            Commit(candidate, $"应用扫描取值 {ScanEngine.Label(Scan.Parameter)}={value}");
            Status = "已应用扫描值，请重新计算。";
        });
    }
}

    public void New(bool demo)

    public void Undo()
    {
        Execute(() =>
        {
            RequireApplied();
            RequireEditable();
            if (undo.Count == 0)
                return;
            redo.Push(ProjectCodec.Clone(Project));
            var prior = undo.Pop();
            PreserveAudit(prior, "撤销上一步");
            Project = prior;
            ResetAfterMutation();
            Status = "已撤销；审计保留，结果需重新核对。";
        });
    }

    public void Redo()
    {
        Execute(() =>
        {
            RequireApplied();
            RequireEditable();
            if (redo.Count == 0)
                return;
            undo.Push(ProjectCodec.Clone(Project));
            var next = redo.Pop();
            PreserveAudit(next, "重做上一步");
            Project = next;
            ResetAfterMutation();
            Status = "已重做。";
        });
    }

    private void PreserveAudit(Project candidate, string action)
    {
        candidate.Revision = Project.Revision + 1;
        candidate.Audit = Project.Audit.TakeLast(999).ToList();
        candidate.Audit.Add(new AuditEntry { Revision = candidate.Revision, Action = action });
    }

    private void Commit(Project candidate, string action)
    {
        var issues = Validation.Check(candidate);
        if (issues.Count > 0)
            throw new ArgumentException(string.Join("\n", issues));
        undo.Push(ProjectCodec.Clone(Project));
        if (undo.Count > 10)
        {
            var keep = undo.Take(10).Reverse().ToArray();
            undo.Clear();
            foreach (var item in keep)
                undo.Push(item);
        }
        redo.Clear();
        PreserveAudit(candidate, action);
        Project = candidate;
        ResetAfterMutation();
    }

    private void ResetAfterMutation()
    {
        ResetDraft();
        if (!Project.Speakers.Any(x => x.Id == SelectedSpeakerId))
            SelectedSpeakerId = Project.Speakers.FirstOrDefault()?.Id;
        if (!Project.Listeners.Any(x => x.Id == SelectedListenerId))
            SelectedListenerId = Project.Listeners.FirstOrDefault()?.Id;
        ScanRequest.SpeakerId = SelectedSpeakerId ?? Guid.Empty;
        Preview = null;
        AudioSpectrum = null;
        Comparison = null;
        Scan = null;
        Transect = null;
        ListenerPreview = null;
        ClearAudio();
        ClearDownload();
    }

    private void RequireEditable()
    {
        if (Project.Archived)
            throw new ArgumentException("当前项目为归档只读，请先在项目页解除归档");
    }

    private void RequireApplied()
    {
        RequireValidBindings();
        if (HasDraft)
            throw new ArgumentException("存在未应用字段，请点击“应用编辑”或“放弃编辑”后继续");
    }

    public void Execute(Action action)
    {
        Errors = [];
        try
        {
            if (Busy)
                throw new ArgumentException("任务正在运行，请等待完成");
            action();
        }
        catch (OperationCanceledException)
        {
            Status = "已取消文件操作，原有输入保留。";
        }
        catch (Exception error) when (error is ArgumentException or IOException or UnauthorizedAccessException or InvalidOperationException or FormatException or OverflowException)
        {
            Errors = error.Message.Split('\n').ToList();
            Status = "操作未完成，原有输入保留；请按提示修正。";
        }
        Notify();
    }

    public void Failure(string message)
    {
        Errors = [message];
        Status = "操作未完成，原有输入保留。";
        Notify();
    }

    private void ClearAudio()
    {
        Audition = null;
        AudioUrl = null;
    }

    private void ClearDownload()
    {
        LastDownload = null;
        LastFileName = null;
    }
