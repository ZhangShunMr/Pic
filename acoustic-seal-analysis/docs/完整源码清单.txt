0029  def energy_mean(levels):
0030      """Equal-weight energetic mean, stable across the allowed level range."""
0031      if not levels:
0032          return None
0033      peak = max(levels)
0034      scaled = math.fsum(10 ** ((level - peak) / 10) for level in levels)
0035      return peak + 10 * math.log10(scaled / len(levels))

0038  def energy_sum(levels):
0039      """Sum of nonoverlapping band energies; never a sum of effect differences."""
0040      if not levels:
0041          return None
0042      peak = max(levels)
0043      return peak + 10 * math.log10(math.fsum(10 ** ((level - peak) / 10) for level in levels))

0069  def paired_summary(differences, independent=True):
0070      count = len(differences)
0071      mean = statistics.mean(differences) if count else None
0072      sd = statistics.stdev(differences) if count > 1 else None
0073      result = {
0074          "n": count,
0075          "meanDifference": mean,
0076          "sdDifference": sd,
0077          "ci95": None,
0078          "ciNote": "n < 2，不生成置信区间",
0079      }
0080      if count > 1 and not independent:
0081          result["ciNote"] = "配对复用试验，独立性不满足，不生成置信区间"
0082      elif count > 1:
0083          # For df > 30 use the conservative df=30 critical value explicitly.
0084          critical = T975[min(count - 1, 30) - 1]
0085          half = critical * sd / math.sqrt(count)
0086          result["ci95"] = [mean - half, mean + half]
0087          result["ciNote"] = (
0088              "基于独立配对差近似正态的双侧 95% Student t 区间；"
0089              + ("自由度大于30时保守使用df=30" if count > 31 else f"df={count - 1}")
0090              + "；未计仪器系统不确定性"
0091          )
0092      return result

0095  def compatible(left, right, parameters):
0096      reasons = []
0097      for key, label in CONDITION_TEXT.items():
0098          a = left[key]
0099          b = right[key]
0100          if not a or not b:
0101              reasons.append(f"{label}缺失，无法确认可比性")
0102          elif a != b:
0103              reasons.append(f"{label}不同：{a} / {b}")
0104      tolerances = {
0105          "speed": ("speedTolerance", "车速", "km/h"),
0106          "rpm": ("rpmTolerance", "转速", "r/min"),
0107          "temperature": ("temperatureTolerance", "温度", "°C"),
0108          "duration": ("durationTolerance", "积分时长", "s"),
0109      }
0110      for field, (parameter, label, unit) in tolerances.items():
0111          a = left[field]
0112          b = right[field]
0113          if a is None or b is None:
0114              reasons.append(f"{label}缺失，无法确认可比性")
0115          elif abs(a - b) > parameters[parameter] + 1e-10:
0116              reasons.append(f"{label}差 {abs(a - b):.6g} {unit} 超过容差 {parameters[parameter]:.6g}")
0117      if left["independentBands"] != right["independentBands"]:
0118          reasons.append("互不重叠频带声明不同")
0119      return reasons

0140  def compare_pair(pair, trial_map, condition_map, parameters):
0141      baseline = trial_map[pair["baselineId"]]
0142      target = trial_map[pair["targetId"]]
0143      left_condition = condition_map[baseline["conditionId"]]
0144      right_condition = condition_map[target["conditionId"]]
0145      reasons = compatible(left_condition, right_condition, parameters)
0146      source_warning = None
0147      if baseline["sourceKind"] != target["sourceKind"]:
0148          reasons.append("来源类别不同，禁止将模拟、人工和实测混合作为同一配对")
0149      if baseline["sourceKind"] != "measured":
0150          source_warning = "包含模拟或人工填写数据，不能证明实际应用效果"
0151      bands = []
0152      valid_count = 0
0153      for frequency in sorted_bands(baseline, target):
0154          before = baseline["bands"].get(frequency)
0155          after = target["bands"].get(frequency)
0156          missing = before is None or after is None
0157          difference = None
0158          if not missing and not reasons:
0159              difference = before - after
0160              valid_count += 1
0161          bands.append({
0162              "frequency": float(frequency),
0163              "baseline": before,
0164              "target": after,
0165              "difference": difference,
0166              "status": "blocked" if reasons else "missing" if missing else "valid",
0167          })
0168      base_total, base_note = trial_total(baseline, left_condition)
0169      target_total, target_note = trial_total(target, right_condition)
0170      total_difference = None
0171      total_note = base_note
0172      if set(baseline["bands"]) != set(target["bands"]):
0173          total_note = "两次试验频带集合不同，不比较合成声级"
0174      elif reasons:
0175          total_note = "工况或来源不可比，不比较合成声级"
0176      elif base_total is not None and target_total is not None:
0177          total_difference = base_total - target_total
0178      elif target_total is None:
0179          total_note = target_note
0180      timestamp_order = (
0181          datetime.fromisoformat(target["timestamp"].replace("Z", "+00:00"))
0182          - datetime.fromisoformat(baseline["timestamp"].replace("Z", "+00:00"))
0183      ).total_seconds()
0184      warnings = []
0185      if source_warning:
0186          warnings.append(source_warning)
0187      if timestamp_order < 0:
0188          warnings.append("目标时间早于基线，检查试验顺序")
0189      if any(band["status"] == "missing" for band in bands):
0190          warnings.append("缺失频带已保留为空；仅计算共同有效频带")
0191      if not valid_count and not reasons:
0192          warnings.append("没有共同有效频带")
0193      screening = None
0194      if pair["purpose"] in ("drift", "recovery"):
0195          threshold_key = "driftThreshold" if pair["purpose"] == "drift" else "recoveryThreshold"
0196          values = [abs(band["difference"]) for band in bands if band["difference"] is not None]
0197          screening = {
0198              "threshold": parameters[threshold_key],
0199              "maximumAbsoluteDifference": max(values) if values else None,
0200              "status": "unavailable" if not values else "alert" if max(values) > parameters[threshold_key] else "within",
0201              "meaning": "用户工程筛查阈值，不是标准合格判定",
0202          }
0203      return {
0204          "id": pair["id"],
0205          "baselineId": baseline["id"],
0206          "targetId": target["id"],
0207          "baselineName": baseline["name"],
0208          "targetName": target["name"],
0209          "targetType": target["type"],
0210          "siteIds": target["siteIds"],
0211          "purpose": pair["purpose"],
0212          "sourceKind": baseline["sourceKind"],
0213          "baselineSource": baseline["sourceNote"],
0214          "targetSource": target["sourceNote"],
0215          "status": "blocked" if reasons else "valid" if valid_count else "empty",
0216          "reasons": reasons,
0217          "warnings": warnings,
0218          "bands": bands,
0219          "baselineTotal": base_total,
0220          "targetTotal": target_total,
0221          "totalDifference": total_difference,
0222          "totalNote": total_note,
0223          "elapsedSeconds": timestamp_order,
0224          "screening": screening,
0225      }

0228  def summarize_effects(results):
0229      groups = defaultdict(list)
0230      for result in results:
0231          if result["purpose"] == "effect" and result["status"] == "valid":
0232              # Different source kinds cannot be pooled. Conditions must also be

0234              key = (tuple(sorted(result["siteIds"])), result["sourceKind"])

0239  def effect_statistics(results, trial_map, condition_map, parameters):
0240      summaries = []
0241      for (site_ids, source), group in summarize_effects(results).items():
0242          first = group[0]
0243          anchor = condition_map[trial_map[first["baselineId"]]["conditionId"]]
0244          reasons = []
0245          for result in group[1:]:
0246              other = condition_map[trial_map[result["baselineId"]]["conditionId"]]
0247              reasons.extend(compatible(anchor, other, parameters))
0248          used_ids = [entry[key] for entry in group for key in ("baselineId", "targetId")]
0249          independent = len(set(used_ids)) == len(used_ids)
0250          band_keys = sorted({band["frequency"] for result in group for band in result["bands"]})
0251          band_results = []
0252          for frequency in band_keys:
0253              before = []
0254              after = []
0255              differences = []
0256              for result in group:
0257                  for band in result["bands"]:
0258                      if band["frequency"] == frequency and band["difference"] is not None:
0259                          before.append(band["baseline"])
0260                          after.append(band["target"])
0261                          differences.append(band["difference"])
0262              summary = paired_summary(differences, independent)
0263              summary["frequency"] = frequency
0264              summary["groupEnergyMeanDifference"] = (
0265                  energy_mean(before) - energy_mean(after) if before else None
0266              )
0267              if reasons:
0268                  summary["meanDifference"] = None
0269                  summary["sdDifference"] = None
0270                  summary["ci95"] = None
0271                  summary["groupEnergyMeanDifference"] = None
0272                  summary["ciNote"] = "跨配对工况不一致，禁止汇总"
0273              band_results.append(summary)
0274          summaries.append({
0275              "siteIds": list(site_ids),
0276              "sourceKind": source,
0277              "pairIds": [result["id"] for result in group],
0278              "status": "blocked" if reasons else "valid",
0279              "reasons": sorted(set(reasons)),
0280              "independentPairs": independent,
0281              "bands": band_results,
0282          })
0283      return summaries

0286  def repeatability(project, condition_map):
0287      groups = defaultdict(list)
0288      for trial in project["trials"]:
0289          if trial["repeatGroup"]:
0290              key = (trial["repeatGroup"], trial["type"], tuple(sorted(trial["siteIds"])), trial["sourceKind"])
0291              groups[key].append(trial)
0292      output = []
0293      for (name, kind, sites, source), trials in groups.items():
0294          reference = condition_map[trials[0]["conditionId"]]
0295          reasons = []
0296          for trial in trials[1:]:
0297              reasons.extend(compatible(reference, condition_map[trial["conditionId"]], project["parameters"]))
0298          rows = []
0299          for frequency in sorted_bands(*trials):
0300              values = [trial["bands"].get(frequency) for trial in trials]
0301              available = [value for value in values if value is not None]
0302              summary = describe(available)
0303              summary["frequency"] = float(frequency)
0304              summary["missing"] = len(values) - len(available)
0305              summary["status"] = (
0306                  "blocked" if reasons else "insufficient" if len(available) < 2
0307                  else "alert" if summary["sd"] > project["parameters"]["repeatabilityThreshold"] else "within"
0308              )
0309              if reasons:
0310                  for key in ("mean", "energyMean", "sd", "range", "minimum", "maximum"):
0311                      summary[key] = None
0312              rows.append(summary)
0313          output.append({
0314              "name": name,
0315              "type": kind,
0316              "siteIds": list(sites),
0317              "sourceKind": source,
0318              "trialIds": [trial["id"] for trial in trials],
0319              "reasons": sorted(set(reasons)),
0320              "bands": rows,
0321              "threshold": project["parameters"]["repeatabilityThreshold"],
0322          })
0323      return output

0326  def quality_issues(project, results):
0327      issues = []
0328      paired = {pair[key] for pair in project["pairs"] for key in ("baselineId", "targetId")}
0329      for trial in project["trials"]:
0330          if not trial["bands"] or all(value is None for value in trial["bands"].values()):
0331              issues.append({"level": "warning", "object": trial["id"], "message": f"{trial['name']} 没有有效频带"})
0332          if trial["id"] not in paired:
0333              issues.append({"level": "info", "object": trial["id"], "message": f"{trial['name']} 尚未显式配对"})
0334          if trial["sourceKind"] != "measured":
0335              issues.append({"level": "info", "object": trial["id"], "message": f"{trial['name']} 为模拟或人工填写"})
0336      for result in results:
0337          for reason in result["reasons"]:
0338              issues.append({"level": "error", "object": result["id"], "message": reason})
0339          for warning in result["warnings"]:
0340              issues.append({"level": "warning", "object": result["id"], "message": warning})
0341          if result["screening"] and result["screening"]["status"] == "alert":
0342              issues.append({"level": "warning", "object": result["id"], "message": f"{result['targetName']} 超过用户筛查阈值"})
0343      if not any(pair["purpose"] == "drift" for pair in project["pairs"]):
0344          issues.append({"level": "warning", "object": project["id"], "message": "未设置漂移配对，无法评估基线稳定性"})
0345      if not any(pair["purpose"] == "recovery" for pair in project["pairs"]):
0346          issues.append({"level": "warning", "object": project["id"], "message": "未设置恢复配对，无法检查解除封堵后的恢复情况"})
0347      return issues

0030  def upsert(project, collection, payload):
0031      if collection not in COLLECTIONS:
0032          raise ValidationError("对象集合无效", "collection")
0033      if not isinstance(payload, dict):
0034          raise ValidationError("对象必须为 JSON 对象", "payload")
0035      item = copy.deepcopy(payload)
0036      item_id = item.get("id") or identifier()
0037      item["id"] = item_id
0038      found = False
0039      changed = []
0040      for index, existing in enumerate(project[collection]):
0041          if existing["id"] == item_id:
0042              changed = [key for key in set(existing) | set(item) if existing.get(key) != item.get(key)]
0043              project[collection][index] = item
0044              found = True
0045              break
0046      if not found:
0047          project[collection].append(item)
0048      verb = "更新" if found else "新增"
0049      detail = str(item.get("name", item_id))
0050      if changed:
0051          detail += "；修改字段=" + ",".join(sorted(changed))
0052      append_audit(project, verb + collection, detail)
0053      return item_id

0015  def parse_csv(content, trial_map):
0016      if not isinstance(content, str):
0017          raise ValidationError("CSV 内容必须为文本", "csv")
0018      if len(content.encode("utf-8")) > 5 * 1024 * 1024:
0019          raise ValidationError("CSV 不得超过 5 MiB", "csv")
0020      stream = io.StringIO(content.lstrip("\ufeff"), newline="")
0021      reader = csv.DictReader(stream, strict=True)
0022      if reader.fieldnames != list(COLUMNS):
0023          raise ValidationError("表头必须严格为 trial_id,frequency_hz,level_db，使用逗号分隔", "csv.header")
0024      result = {}
0025      seen = set()
0026      missing = 0
0027      row_count = 0
0028      try:
0029          for row_number, row in enumerate(reader, start=2):
0030              row_count += 1
0031              if row_count > 100000:
0032                  raise ValidationError("CSV 最多 100000 行", "csv")
0033              if None in row or any(value is None for value in row.values()):
0034                  raise ValidationError("列数不匹配", f"csv.line[{row_number}]")
0035              trial_id = row["trial_id"].strip()
0036              if trial_id not in trial_map:
0037                  raise ValidationError(f"试验标识不存在：{trial_id}，请先建试验", f"csv.line[{row_number}].trial_id")
0038              try:
0039                  frequency = float(row["frequency_hz"].strip())
0040              except ValueError as error:
0041                  raise ValidationError("频率必须为数值 Hz", f"csv.line[{row_number}].frequency_hz") from error
0042              if not math.isfinite(frequency) or not 1 <= frequency <= 100000:
0043                  raise ValidationError("频率须在 1—100000 Hz", f"csv.line[{row_number}].frequency_hz")
0044              key = format(frequency, ".12g")
0045              row_key = (trial_id, key)
0046              if row_key in seen:
0047                  raise ValidationError("同一试验频带重复，导入已整体取消", f"csv.line[{row_number}]")
0048              seen.add(row_key)
0049              value = row["level_db"].strip()
0050              if value in ("", "NA", "N/A", "null"):
0051                  level = None
0052                  missing += 1
0053              else:
0054                  try:
0055                      level = float(value)
0056                  except ValueError as error:
0057                      raise ValidationError("声级必须为数值或空白/NA/N/A/null", f"csv.line[{row_number}].level_db") from error
0058                  if not math.isfinite(level) or not -100 <= level <= 200:
0059                      raise ValidationError("声级须在 -100—200 dB", f"csv.line[{row_number}].level_db")
0060              result.setdefault(trial_id, {})[key] = level
0061      except csv.Error as error:
0062          raise ValidationError(f"CSV 格式错误：{error}", f"csv.line[{reader.line_num}]") from error
0063      if not row_count:
0064          raise ValidationError("CSV 没有数据行", "csv")
0065      for trial_id, bands in result.items():
0066          if len(bands) > 200:
0067              raise ValidationError("每个试验最多 200 个频带", f"csv.{trial_id}")
0068      return {
0069          "updates": result,
0070          "rows": row_count,
0071          "trials": len(result),
0072          "missing": missing,
0073      }

0251      raw = object_value(raw, path)
0252      fields_only(raw, ["id", "baselineId", "targetId", "purpose", "notes"], path)

0257      if base_id == target_id:

0262      target_type = trial_map[target_id]["type"]
0263      allowed = {
0264          "effect": ("single", "combination"),
0265          "drift": ("baseline",),
0266          "recovery": ("recovery",),
0267      }

0269          raise ValidationError("目标试验类型与配对用途不一致", f"{path}.purpose")
0270      return {
0271          "id": id_value(raw.get("id"), f"{path}.id"),

0287      raw = object_value(raw, "project")
0288      allowed = [
0289          "schemaVersion", "softwareVersion", "id", "name", "owner", "description",
0290          "createdAt", "updatedAt", "revision", "status", "sites", "conditions",
0291          "trials", "pairs", "parameters", "audit",

0298          raise ValidationError("修订号必须为非负整数", "revision")
0299      sites = [
0300          validate_site(item, f"sites[{index}]")
0301          for index, item in enumerate(array_value(raw.get("sites"), "sites", 200))

0324      normalized_parameters = {}
0325      for key, (low, high, default) in PARAMETERS.items():
0326          normalized_parameters[key] = number_value(parameters.get(key, default), f"parameters.{key}", low, high)
0327      audit = array_value(raw.get("audit", []), "audit", 10000)
0328      normalized_audit = []
0329      for index, entry in enumerate(audit):

0331          entry = object_value(entry, path)
0332          fields_only(entry, ["time", "action", "detail"], path)
0333          normalized_audit.append({
0334              "time": date_value(entry.get("time"), f"{path}.time"),
0335              "action": text_value(entry.get("action"), f"{path}.action", limit=120),
0336              "detail": text_value(entry.get("detail", ""), f"{path}.detail", False, 4000),
0337          })
0338      return {
0339          "schemaVersion": SCHEMA_VERSION,
0340          "softwareVersion": text_value(raw.get("softwareVersion"), "softwareVersion", limit=40),
0341          "id": id_value(raw.get("id"), "id"),
0342          "name": text_value(raw.get("name"), "name", limit=160),
0343          "owner": text_value(raw.get("owner", ""), "owner", False, 120),
0344          "description": text_value(raw.get("description", ""), "description", False, 8000),
0345          "createdAt": date_value(raw.get("createdAt"), "createdAt"),
0346          "updatedAt": date_value(raw.get("updatedAt"), "updatedAt"),
0347          "revision": revision,
0348          "status": enum_value(raw.get("status", "active"), ["active", "archived"], "status"),
0349          "sites": sites,
0350          "conditions": conditions,
0351          "trials": trials,
0352          "pairs": pairs,

0028  def level_chart(pair):
0029      rows = [band for band in pair["bands"] if band["difference"] is not None]
0030      if not rows:
0031          return "<p>无可比较数据，不绘图。</p>"
0032      values = [band[key] for band in rows for key in ("baseline", "target")]
0033      minimum = min(values) - 3
0034      span = max(max(values) - minimum + 3, 10)
0035      width = 760
0036      height = 280
0037      left = 64
0038      plot_width = width - left - 20
0039      elements = [
0040          f'<svg viewBox="0 0 {width} {height}" role="img" aria-label="基线和目标频带声级，单位dB">',
0041          '<rect width="100%" height="100%" fill="#f8fafc"/>',
0042      ]
0043      for step in range(5):
0044          value = minimum + span * step / 4
0045          y = 220 - step * 45
0046          elements.append(f'<line x1="{left}" y1="{y}" x2="740" y2="{y}" stroke="#dbe3ec"/>')
0047          elements.append(f'<text x="5" y="{y + 4}" font-size="12">{value:.1f}</text>')
0048      for index, band in enumerate(rows):
0049          x = left + (index + 0.5) * plot_width / len(rows)
0050          for offset, key, color in [(-9, "baseline", "#64748b"), (9, "target", "#0d9488")]:
0051              y = 220 - (band[key] - minimum) / span * 180
0052              elements.append(f'<rect x="{x + offset - 6:.2f}" y="{y:.2f}" width="12" height="{220 - y:.2f}" fill="{color}"/>')
0053          elements.append(f'<text x="{x:.2f}" y="242" text-anchor="middle" font-size="11">{band["frequency"]:g}</text>')
0054      elements.extend([
0055          '<text x="5" y="18" font-size="12">dB re 20 µPa（截断纵轴）</text>',
0056          '<text x="650" y="268" font-size="12">频率 Hz</text>',
0057          '<text x="160" y="268" font-size="12" fill="#64748b">■ 基线</text>',
0058          '<text x="250" y="268" font-size="12" fill="#0d9488">■ 目标</text>',
0059          '</svg>',
0060      ])
0061      return "".join(elements)

0026  def project_coverage(project, results):
0027      """Coverage is descriptive and never an experimental sufficiency guarantee."""
0028      paired_targets = defaultdict(list)
0029      for pair in project["pairs"]:
0030          paired_targets[pair["targetId"]].append(pair["id"])
0031      result_map = {result["id"]: result for result in results}
0032      condition_rows = [condition_completeness(condition) for condition in project["conditions"]]
0033      condition_rows_map = {condition["id"]: condition for condition in condition_rows}
0034      type_counts = {key: 0 for key in TRIAL_TYPES}
0035      source_counts = {key: 0 for key in SOURCES}
0036      frequency_counts = defaultdict(lambda: {"trials": 0, "missing": 0})
0037      timeline = []
0038      for trial in project["trials"]:
0039          type_counts[trial["type"]] += 1
0040          source_counts[trial["sourceKind"]] += 1
0041          condition_rows_map[trial["conditionId"]]["trialCount"] += 1
0042          for frequency, value in trial["bands"].items():
0043              frequency_counts[frequency]["trials"] += 1
0044              if value is None:
0045                  frequency_counts[frequency]["missing"] += 1
0046          valid = sum(value is not None for value in trial["bands"].values())
0047          pair_ids = paired_targets[trial["id"]]
0048          timeline.append({
0049              "id": trial["id"],
0050              "name": trial["name"],
0051              "timestamp": trial["timestamp"],

0053              "sourceKind": trial["sourceKind"],
0054              "conditionId": trial["conditionId"],

0056              "declaredBands": len(trial["bands"]),
0057              "targetPairs": len(pair_ids),
0058              "validTargetPairs": sum(result_map[pair_id]["status"] == "valid" for pair_id in pair_ids),
0059          })
0060      timeline.sort(key=lambda trial: datetime.fromisoformat(trial["timestamp"].replace("Z", "+00:00")))
0061      sites = []
0062      for site in project["sites"]:
0063          single = [trial for trial in project["trials"] if trial["type"] == "single" and site["id"] in trial["siteIds"]]
0064          combination = [trial for trial in project["trials"] if trial["type"] == "combination" and site["id"] in trial["siteIds"]]
0065          valid_effects = [result for result in results if result["purpose"] == "effect" and result["status"] == "valid" and site["id"] in result["siteIds"]]
0066          sites.append({
0067              "id": site["id"],
0068              "name": site["name"],
0069              "singleTrials": len(single),
0070              "combinationTrials": len(combination),
0071              "validEffectPairs": len(valid_effects),
0072              "missingSingle": not single,
0073              "missingCombination": not combination,
0074          })

0077          frequencies.append({

0115          if not self.allowed_request():
0116              return
0117          route = urlparse(self.path).path
0118          if route == "/api/health":
0119              self.respond(200, {"name": SOFTWARE_NAME, "version": VERSION, "status": "ready"})
0120              return
0121          if route == "/":
0122              route = "/index.html"

0124          if WEB_ROOT not in target.parents or not target.is_file():
0125              self.respond(404, {"error": {"path": "route", "message": "文件不存在"}})
0126              return

0128          content_type = mimetypes.guess_type(target)[0] or "application/octet-stream"
0129          if content_type.startswith("text/") or target.suffix == ".js":
0130              content_type += ";charset=utf-8"

0134      def do_POST(self):
0135          if not self.allowed_request():
0136              return
0137          try:
0138              if self.headers.get("Content-Type", "").split(";")[0] != "application/json":
0139                  raise ValidationError("请求须使用 application/json", "contentType")

0003  export class ApiService {
0004    async post(route, body = {}) {
0005      let response;
0006      try {
0007        response = await fetch(`/api/${route}`, {
0008          method: 'POST',
0009          headers: { 'Content-Type': 'application/json' },
0010          body: JSON.stringify(body),
0011        });
0012      } catch (error) {
0013        throw new Error('无法连接本地服务。请保留当前页面，确认 Python 服务仍在运行后重试。');
0014      }
0015      let data;
0016      try {
0017        data = await response.json();
0018      } catch (error) {
0019        throw new Error('服务返回无效内容，请检查终端日志。');
0020      }
0021      if (!response.ok) {
0022        const problem = data.error;
0023        throw new Error(problem ? `${problem.path}：${problem.message}` : `请求失败 ${response.status}`);
0024      }
0025      return data;
0026    }
0027
0028    async open(file) {
0029      if (file.size > 12 * 1024 * 1024) {
0030        throw new Error('项目文件不得超过 12 MiB。');
0031      }
0032      try {
0033        const buffer = await file.arrayBuffer();
0034        const text = new TextDecoder('utf-8', { fatal: true }).decode(buffer);
0035        // Send text as an API string so the Python strict JSON decoder can detect
0036        // duplicate fields rather than losing them through browser JSON.parse.
0037        return await this.post('open-text', { content: text });
0038      } catch (error) {
0039        throw new Error(`打开失败：${error.message}`);
0040      }
0041    }
0042
0043    async export(project, kind) {
0044      return this.post('export', { project, kind });
0045    }
0046  }
0047
0048  export function download(content, filename, mime) {
0049    const blob = new Blob([content], { type: mime });
0050    const url = URL.createObjectURL(blob);
0051    const anchor = document.createElement('a');
0052    anchor.href = url;
0053    anchor.download = filename;
0054    document.body.append(anchor);
0055    anchor.click();
0056    anchor.remove();
0057    setTimeout(() => URL.revokeObjectURL(url), 30000);
0058  }
0059
0060  export function filename(value) {
0061    return value.replace(/[<>:"/\\|?*\x00-\x1f]/g, '_').slice(0, 100) || '声学项目';
0062  }

0064  export async function readCsv(file) {
0065    if (file.size > 5 * 1024 * 1024) {
0066      throw new Error('CSV 文件不得超过 5 MiB。');
0067    }
0068    const buffer = await file.arrayBuffer();

0071    } catch (error) {
0072      throw new Error('CSV 必须为 UTF-8 编码。请在表格软件中另存为 CSV UTF-8 后重试。');
0073    }
0074  }

0009      '>': '&gt;',
0010      '"': '&quot;',

0286  function pairChart(pair) {
0287    const bands = pair.bands.filter(band => band.difference !== null);
0288    if (!bands.length) {
0289      return empty('无可绘制结果', pair.reasons.join('；') || '请录入共同有效频带。');
0290    }
0291    const width = 760;
0292    const height = 240;
0293    const left = 55;
0294    const right = 740;
0295    const baselineY = 132;
0296    const maximum = Math.max(1, ...bands.map(band => Math.abs(band.difference)));
0297    const scale = 85 / maximum;
0298    const parts = [
0299      `<svg class="chart" viewBox="0 0 ${width} ${height}" role="img" aria-label="封堵前后测得声级差，横轴Hz，纵轴dB">`,
0300      `<line x1="${left}" y1="${baselineY}" x2="${right}" y2="${baselineY}" class="zero-line"/>`,
0301      '<text x="5" y="20">ΔL / dB</text>',
0302      `<text x="8" y="50">${number(maximum, 1)}</text>`,
0303      `<text x="20" y="136">0</text>`,
0304      `<text x="5" y="220">${number(-maximum, 1)}</text>`,
0305    ];
0306    for (const [index, band] of bands.entries()) {
0307      const x = left + (index + 0.5) * (right - left) / bands.length;
0308      const barWidth = Math.min(50, (right - left) / bands.length * 0.55);
0309      const barHeight = Math.abs(band.difference) * scale;
0310      const y = band.difference >= 0 ? baselineY - barHeight : baselineY;
0311      parts.push(`<rect x="${x - barWidth / 2}" y="${y}" width="${barWidth}" height="${barHeight}" class="${band.difference >= 0 ? 'positive-bar' : 'negative-bar'}"><title>${band.frequency} Hz：${number(band.difference)} dB</title></rect>`);
0312      parts.push(`<text x="${x}" y="${band.difference >= 0 ? y - 8 : y + barHeight + 15}" text-anchor="middle">${number(band.difference, 2)}</text>`);
0313      parts.push(`<text x="${x}" y="238" text-anchor="middle">${band.frequency}</text>`);
0314    }
0315    parts.push('<text x="705" y="20">Hz</text>', '</svg>');
0316    return parts.join('');
0317  }
