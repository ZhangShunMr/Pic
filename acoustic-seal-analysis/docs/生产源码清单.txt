namespace AcousticSeal.Core.Analysis;

public static class AcousticMath
{
    private static readonly double[] Critical95 =
    [
        12.706204736, 4.302652730, 3.182446305, 2.776445105, 2.570581836,
        2.446911851, 2.364624252, 2.306004135, 2.262157163, 2.228138852,
        2.200985160, 2.178812830, 2.160368656, 2.144786688, 2.131449546,
        2.119905299, 2.109815578, 2.100922040, 2.093024054, 2.085963447,
        2.079613845, 2.073873068, 2.068657610, 2.063898562, 2.059538553,
        2.055529439, 2.051830516, 2.048407142, 2.045229642, 2.042272456
    ];

    public static double? EnergyMean(IEnumerable<double> input)
    {
        var values = Checked(input);
        if (values.Count == 0)
        {
            return null;
        }

        var maximum = values.Max();
        var sum = CompensatedSum(values.Select(value => Math.Pow(10, (value - maximum) / 10)));
        return maximum + 10 * Math.Log10(sum / values.Count);
    }

    public static double? EnergySum(IEnumerable<double> input)
    {
        var values = Checked(input);
        if (values.Count == 0)
        {
            return null;
        }

        var maximum = values.Max();
        var sum = CompensatedSum(values.Select(value => Math.Pow(10, (value - maximum) / 10)));
        return maximum + 10 * Math.Log10(sum);
    }

    public static double? Mean(IEnumerable<double> input)
    {
        var values = Checked(input);
        return values.Count == 0 ? null : CompensatedSum(values) / values.Count;
    }

    public static double? SampleStandardDeviation(IEnumerable<double> input)
    {
        var values = Checked(input);
        if (values.Count < 2)
        {
            return null;
        }

        var mean = Mean(values)!.Value;
        var sum = CompensatedSum(values.Select(value => (value - mean) * (value - mean)));
        return Math.Sqrt(sum / (values.Count - 1));
    }

    public static DescriptiveStatistics Describe(IEnumerable<double> input)
    {
        var values = Checked(input);
        if (values.Count == 0)
        {
            return new DescriptiveStatistics();
        }

        return new DescriptiveStatistics
        {
            Count = values.Count,
            ArithmeticMean = Mean(values),
            EnergyMean = EnergyMean(values),
            StandardDeviation = SampleStandardDeviation(values),
            Minimum = values.Min(),
            Maximum = values.Max(),
            Range = values.Max() - values.Min()
        };
    }

    public static PairedStatistics Paired(
        IReadOnlyList<double> baseline,
        IReadOnlyList<double> target,
        bool independent)
    {
        if (baseline.Count != target.Count)
        {
            throw new ArgumentException("配对基线与目标数量须相同");
        }

        var differences = baseline.Zip(target, (before, after) => before - after).ToList();
        var mean = Mean(differences);
        var deviation = SampleStandardDeviation(differences);
        double? lower = null;
        double? upper = null;
        var note = "n<2，不生成置信区间";
        if (differences.Count > 1 && !independent)
        {
            note = "同组配对复用试验，独立性不满足，不生成置信区间";
        }
        else if (differences.Count > 1)
        {
            var freedom = differences.Count - 1;
            var critical = Critical95[Math.Min(freedom, 30) - 1];
            var half = critical * deviation!.Value / Math.Sqrt(differences.Count);
            lower = mean - half;
            upper = mean + half;
            note = "独立配对差近似正态的双侧95% Student t均值区间；" +
                (freedom > 30 ? "df>30保守使用df=30" : "df=" + freedom) +
                "；不含仪器系统不确定性";
        }

        return new PairedStatistics
        {
            Count = differences.Count,
            MeanDifference = mean,
            StandardDeviation = deviation,
            GroupEnergyMeanDifference = baseline.Count == 0 ? null : EnergyMean(baseline) - EnergyMean(target),
            Lower95 = lower,
            Upper95 = upper,
            IntervalNote = note
        };
    }

    private static List<double> Checked(IEnumerable<double> values)
    {
        var list = values.ToList();
        if (list.Any(value => !double.IsFinite(value)))
        {
            throw new ArgumentException("计算输入须为有限数值；缺失值应在明确口径后筛选");
        }

        return list;
    }

    private static double CompensatedSum(IEnumerable<double> values)
    {
        double total = 0;
        double correction = 0;
        foreach (var value in values)
        {
            var adjusted = value - correction;
            var next = total + adjusted;
            correction = (next - total) - adjusted;
            total = next;
        }

        return total;
    }
}

using AcousticSeal.Core.Domain;

namespace AcousticSeal.Core.Analysis;

public sealed class AnalysisEngine
{
    private readonly ProjectValidator validator = new();
    private readonly Comparability comparability = new();

    public AnalysisReport Calculate(AcousticProject project)
    {
        validator.Ensure(project);
        var conditions = project.Conditions.ToDictionary(condition => condition.Id);
        var trials = project.Trials.ToDictionary(trial => trial.Id);
        var pairs = project.Pairs.Select(pair => Compare(pair, trials, conditions, project.Settings)).ToList();
        return new AnalysisReport
        {
            ProjectId = project.Id,
            Revision = project.Revision,
            Pairs = pairs,
            Effects = Summarize(pairs, trials, conditions, project.Settings),
            Repeatability = Repeatability(project, conditions),
            Coverage = BuildCoverage(project, pairs),
            Issues = new QualityInspector().Inspect(project, pairs)
        };
    }

    private PairResult Compare(
        TrialPair pair,
        Dictionary<Guid, AcousticTrial> trials,
        Dictionary<Guid, MeasurementCondition> conditions,
        AnalysisSettings settings)
    {
        var baseline = trials[pair.BaselineId];
        var target = trials[pair.TargetId];
        var left = conditions[baseline.ConditionId];
        var right = conditions[target.ConditionId];
        var reasons = comparability.Inspect(left, right, settings);
        var warnings = new List<string>();
        if (baseline.Source != target.Source)
        {
            reasons.Add("来源类别不同，禁止混合模拟、实测、人工或估算作为同一配对");
        }

        if (baseline.Source != SourceKind.Measured)
        {
            warnings.Add("包含" + baseline.Source.Text() + "数据，不能证明真实应用效果");
        }

        var elapsed = (target.Timestamp - baseline.Timestamp).TotalSeconds;
        if (elapsed < 0)
        {
            warnings.Add("目标时间早于基线，核对时间与配对用途");
        }

        var frequencies = Frequencies([baseline, target]);
        var bands = new List<BandDifference>();
        foreach (var frequency in frequencies)
        {
            var before = baseline.LevelAt(frequency);
            var after = target.LevelAt(frequency);
            var status = reasons.Count > 0 ? ResultStatus.Blocked :
                !before.HasValue || !after.HasValue ? ResultStatus.Missing : ResultStatus.Valid;
            var difference = status == ResultStatus.Valid ? before - after : null;
            var interpretation = difference switch
            {
                null => status.Text(),
                > 0 => "目标声级低于基线",
                < 0 => "目标声级高于基线",
                _ => "两者声级相同"
            };
            if (difference.HasValue && Math.Abs(difference.Value) < settings.MinimumEffectDisplay)
            {
                interpretation += "；低于用户变化标记阈值";
            }

            bands.Add(new BandDifference
            {
                Frequency = frequency,
                Baseline = before,
                Target = after,
                Difference = difference,
                Status = status,
                Interpretation = interpretation
            });
        }

        if (bands.Any(band => band.Status == ResultStatus.Missing))
        {
            warnings.Add("部分频带缺失，仅计算共同有效频带，不补零");
        }

        var validCount = bands.Count(band => band.Difference.HasValue);
        var baseTotal = Total(baseline, left);
        var targetTotal = Total(target, right);
        var sameFrequencies = baseline.Bands.Select(band => band.Frequency).Order().SequenceEqual(
            target.Bands.Select(band => band.Frequency).Order());
        var totalNote = "只表示已录入频带范围的能量和，不是全频总声级";
        double? totalDifference = null;
        if (reasons.Count > 0)
        {
            totalNote = "条件或来源不可比，不比较合成声级";
        }
        else if (!sameFrequencies)
        {
            totalNote = "频带集合不同，不比较合成声级";
        }
        else if (!left.NonOverlappingBands || !right.NonOverlappingBands)
        {
            totalNote = "未声明频带互不重叠，不计算合成声级";
        }
        else if (!baseTotal.HasValue || !targetTotal.HasValue)
        {
            totalNote = "频带缺失或为空，不计算合成声级";
        }
        else
        {
            totalDifference = baseTotal - targetTotal;
        }

        ScreeningResult? screening = null;
        if (pair.Purpose != PairPurpose.Effect)
        {
            var threshold = pair.Purpose == PairPurpose.Drift ? settings.DriftThreshold : settings.RecoveryThreshold;
            var worst = bands.Where(band => band.Difference.HasValue)
                .OrderByDescending(band => Math.Abs(band.Difference!.Value)).FirstOrDefault();
            var maximum = worst is null ? (double?)null : Math.Abs(worst.Difference!.Value);
            screening = new ScreeningResult(
                threshold,
                maximum,
                worst?.Frequency,
                maximum is null ? ResultStatus.Empty : maximum > threshold ? ResultStatus.Alert : ResultStatus.Within);
        }

        return new PairResult
        {
            PairId = pair.Id,
            BaselineId = baseline.Id,
            TargetId = target.Id,
            BaselineName = baseline.Name,
            TargetName = target.Name,
            Purpose = pair.Purpose,
            TargetKind = target.Kind,
            Source = baseline.Source,
            LocationIds = [.. target.LocationIds],
            Status = reasons.Count > 0 ? ResultStatus.Blocked : validCount > 0 ? ResultStatus.Valid : ResultStatus.Empty,
            BlockingReasons = reasons,
            Warnings = warnings,
            Bands = bands,
            BaselineTotal = baseTotal,
            TargetTotal = targetTotal,
            TotalDifference = totalDifference,
            TotalNote = totalNote,
            ElapsedSeconds = elapsed,
            Screening = screening,
            BaselineSource = baseline.SourceNote,
            TargetSource = target.SourceNote
        };
    }

    private List<EffectSummary> Summarize(
        List<PairResult> pairs,
        Dictionary<Guid, AcousticTrial> trials,
        Dictionary<Guid, MeasurementCondition> conditions,
        AnalysisSettings settings)
    {
        var groups = pairs.Where(pair => pair.Purpose == PairPurpose.Effect && pair.Status == ResultStatus.Valid)
            .GroupBy(pair => Key(pair.LocationIds) + "|" + pair.Source);
        var summaries = new List<EffectSummary>();
        foreach (var group in groups)
        {
            var entries = group.ToList();
            var reasons = new List<string>();
            // Check all pairs, rather than only one anchor, so tolerances do not
            // silently accumulate through a nontransitive matching chain.
            for (var index = 0; index < entries.Count; index++)
            {
                for (var other = index + 1; other < entries.Count; other++)
                {
                    reasons.AddRange(comparability.Inspect(
                        conditions[trials[entries[index].BaselineId].ConditionId],
                        conditions[trials[entries[other].BaselineId].ConditionId], settings));
                }
            }

            var ids = entries.SelectMany(pair => new[] { pair.BaselineId, pair.TargetId }).ToList();
            var independent = ids.Distinct().Count() == ids.Count;
            var bands = new List<EffectBandSummary>();
            foreach (var frequency in entries.SelectMany(pair => pair.Bands.Select(band => band.Frequency)).Distinct().Order())
            {
                var valid = entries.SelectMany(pair => pair.Bands)
                    .Where(band => band.Frequency == frequency && band.Difference.HasValue).ToList();
                var statistics = reasons.Count > 0
                    ? new PairedStatistics { Count = valid.Count, IntervalNote = "跨配对工况不同，禁止汇总" }
                    : AcousticMath.Paired(
                        valid.Select(band => band.Baseline!.Value).ToList(),
                        valid.Select(band => band.Target!.Value).ToList(), independent);
                bands.Add(new EffectBandSummary { Frequency = frequency, Statistics = statistics });
            }

            summaries.Add(new EffectSummary
            {
                Key = group.Key,
                LocationIds = [.. entries[0].LocationIds],
                Source = entries[0].Source,
                IndependentPairs = independent,
                PairIds = entries.Select(pair => pair.PairId).ToList(),
                BlockingReasons = reasons.Distinct().ToList(),
                Bands = bands
            });
        }

        return summaries;
    }

    private List<RepeatabilityGroup> Repeatability(AcousticProject project, Dictionary<Guid, MeasurementCondition> conditions)
    {
        var groups = project.Trials.Where(trial => !string.IsNullOrWhiteSpace(trial.RepeatGroup))
            .GroupBy(trial => (trial.RepeatGroup, trial.Kind, Locations: Key(trial.LocationIds), trial.Source));
        var output = new List<RepeatabilityGroup>();
        foreach (var group in groups)
        {
            var entries = group.ToList();
            var reasons = comparability.MissingFields(conditions[entries[0].ConditionId], project.Settings);
            for (var index = 0; index < entries.Count; index++)
            {
                for (var other = index + 1; other < entries.Count; other++)
                {
                    reasons.AddRange(comparability.Inspect(conditions[entries[index].ConditionId],
                        conditions[entries[other].ConditionId], project.Settings));
                }
            }

            var bands = new List<RepeatabilityBand>();
            foreach (var frequency in Frequencies(entries))
            {
                var levels = entries.Select(trial => trial.LevelAt(frequency)).Where(value => value.HasValue)
                    .Select(value => value!.Value).ToList();
                var statistics = reasons.Count > 0 ? new DescriptiveStatistics { Count = levels.Count } : AcousticMath.Describe(levels);
                bands.Add(new RepeatabilityBand
                {
                    Frequency = frequency,
                    MissingCount = entries.Count - levels.Count,
                    Statistics = statistics,
                    Status = reasons.Count > 0 ? ResultStatus.Blocked : levels.Count < 2 ? ResultStatus.Insufficient :
                        statistics.StandardDeviation > project.Settings.RepeatabilityThreshold ? ResultStatus.Alert : ResultStatus.Within
                });
            }

            output.Add(new RepeatabilityGroup
            {
                Name = group.Key.RepeatGroup,
                Kind = group.Key.Kind,
                Source = group.Key.Source,
                LocationIds = [.. entries[0].LocationIds],
                TrialIds = entries.Select(trial => trial.Id).ToList(),
                BlockingReasons = reasons.Distinct().ToList(),
                Threshold = project.Settings.RepeatabilityThreshold,
                Bands = bands
            });
        }

        return output;
    }

    private CoverageReport BuildCoverage(AcousticProject project, List<PairResult> pairs)
    {
        return new CoverageReport
        {
            TrialTypes = Enum.GetValues<TrialKind>().ToDictionary(kind => kind, kind => project.Trials.Count(trial => trial.Kind == kind)),
            Conditions = project.Conditions.Select(condition => new ConditionCoverage(condition.Id, condition.Name,
                project.Trials.Count(trial => trial.ConditionId == condition.Id), comparability.MissingFields(condition, project.Settings))).ToList(),
            Locations = project.Locations.Select(location => new LocationCoverage(
                location.Id, location.Name,
                project.Trials.Count(trial => trial.Kind == TrialKind.Single && trial.LocationIds.Contains(location.Id)),
                project.Trials.Count(trial => trial.Kind == TrialKind.Combination && trial.LocationIds.Contains(location.Id)),
                pairs.Count(pair => pair.Purpose == PairPurpose.Effect && pair.Status == ResultStatus.Valid && pair.LocationIds.Contains(location.Id)))).ToList(),
            Frequencies = Frequencies(project.Trials).Select(frequency =>
            {
                var declared = project.Trials.Where(trial => trial.Bands.Any(band => band.Frequency == frequency)).ToList();
                var valid = declared.Count(trial => trial.LevelAt(frequency).HasValue);
                return new FrequencyCoverage(frequency, valid, declared.Count - valid, project.Trials.Count - declared.Count);
            }).ToList(),
            Timeline = project.Trials.OrderBy(trial => trial.Timestamp).Select(trial => new TimelineEntry(
                trial.Id, trial.Name, trial.Kind, trial.Source, trial.Timestamp, trial.ValidBandCount,
                pairs.Count(pair => pair.TargetId == trial.Id))).ToList()
        };
    }

    public static List<double> Frequencies(IEnumerable<AcousticTrial> trials)
    {
        return trials.SelectMany(trial => trial.Bands.Select(band => band.Frequency)).Distinct().Order().ToList();
    }

    public static string Key(IEnumerable<Guid> ids)
    {
        return string.Join("+", ids.Order());
    }

    public static double? Total(AcousticTrial trial, MeasurementCondition condition)
    {
        if (!condition.NonOverlappingBands || trial.Bands.Count == 0 || trial.Bands.Any(band => !band.Level.HasValue))
        {
            return null;
        }

        return AcousticMath.EnergySum(trial.Bands.Select(band => band.Level!.Value));
    }
}

using AcousticSeal.Core.Domain;

namespace AcousticSeal.Core.Analysis;

public sealed class Comparability
{
    public List<string> Inspect(MeasurementCondition left, MeasurementCondition right, AnalysisSettings settings)
    {
        var reasons = new List<string>();
        var rightFields = right.TextFields().ToDictionary(field => field.Key);
        foreach (var field in left.TextFields())
        {
            var other = rightFields[field.Key];
            CompareText(field.Label, field.Value, other.Value, reasons);
        }

        CompareNumber("车速", "km/h", left.Speed, right.Speed, settings.SpeedTolerance, reasons);
        CompareNumber("转速", "r/min", left.Rpm, right.Rpm, settings.RpmTolerance, reasons);
        CompareNumber("温度", "°C", left.Temperature, right.Temperature, settings.TemperatureTolerance, reasons);
        CompareNumber("积分时长", "s", left.Duration, right.Duration, settings.DurationTolerance, reasons);
        if (left.NonOverlappingBands != right.NonOverlappingBands)
        {
            reasons.Add("互不重叠频带声明不同");
        }

        if (settings.RequireSampleRateMatch)
        {
            CompareNumber("采样率", "Hz", left.SampleRate, right.SampleRate, 0, reasons);
        }

        if (settings.RequireSynchronizationMatch)
        {
            CompareText("时钟与同步定义", left.Synchronization, right.Synchronization, reasons);
        }

        return reasons;
    }

    public List<string> MissingFields(MeasurementCondition condition, AnalysisSettings settings)
    {
        return Inspect(condition, condition, settings);
    }

    private static void CompareText(string label, string? left, string? right, List<string> reasons)
    {
        if (string.IsNullOrWhiteSpace(left) || string.IsNullOrWhiteSpace(right))
        {
            reasons.Add(label + "缺失，无法确认可比性");
        }
        else if (!string.Equals(left.Trim(), right.Trim(), StringComparison.Ordinal))
        {
            reasons.Add(label + "不同：" + left + " / " + right);
        }
    }

    private static void CompareNumber(string label, string unit, double? left, double? right, double tolerance, List<string> reasons)
    {
        if (!left.HasValue || !right.HasValue)
        {
            reasons.Add(label + "缺失，无法确认可比性");
            return;
        }

        var difference = Math.Abs(left.Value - right.Value);
        if (difference > tolerance + 1e-10)
        {
            reasons.Add($"{label}绝对差{difference:G6} {unit}超过容差{tolerance:G6}");
        }
    }
}

public sealed record WorkbookColumn(Guid TrialId, string Name, string Condition, SourceKind Source, string SourceNote);
public sealed record WorkbookCell(double? Level, bool Declared, string Note)
{
    public string Status => !Declared ? "未声明" : Level.HasValue ? "有效" : "明确缺失";
}
public sealed record WorkbookRow(double Frequency, List<WorkbookCell> Cells);

public sealed class InputWorkbookResult
{
    public List<WorkbookColumn> Columns { get; init; } = [];
    public List<WorkbookRow> Rows { get; init; } = [];
    public List<string> Warnings { get; init; } = [];
    public int CompleteRows => Rows.Count(row => row.Cells.All(cell => cell.Level.HasValue));
    public string Purpose { get; init; } = "原始频带人工核对，不自动配对、不计算封堵差、不证明跨列可比。";
}

public sealed class InputWorkbook
{
    public InputWorkbookResult Build(AcousticProject project, IReadOnlyList<Guid> ids, double? minimum, double? maximum)
    {
        if (ids.Count == 0 || ids.Count > 20 || ids.Distinct().Count() != ids.Count)
        {
            throw new DomainValidationException("workbook", "请显式选择1至20条不重复试验");
        }

        if (minimum > maximum || minimum is < 1 or > 100000 || maximum is < 1 or > 100000)
        {
            throw new DomainValidationException("workbook.frequency", "频带范围无效");
        }

        var trials = new List<AcousticTrial>();
        foreach (var id in ids)
        {
            trials.Add(project.Trials.FirstOrDefault(trial => trial.Id == id)
                ?? throw new DomainValidationException("workbook.trial", "选中的试验已不存在，请重新选择"));
        }

        var rows = new List<WorkbookRow>();
        foreach (var frequency in AnalysisEngine.Frequencies(trials).Where(frequency =>
            (!minimum.HasValue || frequency >= minimum) && (!maximum.HasValue || frequency <= maximum)))
        {
            var cells = new List<WorkbookCell>();
            foreach (var trial in trials)
            {
                var band = trial.Bands.FirstOrDefault(band => band.Frequency == frequency);
                cells.Add(new WorkbookCell(band?.Level, band is not null, band?.Note ?? ""));
            }

            rows.Add(new WorkbookRow(frequency, cells));
        }

        var warnings = new List<string>();
        if (trials.Select(trial => trial.Source).Distinct().Count() > 1)
        {
            warnings.Add("所选试验包含不同来源类别，只供原始输入核对，不能混合统计。");
        }

        var conditions = project.Conditions.ToDictionary(condition => condition.Id);
        var comparator = new Comparability();
        for (var index = 0; index < trials.Count; index++)
        {
            for (var other = index + 1; other < trials.Count; other++)
            {
                var reasons = comparator.Inspect(conditions[trials[index].ConditionId], conditions[trials[other].ConditionId], project.Settings);
                if (reasons.Count > 0)
                {
                    warnings.Add(trials[index].Name + " / " + trials[other].Name + "：" + string.Join("；", reasons));
                }
            }
        }

        return new InputWorkbookResult
        {
            Columns = trials.Select(trial => new WorkbookColumn(trial.Id, trial.Name,
                conditions[trial.ConditionId].Name, trial.Source, trial.SourceNote)).ToList(),
            Rows = rows,
            Warnings = warnings.Distinct().ToList()
        };
    }

    public byte[] Csv(InputWorkbookResult workbook)
    {
        var output = new StringBuilder("\uFEFF");
        var header = new List<string> { "frequency_hz" };
        foreach (var column in workbook.Columns)
        {
            header.Add(column.TrialId + "_level_db");
            header.Add(column.TrialId + "_state");
        }

        output.AppendLine(string.Join(",", header));
        foreach (var row in workbook.Rows)
        {
            var cells = new List<string> { row.Frequency.ToString("G17", CultureInfo.InvariantCulture) };
            foreach (var cell in row.Cells)
            {
                cells.Add(cell.Level?.ToString("G17", CultureInfo.InvariantCulture) ?? "");
                cells.Add(cell.Status);
            }

            output.AppendLine(string.Join(",", cells));
        }

        output.AppendLine();
        output.AppendLine("trial_id,name,condition,source_kind,source_note");
        foreach (var column in workbook.Columns)
        {
            output.AppendLine(string.Join(",", new[] { column.TrialId.ToString(), column.Name, column.Condition, column.Source.Text(), column.SourceNote }
                .Select(CsvCell)));
        }

        return Encoding.UTF8.GetBytes(output.ToString());
    }

    private static string CsvCell(string text)
    {
        if (text.Length > 0 && "=+-@\t\r".Contains(text[0]))
        {
            text = "'" + text;
        }

        return "\"" + text.Replace("\"", "\"\"") + "\"";
    }
}

using AcousticSeal.Core.Domain;

namespace AcousticSeal.Core.Analysis;

public sealed class QualityInspector
{
    public List<QualityIssue> Inspect(AcousticProject project, List<PairResult> results)
    {
        var issues = new List<QualityIssue>();
        var pairedIds = project.Pairs.SelectMany(pair => new[] { pair.BaselineId, pair.TargetId }).ToHashSet();
        var comparability = new Comparability();
        foreach (var condition in project.Conditions)
        {
            foreach (var missing in comparability.MissingFields(condition, project.Settings))
            {
                issues.Add(new QualityIssue(IssueSeverity.Warning, "condition.incomplete", "condition", condition.Id,
                    condition.Name, missing, "打开工况，按原始试验记录补充该字段后应用；不要用零代替未知"));
            }
        }

        foreach (var trial in project.Trials)
        {
            if (trial.ValidBandCount == 0)
            {
                issues.Add(new QualityIssue(IssueSeverity.Warning, "trial.empty", "trial", trial.Id,
                    trial.Name, "没有有效频带", "打开试验频带表，录入声级或通过CSV预检导入"));
            }

            if (trial.MissingBandCount > 0)
            {
                issues.Add(new QualityIssue(IssueSeverity.Warning, "trial.missing", "trial", trial.Id,
                    trial.Name, $"存在{trial.MissingBandCount}个明确缺失频带", "核对仪器导出记录；确实缺失时保留空值，不补零"));
            }

            if (!pairedIds.Contains(trial.Id))
            {
                issues.Add(new QualityIssue(IssueSeverity.Information, "trial.unpaired", "trial", trial.Id,
                    trial.Name, "尚未被显式配对引用", "按试验计划建立基线与目标对应关系，或说明保留为参考数据"));
            }

            if (trial.Source != SourceKind.Measured)
            {
                issues.Add(new QualityIssue(IssueSeverity.Information, "trial.source", "trial", trial.Id,
                    trial.Name, "数据来源为" + trial.Source.Text(), "输出时继续保留来源标识；本记录不能证明真实车辆应用效果"));
            }
        }

        foreach (var result in results)
        {
            foreach (var reason in result.BlockingReasons)
            {
                issues.Add(new QualityIssue(IssueSeverity.Error, "pair.blocked", "pair", result.PairId,
                    result.Title, reason, "打开配对核对基线与目标，再进入对应工况或试验按真实记录修正"));
            }

            foreach (var warning in result.Warnings)
            {
                issues.Add(new QualityIssue(IssueSeverity.Warning, "pair.warning", "pair", result.PairId,
                    result.Title, warning, "展开配对来源和时间详情，确认输入或保留说明"));
            }

            if (result.Screening?.Status == ResultStatus.Alert)
            {
                issues.Add(new QualityIssue(IssueSeverity.Warning, "screening.alert", "pair", result.PairId,
                    result.Title, "最大频带绝对差超过用户" + result.Purpose.Text() + "阈值",
                    "检查封堵恢复状态、仪器标定、声源稳定性和环境条件，再决定是否补测；不要仅改阈值消除警示"));
            }
        }

        foreach (var purpose in new[] { PairPurpose.Drift, PairPurpose.Recovery })
        {
            if (project.Pairs.All(pair => pair.Purpose != purpose))
            {
                issues.Add(new QualityIssue(IssueSeverity.Warning, "coverage." + purpose, "project", project.Id,
                    project.Name, "尚无" + purpose.Text() + "配对", "先采集对应试验，再明确指定基线和目标；不能推断稳定或已恢复"));
            }
        }

        return issues.OrderByDescending(issue => issue.Severity).ThenBy(issue => issue.Title, StringComparer.Ordinal).ToList();
    }

public sealed class RevisionComparison
{
    public RevisionComparisonResult Compare(AcousticProject left, AcousticProject right)
    {
        var validator = new ProjectValidator();
        validator.Ensure(left);
        validator.Ensure(right);
        using var before = JsonDocument.Parse(JsonSerializer.Serialize(left, ProjectCodec.Options));
        using var after = JsonDocument.Parse(JsonSerializer.Serialize(right, ProjectCodec.Options));
        var changes = new List<RevisionChange>();
        var ignored = new HashSet<string>(StringComparer.Ordinal)
        {
            "audit", "reviews", "createdAt", "updatedAt", "softwareVersion", "revision", "id", "schemaVersion"
        };
        foreach (var property in before.RootElement.EnumerateObject())
        {
            if (!ignored.Contains(property.Name))
            {
                Walk(property.Value, after.RootElement.GetProperty(property.Name), property.Name, "project", "", changes);
            }
        }

        var notes = new List<string>
        {
            "只按对象GUID和频率键核对，不按名称猜测对象对应关系。",
            "这里是两份已录入项目内容的变化，不是封堵效果或物理路径贡献。",
            "审计与复核记录只比较数量，详细历史在各项目文件内查看；时间、修订和软件版本列为身份信息。"
        };
        if (left.Id != right.Id)
        {
            notes.Add("两侧项目标识不同：只核对仍具有相同GUID的对象，不能断言属于同一次试验版本。");
        }

        if (left.Audit.Count != right.Audit.Count || left.Reviews.Count != right.Reviews.Count)
        {
            notes.Add($"审计数量{left.Audit.Count}→{right.Audit.Count}；复核数量{left.Reviews.Count}→{right.Reviews.Count}。");
        }

        return new RevisionComparisonResult
        {
            LeftProjectId = left.Id,
            RightProjectId = right.Id,
            LeftName = left.Name,
            RightName = right.Name,
            LeftRevision = left.Revision,
            RightRevision = right.Revision,
            LeftHash = ProjectCodec.Fingerprint(left),
            RightHash = ProjectCodec.Fingerprint(right),
            Notes = notes,
            Changes = changes
        };
    }

    private static void Walk(JsonElement left, JsonElement right, string path, string collection, string entityId, List<RevisionChange> changes)
    {
        if (left.ValueKind != right.ValueKind)
        {
            Add(ChangeKind.Modified, left, right, path, collection, entityId, changes);
            return;
        }

        if (left.ValueKind == JsonValueKind.Object)
        {
            foreach (var property in left.EnumerateObject())
            {
                if (right.TryGetProperty(property.Name, out var other))
                {
                    Walk(property.Value, other, path + "." + property.Name, collection, entityId, changes);
                }
                else
                {
                    changes.Add(new RevisionChange(ChangeKind.Removed, collection, entityId,
                        path + "." + property.Name, Show(property.Value), "未提供"));
                }
            }

            foreach (var property in right.EnumerateObject())
            {
                if (!left.TryGetProperty(property.Name, out _))
                {
                    changes.Add(new RevisionChange(ChangeKind.Added, collection, entityId,
                        path + "." + property.Name, "未提供", Show(property.Value)));
                }
            }

            return;
        }

        if (left.ValueKind == JsonValueKind.Array)
        {
            var a = left.EnumerateArray().ToList();
            var b = right.EnumerateArray().ToList();
            var keyed = a.Concat(b).All(item => item.ValueKind == JsonValueKind.Object && Key(item) is not null);
            if (keyed)
            {
                var leftMap = a.ToDictionary(item => Key(item)!);
                var rightMap = b.ToDictionary(item => Key(item)!);
                foreach (var key in leftMap.Keys.Union(rightMap.Keys).Order(StringComparer.Ordinal))
                {
                    var foundLeft = leftMap.TryGetValue(key, out var first);
                    var foundRight = rightMap.TryGetValue(key, out var second);
                    var location = path + "[" + key + "]";
                    if (!foundLeft)
                    {
                        changes.Add(new RevisionChange(ChangeKind.Added, path, key, location, "未提供", Show(second)));
                    }
                    else if (!foundRight)
                    {
                        changes.Add(new RevisionChange(ChangeKind.Removed, path, key, location, Show(first), "未提供"));
                    }
                    else
                    {
                        Walk(first, second, location, path, key, changes);
                    }
                }
            }
            else
            {
                var leftValues = a.Select(item => item.GetRawText()).Order(StringComparer.Ordinal);
                var rightValues = b.Select(item => item.GetRawText()).Order(StringComparer.Ordinal);
                if (!leftValues.SequenceEqual(rightValues))
                {
                    Add(ChangeKind.Modified, left, right, path, collection, entityId, changes);
                }
            }

            return;
        }

        if (left.GetRawText() != right.GetRawText())
        {
            Add(ChangeKind.Modified, left, right, path, collection, entityId, changes);
        }
    }

    private static string? Key(JsonElement value)
    {
        if (value.TryGetProperty("id", out var id))
        {
            return id.GetString();
        }

        if (value.TryGetProperty("frequency", out var frequency))
        {
            return frequency.GetDouble().ToString("G17", CultureInfo.InvariantCulture);
        }

        return null;
    }

    private static void Add(ChangeKind kind, JsonElement before, JsonElement after, string path,
        string collection, string entityId, List<RevisionChange> changes)
    {
        changes.Add(new RevisionChange(kind, collection, entityId, path, Show(before), Show(after)));
    }

    private static string Show(JsonElement value)
    {
        return value.ValueKind switch
        {
            JsonValueKind.Null => "缺失(null)",
            JsonValueKind.String => value.GetString() ?? "",
            _ => value.GetRawText()
        };
    }
}

using System.Globalization;
using System.Text;
using AcousticSeal.Core.Domain;

namespace AcousticSeal.Core.Analysis;

public static class SpectrumChart
{
    public static string Build(PairResult pair)
    {
        if (pair.Status == ResultStatus.Blocked)
        {
            return "<p>配对条件已阻断，不绘制可比频谱图。原始声级仍在数值表中保留。</p>";
        }

        var bands = pair.Bands.Where(band => band.Baseline.HasValue || band.Target.HasValue).ToList();
        if (bands.Count == 0)
        {
            return "<p>没有有效声级，无法绘制频谱。</p>";
        }

        if (bands.Count > 30)
        {
            return "<p>超过30个频带，请先缩窄界面频带扫描范围，或使用完整数值表。</p>";
        }

        var levels = bands.SelectMany(band => new[] { band.Baseline, band.Target }).Where(value => value.HasValue)
            .Select(value => value!.Value).ToList();
        var low = Math.Floor(levels.Min() / 5) * 5 - 5;
        var high = Math.Ceiling(levels.Max() / 5) * 5 + 5;
        var minFrequency = bands.Min(band => band.Frequency);
        var maxFrequency = bands.Max(band => band.Frequency);
        var minimumLog = Math.Log10(minFrequency);
        var logSpan = Math.Log10(maxFrequency) - minimumLog;
        double X(double frequency)
        {
            return logSpan == 0 ? 430 : 65 + (Math.Log10(frequency) - minimumLog) / logSpan * 710;
        }

        double Y(double level)
        {
            return 225 - (level - low) / (high - low) * 175;
        }

        var builder = new StringBuilder("<svg viewBox=\"0 0 820 300\" role=\"img\" aria-label=\"基线与目标原始频带声级图，横轴对数频率Hz，纵轴声级dB\">");
        builder.Append("<text x=\"8\" y=\"18\" font-size=\"11\">声级 dB re 20 µPa · 截断纵轴</text>");
        for (var index = 0; index < 5; index++)
        {
            var value = low + index * (high - low) / 4;
            var y = Y(value);
            builder.Append("<line x1=\"60\" x2=\"790\" y1=\"").Append(C(y)).Append("\" y2=\"").Append(C(y)).Append("\" stroke=\"#dce6e8\"/>");
            builder.Append("<text x=\"8\" y=\"").Append(C(y + 4)).Append("\" font-size=\"10\">").Append(C(value)).Append("</text>");
        }

        foreach (var band in bands)
        {
            builder.Append("<text x=\"").Append(C(X(band.Frequency))).Append("\" y=\"248\" text-anchor=\"middle\" font-size=\"10\">").Append(C(band.Frequency)).Append("</text>");
        }

        DrawSeries(builder, bands, band => band.Baseline, "#617b8a", "基线", X, Y);
        DrawSeries(builder, bands, band => band.Target, "#138d82", "目标", X, Y);
        builder.Append("<text x=\"140\" y=\"278\" fill=\"#617b8a\" font-size=\"12\">● 基线</text>");
        builder.Append("<text x=\"240\" y=\"278\" fill=\"#138d82\" font-size=\"12\">● 目标</text>");
        builder.Append("<text x=\"440\" y=\"278\" font-size=\"11\">缺失处断线；连线仅引导阅读，不插值补数</text>");
        builder.Append("<text x=\"755\" y=\"295\" font-size=\"11\">Hz（对数）</text></svg>");
        return builder.ToString();
    }

    private static void DrawSeries(StringBuilder output, List<BandDifference> bands, Func<BandDifference, double?> value,
        string color, string label, Func<double, double> x, Func<double, double> y)
    {
        var segment = new List<string>();
        void Flush()
        {
            if (segment.Count > 1)
            {
                output.Append("<polyline fill=\"none\" stroke=\"").Append(color).Append("\" stroke-width=\"2\" points=\"")
                    .Append(string.Join(" ", segment)).Append("\"/>");
            }

            segment.Clear();
        }

        foreach (var band in bands)
        {
            var level = value(band);
            if (!level.HasValue)
            {
                Flush();
                continue;
            }

            var point = C(x(band.Frequency)) + "," + C(y(level.Value));
            segment.Add(point);
            output.Append("<circle cx=\"").Append(C(x(band.Frequency))).Append("\" cy=\"").Append(C(y(level.Value)))
                .Append("\" r=\"4\" fill=\"").Append(color).Append("\"><title>").Append(label).Append(" ")
                .Append(C(band.Frequency)).Append("Hz ").Append(C(level.Value)).Append("dB</title></circle>");
        }

        Flush();
    }

    private static string C(double value)
    {
        return value.ToString("0.###", CultureInfo.InvariantCulture);
    }
}

using System.Globalization;

namespace AcousticSeal.Core.Domain;

public sealed record ValidationMessage(string Path, string Message)
{
    public override string ToString()
    {
        return Path + "：" + Message;
    }
}

public sealed class DomainValidationException : Exception
{
    public IReadOnlyList<ValidationMessage> Problems { get; }

    public DomainValidationException(string path, string message)
        : this([new ValidationMessage(path, message)])
    {
    }

    public DomainValidationException(IReadOnlyList<ValidationMessage> problems)
        : base(string.Join("；", problems.Select(problem => problem.ToString())))
    {
        Problems = problems;
    }
}

public sealed class ProjectValidator
{
    public IReadOnlyList<ValidationMessage> Inspect(AcousticProject project)
    {
        var errors = new List<ValidationMessage>();
        if (project.SchemaVersion != Software.SchemaVersion)
        {
            errors.Add(new("schemaVersion", "格式版本不受支持，请保留原文件"));
        }

        CheckId(project.Id, "project.id", errors);
        CheckText(project.Name, "project.name", 160, true, errors);
        CheckText(project.Owner, "project.owner", 120, false, errors);
        CheckText(project.VehicleProgramme, "project.vehicleProgramme", 200, false, errors);
        CheckText(project.Objective, "project.objective", 4000, false, errors);
        CheckText(project.Description, "project.description", 8000, false, errors);
        CheckText(project.SoftwareVersion, "softwareVersion", 40, true, errors);
        CheckEnum(project.Status, "project.status", errors);
        if (project.Revision < 0)
        {
            errors.Add(new("revision", "修订号不得为负"));
        }

        CheckDate(project.CreatedAt, "createdAt", errors);
        CheckDate(project.UpdatedAt, "updatedAt", errors);
        if (project.Locations is null || project.Conditions is null || project.Trials is null ||
            project.Pairs is null || project.Settings is null || project.Audit is null || project.Reviews is null)
        {
            errors.Add(new("project", "对象集合、设置、审计或复核不得为null"));
            return errors;
        }

        CheckCount(project.Locations.Count, 200, "locations", errors);
        CheckCount(project.Conditions.Count, 500, "conditions", errors);
        CheckCount(project.Trials.Count, Software.MaximumTrials, "trials", errors);
        CheckCount(project.Pairs.Count, Software.MaximumPairs, "pairs", errors);
        CheckCount(project.Audit.Count, 10000, "audit", errors);
        CheckCount(project.Reviews.Count, 2000, "reviews", errors);
        if (project.Locations.Any(item => item is null) || project.Conditions.Any(item => item is null) ||
            project.Trials.Any(item => item is null) || project.Pairs.Any(item => item is null) ||
            project.Audit.Any(item => item is null) || project.Reviews.Any(item => item is null))
        {
            errors.Add(new("project", "集合中不可包含null对象"));
            return errors;
        }

        CheckUnique(project.Locations.Select(item => item.Id), "locations", errors);
        CheckUnique(project.Conditions.Select(item => item.Id), "conditions", errors);
        CheckUnique(project.Trials.Select(item => item.Id), "trials", errors);
        CheckUnique(project.Pairs.Select(item => item.Id), "pairs", errors);
        CheckUnique(project.Audit.Select(item => item.Id), "audit", errors);
        CheckUnique(project.Reviews.Select(item => item.Id), "reviews", errors);
        foreach (var location in project.Locations)
        {
            ValidateLocation(location, errors);
        }

        foreach (var condition in project.Conditions)
        {
            ValidateCondition(condition, errors);
        }

        var locationIds = project.Locations.Select(item => item.Id).ToHashSet();
        var conditionIds = project.Conditions.Select(item => item.Id).ToHashSet();
        foreach (var trial in project.Trials)
        {
            ValidateTrial(trial, locationIds, conditionIds, errors);
        }

        var trialMap = project.Trials.GroupBy(item => item.Id)
            .ToDictionary(group => group.Key, group => group.First());
        foreach (var pair in project.Pairs)
        {
            ValidatePair(pair, trialMap, errors);
        }

        var duplicatePair = project.Pairs.GroupBy(pair => (pair.BaselineId, pair.TargetId, pair.Purpose))
            .FirstOrDefault(group => group.Count() > 1);
        if (duplicatePair is not null)
        {
            errors.Add(new("pairs", "同一用途的相同基线和目标不可重复"));
        }

        foreach (var setting in project.Settings.Describe())
        {
            CheckNumber(setting.Value, setting.Minimum, setting.Maximum, "settings." + setting.Key, false, errors);
        }

        foreach (var entry in project.Audit)
        {
            CheckId(entry.Id, "audit.id", errors);
            CheckDate(entry.Time, "audit.time", errors);
            CheckText(entry.Action, "audit.action", 120, true, errors);
            CheckText(entry.Entity, "audit.entity", 200, false, errors);
            CheckText(entry.Detail, "audit.detail", 16000, false, errors);
            CheckText(entry.InputHash, "audit.inputHash", 64, false, errors);
            if (entry.Revision < 0 || entry.Revision > project.Revision)
            {
                errors.Add(new("audit.revision", "审计修订号超出项目范围"));
            }
        }

        foreach (var review in project.Reviews)
        {
            CheckId(review.Id, "reviews.id", errors);
            CheckDate(review.Time, "reviews.time", errors);
            CheckText(review.Reviewer, "reviews.reviewer", 120, true, errors);
            CheckText(review.Comment, "reviews.comment", 4000, false, errors);
            CheckText(review.SettingsJson, "reviews.settingsJson", 8000, true, errors);
            CheckText(review.InputHash, "reviews.inputHash", 64, true, errors);
            if (review.BlockingReasons is null)
            {
                errors.Add(new("reviews.blockingReasons", "复核阻断原因列表不得为null"));
            }
            else
            {
                CheckCount(review.BlockingReasons.Count, 5000, "reviews.blockingReasons", errors);
                foreach (var reason in review.BlockingReasons)
                {
                    CheckText(reason, "reviews.blockingReasons", 2000, true, errors);
                }
            }
        }

        return errors;
    }

    public void Ensure(AcousticProject project)
    {
        var errors = Inspect(project);
        if (errors.Count != 0)
        {
            throw new DomainValidationException(errors);
        }
    }

    private static void ValidateLocation(SealLocation location, List<ValidationMessage> errors)
    {
        var path = "locations[" + location.Id + "]";
        CheckText(location.Name, path + ".name", 120, true, errors);
        CheckText(location.Zone, path + ".zone", 120, false, errors);
        CheckText(location.Material, path + ".material", 1000, false, errors);
        CheckText(location.Extent, path + ".extent", 1000, false, errors);
        CheckText(location.Procedure, path + ".procedure", 4000, false, errors);
        CheckText(location.RecoveryProcedure, path + ".recoveryProcedure", 4000, false, errors);
        CheckText(location.Reference, path + ".reference", 1000, false, errors);
        CheckText(location.Notes, path + ".notes", 4000, false, errors);
    }

    private static void ValidateCondition(MeasurementCondition condition, List<ValidationMessage> errors)
    {
        var path = "conditions[" + condition.Id + "]";
        CheckText(condition.Name, path + ".name", 120, true, errors);
        foreach (var field in condition.TextFields())
        {
            CheckText(field.Value, path + "." + field.Key, 500, false, errors);
        }

        CheckNumber(condition.Temperature, -80, 100, path + ".temperature", true, errors);
        CheckNumber(condition.Duration, 0.001, 86400, path + ".duration", true, errors);
        CheckNumber(condition.SampleRate, 1, 1000000, path + ".sampleRate", true, errors);
        CheckText(condition.Synchronization, path + ".synchronization", 500, false, errors);
        CheckText(condition.Notes, path + ".notes", 4000, false, errors);
        if (condition.Weighting is not ("" or "A" or "C" or "Z"))

            errors.Add(new(path + ".weighting", "频率计权仅支持A、C、Z或草稿空白"));

            errors.Add(new(path + ".bandType", "频带制须为octave、third-octave或custom"));

        if (condition.Unit is not ("" or Software.LevelUnit))

            errors.Add(new(path + ".unit", "仅支持" + Software.LevelUnit));

    private static void ValidateTrial(
        AcousticTrial trial,
        HashSet<Guid> locations,
        HashSet<Guid> conditions,
        List<ValidationMessage> errors)
    {
        var path = "trials[" + trial.Id + "]";
        CheckText(trial.Name, path + ".name", 160, true, errors);
        CheckEnum(trial.Kind, path + ".kind", errors);
        CheckEnum(trial.Source, path + ".source", errors);
        CheckDate(trial.Timestamp, path + ".timestamp", errors);
        CheckText(trial.RepeatGroup, path + ".repeatGroup", 120, false, errors);
        CheckText(trial.SourceNote, path + ".sourceNote", 4000, true, errors);
        CheckText(trial.OriginalFile, path + ".originalFile", 1000, false, errors);
        CheckText(trial.Operator, path + ".operator", 120, false, errors);
        CheckText(trial.Preparation, path + ".preparation", 4000, false, errors);
        CheckText(trial.Notes, path + ".notes", 4000, false, errors);
        if (!conditions.Contains(trial.ConditionId))
        {
            errors.Add(new(path + ".conditionId", "引用的工况不存在"));
        }

        if (trial.LocationIds is null || trial.Bands is null)
        {
            errors.Add(new(path, "封堵位置集合与频带集合不得为null"));
            return;
        }

        if (trial.LocationIds.Distinct().Count() != trial.LocationIds.Count)
        {
            errors.Add(new(path + ".locationIds", "封堵位置不可重复"));
        }

        if (trial.LocationIds.Any(id => !locations.Contains(id)))
        {
            errors.Add(new(path + ".locationIds", "引用的封堵位置不存在"));
        }

        var count = trial.LocationIds.Count;
        if (trial.Kind is TrialKind.Baseline or TrialKind.Recovery && count != 0)
        {
            errors.Add(new(path + ".locationIds", "基线和恢复不得选择封堵位置"));
        }

        if (trial.Kind == TrialKind.Single && count != 1)
        {
            errors.Add(new(path + ".locationIds", "单项封堵须选择一个位置"));
        }

        if (trial.Kind == TrialKind.Combination && count < 2)
        {
            errors.Add(new(path + ".locationIds", "组合封堵至少选择两个位置"));
        }

        CheckCount(trial.Bands.Count, Software.MaximumBands, path + ".bands", errors);
        if (trial.Bands.Any(band => band is null))
        {
            errors.Add(new(path + ".bands", "频带不可为null对象"));
            return;
        }

        foreach (var band in trial.Bands)
        {
            var bandPath = path + ".bands[" + band.Frequency.ToString(CultureInfo.InvariantCulture) + "]";
            CheckNumber(band.Frequency, 1, 100000, bandPath + ".frequency", false, errors);
            CheckNumber(band.Level, -100, 200, bandPath + ".level", true, errors);
            CheckText(band.Note, bandPath + ".note", 1000, false, errors);
        }

        if (trial.Bands.GroupBy(band => band.Frequency).Any(group => group.Count() > 1))
        {
            errors.Add(new(path + ".bands", "同一试验频率不可重复"));
        }
    }

    private static void ValidatePair(TrialPair pair, Dictionary<Guid, AcousticTrial> trials, List<ValidationMessage> errors)
    {
        var path = "pairs[" + pair.Id + "]";
        CheckText(pair.Notes, path + ".notes", 4000, false, errors);
        CheckEnum(pair.Purpose, path + ".purpose", errors);
        if (!trials.TryGetValue(pair.BaselineId, out var baseline) ||
            !trials.TryGetValue(pair.TargetId, out var target))
        {
            errors.Add(new(path, "配对引用的试验不存在"));
            return;
        }

        if (baseline.Id == target.Id)
        {
            errors.Add(new(path, "不能与自身配对"));
        }

        if (baseline.Kind != TrialKind.Baseline)
        {
            errors.Add(new(path + ".baselineId", "基准须为基线试验"));
        }

        var allowed = pair.Purpose switch
        {
            PairPurpose.Effect => target.Kind is TrialKind.Single or TrialKind.Combination,
            PairPurpose.Drift => target.Kind == TrialKind.Baseline,
            PairPurpose.Recovery => target.Kind == TrialKind.Recovery,
            _ => false
        };
        if (!allowed)
        {
            errors.Add(new(path + ".targetId", "目标试验类型与配对用途不匹配"));
        }
    }

    private static void CheckText(string? value, string path, int maximum, bool required, List<ValidationMessage> errors)
    {
        if (value is null)
        {
            errors.Add(new(path, "文本不得为null"));
            return;
        }

        if (required && string.IsNullOrWhiteSpace(value))
        {
            errors.Add(new(path, "不能为空"));
        }

        if (value.Length > maximum)
        {
            errors.Add(new(path, $"最多{maximum}字符"));
        }

        if (value.Any(character => char.IsControl(character) && character is not ('\r' or '\n' or '\t')))
        {
            errors.Add(new(path, "包含不可用控制字符"));
        }
    }

    private static void CheckNumber(double? value, double minimum, double maximum, string path, bool nullable, List<ValidationMessage> errors)
    {
        if (!value.HasValue)
        {
            if (!nullable)
            {
                errors.Add(new(path, "数值不能为空"));
            }

            return;
        }

        if (!double.IsFinite(value.Value) || value < minimum || value > maximum)
        {
            errors.Add(new(path, $"须为{minimum}至{maximum}的有限数值"));
        }
    }

    private static void CheckId(Guid id, string path, List<ValidationMessage> errors)
    {
        if (id == Guid.Empty)
        {
            errors.Add(new(path, "对象标识不可为空"));
        }
    }

    private static void CheckUnique(IEnumerable<Guid> ids, string path, List<ValidationMessage> errors)
    {
        var values = ids.ToList();
        if (values.Contains(Guid.Empty) || values.Distinct().Count() != values.Count)
        {
            errors.Add(new(path, "对象标识为空或重复"));
        }
    }

    private static void CheckDate(DateTimeOffset date, string path, List<ValidationMessage> errors)
    {
        if (date.Year < 1900 || date.Year > 2200)
        {
            errors.Add(new(path, "日期范围须为1900至2200年，保留时区"));
        }
    }

    private static void CheckCount(int count, int maximum, string path, List<ValidationMessage> errors)
    {
        if (count > maximum)
        {
            errors.Add(new(path, $"最多允许{maximum}项"));
        }
    }

    private static void CheckEnum<T>(T value, string path, List<ValidationMessage> errors) where T : struct, Enum
    {
        if (!Enum.IsDefined(value))
        {
            errors.Add(new(path, "枚举值无效"));
        }
    }

    public ImportPreview Preview(AcousticProject project, string content, ImportMode mode, char delimiter = ',')
    {
        if (delimiter is not (',' or ';' or '\t'))
        {
            throw new DomainValidationException("delimiter", "仅支持逗号、分号或制表符");
        }

        if (Encoding.UTF8.GetByteCount(content) > 5 * 1024 * 1024)
        {
            throw new DomainValidationException("csv", "CSV不得超过5MiB");
        }

        var records = CsvParser.Read(content.TrimStart('\uFEFF'), delimiter);
        var problems = new List<ValidationMessage>();
        var expected = new[] { "trial_id", "frequency_hz", "level_db" };
        if (records.Count == 0 || !records[0].Cells.SequenceEqual(expected))
        {
            throw new DomainValidationException("csv.header", "表头必须为trial_id、frequency_hz、level_db三列，分隔符须与选择一致");
        }

        if (records.Count > 100001)
        {
            throw new DomainValidationException("csv", "最多允许100000数据行");
        }

        var trials = project.Trials.ToDictionary(trial => trial.Id);
        var updates = new Dictionary<Guid, List<BandMeasurement>>();
        var seen = new HashSet<(Guid, double)>();
        foreach (var record in records.Skip(1))
        {
            var path = "CSV第" + record.Line + "行";
            if (record.Cells.Count != 3)
            {
                problems.Add(new(path, "列数不是3"));
                continue;
            }

            if (!Guid.TryParse(record.Cells[0].Trim(), out var id) || !trials.ContainsKey(id))
            {
                problems.Add(new(path + ".trial_id", "试验标识不存在；先建立试验并使用其GUID"));
                continue;
            }

            if (!double.TryParse(record.Cells[1].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out var frequency) ||
                !double.IsFinite(frequency) || frequency < 1 || frequency > 100000)
            {
                problems.Add(new(path + ".frequency_hz", "频率须为1至100000Hz有限数值"));
                continue;
            }

            if (!seen.Add((id, frequency)))
            {
                problems.Add(new(path + ".frequency_hz", "同一试验频率重复，整批不可应用"));
                continue;
            }

            var text = record.Cells[2].Trim();
            double? level = null;
            if (text is not ("" or "NA" or "N/A" or "null"))
            {
                if (!double.TryParse(text, NumberStyles.Float, CultureInfo.InvariantCulture, out var value) ||
                    !double.IsFinite(value) || value < -100 || value > 200)
                {
                    problems.Add(new(path + ".level_db", "声级须为-100至200dB，或空白/NA/N/A/null"));
                    continue;
                }

                level = value;
            }

            if (!updates.TryGetValue(id, out var bands))
            {
                bands = [];
                updates[id] = bands;
            }

            bands.Add(new BandMeasurement { Frequency = frequency, Level = level });
        }

        var changes = new List<ImportChange>();
        foreach (var (id, bands) in updates)
        {
            var trial = trials[id];
            var original = trial.Bands.Select(band => band.Frequency).ToHashSet();
            var resultCount = mode == ImportMode.Replace ? bands.Count : original.Union(bands.Select(band => band.Frequency)).Count();
            if (resultCount > Software.MaximumBands)
            {
                problems.Add(new("csv." + trial.Name, "应用后超过每试验200频带"));
            }

            changes.Add(new ImportChange(id, trial.Name, bands.Count(band => !original.Contains(band.Frequency)),
                bands.Count(band => original.Contains(band.Frequency)), bands.Count(band => !band.Level.HasValue), bands));
        }

        if (records.Count == 1)
        {
            problems.Add(new("csv", "没有数据行"));
        }

        return new ImportPreview
        {
            ProjectHash = ProjectCodec.Fingerprint(project),
            Mode = mode,
            Delimiter = delimiter,
            Rows = records.Count - 1,
            Problems = problems,
            Changes = changes
        };
    }

    public void Apply(ProjectSession session, ImportPreview preview)
    {
        if (!preview.CanApply)
        {
            throw new DomainValidationException("csv", "预检未通过，数据没有改变");
        }

        if (ProjectCodec.Fingerprint(session.Project) != preview.ProjectHash)
        {
            throw new DomainValidationException("csv", "预检后项目已改变，请重新预检，禁止应用过期结果");
        }

        session.Apply("CSV整体导入：" + preview.Mode, session.Project.Id.ToString(), project =>
        {
            foreach (var change in preview.Changes)
            {
                var trial = project.Trials.Single(trial => trial.Id == change.TrialId);
                if (preview.Mode == ImportMode.Replace)
                {
                    trial.Bands = ProjectCodec.Clone(change.Bands);
                }
                else
                {
                    var updated = change.Bands.Select(band => band.Frequency).ToHashSet();
                    trial.Bands.RemoveAll(band => updated.Contains(band.Frequency));
                    trial.Bands.AddRange(ProjectCodec.Clone(change.Bands));
                }

                trial.Bands = trial.Bands.OrderBy(band => band.Frequency).ToList();
            }
        });
    }

public static class CsvParser
{
    public static List<CsvRecord> Read(string content, char delimiter)
    {
        var records = new List<CsvRecord>();
        var cells = new List<string>();
        var cell = new StringBuilder();
        var quoted = false;
        var closedQuote = false;
        var line = 1;
        var startLine = 1;
        for (var index = 0; index < content.Length; index++)
        {
            var character = content[index];
            if (quoted)
            {
                if (character == '"' && index + 1 < content.Length && content[index + 1] == '"')
                {
                    cell.Append('"');
                    index++;
                }
                else if (character == '"')
                {
                    quoted = false;
                    closedQuote = true;
                }
                else
                {
                    cell.Append(character);
                    if (character == '\n')
                    {
                        line++;
                    }
                }

                continue;
            }

            if (character == delimiter)
            {
                cells.Add(cell.ToString());
                cell.Clear();
                closedQuote = false;
            }
            else if (character is '\r' or '\n')
            {
                if (character == '\r' && index + 1 < content.Length && content[index + 1] == '\n')
                {
                    index++;
                }

                cells.Add(cell.ToString());
                if (cells.Any(value => value.Length > 0))
                {
                    records.Add(new CsvRecord(startLine, cells));
                }

                cells = [];
                cell.Clear();
                closedQuote = false;
                line++;
                startLine = line;
            }
            else if (character == '"' && cell.Length == 0 && !closedQuote)
            {
                quoted = true;
            }
            else if (closedQuote || character == '"')
            {
                throw new DomainValidationException("CSV第" + line + "行", "引号位置错误或闭引号后仍有内容");
            }
            else
            {
                cell.Append(character);
            }
        }

        if (quoted)
        {
            throw new DomainValidationException("CSV第" + startLine + "行", "引号未闭合");
        }

        if (cell.Length > 0 || cells.Count > 0 || closedQuote)
        {
            cells.Add(cell.ToString());
            records.Add(new CsvRecord(startLine, cells));
        }

        return records;
    }
}

    {
        if (bytes.Length == 0 || bytes.Length > Software.MaximumFileBytes)
        {
            throw new DomainValidationException("file", "项目文件须为1字节至12MiB");
        }

        string content;
        try
        {
            content = new UTF8Encoding(false, true).GetString(bytes).TrimStart('\uFEFF');
        }
        catch (DecoderFallbackException)
        {
            throw new DomainValidationException("file.encoding", "文件不是有效UTF-8，请保留原文件");
        }

        try
        {
            using var document = JsonDocument.Parse(content, new JsonDocumentOptions { MaxDepth = 64 });
            InspectDuplicates(document.RootElement, "project");
            if (document.RootElement.ValueKind != JsonValueKind.Object ||
                !document.RootElement.TryGetProperty("schemaVersion", out var version) ||
                version.ValueKind != JsonValueKind.Number || !version.TryGetInt32(out var number))
            {
                throw new DomainValidationException("schemaVersion", "缺少整数格式版本");
            }

            if (number == 1)
            {
                var migrated = LegacyMigration.Convert(document.RootElement);
                validator.Ensure(migrated.Project);
                return migrated;
            }

            if (number != Software.SchemaVersion)
            {
                throw new DomainValidationException("schemaVersion", "不支持此版本，不修改原文件");
            }

            foreach (var required in new[] { "id", "name", "createdAt", "updatedAt", "revision", "status", "locations", "conditions", "trials", "pairs", "settings", "audit", "reviews" })
            {
                if (!document.RootElement.TryGetProperty(required, out _))
                {
                    throw new DomainValidationException(required, "项目缺少必要字段");
                }
            }

            InspectRequiredObjects(document.RootElement);
            var project = JsonSerializer.Deserialize<AcousticProject>(content, Options)
                ?? throw new DomainValidationException("project", "项目内容为空");
            validator.Ensure(project);
            return new DecodedProject(project, false, []);
        }
        catch (Exception error) when (error is InvalidOperationException or KeyNotFoundException or FormatException)
        {
            throw new DomainValidationException("file.structure", "项目字段缺失或类型不匹配：" + error.Message);
        }
        catch (JsonException error)
        {
            throw new DomainValidationException(error.Path ?? "json", "JSON格式或类型错误：" + error.Message);
        }
    }

    public static T Clone<T>(T value)
    {
        var bytes = JsonSerializer.SerializeToUtf8Bytes(value, Options);
        return JsonSerializer.Deserialize<T>(bytes, Options)!;
    }

    public static string Fingerprint(AcousticProject project)
    {
        var copy = Clone(project);
        copy.Audit = [];
        copy.Reviews = [];
        copy.Revision = 0;
        copy.UpdatedAt = copy.CreatedAt;
        return Convert.ToHexStringLower(SHA256.HashData(JsonSerializer.SerializeToUtf8Bytes(copy, Options)));
    }

    private static void InspectDuplicates(JsonElement element, string path)
    {
        if (element.ValueKind == JsonValueKind.Object)
        {
            var seen = new HashSet<string>(StringComparer.Ordinal);
            foreach (var property in element.EnumerateObject())
            {
                if (!seen.Add(property.Name))
                {
                    throw new DomainValidationException(path, "重复JSON字段：" + property.Name);
                }

                InspectDuplicates(property.Value, path + "." + property.Name);
            }
        }
        else if (element.ValueKind == JsonValueKind.Array)
        {
            var index = 0;
            foreach (var item in element.EnumerateArray())
            {
                InspectDuplicates(item, path + "[" + index++ + "]");
            }
        }
    }

    private static void InspectRequiredObjects(JsonElement root)
    {
        var requirements = new Dictionary<string, string[]>
        {
            ["locations"] = ["id", "name"],
            ["conditions"] = ["id", "name"],
            ["trials"] = ["id", "name", "kind", "conditionId", "locationIds", "timestamp", "source", "sourceNote", "bands"],
            ["pairs"] = ["id", "baselineId", "targetId", "purpose"],
            ["audit"] = ["id", "time", "revision", "action", "entity", "detail", "inputHash"],
            ["reviews"] = ["id", "time", "inputRevision", "reviewer", "validPairs", "blockedPairs", "blockingReasons", "settingsJson", "inputHash"]
        };
        foreach (var (collection, names) in requirements)
        {
            var array = root.GetProperty(collection);
            if (array.ValueKind != JsonValueKind.Array)
            {
                throw new DomainValidationException(collection, "必要对象集合须为数组");
            }

            var index = 0;
            foreach (var item in array.EnumerateArray())
            {
                RequireProperties(item, names, collection + "[" + index++ + "]");
            }
        }

        var settings = new AnalysisSettings().Describe().Select(setting => JsonNamingPolicy.CamelCase.ConvertName(setting.Key)).ToList();
        settings.Add("requireSampleRateMatch");
        settings.Add("requireSynchronizationMatch");
        RequireProperties(root.GetProperty("settings"), settings, "settings");
    }

    private static void RequireProperties(JsonElement element, IEnumerable<string> names, string path)
    {
        if (element.ValueKind != JsonValueKind.Object)
        {
            throw new DomainValidationException(path, "对象缺失或类型无效");
        }

        foreach (var name in names)
        {
            if (!element.TryGetProperty(name, out _))
            {
                throw new DomainValidationException(path + "." + name, "缺少必要字段，不自动生成标识或猜测来源");
            }
        }
    }

    private static JsonSerializerOptions CreateOptions()
    {
        var options = new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
            WriteIndented = true,
            MaxDepth = 64,
            UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow,
            Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping
        };
        options.Converters.Add(new JsonStringEnumConverter(JsonNamingPolicy.CamelCase, allowIntegerValues: false));
        options.Converters.Add(new OffsetDateConverter());
        return options;
    }
}

public sealed class OffsetDateConverter : JsonConverter<DateTimeOffset>
{
    public override DateTimeOffset Read(ref Utf8JsonReader reader, Type type, JsonSerializerOptions options)
    {
        var text = reader.GetString();
        if (text is null || !Regex.IsMatch(text, "(Z|[+-][0-9]{2}:[0-9]{2})$", RegexOptions.CultureInvariant) ||
            !DateTimeOffset.TryParse(text, System.Globalization.CultureInfo.InvariantCulture,
                System.Globalization.DateTimeStyles.None, out var value))
        {
            throw new JsonException("日期时间必须带时区，例如2026-01-01T10:00:00+08:00");
        }

        return value;
    }

    public override void Write(Utf8JsonWriter writer, DateTimeOffset value, JsonSerializerOptions options)
    {
        writer.WriteStringValue(value.ToString("O", System.Globalization.CultureInfo.InvariantCulture));
    }
}

public sealed class ProjectStore
{
    private readonly string directory;
    private readonly ProjectCodec codec = new();
    private readonly SemaphoreSlim gate = new(1, 1);

    public ProjectStore(string directory)
    {
        this.directory = System.IO.Path.GetFullPath(directory);
        Directory.CreateDirectory(this.directory);
    }

    public string DirectoryPath => directory;

    public async Task<List<ProjectCard>> ListAsync(CancellationToken cancellation = default)
    {
        var cards = new List<ProjectCard>();
        foreach (var file in Directory.EnumerateFiles(directory, "*" + Software.Extension))
        {
            cancellation.ThrowIfCancellationRequested();
            var filename = System.IO.Path.GetFileNameWithoutExtension(file);
            if (!Guid.TryParseExact(filename, "N", out var id))
            {
                continue;
            }

            try
            {
                var stored = await LoadAsync(id, cancellation);
                cards.Add(new ProjectCard(id, stored.Project.Name, stored.Project.Revision,
                    stored.Project.UpdatedAt, file, File.Exists(file + ".bak"), ""));
            }
            catch (Exception error) when (error is IOException or DomainValidationException or UnauthorizedAccessException)
            {
                cards.Add(new ProjectCard(id, "无法读取的项目", 0, new DateTimeOffset(File.GetLastWriteTimeUtc(file)),
                    file, File.Exists(file + ".bak"), error.Message));
            }
        }

        return cards.OrderByDescending(card => card.Modified).ToList();
    }

    public async Task<StoredProject> LoadAsync(Guid id, CancellationToken cancellation = default)
    {
        await gate.WaitAsync(cancellation);
        try
        {
            var file = PathFor(id);
            var bytes = await ReadBoundedAsync(file, cancellation);
            var decoded = codec.Decode(bytes);
            if (decoded.Project.Id != id)
            {
                throw new DomainValidationException("file.id", "文件名与内嵌项目标识不一致，请通过导入项目另存");
            }

            return new StoredProject(decoded.Project, Hash(bytes), file);
        }
        finally
        {
            gate.Release();
        }
    }

    public async Task<StoredProject> LoadBackupAsync(Guid id, CancellationToken cancellation = default)
    {
        await gate.WaitAsync(cancellation);
        try
        {
            var file = PathFor(id);
            var backup = file + ".bak";
            var bytes = await ReadBoundedAsync(backup, cancellation);
            var decoded = codec.Decode(bytes);
            if (decoded.Project.Id != id)
            {
                throw new DomainValidationException("backup.id", "备份项目标识与文件名不一致");
            }

            var current = File.Exists(file) ? Hash(await ReadBoundedAsync(file, cancellation)) : "";
            return new StoredProject(decoded.Project, current, backup, true);
        }
        finally
        {
            gate.Release();
        }
    }

    public async Task<SaveReceipt> SaveAsync(AcousticProject project, string? expectedETag, CancellationToken cancellation = default)
    {
        var bytes = codec.Encode(project);
        if (bytes.Length > Software.MaximumFileBytes)
        {
            throw new DomainValidationException("file", "项目超过12MiB，请减少记录或分项目保存");
        }

        await gate.WaitAsync(cancellation);
        string? temporary = null;
        try
        {
            var file = PathFor(project.Id);
            // Share.None coordinates cooperating application instances before the
            // ETag check. The persistent lock file avoids unlink/recreate races.
            using var fileLock = new FileStream(file + ".lock", FileMode.OpenOrCreate,
                FileAccess.ReadWrite, FileShare.None);
            var exists = File.Exists(file);
            var actual = exists ? Hash(await ReadBoundedAsync(file, cancellation)) : null;
            var expected = string.IsNullOrEmpty(expectedETag) ? null : expectedETag;
            if (actual != expected)
            {
                throw new ProjectConflictException();
            }

            temporary = file + "." + Guid.NewGuid().ToString("N") + ".tmp";
            await using (var stream = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None,
                81920, FileOptions.Asynchronous | FileOptions.WriteThrough))
            {
                await stream.WriteAsync(bytes, cancellation);
                await stream.FlushAsync(cancellation);
                stream.Flush(flushToDisk: true);
            }

            if (exists)
            {
                File.Copy(file, file + ".bak", overwrite: true);
            }

            File.Move(temporary, file, overwrite: true);
            temporary = null;
            return new SaveReceipt(file, exists ? file + ".bak" : "", Hash(bytes), bytes.Length);
        }
        finally
        {
            if (temporary is not null && File.Exists(temporary))
            {
                File.Delete(temporary);
            }

            gate.Release();
        }
    }

    public string PathFor(Guid id)
    {
        if (id == Guid.Empty)
        {
            throw new DomainValidationException("project.id", "项目标识不能为空");
        }

        return System.IO.Path.Combine(directory, id.ToString("N") + Software.Extension);
    }

    private static async Task<byte[]> ReadBoundedAsync(string file, CancellationToken cancellation)
    {
        var info = new FileInfo(file);
        if (info.Length > Software.MaximumFileBytes)
        {
            throw new DomainValidationException("file", "文件超过12MiB，拒绝读取");
        }

        return await File.ReadAllBytesAsync(file, cancellation);
    }

    private static string Hash(byte[] bytes)
    {
        return Convert.ToHexStringLower(SHA256.HashData(bytes));
    }
}

    public static string SafeName(string value)
    {
        var invalid = "<>:\"/\\|?*";
        var name = new string(value.Select(character => char.IsControl(character) || invalid.Contains(character) ? '_' : character).ToArray());
        name = name.Trim().TrimEnd('.');
        return string.IsNullOrEmpty(name) ? "声学项目" : name[..Math.Min(name.Length, 100)];
    }

    private static ExportDocument Csv(AcousticProject project, string extension, IEnumerable<IEnumerable<object?>> rows)
    {
        var builder = new StringBuilder("\uFEFF");
        foreach (var row in rows)
        {
            builder.AppendLine(string.Join(",", row.Select(Cell)));
        }

        return new ExportDocument(SafeName(project.Name) + extension, "text/csv; charset=utf-8", Encoding.UTF8.GetBytes(builder.ToString()));
    }

    private static string Cell(object? value)
    {
        if (value is null)
        {
            return "";
        }

        var text = value is IFormattable formatted
            ? formatted.ToString(null, CultureInfo.InvariantCulture)
            : value.ToString() ?? "";
        if (value is string && text.Length > 0 && "=+-@\t\r".Contains(text[0]))
        {
            text = "'" + text;
        }

        return "\"" + text.Replace("\"", "\"\"") + "\"";
    }

    private void Matrix(AcousticProject project, AnalysisReport report)

        var pairs = report.Pairs.Where(pair => pair.Purpose == PairPurpose.Effect).ToList();
        var columns = new List<string> { "Hz" };
        columns.AddRange(pairs.Select(pair => pair.TargetName + "；基线=" + pair.BaselineName + "；" + pair.Source.Text()));
        var rows = new List<object?[]>();
        foreach (var frequency in pairs.SelectMany(pair => pair.Bands.Select(band => band.Frequency)).Distinct().Order())
        {

            foreach (var pair in pairs)
            {
                var band = pair.Bands.FirstOrDefault(band => band.Frequency == frequency);
                cells.Add(band?.Difference is { } value ? N(value) : band?.Status.Text() ?? "未声明");
            }

            rows.Add(cells.ToArray());
        }

        Table(columns, rows);
    }

    public static string Chart(PairResult pair)
    {
        var bands = pair.Bands.Where(band => band.Difference.HasValue).ToList();
        if (bands.Count == 0)
        {
            return "<p>无可绘制差值，未生成图表。</p>";
        }

        if (bands.Count > 30)
        {
            return "<p>频带超过30个，为保证文字可读，报告使用完整数值表；界面可扫描较窄频率范围查看图表。</p>";
        }

        var maximum = Math.Max(1, bands.Max(band => Math.Abs(band.Difference!.Value)));
        var svg = new StringBuilder("<svg viewBox=\"0 0 820 270\" role=\"img\" aria-label=\"测得声级差图，横轴Hz，纵轴dB\"><text x=\"8\" y=\"20\" font-size=\"12\">ΔL / dB</text><line x1=\"60\" y1=\"130\" x2=\"800\" y2=\"130\" stroke=\"#a5bac8\"/>");
        svg.Append("<text x=\"8\" y=\"45\" font-size=\"11\">").Append(N(maximum)).Append("</text><text x=\"20\" y=\"135\" font-size=\"11\">0</text>");
        for (var index = 0; index < bands.Count; index++)
        {
            var band = bands[index];
            var x = 60 + (index + 0.5) * 740 / bands.Count;
            var width = Math.Min(45, 500.0 / bands.Count);
            var height = Math.Abs(band.Difference!.Value) / maximum * 85;
            var y = band.Difference >= 0 ? 130 - height : 130;
            svg.Append("<rect x=\"").Append(C(x - width / 2)).Append("\" y=\"").Append(C(y)).Append("\" width=\"").Append(C(width))
                .Append("\" height=\"").Append(C(height)).Append("\" fill=\"").Append(band.Difference >= 0 ? "#148c86" : "#be792e").Append("\"/>");
            svg.Append("<text x=\"").Append(C(x)).Append("\" y=\"245\" text-anchor=\"middle\" font-size=\"10\">").Append(C(band.Frequency)).Append("</text>");
            svg.Append("<text x=\"").Append(C(x)).Append("\" y=\"").Append(C(band.Difference >= 0 ? y - 8 : y + height + 15))
                .Append("\" text-anchor=\"middle\" font-size=\"11\">").Append(N(band.Difference)).Append("</text>");
        }

        return svg.Append("<text x=\"775\" y=\"265\" font-size=\"12\">Hz</text></svg>").ToString();
    }

    {
        output.Append("<h2>").Append(E(text)).AppendLine("</h2>");
    }

    {
        output.Append("<h3>").Append(E(text)).AppendLine("</h3>");
    }

    {

    }

    }

    private void Table(IEnumerable<string> headings, IEnumerable<object?[]> rows)
    {
        var columns = headings.ToList();
        output.Append("<table><thead><tr>");
        foreach (var heading in columns)
        {
            output.Append("<th>").Append(E(heading)).Append("</th>");
        }

        output.Append("</tr></thead><tbody>");
        var count = 0;
        foreach (var row in rows)
        {
            output.Append("<tr>");
            foreach (var cell in row)
            {
                output.Append("<td>").Append(E(cell?.ToString() ?? "—")).Append("</td>");
            }

            output.Append("</tr>");
            count++;
        }

        if (count == 0)
        {
            output.Append("<tr><td colspan=\"").Append(columns.Count).Append("\">暂无记录</td></tr>");
        }

        output.AppendLine("</tbody></table>");
    }

    {
        return WebUtility.HtmlEncode(value);
    }

    private static string N(double? value)
    {
        return Software.Format(value, 6);
    }

    {
        return value.ToString("0.###", CultureInfo.InvariantCulture);
    }

public sealed class BatchPairing
{
    public PairingPreview Preview(AcousticProject project, Guid baselineId, IReadOnlyList<Guid> targetIds, PairPurpose purpose)
    {
        var baseline = project.Trials.FirstOrDefault(trial => trial.Id == baselineId && trial.Kind == TrialKind.Baseline)
            ?? throw new DomainValidationException("baseline", "先明确选择一个基线试验");
        if (targetIds.Count == 0 || targetIds.Distinct().Count() != targetIds.Count)
        {
            throw new DomainValidationException("targets", "目标列表不能为空或重复");
        }

        var rows = new List<ProposedPair>();
        var validator = new ProjectValidator();
        var comparator = new Comparability();
        foreach (var id in targetIds)
        {
            var target = project.Trials.FirstOrDefault(trial => trial.Id == id)
                ?? throw new DomainValidationException("target", "目标试验不存在");
            var pair = new TrialPair
            {
                BaselineId = baseline.Id,
                TargetId = id,
                Purpose = purpose,
                Notes = "用户显式勾选目标并确认批量配对；共享基线时不作为独立配对估计区间"
            };
            var duplicate = project.Pairs.Any(item => item.BaselineId == pair.BaselineId && item.TargetId == pair.TargetId && item.Purpose == pair.Purpose);
            var candidate = ProjectCodec.Clone(project);
            candidate.Pairs.Add(pair);
            var structural = validator.Inspect(candidate).Where(error => error.Path.StartsWith("pairs", StringComparison.Ordinal)).ToList();
            if (structural.Count > 0 && !duplicate)
            {
                throw new DomainValidationException(structural);
            }

            var left = project.Conditions.Single(condition => condition.Id == baseline.ConditionId);
            var right = project.Conditions.Single(condition => condition.Id == target.ConditionId);
            var reasons = comparator.Inspect(left, right, project.Settings);
            if (baseline.Source != target.Source)
            {
                reasons.Add("来源类别不一致");
            }

            rows.Add(new ProposedPair(pair, baseline.Name, target.Name, reasons, duplicate));
        }

        return new PairingPreview(ProjectCodec.Fingerprint(project), rows);
    }

    public void Apply(ProjectSession session, PairingPreview preview)
    {
        if (!preview.CanApply || ProjectCodec.Fingerprint(session.Project) != preview.ProjectHash)
        {
            throw new DomainValidationException("pairing", "预检重复、为空或已过期，请重新选择目标预检");
        }

        session.Apply("用户确认批量配对", session.Project.Id.ToString(), project =>
            project.Pairs.AddRange(preview.Rows.Select(row => ProjectCodec.Clone(row.Pair))));
    }
}

using System.Text.Json;
using AcousticSeal.Core.Analysis;
using AcousticSeal.Core.Domain;
using AcousticSeal.Core.IO;

namespace AcousticSeal.Core.Operations;

public sealed class ProjectSession
{
    private readonly ProjectValidator validator = new();
    private readonly AnalysisEngine engine = new();
    private readonly List<HistoryFrame> undo = [];
    private readonly List<HistoryFrame> redo = [];
    public AcousticProject Project { get; private set; } = AcousticProject.Empty();
    public AnalysisReport Analysis { get; private set; }
    public bool Dirty { get; private set; }
    public string? SavedETag { get; private set; }
    public string SavedPath { get; private set; } = "";
    public bool CanUndo => undo.Count > 0;
    public bool CanRedo => redo.Count > 0;
    public string UndoLabel => undo.Count > 0 ? undo[^1].Action : "";
    public string RedoLabel => redo.Count > 0 ? redo[^1].Action : "";

    public ProjectSession()
    {
        Analysis = engine.Calculate(Project);
    }

    public void Open(AcousticProject project, string? etag = null, string path = "", bool changed = false)
    {
        validator.Ensure(project);
        var copy = ProjectCodec.Clone(project);
        var analysis = engine.Calculate(copy);
        Project = copy;
        Analysis = analysis;
        Dirty = changed;
        SavedETag = etag;
        SavedPath = path;
        undo.Clear();
        redo.Clear();
    }

    public void MarkSaved(SaveReceipt receipt)
    {
        SavedETag = receipt.ETag;
        SavedPath = receipt.Path;
        Dirty = false;
    }

    public void Apply(string action, string entity, Action<AcousticProject> change, bool allowArchived = false)
    {
        if (Project.Status == ProjectStatus.Archived && !allowArchived)
        {
            throw new DomainValidationException("status", "项目已归档，请先解除归档或创建副本");
        }

        var before = ProjectCodec.Clone(Project);
        var candidate = ProjectCodec.Clone(Project);
        change(candidate);
        var detail = DescribeChanges(before, candidate);
        Stamp(candidate, action, entity, detail);
        validator.Ensure(candidate);
        var analysis = engine.Calculate(candidate);
        undo.Add(new HistoryFrame(before, ProjectCodec.Clone(candidate), action));
        if (undo.Count > 20)
        {
            undo.RemoveAt(0);
        }

        redo.Clear();
        Project = candidate;
        Analysis = analysis;
        Dirty = true;
    }

    public void Undo()
    {
        if (!CanUndo)
        {
            return;
        }

        var frame = undo[^1];
        RestoreHistory(frame.Before, "撤销：" + frame.Action);
        undo.RemoveAt(undo.Count - 1);
        redo.Add(frame);
    }

    public void Redo()
    {
        if (!CanRedo)
        {
            return;
        }

        var frame = redo[^1];
        RestoreHistory(frame.After, "重做：" + frame.Action);
        redo.RemoveAt(redo.Count - 1);
        undo.Add(frame);
    }

    public void MakeCopy()
    {
        var copy = ProjectCodec.Clone(Project);
        var original = copy.Id;
        copy.Id = Guid.NewGuid();
        copy.Name = (copy.Name.Length > 150 ? copy.Name[..150] : copy.Name) + " 副本";
        copy.Status = ProjectStatus.Active;
        Stamp(copy, "复制项目", copy.Id.ToString(), "来源项目=" + original + "；测量来源仍沿用原记录，不代表重新采集");
        Open(copy, changed: true);
    }

    public void RecordReview(string reviewer, string comment)
    {
        var snapshot = Analysis;
        var input = ProjectCodec.Fingerprint(Project);
        Apply("记录分析复核", Project.Id.ToString(), project =>
        {
            project.Reviews.Add(new ReviewRecord
            {
                InputRevision = project.Revision,
                Reviewer = reviewer.Trim(),
                Comment = comment.Trim(),
                ValidPairs = snapshot.ValidPairCount,
                BlockedPairs = snapshot.BlockedPairCount,
                BlockingReasons = snapshot.Pairs.SelectMany(pair => pair.BlockingReasons.Select(reason => pair.Title + "：" + reason)).ToList(),
                SettingsJson = JsonSerializer.Serialize(project.Settings, ProjectCodec.Options),
                InputHash = input
            });
        });
    }

    private void RestoreHistory(AcousticProject state, string action)
    {
        var candidate = ProjectCodec.Clone(state);
        candidate.Audit = ProjectCodec.Clone(Project.Audit);
        candidate.Reviews = ProjectCodec.Clone(Project.Reviews);
        candidate.Revision = Project.Revision;
        Stamp(candidate, action, candidate.Id.ToString(), "恢复数据状态，保留累计审计与复核；撤销记录仅限当前会话20步");
        validator.Ensure(candidate);
        var analysis = engine.Calculate(candidate);
        Project = candidate;
        Analysis = analysis;
        Dirty = true;
    }

    private static void Stamp(AcousticProject project, string action, string entity, string detail)
    {
        project.Revision++;
        project.UpdatedAt = DateTimeOffset.UtcNow;
        project.SoftwareVersion = Software.Version;
        project.Audit.Add(new AuditEntry
        {
            Revision = project.Revision,
            Action = action,
            Entity = entity,
            Detail = detail.Length > 16000 ? detail[..16000] : detail,
            InputHash = ProjectCodec.Fingerprint(project)
        });
        if (project.Audit.Count > 10000)
        {
            project.Audit.RemoveRange(0, project.Audit.Count - 10000);
        }
    }

    private static string DescribeChanges(AcousticProject before, AcousticProject after)
    {
        using var left = JsonDocument.Parse(JsonSerializer.Serialize(before, ProjectCodec.Options));
        using var right = JsonDocument.Parse(JsonSerializer.Serialize(after, ProjectCodec.Options));
        var changes = new List<string>();
        foreach (var property in right.RootElement.EnumerateObject())
        {
            if (property.Name is "audit" or "reviews")
            {
                continue;
            }

            if (!left.RootElement.TryGetProperty(property.Name, out var previous) || previous.GetRawText() != property.Value.GetRawText())
            {
                changes.Add(property.Name);
            }
        }

        return changes.Count == 0 ? "记录复核或状态操作" : "修改字段/集合：" + string.Join("、", changes);
    }

    private sealed record HistoryFrame(AcousticProject Before, AcousticProject After, string Action);
}

    }
    private void OnClosing(object? sender, CancelEventArgs e)
    {
        // Ensure the focused native editor transfers its last keystroke before the save decision.
        FocusManagerCommit();
        e.Cancel = !workspace.CanClose();
    }
    private void FocusManagerCommit()
    {
        if (System.Windows.Input.Keyboard.FocusedElement is System.Windows.Controls.TextBox box)
            box.GetBindingExpression(System.Windows.Controls.TextBox.TextProperty)?.UpdateSource();
    }

        if (args.OldValue is INotifyCollectionChanged oldItems) oldItems.CollectionChanged -= plot.CollectionChanged;
        if (args.NewValue is INotifyCollectionChanged newItems) newItems.CollectionChanged += plot.CollectionChanged;

    protected override void OnRender(DrawingContext dc)
    {

        dc.DrawRectangle(Brushes.White, null, new Rect(0, 0, ActualWidth, ActualHeight));
        var bands = Items?.Cast<BandDifference>().OrderBy(item => item.Frequency).ToList() ?? [];
        var left = 66.0; var top = 38.0;
        var width = Math.Max(10, ActualWidth - left - 30);
        var height = Math.Max(10, ActualHeight - top - 58);

        var copper = new SolidColorBrush(Color.FromRgb(182, 93, 51));
        void Text(string text, double x, double y, Brush? brush = null, double size = 12)
        {

            dc.DrawText(formatted, new Point(x, y));
        }
        Text(Original ? "基线（青） / 目标（铜） · dB re 20 µPa · 纵轴截断" : "测得声级差 ΔL = 基线 − 目标 (dB) · 零基准", 10, 8, teal);
        if (bands.Count == 0 || bands.All(item => Original ? !item.Baseline.HasValue && !item.Target.HasValue : !item.Difference.HasValue))
        {
            Text("无可绘制数据；检查配对状态、缺失频带或显示范围。", left, top + 50);
            return;
        }
        var values = bands.SelectMany(item => Original ? new[] { item.Baseline, item.Target } : new[] { item.Difference }).Where(value => value.HasValue).Select(value => value!.Value).ToList();
        var min = Original ? Math.Floor(values.Min() - 1) : Math.Min(0, Math.Floor(values.Min() - 0.5));
        var max = Original ? Math.Ceiling(values.Max() + 1) : Math.Max(0, Math.Ceiling(values.Max() + 0.5));
        if (max <= min) max = min + 1;
        var xMin = Math.Log10(bands.Min(item => item.Frequency));
        var xMax = Math.Log10(bands.Max(item => item.Frequency));
        if (xMax <= xMin) { xMin -= 0.2; xMax += 0.2; }
        Point Position(double frequency, double value) => new(left + (Math.Log10(frequency) - xMin) / (xMax - xMin) * width, top + height - (value - min) / (max - min) * height);
        for (var tick = 0; tick <= 4; tick++)
        {
            var value = min + (max - min) * tick / 4;
            var y = top + height - height * tick / 4;

            Text(value.ToString("0.##", CultureInfo.InvariantCulture), 8, y - 7);

        {
            var x = left + width * tick / 4;
            Text(Math.Pow(10, xMin + (xMax - xMin) * tick / 4).ToString("0.#", CultureInfo.InvariantCulture), x - 10, top + height + 8);
        }
        Text("Hz（对数坐标）；缺失处断线，不插值。", left, ActualHeight - 20);
        void Series(Func<BandDifference, double?> get, Brush brush)
        {
            Point? previous = null;
            foreach (var band in bands)
            {
                var value = get(band);
                if (!value.HasValue) { previous = null; continue; }
                var point = Position(band.Frequency, value.Value);

                previous = point;
            }
        }
        if (Original) { Series(item => item.Baseline, teal); Series(item => item.Target, copper); }
        else Series(item => item.Difference, teal);

    }

    public AcousticProject Open(string path)
    {
        var full = System.IO.Path.GetFullPath(path);
        var bytes = Read(full);
        var decoded = codec.Decode(bytes);
        var separate = decoded.Migrated || full.EndsWith(".bak", StringComparison.OrdinalIgnoreCase);
        Path = separate ? "" : full;
        expectedHash = separate ? null : Hash(bytes);
        OpenNotes = string.Join("；", decoded.Notes) + (separate ? "；旧格式或备份请另存为新的.sealproj，原文件保持不变。" : "");
        return decoded.Project;
    }
    public void Reset()
    {
        Path = "";
        expectedHash = null;
    }
    public SaveReceipt Save(AcousticProject project, string path, bool overwriteConfirmed)
    {
        var full = System.IO.Path.GetFullPath(path);
        var bytes = codec.Encode(project);
        if (bytes.Length > Software.MaximumFileBytes) throw new DomainValidationException("项目", "文件超过12MiB");
        Directory.CreateDirectory(System.IO.Path.GetDirectoryName(full)!);
        using var fileLock = new FileStream(full + ".lock", FileMode.OpenOrCreate, FileAccess.ReadWrite, FileShare.None);
        var exists = File.Exists(full);
        var actual = exists ? Hash(Read(full)) : null;
        var samePath = string.Equals(full, Path, OperatingSystem.IsWindows() ? StringComparison.OrdinalIgnoreCase : StringComparison.Ordinal);
        if (samePath && expectedHash != actual) throw new ProjectConflictException();
        if (!samePath && exists && !overwriteConfirmed) throw new IOException("目标已存在，必须确认覆盖或改用新文件名。");
        var temporary = full + "." + Guid.NewGuid().ToString("N") + ".tmp";
        try
        {
            using (var stream = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None))
            {
                stream.Write(bytes);
                stream.Flush(true);
            }
            if (exists) File.Copy(full, full + ".bak", true);
            File.Move(temporary, full, true);
        }
        finally
        {
            if (File.Exists(temporary)) File.Delete(temporary);
        }
        Path = full;
        expectedHash = Hash(bytes);
        return new SaveReceipt(full, exists ? full + ".bak" : "", expectedHash, bytes.Length);
    }
    public static void Export(string path, byte[] bytes)
    {
        var full = System.IO.Path.GetFullPath(path);
        var temporary = full + "." + Guid.NewGuid().ToString("N") + ".tmp";
        try
        {
            File.WriteAllBytes(temporary, bytes);
            File.Move(temporary, full, true);
        }
        finally { if (File.Exists(temporary)) File.Delete(temporary); }
    }
    public static byte[] Read(string path)
    {
        var info = new FileInfo(path);
        if (info.Length > Software.MaximumFileBytes) throw new DomainValidationException("文件", "文件超过12MiB，未读取");
        return File.ReadAllBytes(path);
    }
    private static string Hash(byte[] bytes) => Convert.ToHexStringLower(SHA256.HashData(bytes));
}

    {
        // Work on a fresh copy so a rejected transaction leaves both saved inputs and draft intact.

                    throw new DomainValidationException(field.Label, "请输入带时区ISO时间，例如2026-11-04T10:00:00+08:00");
                value = time;
            }
            else value = text == "" && Nullable.GetUnderlyingType(property.PropertyType) is not null ? null : ParseNumber(text, field.Label);
            property.SetValue(copy, value);
        }
        if (copy is AcousticTrial trial)
        {
            trial.LocationIds = Locations.Where(item => item.Selected).Select(item => item.Id).ToList();

    public Workspace(IUserDialogs dialogs)
    {
        this.dialogs = dialogs;
        ActionCommand Command(Action action) => new(() => { action(); return Task.CompletedTask; }, Fail);
        NewCommand = Command(() => Replace(AcousticProject.Empty(), "已新建空白项目。"));
        DemoCommand = Command(() => Replace(Demonstration.Create(), "已载入独立模拟案例；全部数据为模拟，不代表实车效果。"));
        OpenCommand = Command(Open);
        SaveCommand = Command(() => Save(false));
        SaveAsCommand = Command(() => Save(true));
        ApplyCommand = Command(Apply);
        RevertCommand = Command(() => { if (Editor?.Dirty != true || dialogs.Confirm("放弃编辑", "放弃当前未应用编辑？")) LoadEditor(); });
        AddCommand = Command(Add);
        DeleteCommand = Command(Delete);
        CopyCommand = Command(CopyProject);
        DuplicateTrialCommand = Command(DuplicateTrial);
        UndoCommand = Command(() => { if (!ResolveDraft()) return; session.Undo(); Refresh(); Status = "已撤销输入；累计审计保留。"; });
        RedoCommand = Command(() => { if (!ResolveDraft()) return; session.Redo(); Refresh(); Status = "已重做输入。"; });
        AddBandCommand = Command(() => Editor?.AddBand());
        RemoveBandCommand = Command(() => Editor?.RemoveBand(SelectedBand));
        PasteCommand = Command(() => { Editor?.Paste(PasteText); Status = "两列数据已放入草稿，点击应用编辑保存到项目。"; });
        ImportFileCommand = Command(ReadCsv);
        PreviewCommand = Command(Preview);
        ImportCommand = Command(Import);
        ReportCommand = Command(() => Export(new ResultExport().Report(session.Project, session.Analysis), "HTML报告|*.html"));
        ResultsCommand = Command(() => Export(new ResultExport().Results(session.Project, session.Analysis), "结果CSV|*.csv"));
        MeasurementsCommand = Command(() => Export(new ResultExport().Measurements(session.Project), "频带CSV|*.csv"));
        JsonCommand = Command(() => Export(new ResultExport().AnalysisJson(session.Project, session.Analysis), "分析JSON|*.json"));
        ReviewCommand = Command(() => { if (!ResolveDraft()) return; session.RecordReview(Reviewer, ReviewComment); Refresh(); Status = "已记录输入指纹及当前分析快照，请保存项目。"; });
        FilterCommand = Command(Filter);
        WorkbookCommand = Command(Workbook);
        RevisionCommand = Command(CompareRevision);
        ScenarioCommand = Command(Scenario);
        ArchiveCommand = Command(() => { if (!ResolveDraft()) return; ProjectCommands.SetArchived(session, session.Project.Status != ProjectStatus.Archived); Refresh(); Status = "项目状态：" + session.Project.Status + "；归档后可查看和导出。"; });
        selectedModule = Modules[0];
        Refresh();
    }

    private void Fail(Exception error)
    {
        Status = "操作未完成：" + error.Message;
        dialogs.ShowError(Status);
        Notify(nameof(Title));
    }
    private bool ResolveDraft()
    {
        if (Editor?.Dirty != true) return true;
        var answer = dialogs.SaveChanges("当前有未应用编辑。选择“是”应用编辑，“否”放弃编辑，“取消”保留当前工作页。");
        if (answer is null) return false;
        if (answer is true)
        {
            try { Apply(); }
            catch (Exception error) { Fail(error); return false; }
        }
        else LoadEditor();
        return true;
    }
    public bool CanClose()
    {
        try
        {
            if (!ResolveDraft()) return false;
            if (!session.Dirty) return true;
            var answer = dialogs.SaveChanges("项目有未保存修改。是否保存到磁盘后继续？");
            if (answer is null) return false;
            return answer is false || Save(false);
        }
        catch (Exception error) { Fail(error); return false; }
    }
    private void Replace(AcousticProject project, string message)
    {
        if (!CanClose()) return;
        session.Open(project, changed: project.Trials.Count > 0);
        files.Reset();
        selectedModule = Modules[0];
        selectedItem = null;
        Refresh();
        Status = message;
    }
    private void Open()
    {
        if (!CanClose()) return;
        var path = dialogs.OpenFile("打开项目或备份", "封堵项目/备份|*.sealproj;*.bak;*.aslp.json|所有文件|*.*");
        if (path is null) return;
        var project = files.Open(path);
        session.Open(project, changed: files.Path.Length == 0);
        selectedModule = Modules[0];
        selectedItem = null;
        Refresh();
        Status = "已打开：" + path + files.OpenNotes;
    }
    private bool Save(bool saveAs)
    {
        if (!ResolveDraft()) return false;
        var target = !saveAs && files.Path.Length > 0 ? files.Path : dialogs.SaveFile("保存项目", "封堵项目|*.sealproj", ResultExport.SafeName(session.Project.Name) + Software.Extension);
        if (target is null) return false;
        var different = !string.Equals(target, files.Path, StringComparison.OrdinalIgnoreCase);
        var overwrite = different && File.Exists(target);
        if (overwrite && !dialogs.Confirm("覆盖项目", "目标存在，将保留.bak备份并覆盖：" + target)) return false;
        var receipt = files.Save(session.Project, target, overwrite);
        session.MarkSaved(receipt);
        NotifyState();
        Status = "已保存：" + receipt.Path + (receipt.BackupPath.Length > 0 ? "；备份：" + receipt.BackupPath : "");
        return true;
    }
    public void Apply()
    {
        if (Editor is null) return;
        var model = Editor.Read();
        switch (model)
        {
            case AcousticProject metadata:
                session.Apply("应用项目档案", metadata.Id.ToString(), project =>
                {
                    project.Name = metadata.Name; project.Owner = metadata.Owner;
                    project.VehicleProgramme = metadata.VehicleProgramme; project.Objective = metadata.Objective;
                    project.Description = metadata.Description;
                });
                break;
            case SealLocation location: ProjectCommands.SaveLocation(session, location); break;
            case MeasurementCondition condition: ProjectCommands.SaveCondition(session, condition); break;
            case AcousticTrial trial: ProjectCommands.SaveTrial(session, trial); break;
            case TrialPair pair: ProjectCommands.SavePair(session, pair); break;
            case AnalysisSettings settings: ProjectCommands.SaveSettings(session, settings); break;
        }
        var id = Guid.TryParse(Editor.Identity, out var key) ? key : Guid.Empty;
        Refresh(id);
        Status = "已应用编辑并重新计算。项目尚需保存到磁盘。";
    }
    private void Add()
    {
        if (!ResolveDraft()) return;
        object model = SelectedModule?.Key switch
        {
            "location" => new SealLocation(),
            "condition" => new MeasurementCondition(),
            "trial" => new AcousticTrial { ConditionId = session.Project.Conditions.FirstOrDefault()?.Id ?? Guid.Empty },
            "pair" => new TrialPair(),
            _ => throw new DomainValidationException("工作页", "仅位置、工况、试验、配对可新增")
        };
        selectedItem = null;
        Notify(nameof(SelectedItem));
        SetEditor(model);
        Status = "新增草稿，必填项完成后点击应用编辑；来源说明不可空白。";
    }
    private void Delete()
    {
        if (SelectedItem is null) throw new DomainValidationException("删除", "先选择列表中的对象");
        if (!dialogs.Confirm("删除对象", "删除“" + SelectedItem.Title + "”？关联引用存在时会阻止删除。")) return;
        var id = SelectedItem.Id;
        switch (SelectedModule?.Key)
        {
            case "location": ProjectCommands.DeleteLocation(session, id); break;
            case "condition": ProjectCommands.DeleteCondition(session, id); break;
            case "trial": ProjectCommands.DeleteTrial(session, id); break;
            case "pair": ProjectCommands.DeletePair(session, id); break;
        }
        selectedItem = null;
        Refresh();
        Status = "已删除；可通过撤销恢复，保存前磁盘文件未改变。";
    }
    private void CopyProject()
    {
        if (!ResolveDraft()) return;
        session.MakeCopy(); files.Reset(); selectedItem = null; Refresh();
        Status = "已创建新的项目标识，来源仍保留原记录；请另存。";
    }
    private void DuplicateTrial()
    {
        if (!ResolveDraft()) return;
        if (SelectedModule?.Key == "trial" && SelectedItem is not null)
        {
            var id = ProjectCommands.DuplicateTrial(session, SelectedItem.Id);
            Refresh(id);
            Status = "已复制试验结构；未复制频带测量和配对，请核对时间和来源。";
        }
        else throw new DomainValidationException("复制试验结构", "先在试验工作表选择一条已应用试验");
    }
    private void Refresh(Guid? focus = null)
    {
        changing = true;
        var id = focus ?? selectedItem?.Id;
        Items.Clear();
        IEnumerable<ObjectItem> values = SelectedModule?.Key switch
        {
            "location" => session.Project.Locations.Select(item => new ObjectItem(item.Id, item.Name, item.Zone + " · " + item.Material)),
            "condition" => session.Project.Conditions.Select(item => new ObjectItem(item.Id, item.Name, item.Vehicle + " · " + item.Weighting)),
            "trial" => session.Project.Trials.Select(item => new ObjectItem(item.Id, item.Name, item.Kind.Text() + " · " + item.Source.Text())),
            "pair" => session.Project.Pairs.Select(item => new ObjectItem(item.Id, session.Project.TrialName(item.BaselineId) + " → " + session.Project.TrialName(item.TargetId), item.Purpose.Text())),
            _ => []
        };
        foreach (var value in values) Items.Add(value);
        selectedItem = Items.FirstOrDefault(item => item.Id == id) ?? Items.FirstOrDefault();
        var resultId = selectedResult?.PairId;
        Results.Clear();
        foreach (var result in session.Analysis.Pairs) Results.Add(result);
        selectedResult = Results.FirstOrDefault(item => item.PairId == resultId) ?? Results.FirstOrDefault();
        Quality.Clear();
        foreach (var issue in session.Analysis.Issues) Quality.Add(new(issue.Severity.Text(), issue.Title, issue.Reason, issue.Repair));
        foreach (var repeat in session.Analysis.Repeatability)
            foreach (var band in repeat.Bands)
                Quality.Add(new(band.Status.Text(), repeat.Name + " / " + band.Frequency + " Hz", $"n={band.Statistics.Count}, SD={Software.Format(band.Statistics.StandardDeviation)} dB, 缺失={band.MissingCount}；" + string.Join("；", repeat.BlockingReasons), "同组工况须两两可比，n<2不产生样本标准差。"));
        Audit.Clear(); foreach (var entry in session.Project.Audit.AsEnumerable().Reverse()) Audit.Add(entry);
        Reviews.Clear(); foreach (var review in session.Project.Reviews.AsEnumerable().Reverse()) Reviews.Add(review);
        InvalidateImport();
        changing = false;
        LoadEditor();
        LoadResult();
        NotifyState();
    }
    private void NotifyState()
    {
        foreach (var property in new[] { nameof(Title), nameof(ProjectSummary), nameof(SourceSummary), nameof(FilePath), nameof(PageTitle), nameof(PageHint), nameof(Inspector), nameof(SelectedModule), nameof(SelectedItem), nameof(SelectedResult), nameof(IsList), nameof(IsResult), nameof(IsQuality), nameof(IsAudit), nameof(IsImport), nameof(IsTrial), nameof(IsPair), nameof(ScenarioParameters) }) Notify(property);
    }
    private void LoadEditor()
    {
        object? model = SelectedModule?.Key switch
        {
            "project" => ProjectCodec.Clone(session.Project),
            "settings" => ProjectCodec.Clone(session.Project.Settings),
            "location" => session.Project.Locations.FirstOrDefault(item => item.Id == selectedItem?.Id) is { } location ? ProjectCodec.Clone(location) : null,
            "condition" => session.Project.Conditions.FirstOrDefault(item => item.Id == selectedItem?.Id) is { } condition ? ProjectCodec.Clone(condition) : null,
            "trial" => session.Project.Trials.FirstOrDefault(item => item.Id == selectedItem?.Id) is { } trial ? ProjectCodec.Clone(trial) : null,
            "pair" => session.Project.Pairs.FirstOrDefault(item => item.Id == selectedItem?.Id) is { } pair ? ProjectCodec.Clone(pair) : null,
            _ => null
        };
        SetEditor(model);
    }
    private void SetEditor(object? model)
    {
        Editor = model is null ? null : new EntityEditor(model, session.Project);
        if (Editor is not null) Editor.PropertyChanged += (_, _) => Notify(nameof(Title));
    }
    private void ReadCsv()
    {
        var path = dialogs.OpenFile("导入UTF-8频带CSV", "CSV|*.csv|所有文件|*.*");
        if (path is null) return;
        var info = new FileInfo(path);
        if (info.Length > 5 * 1024 * 1024) throw new DomainValidationException("CSV", "文件超过5MiB");
        ImportText = CsvImportService.DecodeUtf8(File.ReadAllBytes(path));
        Status = "已读取CSV，请预检并核对覆盖数量：" + path;
    }
    private void InvalidateImport()
    {
        importPreview = null;
        ImportStatus = "当前内容尚未预检。修改内容、分隔符、模式或项目后须重新预检。";
    }
    private void Preview()
    {
        importPreview = importer.Preview(session.Project, ImportText, ImportMode, Delimiter == "tab" ? '\t' : Delimiter[0]);
        ImportStatus = $"数据 {importPreview.Rows} 行；覆盖 {importPreview.Replaced}；缺失 {importPreview.Missing}；" +
            (importPreview.CanApply ? "可以应用。" : "预检失败，整批未应用。") + "\n" +
            string.Join("\n", importPreview.Changes.Select(item => item.TrialName + $"：新增 {item.Added}，覆盖 {item.Replaced}，缺失 {item.Missing}")) + "\n" +
            string.Join("\n", importPreview.Problems.Select(item => item.ToString()));
    }
    private void Import()
    {
        if (importPreview is null) throw new DomainValidationException("导入", "先预检当前内容");
        if (!importPreview.CanApply) throw new DomainValidationException("导入", "预检未通过，请修复指定行");
        if (!dialogs.Confirm("整体应用导入", ImportStatus + "\n确认应用？替换模式将清空涉及试验的旧频带。")) return;
        importer.Apply(session, importPreview);
        Refresh(); Status = "已整体应用导入并重新计算，请保存项目。";
    }
    private void Export(ExportDocument document, string filter)
    {
        if (Editor?.Dirty == true) throw new DomainValidationException("导出", "先应用或放弃当前编辑，避免导出旧输入");
        var path = dialogs.SaveFile("导出成果", filter, document.Filename);
        if (path is null) return;
        DesktopFiles.Export(path, document.Bytes);
        Status = "已导出：" + path;
    }
    private void Filter()
    {
        double? Parse(string text) => string.IsNullOrWhiteSpace(text) ? null :
            double.TryParse(text, NumberStyles.Float, CultureInfo.InvariantCulture, out var value) && double.IsFinite(value) && value is >= 1 and <= 100000 ? value : throw new DomainValidationException("频带范围", "请输入1—100000Hz或空白");
        var low = Parse(Minimum); var high = Parse(Maximum);
        if (low > high) throw new DomainValidationException("频带范围", "下限不得大于上限");
        appliedMinimum = low; appliedMaximum = high; LoadResult();
        Status = "已筛选显示频带，完整报告仍导出全部频带。";
    }
    private void LoadResult()
    {
        ResultBands.Clear();
        foreach (var band in selectedResult?.Bands ?? [])
            if ((!appliedMinimum.HasValue || band.Frequency >= appliedMinimum) && (!appliedMaximum.HasValue || band.Frequency <= appliedMaximum)) ResultBands.Add(band);
        Notify(nameof(ResultNote));
        Notify(nameof(SelectedResult));
    }
    private void Workbook()
    {
        if (selectedResult is null) throw new DomainValidationException("原始核对", "请先建立并选择一个显式配对");
        var ids = new List<Guid> { selectedResult.BaselineId, selectedResult.TargetId };
        var workbook = new InputWorkbook().Build(session.Project, ids, appliedMinimum, appliedMaximum);
        Export(new ExportDocument("原始频带核对.csv", "text/csv", new InputWorkbook().Csv(workbook)), "CSV核对表|*.csv");
    }
    private void CompareRevision()
    {
        var path = dialogs.OpenFile("选择用于输入内容核对的项目（当前工作项目保持原样）", "项目|*.sealproj;*.bak|所有文件|*.*");
        if (path is null) return;
        var right = new ProjectCodec().Decode(DesktopFiles.Read(path)).Project;
        var result = new RevisionComparison().Compare(session.Project, right);
        AuxiliaryResult = JsonSerializer.Serialize(result, ProjectCodec.Options);
        Notify(nameof(AuxiliaryResult));
        Export(new ExportDocument("项目输入版本核对.json", "application/json", Encoding.UTF8.GetBytes(AuxiliaryResult)), "JSON核对|*.json");
    }
    private void Scenario()
    {
        var values = ScenarioValues.Split(',').Select(text => double.Parse(text.Trim(), CultureInfo.InvariantCulture)).ToList();
        var service = new SettingsScenarios();
        var result = service.Compare(session.Project, ScenarioKey, values);
        AuxiliaryResult = result.Note + "\n" + string.Join("\n", result.Scenarios.Select(item => $"{item.Value}：可比{item.ValidPairs} / 阻断{item.BlockedPairs} / 漂移恢复警示{item.ScreeningAlerts} / 重复性警示{item.RepeatabilityAlerts}"));
        Notify(nameof(AuxiliaryResult));
        Export(service.Export(result), "规则情景JSON|*.json");
    }
}
