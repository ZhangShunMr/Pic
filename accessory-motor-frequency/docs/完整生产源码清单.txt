using System.Globalization;
using MotorFrequency.Core.Domain;
using MotorFrequency.Core.Services;

namespace MotorFrequency.App.ViewModels;

public sealed class MotorDraft
{
    public string Name { get; set; } = "当前电机";
    public MotorTopology Topology { get; set; }
    public string PolePairs { get; set; } = "2";
    public string Slots { get; set; } = "12";
    public string Segments { get; set; } = "10";
    public string MinimumRpm { get; set; } = "0";
    public string MaximumRpm { get; set; } = "6000";
    public string StepRpm { get; set; } = "100";
    public string InspectRpm { get; set; } = "1800";
    public string MaximumHz { get; set; } = "1500";
    public bool IncludeSlotPass { get; set; } = true;
    public bool IncludeCogging { get; set; }
    public bool Symmetric { get; set; }
    public string Assumptions { get; set; } = "";

    public static MotorDraft From(MotorConfiguration motor)
    {
        return new MotorDraft
        {
            Name = motor.Name,
            Topology = motor.Topology,
            PolePairs = Labels.Number(motor.PolePairs),
            Slots = Labels.Number(motor.Slots),
            Segments = Labels.Number(motor.CommutatorSegments),
            MinimumRpm = Labels.Number(motor.MinimumRpm),
            MaximumRpm = Labels.Number(motor.MaximumRpm),
            StepRpm = Labels.Number(motor.StepRpm),
            InspectRpm = Labels.Number(motor.InspectRpm),
            MaximumHz = Labels.Number(motor.MaximumHz),
            IncludeSlotPass = motor.IncludeSlotPass,
            IncludeCogging = motor.IncludeIdealCogging,
            Symmetric = motor.SymmetricUnskewedAssumption,
            Assumptions = motor.AssumptionNotes
        };
    }

    public MotorConfiguration Build(MotorConfiguration previous)
    {
        var result = ProjectCodec.Clone(previous);
        result.Name = Name.Trim();
        result.Topology = Topology;
        result.PolePairs = ParseInt(PolePairs, "极对数");
        result.Slots = ParseInt(Slots, "槽数");
        result.CommutatorSegments = ParseInt(Segments, "换向片数");
        result.MinimumRpm = Parse(MinimumRpm, "起始转速");
        result.MaximumRpm = Parse(MaximumRpm, "终止转速");
        result.StepRpm = Parse(StepRpm, "扫描步长");
        result.InspectRpm = Parse(InspectRpm, "观察转速");
        result.MaximumHz = Parse(MaximumHz, "频率上限");
        result.IncludeSlotPass = IncludeSlotPass;
        result.IncludeIdealCogging = IncludeCogging;
        result.SymmetricUnskewedAssumption = Symmetric;
        result.AssumptionNotes = Assumptions.Trim();
        return result;
    }

    public static double Parse(string value, string name)
    {
        if (!double.TryParse(value, NumberStyles.Float, CultureInfo.InvariantCulture, out var number) || !double.IsFinite(number))
        {
            throw new InvalidDataException(name + "：请输入有限数值，小数点用英文句点。");
        }
        return number;
    }

    public static int ParseInt(string value, string name)
    {
        if (!int.TryParse(value, NumberStyles.Integer, CultureInfo.InvariantCulture, out var number))
        {
            throw new InvalidDataException(name + "：请输入整数。");
        }
        return number;
    }
}

using MotorFrequency.Core.Domain;
using MotorFrequency.Core.Services;

namespace MotorFrequency.App.ViewModels;

public sealed class SweepViewModel
{
    public string PolePairs { get; set; } = "1,2,3";
    public string Slots { get; set; } = "6,12,18";
    public string Segments { get; set; } = "8,10,12";
    public string ReferenceRpm { get; set; } = "1800";
    public string ObservedHz { get; set; } = "60";
    public string ToleranceHz { get; set; } = "3";
    public string InputHash { get; private set; } = "";
    public string Summary { get; private set; } = "尚未运行参数扫查。";
    public IReadOnlyList<SweepRow> Rows { get; private set; } = [];
    public IReadOnlyList<ReverseRow> ReverseRows { get; private set; } = [];
    public bool Error { get; private set; }
    public bool OnlyNearby { get; set; }
    public SweepRequest? LastRequest { get; private set; }

    public IReadOnlyList<SweepRow> DisplayRows => OnlyNearby ? Rows.Where(x => x.WithinTolerance).ToList() : Rows;

    public bool Stale(Project project)
    {
        return InputHash.Length > 0 && InputHash != ContextHash(project);
    }

    private static string ContextHash(Project project)
    {
        return ParameterSweep.ContextHash(project);
    }

    public void Run(Project project)
    {
        try
        {
            var request = new SweepRequest
            {
                ReferenceRpm = MotorDraft.Parse(ReferenceRpm, "参考转速"),
                ObservedHz = MotorDraft.Parse(ObservedHz, "观察频率"),
                ToleranceHz = MotorDraft.Parse(ToleranceHz, "容差")
            };
            if (project.Motor.Topology == MotorTopology.BrushlessPermanentMagnet)
            {
                request.PolePairs = ParameterSweep.ParseIntegers(PolePairs, "极对数组", 1, 64).ToList();
                request.Slots = ParameterSweep.ParseIntegers(Slots, "槽数组", 1, 512).ToList();
            }
            else
            {
                request.Segments = ParameterSweep.ParseIntegers(Segments, "换向片数组", 2, 512).ToList();
            }
            Rows = ParameterSweep.Calculate(project, request);
            ReverseRows = ParameterSweep.Reverse(project, request.ObservedHz, request.ToleranceHz);
            LastRequest = request;
            InputHash = ContextHash(project);
            Error = false;
            Summary = $"扫查完成：{Rows.Count} 条候选参考，{Rows.Count(x => x.WithinTolerance)} 条在 ±{Labels.Number(request.ToleranceHz)} Hz 内；不作可制造性或噪声优劣排序。";
        }
        catch (Exception ex) when (ex is IOException or InvalidDataException or ArgumentException)
        {
            Rows = [];
            ReverseRows = [];
            LastRequest = null;
            InputHash = "";
            Error = true;
            Summary = ex.Message;
        }
    }

    public void Clear()
    {
        Rows = [];
        ReverseRows = [];
        InputHash = "";
        LastRequest = null;
        Error = false;
        Summary = "扫查结果已清除；项目输入没有改变。";
    }
}

    private void SetCurrent(Project project, string? readRevision)
    {

    private void SyncDrafts()
    {
        if (Current is null) return;

        ProjectName = Current.Name;
        Owner = Current.Owner;
        Purpose = Current.Purpose;
        Tolerance = Labels.Number(Current.Review.MatchToleranceHz);

    private void Mutate(string action, string detail, Action<Project> change)
    {
        Execute(() =>
        {
            if (Current is null) throw new InvalidDataException("请先新建项目。");
            if (Current.Archived && action != "取消归档") throw new InvalidDataException("项目已归档；请先取消归档，或复制项目。");
            var before = ProjectCodec.Serialize(Current);
            var next = ProjectCodec.Clone(Current);
            change(next);
            next.ModifiedAt = DateTimeOffset.UtcNow;
            next.History.Add(new AuditEvent { Action = action, Detail = detail });
            if (next.History.Count > 2000) next.History.RemoveRange(0, next.History.Count - 2000);
            ProjectValidator.ThrowIfInvalid(next);
            undo.Push(before);
            if (undo.Count > 30)
            {
                var keep = undo.Take(30).Reverse().ToList();
                undo.Clear();
                foreach (var text in keep) undo.Push(text);
            }
            redo.Clear();
            Current = next;
        }, action + "完成；结果已按当前输入更新。" );
    }

    public async Task ReadCsv(IBrowserFile file)
    {
        await ExecuteAsync(async () =>
        {
            if (!file.Name.EndsWith(".csv", StringComparison.OrdinalIgnoreCase)) throw new InvalidDataException("请选择 .csv 文件。");
            using var stream = file.OpenReadStream(SpectrumCsv.MaximumBytes);
            using var reader = new StreamReader(stream, new UTF8Encoding(false, true));
            CsvText = await reader.ReadToEndAsync();
            CsvFileName = file.Name;
            CsvPickerVersion++;
            if (string.IsNullOrWhiteSpace(SpectrumName)) SpectrumName = Path.GetFileNameWithoutExtension(file.Name);
            Preview = null;
        }, "CSV 已读取。填写元数据后点击预检，确认后导入。" );
    }

    public void Undo()
    {
        Execute(() =>
        {
            if (Current is null || undo.Count == 0) return;
            redo.Push(ProjectCodec.Serialize(Current));
            var restored = ProjectCodec.Parse(undo.Pop());
            restored.History = ProjectCodec.Clone(Current.History);
            restored.History.Add(new AuditEvent { Action = "撤销", Detail = "恢复上一个输入状态；审计记录保留" });
            if (restored.History.Count > 2000) restored.History.RemoveRange(0, restored.History.Count - 2000);
            restored.ModifiedAt = DateTimeOffset.UtcNow;
            Current = restored;
            SyncDrafts();
        }, "已撤销；请检查后保存。" );
    }

    {

        {

            undo.Push(ProjectCodec.Serialize(Current));
            var restored = ProjectCodec.Parse(redo.Pop());
            restored.History = ProjectCodec.Clone(Current.History);
            restored.History.Add(new AuditEvent { Action = "重做", Detail = "恢复下一输入状态；审计记录保留" });
            if (restored.History.Count > 2000) restored.History.RemoveRange(0, restored.History.Count - 2000);
            restored.ModifiedAt = DateTimeOffset.UtcNow;
            Current = restored;
            SyncDrafts();
        }, "已重做；请检查后保存。" );
    }

    public async Task Export(string kind)
    {
        await ExecuteAsync(async () =>
        {
            if (kind == "template")
            {
                await Download("频谱导入模板.csv", "text/csv;charset=utf-8", SpectrumCsv.Template(), true);
                return;
            }
            if (Current is null) throw new InvalidDataException("请先打开项目。");
            var name = ExportService.SafeName(Current.Name);
            switch (kind)
            {
                case "review":
                    await Download(name + "-输入核对.json", "application/json", ProjectReview.Json(Current));
                    break;
                case "project":
                    await Download(name + Software.Extension, "application/json", ProjectCodec.Serialize(Current));
                    break;
                case "frequency":
                    await Download(name + "-频率扫描.csv", "text/csv;charset=utf-8", ExportService.FrequencyCsv(Current), true);
                    break;
                case "crossing":
                    await Download(name + "-频段交叠.csv", "text/csv;charset=utf-8", ExportService.CrossingCsv(Current), true);
                    break;
                case "evidence":
                    await Download(name + "-谱峰核对.csv", "text/csv;charset=utf-8", ExportService.EvidenceCsv(Current), true);
                    break;
                case "comparison":
                    if (SelectedScenario is null) throw new InvalidDataException("请先选中方案快照。");
                    await Download(name + "-方案比较.csv", "text/csv;charset=utf-8", ExportService.ComparisonCsv(Current, SelectedScenario), true);
                    break;
                case "svg":
                    await Download(name + "-频率图.svg", "image/svg+xml;charset=utf-8", new ChartGeometry(Current.Motor).Svg(Current));
                    break;
                case "html":
                    await Download(name + "-复核报告.html", "text/html;charset=utf-8", ExportService.Html(Current));
                    break;
                case "raw":
                    if (SelectedSpectrum is null) throw new InvalidDataException("请先选中频谱。");
                    await Download(ExportService.SafeName(SelectedSpectrum.Name) + "-原始频点.csv", "text/csv;charset=utf-8", SpectrumCsv.Export(SelectedSpectrum), true);
                    break;
                default:
                    throw new InvalidDataException("不支持的导出类型。");
            }
        }, "已发起浏览器下载；目标为浏览器下载目录，请在下载列表核实文件。导出使用已应用输入。" );
    }

    private async Task Download(string name, string mime, string text, bool bom = false)
    {
        await js.InvokeVoidAsync("motorFiles.download", name, mime, (bom ? "\uFEFF" : "") + text);
    }

    private void Ask(string text, Action action)
    {
        Confirmation = text;
        pending = action;
        Notify();
    }

    public void Confirm()
    {
        var action = pending;
        pending = null;
        Confirmation = "";
        if (action is not null)
        {
            Execute(action, null);
        }
    }

    public void Cancel()
    {
        pending = null;
        pendingProjectText = null;
        Confirmation = "";
        Notify();
    }

    private void Execute(Action action, string? success)
    {
        try
        {
            action();
            if (success is not null)
            {
                Message = success;
                IsError = false;
            }
        }
        catch (Exception ex) when (ex is IOException or InvalidDataException or UnauthorizedAccessException or ArgumentException or InvalidOperationException)
        {
            Message = ex.Message;
            IsError = true;
        }
        Notify();
    }

    private async Task ExecuteAsync(Func<Task> action, string success)
    {
        try
        {
            await action();
            Message = success;
            IsError = false;
        }
        catch (Exception ex) when (ex is IOException or InvalidDataException or UnauthorizedAccessException or ArgumentException or InvalidOperationException or JSException)
        {
            Message = ex.Message;
            IsError = true;
        }
        Notify();
    }

    private void Notify()
    {
        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(null));
    }
}

    dirty: false,
    setDirty(value) {
        this.dirty = value;
    },
    download(filename, mime, text) {
        const blob = new Blob([text], { type: mime });
        const url = URL.createObjectURL(blob);
        const anchor = document.createElement('a');
        anchor.href = url;
        anchor.download = filename;
        document.body.appendChild(anchor);
        anchor.click();
        anchor.remove();
        setTimeout(() => URL.revokeObjectURL(url), 30000);
    }
};

    if (window.motorFiles.dirty) {
        event.preventDefault();
        event.returnValue = '';
    }
});

using System.Globalization;
using System.Text;
using MotorFrequency.Core.Domain;

namespace MotorFrequency.Core.Services;

public sealed class ChartGeometry
{
    public const double Width = 900;
    public const double Height = 370;
    public const double Left = 68;
    public const double Right = 860;
    public const double Top = 24;
    public const double Bottom = 316;
    private static readonly string[] palette = ["#1d6860", "#b65f32", "#566da1", "#9a7c27", "#85608b", "#4b7c8b", "#6d7743"];
    public MotorConfiguration Motor { get; }

    public ChartGeometry(MotorConfiguration motor)
    {
        Motor = motor;
    }

    public double X(double rpm)
    {
        return Left + (rpm - Motor.MinimumRpm) / (Motor.MaximumRpm - Motor.MinimumRpm) * (Right - Left);
    }

    public double Y(double hz)
    {
        return Bottom - hz / Motor.MaximumHz * (Bottom - Top);
    }

    public static string N(double value)
    {
        return value.ToString("0.###", CultureInfo.InvariantCulture);
    }

    public static string Color(int index)
    {
        return palette[Math.Abs(index) % palette.Length];
    }

    public string CandidateLine(Candidate candidate)
    {
        var lower = Motor.MinimumRpm;
        var upper = Math.Min(Motor.MaximumRpm, 60 * Motor.MaximumHz / candidate.Order);
        if (upper < lower)
        {
            return "";
        }
        return $"M {N(X(lower))} {N(Y(FrequencyEngine.At(candidate.Order, lower)))} L {N(X(upper))} {N(Y(FrequencyEngine.At(candidate.Order, upper)))}";
    }

    public IEnumerable<(double Value, double Position)> SpeedTicks()
    {
        for (var i = 0; i <= 6; i++)
        {
            var value = Motor.MinimumRpm + (Motor.MaximumRpm - Motor.MinimumRpm) * i / 6;
            yield return (value, X(value));
        }
    }

    public IEnumerable<(double Value, double Position)> FrequencyTicks()
    {
        for (var i = 0; i <= 5; i++)
        {
            var value = Motor.MaximumHz * i / 5;
            yield return (value, Y(value));
        }
    }

    public static IReadOnlyList<SpectrumPoint> DisplayPoints(Spectrum spectrum, int maximum = 1000)
    {
        if (spectrum.Points.Count <= maximum)
        {
            return spectrum.Points;
        }
        var output = new List<SpectrumPoint>();
        var bucketCount = maximum / 2;
        for (var bucket = 0; bucket < bucketCount; bucket++)
        {
            var start = bucket * spectrum.Points.Count / bucketCount;
            var end = (bucket + 1) * spectrum.Points.Count / bucketCount;
            var segment = spectrum.Points.GetRange(start, end - start);
            var low = segment.MinBy(x => x.Amplitude)!;
            var high = segment.MaxBy(x => x.Amplitude)!;
            if (low.FrequencyHz <= high.FrequencyHz)
            {
                output.Add(low);
                if (low != high) output.Add(high);
            }
            else
            {
                output.Add(high);
                output.Add(low);
            }
        }
        return output;
    }

    public static string SpectrumPolyline(Spectrum spectrum, double maximumHz, double minimumAmplitude, double maximumAmplitude)
    {
        var range = Math.Max(1e-9, maximumAmplitude - minimumAmplitude);
        return string.Join(" ", DisplayPoints(spectrum).Where(x => x.FrequencyHz <= maximumHz).Select(point =>
        {
            var x = Left + point.FrequencyHz / maximumHz * (Right - Left);
            var y = Bottom - (point.Amplitude - minimumAmplitude) / range * (Bottom - Top);
            return N(x) + "," + N(y);
        }));
    }

    public static string MotorArc(double startDegrees, double endDegrees, double radius, double centerX = 160, double centerY = 160)
    {
        var a = startDegrees * Math.PI / 180;
        var b = endDegrees * Math.PI / 180;
        var x1 = centerX + radius * Math.Cos(a);
        var y1 = centerY + radius * Math.Sin(a);
        var x2 = centerX + radius * Math.Cos(b);
        var y2 = centerY + radius * Math.Sin(b);
        return $"M {N(x1)} {N(y1)} A {N(radius)} {N(radius)} 0 {(endDegrees - startDegrees > 180 ? 1 : 0)} 1 {N(x2)} {N(y2)}";
    }

using MotorFrequency.Core.Domain;

namespace MotorFrequency.Core.Services;

public static class FrequencyEngine
{
    public static IReadOnlyList<Candidate> Candidates(MotorConfiguration motor, IReadOnlyList<CustomOrder> custom)
    {
        var output = new List<Candidate>();
        void Add(string id, string name, double order, string formula, string basis, string category, string key)
        {
            output.Add(new Candidate(id, name, order, formula, basis, category, key, !motor.HiddenCandidates.Contains(id)));
        }
        Add("rotation", "机械转频", 1, "fᵣ = n / 60", "机械轴每转一次的运动学参考；不等同于不平衡已被证实。", "运动学", "speed");
        if (motor.Topology == MotorTopology.BrushlessPermanentMagnet)
        {
            Add("electrical", "电气基频", motor.PolePairs, "fₑ = p × n / 60", "仅适用于同步运行的永磁无刷拓扑，p 为极对数；不是电磁力阶次。", "电气参考", "rotor");
            if (motor.IncludeSlotPass)
            {
                Add("slot", "槽通过参考", motor.Slots, "fₛ = Q × n / 60", "转子参考点每转经过 Q 个等间距定子槽；只表示相对运动周期，不指认声压主峰。", "运动学", "slots");
            }
            if (motor.IncludeIdealCogging && motor.SymmetricUnskewedAssumption && !string.IsNullOrWhiteSpace(motor.AssumptionNotes))
            {
                var order = LeastCommonMultiple(motor.Slots, 2 * motor.PolePairs);
                Add("cogging", "理想齿槽周期候选", order, "f𝚌 = LCM(Q, 2p) × n / 60", "等槽、对称永磁结构的理想周期；不含制造偏差、斜槽、饱和与负载影响。用户依据：" + motor.AssumptionNotes, "假设候选", "slots");
            }
        }
        else
        {
            Add("commutator", "单刷换向片通过参考", motor.CommutatorSegments, "fₖ = K × n / 60", "理想等宽 K 片换向器相对单个固定电刷的片通过；全机脉冲可能重合或出现不同谐波，不乘电刷数。", "运动学", "brush");
        }
        foreach (var order in custom.Where(x => x.Enabled))
        {
            Add("custom:" + order.Id, order.Name, order.Order, "f = k × n / 60", order.Basis, "用户候选", "custom");
        }
        return output;
    }

    public static double At(double order, double rpm)
    {
        if (!double.IsFinite(order) || !double.IsFinite(rpm) || order <= 0 || rpm < 0)
        {
            throw new ArgumentOutOfRangeException(nameof(order), "阶次必须大于零、转速非负且均为有限数值。");
        }
        return order * rpm / 60.0;
    }

    public static int GreatestCommonDivisor(int a, int b)
    {
        if (a <= 0 || b <= 0)
        {
            throw new ArgumentOutOfRangeException(nameof(a));
        }
        while (b != 0)
        {
            (a, b) = (b, a % b);
        }
        return a;
    }

    public static int LeastCommonMultiple(int a, int b)
    {
        return checked(a / GreatestCommonDivisor(a, b) * b);
    }

    public static IReadOnlyList<double> Speeds(MotorConfiguration motor)
    {
        var errors = ProjectValidator.ValidateMotor(motor);
        if (errors.Count > 0)
        {
            throw new InvalidDataException(string.Join("; ", errors.Select(x => x.Message)));
        }
        var values = new List<double>();
        var count = (int)Math.Floor((motor.MaximumRpm - motor.MinimumRpm) / motor.StepRpm);
        for (var index = 0; index <= count; index++)
        {
            values.Add(motor.MinimumRpm + index * motor.StepRpm);
        }
        if (Math.Abs(values[^1] - motor.MaximumRpm) > 1e-7)
        {
            values.Add(motor.MaximumRpm);
        }
        return values;
    }

    public static IReadOnlyList<FrequencyRow> Rows(Project project, bool fullScan)
    {
        var speeds = fullScan ? Speeds(project.Motor) : [project.Motor.InspectRpm];
        var candidates = Candidates(project.Motor, project.CustomOrders).Where(x => x.Visible);
        var result = new List<FrequencyRow>();
        foreach (var candidate in candidates)
        {
            foreach (var rpm in speeds)
            {
                result.Add(new(candidate.Id, candidate.Name, candidate.Order, rpm, At(candidate.Order, rpm), candidate.Formula, candidate.Basis));
            }
        }
        return result;
    }

    public static IReadOnlyList<BandCrossing> Crossings(Project project)
    {
        var result = new List<BandCrossing>();
        foreach (var candidate in Candidates(project.Motor, project.CustomOrders).Where(x => x.Visible))
        {
            foreach (var band in project.Bands.Where(x => x.Enabled))
            {
                var low = Math.Max(project.Motor.MinimumRpm, 60 * band.LowerHz / candidate.Order);
                var high = Math.Min(project.Motor.MaximumRpm, 60 * band.UpperHz / candidate.Order);
                if (low <= high)
                {
                    result.Add(new(candidate.Id, candidate.Name, band.Name, low, high, band.LowerHz, band.UpperHz, band.Basis));
                }
            }
        }
        return result.OrderBy(x => x.LowerRpm).ToList();
    }

    public static IReadOnlyList<PeakMatch> MatchPeaks(Project project, Spectrum spectrum)
    {
        var candidates = Candidates(project.Motor, project.CustomOrders).Where(x => x.Visible).ToList();
        var result = new List<PeakMatch>();
        var points = spectrum.Points;
        var peaks = Enumerable.Range(1, Math.Max(0, points.Count - 2))
            .Where(i => points[i].Amplitude > points[i - 1].Amplitude && points[i].Amplitude >= points[i + 1].Amplitude)
            .Select(i => points[i])
            .OrderByDescending(x => x.Amplitude)
            .Take(project.Review.MaximumPeaks);
        foreach (var peak in peaks)
        {
            var nearby = candidates
                .Select(x => (Candidate: x, Frequency: At(x.Order, spectrum.Rpm)))
                .Where(x => Math.Abs(x.Frequency - peak.FrequencyHz) <= project.Review.MatchToleranceHz)
                .OrderBy(x => Math.Abs(x.Frequency - peak.FrequencyHz))
                .ToList();
            if (nearby.Count == 0)
            {
                result.Add(new(spectrum.Name, spectrum.Rpm, peak.FrequencyHz, peak.Amplitude, spectrum.Unit, "无邻近候选", null, null, "容差内无候选；不排除未建模激励。"));
            }
            foreach (var item in nearby)
            {
                result.Add(new(spectrum.Name, spectrum.Rpm, peak.FrequencyHz, peak.Amplitude, spectrum.Unit, item.Candidate.Name, item.Frequency, peak.FrequencyHz - item.Frequency, "仅频率邻近，不能证明来源；同频候选均保留。"));
            }
        }
        return result;
    }

    public static IReadOnlyList<ComparisonRow> Compare(Project project, Scenario snapshot)
    {
        var current = Candidates(project.Motor, project.CustomOrders).Where(x => x.Visible).ToDictionary(x => x.Id);
        var previous = Candidates(snapshot.Motor, snapshot.CustomOrders).Where(x => x.Visible).ToDictionary(x => x.Id);
        var result = new List<ComparisonRow>();
        foreach (var id in current.Keys.Union(previous.Keys))
        {
            current.TryGetValue(id, out var now);
            previous.TryGetValue(id, out var before);
            double? currentHz = now is null ? null : At(now.Order, project.Motor.InspectRpm);
            double? previousHz = before is null ? null : At(before.Order, project.Motor.InspectRpm);
            var statement = now is null ? "当前未启用" : before is null ? "快照未启用" : "统一按当前观察转速换算；差值=当前−快照";
            if (project.Motor.Topology != snapshot.Motor.Topology)
            {
                statement += "；拓扑不同，只作频率参考对照";
            }
            result.Add(new(now?.Name ?? before!.Name, now?.Order, before?.Order, currentHz, previousHz, currentHz - previousHz, statement));
        }
        return result;
    }

    public static IReadOnlyList<string> QualityNotes(Project project)
    {
        var notes = new List<string>();
        if (project.Demonstration)
        {
            notes.Add("本项目为模拟演示，不能证明实际车辆噪声来源或改善效果。");
        }
        foreach (var spectrum in project.Spectra)
        {
            if (spectrum.Rpm < project.Motor.MinimumRpm || spectrum.Rpm > project.Motor.MaximumRpm)
            {
                notes.Add(spectrum.Name + "：频谱转速在当前扫描范围外，Campbell 图不显示该切片。");
            }
            if (spectrum.Origin == DataOrigin.Simulated)
            {
                notes.Add(spectrum.Name + "：模拟频谱，保留来源标记。");
            }
            if (spectrum.Points.Count > 1 && spectrum.Points.Zip(spectrum.Points.Skip(1)).Any(x => x.Second.FrequencyHz - x.First.FrequencyHz > project.Review.MatchToleranceHz * 2))
            {
                notes.Add(spectrum.Name + "：部分频率间距大于两倍邻近容差，粗分辨率可能漏检；未插值补点。");
            }
        }
        if (project.Bands.Any(x => x.Enabled && x.UpperHz > project.Motor.MaximumHz))
        {
            notes.Add("部分敏感频段超出图表上限，交叠表仍按完整范围计算。");
        }
        var duplicates = Candidates(project.Motor, project.CustomOrders).Where(x => x.Visible).GroupBy(x => x.Order).Where(x => x.Count() > 1);
        foreach (var group in duplicates)
        {
            notes.Add($"{Labels.Number(group.Key)} 阶存在多个候选：{string.Join("、", group.Select(x => x.Name))}；图线可能重合，依据分别保留。");
        }
        notes.Add(Software.Boundary);
        return notes;
    }

    public static string ContextHash(Project project)
    {
        var input = new
        {
            projectId = project.Id,
            projectName = project.Name,
            demonstration = project.Demonstration,
            motor = project.Motor,
            customOrders = project.CustomOrders
        };

    }

    public static IReadOnlyList<int> ParseIntegers(string text, string field, int min, int max)
    {
        if (string.IsNullOrWhiteSpace(text))
        {
            throw new InvalidDataException(field + "不能为空。");
        }
        var values = new List<int>();
        var tokens = text.Split(',', StringSplitOptions.TrimEntries);
        if (tokens.Length > 12)
        {
            throw new InvalidDataException(field + "最多输入 12 个整数，以英文逗号分隔。");
        }
        foreach (var token in tokens)
        {
            if (!int.TryParse(token, NumberStyles.Integer, CultureInfo.InvariantCulture, out var value) || value < min || value > max)
            {
                throw new InvalidDataException($"{field} 中的“{token}”无效，范围 {min}～{max}。");
            }
            if (values.Contains(value))
            {
                throw new InvalidDataException(field + "包含重复值 " + value + "，请删除重复项。");
            }
            values.Add(value);
        }
        return values.Order().ToList();
    }

    public static IReadOnlyList<ValidationIssue> Validate(SweepRequest request, MotorTopology topology)
    {
        var issues = new List<ValidationIssue>();
        if (!double.IsFinite(request.ReferenceRpm) || request.ReferenceRpm < 0 || request.ReferenceRpm > 100000)
        {
            issues.Add(new("referenceRpm", "参考转速需为 0～100000 r/min。"));
        }
        if (!double.IsFinite(request.ObservedHz) || request.ObservedHz < 0 || request.ObservedHz > 1000000)
        {
            issues.Add(new("observedHz", "观察频率需为 0～1000000 Hz。"));
        }
        if (!double.IsFinite(request.ToleranceHz) || request.ToleranceHz < 0 || request.ToleranceHz > 1000)
        {
            issues.Add(new("toleranceHz", "容差需为 0～1000 Hz。"));
        }
        void Check(IReadOnlyList<int>? list, string path, int minimum, int maximum)
        {
            if (list is null || list.Count is < 1 or > 12)
            {
                issues.Add(new(path, "每组需 1～12 个整数。"));
            }
            else if (list.Distinct().Count() != list.Count || list.Any(x => x < minimum || x > maximum))
            {
                issues.Add(new(path, "列表含重复项或超范围整数。"));
            }
        }
        if (topology == MotorTopology.BrushlessPermanentMagnet)
        {
            Check(request.PolePairs, "polePairs", 1, 64);
            Check(request.Slots, "slots", 1, 512);
        }
        else if (topology == MotorTopology.BrushedPermanentMagnet)
        {
            Check(request.Segments, "segments", 2, 512);
        }
        else
        {
            issues.Add(new("topology", "不支持的拓扑。"));
        }
        return issues;
    }

    public static IReadOnlyList<SweepRow> Calculate(Project project, SweepRequest request)
    {
        var errors = Validate(request, project.Motor.Topology);
        if (errors.Count > 0)
        {
            throw new InvalidDataException(string.Join("；", errors.Select(x => x.Path + "：" + x.Message)));
        }
        var result = new List<SweepRow>();
        var poles = project.Motor.Topology == MotorTopology.BrushlessPermanentMagnet ? request.PolePairs : [project.Motor.PolePairs];
        var slots = project.Motor.Topology == MotorTopology.BrushlessPermanentMagnet ? request.Slots : [project.Motor.Slots];
        var segments = project.Motor.Topology == MotorTopology.BrushedPermanentMagnet ? request.Segments : [project.Motor.CommutatorSegments];
        foreach (var pole in poles)
        {
            foreach (var slot in slots)
            {
                foreach (var segment in segments)
                {
                    var configuration = ProjectCodec.Clone(project.Motor);
                    configuration.PolePairs = pole;
                    configuration.Slots = slot;
                    configuration.CommutatorSegments = segment;
                    foreach (var candidate in FrequencyEngine.Candidates(configuration, project.CustomOrders).Where(x => x.Visible))
                    {
                        var frequency = FrequencyEngine.At(candidate.Order, request.ReferenceRpm);
                        var difference = frequency - request.ObservedHz;
                        result.Add(new(configuration.Topology, pole, slot, segment, candidate.Name, candidate.Order, request.ReferenceRpm, frequency, request.ObservedHz, difference, Math.Abs(difference) <= request.ToleranceHz, candidate.Basis));
                    }
                }
            }
        }
        return result;
    }

    public static IReadOnlyList<ReverseRow> Reverse(Project project, double observedHz, double toleranceHz)
    {
        if (!double.IsFinite(observedHz) || observedHz < 0 || observedHz > 1000000)
        {
            throw new InvalidDataException("观察频率需为 0～1000000 Hz。");
        }
        if (!double.IsFinite(toleranceHz) || toleranceHz < 0 || toleranceHz > 1000)
        {
            throw new InvalidDataException("容差需为 0～1000 Hz。");
        }
        var output = new List<ReverseRow>();
        foreach (var candidate in FrequencyEngine.Candidates(project.Motor, project.CustomOrders).Where(x => x.Visible))
        {
            var center = observedHz * 60 / candidate.Order;
            var lower = Math.Max(0, observedHz - toleranceHz) * 60 / candidate.Order;
            var upper = (observedHz + toleranceHz) * 60 / candidate.Order;
            var inScan = center >= project.Motor.MinimumRpm && center <= project.Motor.MaximumRpm;
            output.Add(new(candidate.Name, candidate.Order, center, lower, upper, inScan, candidate.Basis));
        }
        return output.OrderBy(x => x.CenterRpm).ToList();
    }

    public static string Csv(Project project, IReadOnlyList<SweepRow> rows)
    {

"模拟演示" : "人工参数扫查"), ExportService.CsvCell(Labels.Topology(row.Topology)), row.PolePairs, row.Slots, row.Segments, ExportService.CsvCell(row.Candidate), N(row.Order), N(row.ReferenceRpm), N(row.FrequencyHz), N(row.ObservedHz), N(row.DifferenceHz), row.WithinTolerance ? "true" : "false", ExportService.CsvCell(row.Basis)));
        }
        return text.ToString();
    }

using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Text.Json.Serialization;
using MotorFrequency.Core.Domain;

namespace MotorFrequency.Core.Services;

public static class ProjectCodec
{
    public const int MaximumFileBytes = 8 * 1024 * 1024;
    public static JsonSerializerOptions Options { get; } = new()
    {
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        WriteIndented = true,
        Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping,
        Converters = { new JsonStringEnumConverter() },
        UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow,
        MaxDepth = 32
    };

    public static string Serialize(Project project)
    {
        ProjectValidator.ThrowIfInvalid(project);
        return JsonSerializer.Serialize(project, Options);
    }

    public static Project Parse(string text)
    {
        if (Encoding.UTF8.GetByteCount(text) > MaximumFileBytes)
        {
            throw new InvalidDataException("项目超过 8 MiB，未载入。请拆分频谱数据。");
        }
        try
        {
            using var document = JsonDocument.Parse(text, new JsonDocumentOptions { MaxDepth = 32 });
            var root = document.RootElement;
            if (!root.TryGetProperty("formatVersion", out var version) || version.GetInt32() != Software.FormatVersion)
            {
                throw new InvalidDataException("formatVersion 缺失或不支持；仅支持版本 1，原文件未修改。");
            }
            string[] required = ["id", "name", "motor", "customOrders", "bands", "spectra", "scenarios", "history", "review"];
            foreach (var property in required)
            {
                if (!root.TryGetProperty(property, out var value) || value.ValueKind == JsonValueKind.Null)
                {
                    throw new InvalidDataException($"项目必需字段 {property} 缺失或为空。");
                }
            }
            RequiredSchema.Validate(root);
            var project = JsonSerializer.Deserialize<Project>(text, Options)
                ?? throw new InvalidDataException("项目不是有效对象。");
            ProjectValidator.ThrowIfInvalid(project);
            return project;
        }
        catch (Exception ex) when (ex is JsonException or InvalidOperationException or FormatException)
        {
            throw new InvalidDataException("项目 JSON 损坏或字段类型错误：" + ex.Message, ex);
        }
    }

    public static T Clone<T>(T value)
    {
        return JsonSerializer.Deserialize<T>(JsonSerializer.Serialize(value, Options), Options)
            ?? throw new InvalidOperationException("无法复制对象。");
    }

    public static string Hash(string text)
    {
        return Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(text))).ToLowerInvariant();
    }

    public static Project Duplicate(Project project)
    {
        var copy = Clone(project);
        copy.Id = Guid.NewGuid();
        copy.Name = project.Name + "（副本）";
        copy.Archived = false;
        copy.CreatedAt = DateTimeOffset.UtcNow;
        copy.ModifiedAt = copy.CreatedAt;
        copy.History.Add(new AuditEvent { Action = "复制项目", Detail = "来源项目 " + project.Id });
        return copy;
    }
}

using System.Text;
using MotorFrequency.Core.Domain;

namespace MotorFrequency.Core.Services;

public sealed class ProjectStore
{
    private readonly string directory;
    private readonly object gate = new();
    public string DirectoryPath => directory;

    public ProjectStore(string directory)
    {
        this.directory = Path.GetFullPath(directory);
        Directory.CreateDirectory(this.directory);
    }

    private string ProjectPath(Guid id)
    {
        if (id == Guid.Empty)
        {
            throw new InvalidDataException("项目标识不能为空。");
        }
        return Path.Combine(directory, id.ToString("N") + Software.Extension);
    }

    public StoreWrite Save(Project project, string? expectedRevision)
    {
        ProjectValidator.ThrowIfInvalid(project);
        lock (gate)
        {
            var path = ProjectPath(project.Id);
            var exists = File.Exists(path);
            if (exists)
            {
                var oldText = File.ReadAllText(path, Encoding.UTF8);
                var actualRevision = ProjectCodec.Hash(oldText);
                if (expectedRevision is null || actualRevision != expectedRevision)
                {
                    throw new InvalidDataException("磁盘项目已存在或被外部修改，未覆盖。请先从项目库打开核对，或复制为新项目。");
                }
            }
            else if (expectedRevision is not null)
            {
                throw new InvalidDataException("原项目文件已被移动或删除，未写入。请复制项目后保存。");
            }
            var text = ProjectCodec.Serialize(project);
            if (Encoding.UTF8.GetByteCount(text) > ProjectCodec.MaximumFileBytes)
            {
                throw new InvalidDataException("保存内容超过 8 MiB，未写入。");
            }
            var temporary = path + "." + Guid.NewGuid().ToString("N") + ".tmp";
            try
            {
                File.WriteAllText(temporary, text, new UTF8Encoding(false));
                if (exists)
                {
                    File.Copy(path, path + ".bak", true);
                }
                File.Move(temporary, path, true);
            }
            finally
            {
                if (File.Exists(temporary))
                {
                    File.Delete(temporary);
                }
            }
            return new(path, ProjectCodec.Hash(text), exists);
        }
    }

    public StoreRead Open(Guid id)
    {
        lock (gate)
        {
            var path = ProjectPath(id);
            var text = ReadLimited(path);
            var project = ProjectCodec.Parse(text);
            if (project.Id != id)
            {
                throw new InvalidDataException("项目内容标识与文件名不一致，请通过导入入口核对。");
            }
            return new(project, ProjectCodec.Hash(text));
        }
    }

    public Project ReadBackupAsCopy(Guid id)
    {
        lock (gate)
        {
            var text = ReadLimited(ProjectPath(id) + ".bak");
            var project = ProjectCodec.Duplicate(ProjectCodec.Parse(text));
            project.Name += "（备份恢复）";
            project.History.Add(new AuditEvent { Action = "读取备份", Detail = "以新项目恢复，不覆盖原文件" });
            return project;
        }
    }

    public IReadOnlyList<ProjectSummary> List(out IReadOnlyList<string> warnings)
    {
        var result = new List<ProjectSummary>();
        var errors = new List<string>();
        lock (gate)
        {
            var files = Directory.EnumerateFiles(directory, "*" + Software.Extension).Take(501).ToList();
            if (files.Count > 500)
            {
                errors.Add("项目库超过 500 个文件，仅列出前 500 个；请归档到其他目录。");
            }
            foreach (var file in files.Take(500))
            {
                try
                {
                    var p = ProjectCodec.Parse(ReadLimited(file));
                    if (Path.GetFileName(file) != p.Id.ToString("N") + Software.Extension)
                    {
                        errors.Add(Path.GetFileName(file) + "：标识不匹配，未列入可打开项目。");
                        continue;
                    }
                    result.Add(new(p.Id, p.Name, p.Demonstration, p.Archived, p.ModifiedAt, file, File.Exists(file + ".bak")));
                }
                catch (Exception ex) when (ex is IOException or InvalidDataException or UnauthorizedAccessException or DecoderFallbackException)
                {
                    errors.Add(Path.GetFileName(file) + "：" + ex.Message);
                }
            }
        }
        warnings = errors;
        return result.OrderByDescending(x => x.ModifiedAt).ToList();
    }

    private static string ReadLimited(string path)
    {
        var info = new FileInfo(path);
        if (!info.Exists)
        {
            throw new FileNotFoundException("找不到项目文件。", path);
        }
        if (info.Length > ProjectCodec.MaximumFileBytes)
        {
            throw new InvalidDataException("项目文件超过 8 MiB。");
        }
        return File.ReadAllText(path, new UTF8Encoding(false, true));
    }

    private static void ValidateCollections(Project p, List<ValidationIssue> issues)
    {

        }

        Limit(issues, "bands", p.Bands.Count, 30);
        Limit(issues, "spectra", p.Spectra.Count, 12);
        Limit(issues, "scenarios", p.Scenarios.Count, 12);
        Limit(issues, "history", p.History.Count, 2000);
        Unique(issues, "customOrders", p.CustomOrders.Where(x => x is not null).Select(x => x.Id));
        Unique(issues, "bands", p.Bands.Where(x => x is not null).Select(x => x.Id));
        Unique(issues, "spectra", p.Spectra.Where(x => x is not null).Select(x => x.Id));
        Unique(issues, "scenarios", p.Scenarios.Where(x => x is not null).Select(x => x.Id));
        foreach (var order in p.CustomOrders)
        {
            if (order is null)
            {
                issues.Add(new("customOrders", "候选对象不能为空。"));
                continue;
            }
            ValidateOrder(order, "customOrders/" + order.Id, issues);
        }
        foreach (var band in p.Bands)
        {
            if (band is null)

                issues.Add(new("bands", "敏感频段对象不能为空。"));
                continue;
            }
            var path = "bands/" + band.Id;
            Text(issues, path + "/name", band.Name, 100, true);
            Text(issues, path + "/basis", band.Basis, 2000, true);
            Range(issues, path + "/lowerHz", band.LowerHz, 0, 1000000);
            Range(issues, path + "/upperHz", band.UpperHz, 0, 1000000);
            if (band.UpperHz < band.LowerHz)
            {
                issues.Add(new(path, "频段上限不能小于下限。"));
            }
            if (!Enum.IsDefined(band.Origin))
            {
                issues.Add(new(path + "/origin", "未知来源分类。"));
            }
        }
        var totalPoints = 0;
        foreach (var spectrum in p.Spectra)

                issues.Add(new("spectra", "频谱对象不能为空。"));
                continue;

            issues.AddRange(ValidateSpectrum(spectrum));

    public static IReadOnlyList<ValidationIssue> ValidateSpectrum(Spectrum spectrum)
    {
        var issues = new List<ValidationIssue>();
        var path = "spectra/" + spectrum.Id;
        Text(issues, path + "/name", spectrum.Name, 100, true);
        Text(issues, path + "/fileName", spectrum.FileName, 255, false);
        Text(issues, path + "/reference", spectrum.Reference, 200, true);
        Text(issues, path + "/condition", spectrum.Condition, 1000, true);
        Range(issues, path + "/rpm", spectrum.Rpm, 0, 100000);
        if (!Enum.IsDefined(spectrum.Origin))
        {
            issues.Add(new(path + "/origin", "未知来源分类。"));
        }
        if (spectrum.Unit is not ("dB" or "Pa" or "m/s²" or "g" or "A" or "V" or "任意单位"))
        {
            issues.Add(new(path + "/unit", "请选择支持的频谱单位。"));
        }
        if (spectrum.Points is null || spectrum.Points.Count < 2 || spectrum.Points.Count > 10000)
        {
            issues.Add(new(path + "/points", "每条频谱需 2～10000 个点。"));
            return issues;
        }
        double previous = -1;
        for (var i = 0; i < spectrum.Points.Count; i++)
        {
            var point = spectrum.Points[i];
            if (point is null)
            {
                issues.Add(new(path + "/points/" + i, "频谱点不能为空。"));
                continue;
            }
            Range(issues, path + $"/points/{i}/frequencyHz", point.FrequencyHz, 0, 1000000);
            Range(issues, path + $"/points/{i}/amplitude", point.Amplitude, -1000000000, 1000000000);
            if (point.FrequencyHz <= previous)
            {
                issues.Add(new(path + $"/points/{i}", "频率必须严格递增，无重复。"));
            }
            if (spectrum.Unit != "dB" && point.Amplitude < 0)
            {
                issues.Add(new(path + $"/points/{i}", "线性幅值频谱不接受负值。"));
            }
            previous = point.FrequencyHz;
            if (issues.Count > 25)
            {
                issues.Add(new(path, "错误过多，停止继续列出；整条频谱不导入。"));
                break;
            }
        }
        return issues;
    }

using System.Reflection;
using System.Text.Json;
using MotorFrequency.Core.Domain;

namespace MotorFrequency.Core.Services;

public static class RequiredSchema
{
    public static void Validate(JsonElement root)
    {
        var issues = new List<string>();
        Visit(root, typeof(Project), "project", issues);
        if (issues.Count > 0)
        {
            throw new InvalidDataException("项目格式字段不完整：\n" + string.Join("\n", issues.Take(25)));
        }
    }

    private static void Visit(JsonElement element, Type type, string path, List<string> issues)
    {
        if (issues.Count >= 25) return;
        if (element.ValueKind == JsonValueKind.Null)
        {
            issues.Add(path + " 不能为 null。");
            return;
        }
        if (type == typeof(string) || type.IsValueType)
        {
            return;
        }
        if (type.IsGenericType && type.GetGenericTypeDefinition() == typeof(List<>))
        {
            if (element.ValueKind != JsonValueKind.Array)
            {
                issues.Add(path + " 必须为数组。");
                return;
            }
            var itemType = type.GetGenericArguments()[0];
            var index = 0;
            foreach (var child in element.EnumerateArray())
            {
                Visit(child, itemType, path + "/" + index, issues);
                index++;
                if (issues.Count >= 25) break;
            }
            return;
        }
        if (element.ValueKind != JsonValueKind.Object)
        {
            issues.Add(path + " 必须为对象。");
            return;
        }
        foreach (var property in type.GetProperties(BindingFlags.Instance | BindingFlags.Public))
        {
            if (!property.CanRead || !property.CanWrite) continue;
            var name = JsonNamingPolicy.CamelCase.ConvertName(property.Name);
            if (!element.TryGetProperty(name, out var child))
            {
                issues.Add(path + "/" + name + " 缺失。版本 1 不静默填入默认值。");
                continue;
            }
            Visit(child, property.PropertyType, path + "/" + name, issues);
        }
    }
}

using System.Globalization;
using System.Text;
using MotorFrequency.Core.Domain;

namespace MotorFrequency.Core.Services;

public static class SpectrumCsv
{
    public const int MaximumBytes = 1024 * 1024;

    public static CsvImportResult Parse(string content, SpectrumMetadata metadata)
    {
        var issues = new List<ValidationIssue>();
        var points = new List<SpectrumPoint>();
        if (Encoding.UTF8.GetByteCount(content) > MaximumBytes)
        {
            return new(null, [new("file", "CSV 超过 1 MiB；请减少频点。")], 0);
        }
        var lines = content.TrimStart('\uFEFF').Replace("\r\n", "\n").Replace('\r', '\n').Split('\n');
        if (lines.Length < 3)
        {
            return new(null, [new("file", "需表头和至少两行频谱点。")], 0);
        }
        if (lines[0].Trim() != "frequency_hz,amplitude")
        {
            return new(null, [new("line 1", "表头必须为 frequency_hz,amplitude，使用英文逗号。")], 0);
        }
        for (var index = 1; index < lines.Length; index++)
        {
            if (string.IsNullOrWhiteSpace(lines[index]))
            {
                if (index < lines.Length - 1 && lines.Skip(index).Any(x => !string.IsNullOrWhiteSpace(x)))
                {
                    issues.Add(new("line " + (index + 1), "中间空行不接受；请移除后重试。"));
                }
                continue;
            }
            var cells = lines[index].Split(',');
            if (cells.Length != 2)
            {
                issues.Add(new("line " + (index + 1), "每行必须正好两列，不支持千位分隔符。"));
                continue;
            }
            var style = NumberStyles.Float;
            var culture = CultureInfo.InvariantCulture;
            if (!double.TryParse(cells[0], style, culture, out var hz) || !double.IsFinite(hz))
            {
                issues.Add(new("line " + (index + 1) + ", frequency_hz", "频率必须为有限数值，单位 Hz，小数点使用英文句点。"));
                continue;
            }
            if (!double.TryParse(cells[1], style, culture, out var amplitude) || !double.IsFinite(amplitude))
            {
                issues.Add(new("line " + (index + 1) + ", amplitude", "幅值必须为有限数值。"));
                continue;
            }
            points.Add(new SpectrumPoint { FrequencyHz = hz, Amplitude = amplitude });
            if (issues.Count >= 25 || points.Count > 10000)
            {
                issues.Add(new("file", "超过 25 条错误或 10000 点，停止解析。"));
                break;
            }
        }
        var spectrum = new Spectrum
        {
            Name = metadata.Name,
            Rpm = metadata.Rpm,
            Unit = metadata.Unit,
            Reference = metadata.Reference,
            Condition = metadata.Condition,
            Origin = metadata.Origin,
            FileName = metadata.FileName,
            InputSha256 = ProjectCodec.Hash(content),
            Points = points
        };
        issues.AddRange(ProjectValidator.ValidateSpectrum(spectrum));
        return new(issues.Count == 0 ? spectrum : null, issues, points.Count);
    }

    public static string Template()
    {
        return "frequency_hz,amplitude\n0,20\n30,25\n59,32\n60,48\n61,33\n90,21\n";
    }

    public static string Export(Spectrum spectrum)
    {
        var text = new StringBuilder("frequency_hz,amplitude\n");
        foreach (var point in spectrum.Points)
        {
            text.Append(point.FrequencyHz.ToString("R", CultureInfo.InvariantCulture));
            text.Append(',');
            text.AppendLine(point.Amplitude.ToString("R", CultureInfo.InvariantCulture));
        }
        return text.ToString();
    }
}

using MotorFrequency.Core.Domain;

namespace MotorFrequency.Core.Services;

public sealed record SpectrumSummary(
    double MinimumHz,
    double MaximumHz,
    double MinimumSpacingHz,
    double MaximumSpacingHz,
    double MedianSpacingHz,
    bool UniformSpacing,
    double MaximumAmplitude,
    double MaximumAtHz,
    int PointsInsideView,
    int TotalPoints,
    string Statement);

public static class SpectrumStatistics
{
    public static SpectrumSummary Inspect(Spectrum spectrum, double maximumViewHz)
    {
        var issues = ProjectValidator.ValidateSpectrum(spectrum);
        if (issues.Count > 0)
        {
            throw new InvalidDataException("无法汇总无效频谱：" + issues[0].Message);
        }
        var points = spectrum.Points;
        var spacings = new List<double>();
        for (var i = 1; i < points.Count; i++)
        {
            spacings.Add(points[i].FrequencyHz - points[i - 1].FrequencyHz);
        }
        spacings.Sort();
        var middle = spacings.Count / 2;
        var median = spacings.Count % 2 == 0
            ? (spacings[middle - 1] + spacings[middle]) / 2
            : spacings[middle];
        var minimum = spacings[0];
        var maximum = spacings[^1];
        var uniform = maximum - minimum <= Math.Max(1e-9, maximum * 1e-6);
        var maximumPoint = points.MaxBy(x => x.Amplitude)!;
        var inside = points.Count(x => x.FrequencyHz <= maximumViewHz);
        var statement = uniform
            ? "频率间距近似一致；仅汇总输入网格，不推断时域采样率或FFT分辨率。"
            : "频率间距不一致；保留原始网格，不做重采样，峰值邻近结果需结合局部间距阅读。";
        if (maximumPoint == points[0] || maximumPoint == points[^1])
        {
            statement += " 最大输入幅值在端点，不会作为内部谱峰参与邻近核对。";
        }
        if (inside < points.Count)
        {
            statement += " 部分原始频点超出当前图表上限。";
        }
        return new SpectrumSummary(
            points[0].FrequencyHz,
            points[^1].FrequencyHz,
            minimum,
            maximum,
            median,
            uniform,
            maximumPoint.Amplitude,
            maximumPoint.FrequencyHz,
            inside,
            points.Count,
            statement);
    }
}
